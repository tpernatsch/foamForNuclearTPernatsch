36 Foam::tmp<Foam::scalarField>
37 Foam::regionCoupledFvPatchScalarField::weights() const
38 {
39  if (coupled_)
40  {
41  const fvPatch& patch = this->patch();
42 
43  const scalarField deltas
44  (
45  patch.nf() & patch.delta()
46  );
47 
48  const scalarField deltaCoeffs(1/deltas);
49 
50  const fvPatch& nbrPatch = regionCoupledPatch_.neighbFvPatch();
51 
52  const scalarField nbrDeltas
53  (
54  regionCoupledPatch_.regionCoupledPatch().interpolate
55  (
56  nbrPatch.nf() & nbrPatch.delta()
57  )
58  );
59 
60  const scalarField nbrDeltaCoeffs(1/nbrDeltas);
61 
62  tmp<scalarField> tw(new scalarField(deltas.size()));
63  scalarField& w = tw.ref();
64 
65  forAll(deltaCoeffs, facei)
66  {
67  scalar di = deltaCoeffs[facei];
68  scalar dni = nbrDeltaCoeffs[facei];
69 
70  w[facei] = di/(di + dni);
71  }
72 
73  return tw;
74  }
75  else
76  {
77  return patch().weights();
78  }
79 }

104
