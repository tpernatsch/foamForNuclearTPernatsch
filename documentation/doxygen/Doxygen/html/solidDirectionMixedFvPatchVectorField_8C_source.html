119 void solidDirectionMixedFvPatchVectorField::evaluate(const Pstream::commsTypes)
120 {
121  if (!this->updated())
122  {
123  this->updateCoeffs();
124  }
125 
126  bool secondOrder_(false);
127 
128  vectorField n = patch().nf();
129  vectorField delta = patch().delta();
130  vectorField k = delta - n*(n&delta);
131 
132  const fvPatchField<tensor>& gradD =
133  patch().lookupPatchField<volTensorField, tensor>
134  (
135  "grad" + internalField().name()
136  );
137 
138 
139  // Calc limited snGrad correction
140 
141  vectorField snGradCorrection =
142  - (k & gradD.patchInternalField())*patch().deltaCoeffs();
143 
144  if (limitCoeff_ < (1.0 - SMALL))
145  {
146  vectorField uncorrectedSnGrad =
147  (
148  *this
149  - patchInternalField()
150  )*patch().deltaCoeffs();
151 
152  scalarField limiter =
153  (
154  min
155  (
156  limitCoeff_*mag(uncorrectedSnGrad + snGradCorrection)
157  /((1 - limitCoeff_)*mag(snGradCorrection) + SMALL),
158  1.0
159  )
160  );
161 
162  snGradCorrection *= limiter;
163  }
164 
165  Field<vector> normalValue = transform(valueFraction(), refValue());
166 
167  Field<vector> gradValue =
168  this->patchInternalField()
169  - snGradCorrection/patch().deltaCoeffs()
170  //+ (k & gradD.patchInternalField())
171  + refGrad()/patch().deltaCoeffs();
172 
173  if (secondOrder_)
174  {
175  vectorField nGradDP = (n & gradD.patchInternalField());
176 
177  gradValue =
178  patchInternalField()
179  + (k & gradD.patchInternalField())
180  + 0.5*(nGradDP + refGrad())/patch().deltaCoeffs();
181  }
182 
183  Field<vector> transformGradValue =
184  transform(I - valueFraction(), gradValue);
185 
186  Field<vector>::operator=(normalValue + transformGradValue);
