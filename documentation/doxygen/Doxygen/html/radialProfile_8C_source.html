103 Foam::tmp<Foam::scalarField> Foam::radialProfile::radialLocations() const
104 {
105  const fvMesh& referenceMesh =
106  (
107  mesh_.foundObject<fvMesh>("referenceMesh")
108  ) ?
109  mesh_.lookupObject<fvMesh>("referenceMesh")
110  :
111  mesh_;
112 
113  tmp<scalarField> tr(new scalarField(addr_.size()));
114  scalarField& r = tr.ref();
115 
116  const vectorField& C = referenceMesh.C();
117 
118  forAll(addr_, addrI)
119  {
120  const label cellI = addr_[addrI];
121  const vector& p = C[cellI];
122  vector n = p - (p & axialDirection_)*axialDirection_;
123  r[addrI] = mag(n)/rMax_;
124  }
125 
126  return tr;
127 }
