325  //- Lambda function to rotate a vector around an axis by a certain
326  // angle in radians, counter-clockwise
327  auto rotateCCWAroundAxisByAngle = []
328  (
329  vector& v,
330  const vector& axis,
331  scalar angle
332  )
333  {
334  scalar c(Foam::cos(angle));
335  scalar s(Foam::sin(angle));
336  scalar x(axis[0]);
337  scalar y(axis[1]);
338  scalar z(axis[2]);
339  tensor R
340  (
341  c+sqr(x)*(1-c), x*y*(1-c)-z*s, x*z*(1-c)+y*s,
342 
343  y*x*(1-c)+z*s, c+sqr(y)*(1-c), y*z*(1-c)-x*s,
344 
345  z*x*(1-c)-y*s, z*y*(1-c)+x*s, c+sqr(z)*(1-c)
346 
347  );
348  v = R & v;

401  //- The basis change matrix is the transformation to move from
402  // the local reference frame to the global one. It is constructed
403  // by simply arranging the local basis vectors (expressed in
404  // global reference frame coordinates) in columns. Since these are
405  // orthonormal, the matrix is orthonormal and its inverse is equal
