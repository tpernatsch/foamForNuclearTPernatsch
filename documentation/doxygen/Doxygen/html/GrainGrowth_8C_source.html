42  // from [1]
43  // The grain growth kinetic is described by Ainscough et al.,
44  // by means of the following equation:
45  // dD/dt = k * (1/D - f(BU)/Dm)
46  // where
47  // D = grain size (um)
48  // k, the rate constant, is
49  // k = 5.24e+7 * exp(-2.67e+5 / (R*T)) (um^2/h)
50  // R = 8.314 (J/K mol)
51  // T = temperature (K)
52  // Dm, the limiting grain size, is a function of temperature such that
53  // Dm = 2.23e+3 * exp(-7620 / T) (um)
54  // f(Bu) = 1 + 0.002e-03 * Burnup (GWd/teU)
55  // In detail, the presence of this last term in the rate equation
56  // allows to take into account the effect of irradiation on grain growth
57  // (consisting in the increasing retarding effect of fission products)
58  {
59  grain_boundary_mobility = 5.24e+07 * exp(-32114.5 / temperature) / (pow(um_m, 2) * s_h);
60  grain_diameter_limit = 2.23e+03 * exp(-7620.0 / temperature) / um_m;
61  burnup_factor = 1.0 + 0.002 * burnup / U_UO2 * 1.0e-03;
62  source_term = - (grain_boundary_mobility * burnup_factor) / grain_diameter_limit;
63  // std::cout << "grain_diameter " << grain_diameter << ", ";
64  // std::cout << "grain_boundary_mobility " << grain_boundary_mobility << ", ";
65  // std::cout << "grain_diameter_limit " << grain_diameter_limit << ", ";
