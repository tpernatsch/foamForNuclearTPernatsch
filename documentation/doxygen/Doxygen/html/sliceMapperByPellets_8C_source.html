56 void Foam::sliceMapperByPellets::calcAddressing() const
57 {
58  // Initialize number of slices
59  nSlices_ = 0;
60 
61  sliceAddrList_.clear();
62  const labelListList& matAddrList(mat_.matAddrList());
63 
64  // Reference to sliceField
65  scalarField& sliceFieldRef = sliceID_->ref();
66 
67  forAll(mat_.materialsList(), i)
68  {
69  const labelList& addrI(matAddrList[i]);
70 
71  dictionary materialDict(mat_.materialsList()[i].materialModelDict());
72  const word materialName(mat_.materialsList()[i].name());
73 
74  // Find min and max axial coordinate (along pinDirection_) in
75  // current material.
76  scalar hMin(GREAT);
77  scalar hMax(-GREAT);
78 
79  forAll(addrI, j)
80  {
81  const label cellI = addrI[j];
82 
83  // Reference to cell as list of points
84  const labelList& c = mesh_.cellPoints()[cellI];
85 
86  forAll(c, pointI)
87  {
88  // Reference to point
89  const vector& p = mesh_.points()[c[pointI]];
90 
91  scalar hPoint(p&pinDirection_);
92  hMin = min(hMin, hPoint);
93  hMax = max(hMax, hPoint);
94  }
95  }
96 
97  reduce(hMin, minOp<scalar>());
98  reduce(hMax, maxOp<scalar>());
99  scalar materialHeight = (hMax - hMin);
100 
101  // Read number of pellets in current material (1 for non-fuel materials)
102  scalar nPellets(isA<fuelMaterial>(mat_.materialsList()[i])?
103  readInt(materialDict.lookup("nPellets")) : 1);
104 
105  // Set height of each pellet in current material
106  scalarField sliceHeights(nPellets, materialHeight/nPellets);
107 
108  // Reference to cell centers
109  const volVectorField& C = mesh_.C();
110 
111  // Slice addressing for current material
112  List<labelList> sliceAddrListI(nPellets, List<label>(0, 0));
113 
114  // Assign correct slice ID to each cell and collect all cell
115  // addresses for each slice in cellIDsPerSlice_
116  forAll(addrI, j)
117  {
118  const label cellI = addrI[j];
119 
120  scalar hCell = C[cellI]&pinDirection_;
121 
122  // Cell distance from bottom of cellZone
123  scalar deltaH = hCell - hMin;
124 
125  // Prepare sliceID corresponding to current cell
126  scalar currentSliceID(0);
127  scalar totalHeight(0);
128 
129  // SliceN label for present cell
130  for(int m=0; m<nPellets; m++)
131  {
132  totalHeight += sliceHeights[m];
133 
134  if(totalHeight >= deltaH + 1e-6)
135  {
136  currentSliceID = m;
137  break;
138  }
139  }
140 
141  sliceAddrListI[currentSliceID].append(cellI);
142  }
143 
144  // If the cells are taller than the pellet size, some of the slices
145  // will be empty. The following lines pass only the non-empty slices
146  // to the global sliceAddrList_
147  label matSlices(0);
148  forAll(sliceAddrListI, matSliceID)
149  {
150  scalar sizeI(sliceAddrListI[matSliceID].size());
151  reduce(sizeI, sumOp<scalar>());
152 
153  // Create the slice only if the size is bigger than 0 in at least
154  // 1 processor (the local ID list might be empty)
155  if(sizeI>0)
156  {
157  sliceAddrList_.append(new labelList(sliceAddrListI[matSliceID]));
158  isFuel_.append(isA<fuelMaterial>(mat_.materialsList()[i]));
159  forAll(sliceAddrList_[nSlices_ + matSlices], i)
160  {
161  label cellI = sliceAddrList_[nSlices_ + matSlices][i];
162  sliceFieldRef[cellI] = nSlices_ + matSlices;
163  }
164  matSlices += 1;
165  }
166  }
167 
168  // Check for empty slices
169  forAll(sliceAddrList_, sliceI)
170  {
171  scalar sizeI(sliceAddrList_[sliceI].size());
172  reduce(sizeI, sumOp<scalar>());
173 
174  if(sizeI<=0)
175  {
176  FatalErrorIn("sliceMapperByPellets")
177  << "Found empty slice (i.e. a slice with no cells) when "
178  << "creating slices for material " << materialName << "." << nl
179  << "Check the keyword \"nSlices\" in the corresponding materials "
180  << "subdictionary within the solverDict." << nl
181  << "As a general rule, the number of slices should be lower than "
182  << "the number of axial divisions in the mesh."
183  << exit(FatalError);
184  }
185  }
186 
187  // Update total number of slice
188  nSlices_ += matSlices;
189  }
190 
191  sliceID_->correctBoundaryConditions();
192 }
