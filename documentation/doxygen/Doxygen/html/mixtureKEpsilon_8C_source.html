341 template<class BasicTurbulenceModel>
342 void mixtureKEpsilon<BasicTurbulenceModel>::setFluidNames() const
343 {
344  const fvMesh& mesh(this->mesh_);
345  HashTable<const fluid*> fluids(mesh.lookupClass<const fluid>());
346  const fluid& fluid1 = *(fluids[fluids.toc()[0]]);
347  const fluid& fluid2 = *(fluids[fluids.toc()[1]]);
348  if
349  (
350  !(fluid1.isLiquid() and fluid2.isGas()) and
351  !(fluid2.isLiquid() and fluid1.isGas())
352  )
353  {
354  FatalErrorInFunction
355  << "The mixtureKEpsilon turbulence model only works for "
356  << "liquid-gas systems. Set the the stateOfMatter entry in "
357  << "phaseProperties." << fluid1.name() << "Properties and/or "
358  << "phaseProperties." << fluid2.name() << "Properties) to "
359  << "distinguish between gas and liquid"
360  << exit(FatalError);
361  }
362  liquidName_ = (fluid1.isLiquid()) ? fluid1.name() : fluid2.name();
363  gasName_ = (fluid1.isGas()) ? fluid1.name() : fluid2.name();
364 }

411 template<class BasicTurbulenceModel>
412 const Foam::tmp<Foam::volScalarField> mixtureKEpsilon<BasicTurbulenceModel>::Cd()
413 const
414 {
415  tmp<volScalarField> tCd
416  (
417  new volScalarField
418  (
419  IOobject
420  (
421  "Cd",
422  this->mesh_.time().timeName(),
423  this->mesh_
424  ),
425  this->mesh_,
426  dimensionedScalar("", dimless, 1e-3),
427  zeroGradientFvPatchScalarField::typeName
428  )
429  );
430  volScalarField& Cd = tCd.ref();
431 
432  const FFPair& p(pair());
433  const volScalarField& l(liquid());
434  const volScalarField& g(gas());
435 
436  //- Compute Cd from Kd, cell-by-cell as it's faster (I don't really care
437  // about BCs)
438  forAll(Cd, i)
439  {
440  const scalar& li(l[i]);
441  const scalar& gi(g[i]);
442  Cd[i] =
443  (2.0)*p.Kd()[i]*p.DhDispersed()[i]/p.rhoContinuous()[i]/
444  max
445  (
446  (li*gi)/(li+gi)*p.magUr()[i], 1e-3
447  );
448  }
449  Cd.correctBoundaryConditions();
450 
451  return tCd;
452 }

486 template<class BasicTurbulenceModel>
487 tmp<volScalarField> mixtureKEpsilon<BasicTurbulenceModel>::Ct2() const
488 {
489  volScalarField beta
490  (
491  (6*this->Cmu_/(4*sqrt(3.0/2.0)))*
492  pair().Kd()/liquid().rho()*
493  (liquid().turbulence().k()/liquid().turbulence().epsilon())
494  );
495  volScalarField Ct0
496  (
497  (3 + beta)/(1 + beta + 2*gas().rho()/liquid().rho())
498  );
499  volScalarField fAlphad((180 + (-4.71e3 + 4.26e4*gas())*gas())*gas());
500 
501  return sqr(1 + (Ct0 - 1)*exp(-fAlphad));
502 }

573 {
574  tmp<volScalarField> bubbleG
575  (

610 template<class BasicTurbulenceModel>
611 void mixtureKEpsilon<BasicTurbulenceModel>::correct()
612 {
613  // Only solve the mixture turbulence for the gas-phase
614  if (!isGas_)
615  {
616  //- The consistency check is now done for both phases at every call
617  // of initMixtureFields, no need to do it here too
618  return;
619  }
620 
621  if (!this->turbulence_)
622  {
623  return;
624  }
625 
626  // Initialise the mixture fields if they have not yet been constructed
627  initMixtureFields();
628 
629  //- Note: I could modify all this to comply with the new structure
630  // (i.e. using gas() and liquid() for more symmetry and code clarity)
631  // but there is no practical need to and I'd only risk breaking things,
632  // so I won't
633 
634  // Local references to gas-phase properties
635  tmp<surfaceScalarField> phig = this->phi();
636  const volVectorField& Ug = this->U_;
637  const volScalarField& alphag = this->alpha_;
638  volScalarField& kg = this->k_;
639  volScalarField& epsilong = this->epsilon_;
640  volScalarField& nutg = this->nut_;
641 
642  // Local references to liquid-phase properties
643  //- This is a werid way of getting a ref to liquid().turbulence(). Why
644  // so? First, I need it non-const accessible, so I need const_case.
645  // Secondly, I cannot just cast liquid().turbulence() as the casting
646  // fails, so I look for the turbulence model in the mesh registry as
647  // well
648  const fvMesh& mesh(this->mesh_);
649  mixtureKEpsilon<BasicTurbulenceModel>& liquidTurbulence =
650  const_cast<mixtureKEpsilon<BasicTurbulenceModel>&>
651  (
652  mesh.lookupObject<mixtureKEpsilon<BasicTurbulenceModel>>
653  (
654  IOobject::groupName
655  (
656  turbulenceModel::propertiesName,
657  liquidName_
658  )
659  )
660  );
661  tmp<surfaceScalarField> phil = liquidTurbulence.phi();
662  const volVectorField& Ul = liquidTurbulence.U_;
663  const volScalarField& alphal = liquidTurbulence.alpha_;
664  volScalarField& kl = liquidTurbulence.k_;
665  volScalarField& epsilonl = liquidTurbulence.epsilon_;
666  volScalarField& nutl = liquidTurbulence.nut_;
667 
668  // Local references to mixture properties
669  volScalarField& rhom = rhom_();
670  volScalarField& km = km_();
671  volScalarField& epsilonm = epsilonm_();
672 
673  fv::options& fvOptions(fv::options::New(this->mesh_));
674 
675  eddyViscosity<RASModel<BasicTurbulenceModel>>::correct();
676 
677  // Update the effective mixture density
678  rhom = this->rhom();
679 
680  // Mixture flux
681  surfaceScalarField phim("phim", mixFlux(phil, phig));
682 
683  // Mixture velocity divergence
684  volScalarField divUm
685  (
686  mixU
687  (
688  fvc::div(fvc::absolute(phil, Ul)),
689  fvc::div(fvc::absolute(phig, Ug))
690  )
691  );
692 
693  tmp<volScalarField> Gc;
694  {
695  tmp<volTensorField> tgradUl = fvc::grad(Ul);
696  Gc = tmp<volScalarField>
697  (
698  new volScalarField
699  (
700  this->GName(),
701  nutl*(tgradUl() && dev(twoSymm(tgradUl())))
702  )
703  );
704  tgradUl.clear();
705 
706  // Update k, epsilon and G at the wall
707  kl.boundaryFieldRef().updateCoeffs();
708  epsilonl.boundaryFieldRef().updateCoeffs();
709 
710  Gc.ref().checkOut();
711  }
712 
713  tmp<volScalarField> Gd;
714  {
715  tmp<volTensorField> tgradUg = fvc::grad(Ug);
716  Gd = tmp<volScalarField>
717  (
718  new volScalarField
719  (
720  this->GName(),
721  nutg*(tgradUg() && dev(twoSymm(tgradUg())))
722  )
723  );
724  tgradUg.clear();
725 
726  // Update k, epsilon and G at the wall
727  kg.boundaryFieldRef().updateCoeffs();
728  epsilong.boundaryFieldRef().updateCoeffs();
729 
730  Gd.ref().checkOut();
731  }
732 
733  // Mixture turbulence generation
734  volScalarField Gm(mix(Gc, Gd));
735 
736  // Mixture turbulence viscosity
737  volScalarField nutm(mixU(nutl, nutg));
738 
739  // Update the mixture k and epsilon boundary conditions
740  km == mix(kl, kg);
741  bound(km, this->kMin_);
742  epsilonm == mix(epsilonl, epsilong);
743  bound(epsilonm, this->epsilonMin_);
744 
745  // Dissipation equation
746  tmp<fvScalarMatrix> epsEqn
747  (
748  fvm::ddt(epsilonm)
749  + fvm::div(phim, epsilonm)
750  - fvm::Sp(fvc::div(phim), epsilonm)
751  - fvm::laplacian(DepsilonEff(nutm), epsilonm)
752  ==
753  C1_*Gm*epsilonm/km
754  - fvm::SuSp(((2.0/3.0)*C1_)*divUm, epsilonm)
755  - fvm::Sp(C2_*epsilonm/km, epsilonm)
756  + epsilonSource()
757  + fvOptions(epsilonm)
758  );
759 
760  epsEqn.ref().relax();
761  fvOptions.constrain(epsEqn.ref());
762  epsEqn.ref().boundaryManipulate(epsilonm.boundaryFieldRef());
763  solve(epsEqn);
764  fvOptions.correct(epsilonm);
765  bound(epsilonm, this->epsilonMin_);
766 
767 
768  // Turbulent kinetic energy equation
