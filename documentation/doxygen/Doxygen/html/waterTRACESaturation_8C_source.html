92 Foam::scalar Foam::saturationModels::waterTRACE::valuePSat
93 (
94  const label& celli
95 ) const
96 {
97  const scalar& T(iT_[celli]);
98  /*
99  // ----- Stefan Radman Version - Interpolation NIST ----- //
100  return A_*pow(T-B_, C_);
101  */
102 
103  // ----- TRACE Version - different formulas according to temperature ----- //
104  if (T<370.4251)
105  {
106  scalar Ts = T;
107  if (T<=273.15)
108  {
109  Ts = 273.15;
110  }
111  const scalar AAP(24821.0);
112  const scalar BAP(338.0);
113  const scalar CAP(-5.3512);
114  const scalar DAP(20.387);
115  scalar ps(AAP*pow(Ts/BAP,CAP)*exp(DAP*(Ts-BAP)/Ts));
116  return ps;
117  }
118  else if (T<609.62462615967)
119  {
120  const scalar AB(117.8);
121  const scalar BB(1e-5);
122  const scalar CB(0.223);
123  const scalar DB(255.2);
124  scalar ps(BB*pow((T-DB)/AB,1.0/CB));
125  return ps;
126  }
127  else if (T<647.3)
128  {
129  const scalar AC(7.2166948490268e11);
130  const scalar BC(-8529.6481905883);
131  const scalar CC(1166669.3278328);
132  scalar ps(AC*exp((BC+CC/T)/T));
133  return ps;
134  }
135  else
136  {
137  const scalar AD(22.12e6);
138  const scalar BD(7.6084087);
139  const scalar CD(4924.9229);

151  const scalar& ps(p_[celli]);
152 
153  /*
154  // ----- Stefan Radman Version - Interpolation NIST ----- //
155  return C_*A_*pow(T-B_, C_-1.0);
156  */
157 
158  if (T<370.4251)
159  {
160  scalar Ts = T;

162  {

201 Foam::scalar Foam::saturationModels::waterTRACE::valueTSat
202 (
203  const label& celli
204 ) const
205 {
206  const scalar& pi(p_[celli]);
207  /*
208  // ----- Stefan Radman Version - Interpolation NIST ----- //
209  scalar TsRAD(pow(pi/A_, 1.0/C_) + B_ );
210  //Info << "Tsat Interpolation NIST --" << TsRAD << endl;
211  */
212 
213  //-------------------------------------------------------------//
214 
215  // ----- TRACE Version - different formulas according to pressure ----- //
216 
217  if (pi<90.56466*1000)
218  {
219  scalar ps = pi;
220  if (pi<610.8)
221  {
222  ps = 610.8;
223  }
224  const scalar AA(-2263.0);
225  const scalar BA(0.434);
226  const scalar CA(100000.0);
227  const scalar DA(6.064);
228  const scalar AAP(24821.0);
229  const scalar BAP(338.0);
230  const scalar CAP(-5.3512);
231  const scalar DAP(20.387);
232  const scalar AAH(3180619.59);
233  const scalar BAH(2470.2120);
234 
235  scalar Tsapprox(0);
236  scalar psapprox (0);
237  scalar hsapprox(0);
238  scalar Ts(AA/(BA*log(ps/CA)-DA));
239  for (int step=0; step<2; step++)
240  {
241  Tsapprox=Ts;
242  psapprox=AAP*pow(Tsapprox/BAP,CAP)*exp(DAP*(Tsapprox-BAP)/Tsapprox);
243  hsapprox=AAH - BAH*Tsapprox;
244  Ts=Tsapprox/(1-Rv_*Tsapprox/hsapprox*log(ps/psapprox));
245  }
246  return Ts;
247  }
248  else if (pi<13.969971285053*1e6)
249  {
250  const scalar AB(117.8);
251  const scalar BB(1.0*pow(10.0,-5));
252  const scalar CB(0.223);
253  const scalar DB(255.2);
254  scalar Ts(AB*pow(BB*pi,CB)+DB);
255  return Ts;
256  }
257  else if (pi<22.12*1e6)
258  {
259  const scalar AC(4264.8240952941);
260  const scalar BC(13666986.708428);
261  const scalar CC(1166669.3278328);
262  const scalar DC(27.304833093884);
263  scalar Ts((AC+sqrt(-BC+CC*log(pi)))/(DC-log(pi)));
264  return Ts;
265  }
266  else
267  {
268  const scalar AD(4924.9229);
269  const scalar BD(24.520401);
270  scalar Ts(AD/(BD-log(pi)));
271  return Ts;
272  }
273 }
