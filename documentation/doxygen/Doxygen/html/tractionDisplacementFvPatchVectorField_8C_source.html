372 void tractionDisplacementFvPatchVectorField::updateCoeffs()
373 {
374  if (updated())
375  {
376  return;
377  }
378 
379  pressureFromGF_.correctBoundaryConditions();
380 
381  // Update traction
382  updateTraction();
383 
384  // Patch normal
385  vectorField n(patch().nf());
386  vectorField nCurrent(n);
387 
388  if(patch().boundaryMesh().mesh().foundObject<volTensorField>("Finv"))
389  {
390  const tensorField& Finv =
391  patch().boundaryMesh().mesh().foundObject<fvMesh>("referenceMesh") ?
392  patch().lookupPatchField<volTensorField, tensor>("relFinv") :
393  patch().lookupPatchField<volTensorField, tensor>("Finv");
394 
395  // Patch unit normals (deformed configuration)
396  nCurrent = (Finv.T() & n);
397  nCurrent /= mag(nCurrent);
398  };
399 
400 
401  // Face area and total area
402  scalarField magSf(patch().magSf());
403  scalar totalArea(gSum(magSf));
404 
405  // Patch number of faces
406  scalar nFaces(patch().size());
407  scalar nFacesTotal(nFaces);
408  reduce(nFacesTotal, sumOp<scalar>());
409 
410  // Patch fields
411  const fvPatchField<scalar>& mu =
412  patch().lookupPatchField<volScalarField, scalar>("mu");
413 
414  const fvPatchField<scalar>& lambda =
415  patch().lookupPatchField<volScalarField, scalar>("lambda");
416 
417  scalarField twoMuLambda(2*mu + lambda);
418 
419  // Lookup the explicit component of the stress
420  const fvPatchField<tensor>& sigmaExp =
421  patch().lookupPatchField<volTensorField, tensor>("sigmaExp");
422 
423  const fvPatchField<scalar>& pressureFromGF =
424  patch().lookupPatchField<volScalarField, scalar>("pressureFromGF");
425 
426 
427  // Local resistance (defined also for non planestrain cases for consistency)
428  scalarField R_P = 1/patch().deltaCoeffs()/(max(2*mu + lambda, VSMALL));
429 
430  // Calculate the new gradient R_P*deltaCoeff is equivalent to 1/twoMuLambda
431  // for cases with no planeStrain or cases where RP is uniform
432  vectorField newGradient(R_P*patch().deltaCoeffs()*(
433  (traction_ - pressureFromGF*nCurrent) - (n & sigmaExp)));
434 
435  // Force the normal component of the normal gradient (and thus epsilonZZ)
436  // to be constant
437  if(planeStrain_)
438  {
439  // Copy only tangent compnent from standard gradient
440  newGradient = (newGradient & (I - sqr(n)));
441 
442  // Correct RP to consider plane strain approximation
443  R_P = scalarField(nFaces, totalArea/gSum(magSf/R_P));
444 
445  // Calculate average traction and sigmaExp
446  vector tractionEq(1/totalArea*gSum(
447  traction_* magSf - pressureFromGF*nCurrent* magSf));
448 
449  vector sigmaExpEq(1/totalArea*gSum(n & sigmaExp * magSf));
450 
451  newGradient += R_P*patch().deltaCoeffs()*(sqr(n) & (tractionEq - sigmaExpEq));
452  }
453 
454  // Resistance network for spring-dashpot system
455  if(fixedSpring_)
456  {
457  // Spring resistance
458  scalarField R_S = magSf/max(fixedSpringModulus_/nFacesTotal, SMALL);
459 
460  if(planeStrain_)
461  {
462  R_S = scalarField(nFaces, totalArea/gSum(magSf/R_S));
463  }
464 
465  // Dashpot resistance
466  scalarField R_D = magSf/max(dashpotModulus_/nFacesTotal, SMALL);
467 
468  // Spring-dashpot resistance
469  scalarField R_SD = R_S*R_D/(R_S + R_D);
470 
471  // Total resistance
472  scalarField Rtot = R_P + R_SD;
473 
474  // Add dashpot contribution to gradient
475  newGradient += R_P*patch().deltaCoeffs()*(*this/R_D);
476 
477  // Update weight factor
478  w_ = R_SD/Rtot;
479  }
480 
481  // Relax gradient
482  gradient() = (1-relax_)*gradient() + relax_*w_*newGradient;
483 
484  fixedGradientFvPatchVectorField::updateCoeffs();
485 }
