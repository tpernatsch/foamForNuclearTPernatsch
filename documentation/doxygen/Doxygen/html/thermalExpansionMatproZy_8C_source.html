96 void Foam::thermalExpansionMatproZy::correct
97 (
98  symmTensorField& sf,
99  const scalarField& T,
100  const labelList& addr
101 )
102 const
103 {
104  const scalar Tref = Tref_.value();
105  const symmTensor alphaTref = setAlphaT(Tref);
106 
107  forAll(addr, i)
108  {
109  const label cellI = addr[i];
110 
111  const scalar Ti = T[cellI];
112 
113  symmTensor nominalValue(setAlphaT(Ti) - alphaTref);
114 
115  // NOTE: avoids instabilities when trying to simulate a material at
116  // constant temperature equal to Tref
117  if
118  (
119  mag(nominalValue.xx()) < 1e-7 &&
120  mag(nominalValue.yy()) < 1e-7 &&
121  mag(nominalValue.zz()) < 1e-7
122  )
123  {
124  nominalValue *= 0;
125  }
126 
127  sf[cellI] = nominalValue*perturb;
128  }

132 const Foam::symmTensor Foam::thermalExpansionMatproZy::setAlphaT(const scalar& Ti) const
133 {
134  Foam::symmTensor alphaT = I;
135 
136  // The validity range is changed by one degree on each side to allow
137  // for rounding errors
138  // NOTE: lower bound from literature (290) extended to 273 for allowing
139  // thermal contraction below 290 K
140  if (Ti < 272.9 || Ti > 1801)
141  {
142  WarningInFunction
143  << "Supplied temperature, " << Ti << ", is out of range 273 < T < 1800 K";
144  }
145 
146  if (Ti < 1073) // Alpha phase
147  {
148  // // Axial expansion
149  // // NOTE we assume isotrpic behavior
150  // alphaT.zz() = par1*Ti - par2;
151  // Axial expansion
152  alphaT.zz() = par3*Ti - par4;
153  alphaT.xx() = par3*Ti - par4; // Radial expansion
154  alphaT.yy() = par3*Ti - par4; // Radial expansion
155  }
156  else if (Ti < 1273) // Interpolate between alpha and beta values
157  {
158  scalar TAlpha = 1073;
159  scalar TBeta = 1273;
160 
161 
162  // // NOTE we assume isotrpic behavior
163  // scalar axAlpha = par1*TAlpha - par2;
164  // scalar axBeta = par5*TBeta - par6;
165  scalar axAlpha = par3*TAlpha - par4;
166  scalar axBeta = par5*TBeta - par7;
167 
168  scalar radAlpha = par3*TAlpha - par4;
169  scalar radBeta = par5*TBeta - par7;
170 
171 
172  alphaT.zz() = axAlpha*(TBeta - Ti)/(TBeta - TAlpha) + axBeta*(Ti - TAlpha)/(TBeta - TAlpha);
173  alphaT.xx() = radAlpha*(TBeta - Ti)/(TBeta - TAlpha) + radBeta*(Ti - TAlpha)/(TBeta - TAlpha);
174  alphaT.yy() = radAlpha*(TBeta - Ti)/(TBeta - TAlpha) + radBeta*(Ti - TAlpha)/(TBeta - TAlpha);
175  }
176  else // Beta phase
177  {
178  // // Axial expansion
179  // // NOTE we assume isotrpic behavior
180  // alphaT.zz() = par5*Ti - par6;
181  alphaT.zz() = par5*Ti - par7;
182  alphaT.xx() = par5*Ti - par7; // Radial expansion
183  alphaT.yy() = par5*Ti - par7; // Radial expansion
184  }
185 
186  return alphaT;
187 }
