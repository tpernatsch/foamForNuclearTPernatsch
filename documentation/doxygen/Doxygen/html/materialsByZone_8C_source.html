50 void Foam::materialsByZone::checkZoneDefinition()
51 {
52  labelList zoneMap(mesh_.nCells(), -1);
53 
54  forAll(mesh_.cellZones(), zoneI)
55  {
56  const labelList& addr = mesh_.cellZones()[zoneI];
57 
58  forAll(addr, i)
59  {
60  const label cellI = addr[i];
61 
62  if (zoneMap[cellI] != -1)
63  {
64  FatalErrorIn("Foam::materialsByZone::checkZoneDefinition()")
65  << "Cell zones " << mesh_.cellZones()[zoneI].name()
66  << " and " << mesh_.cellZones()[zoneMap[cellI]].name()
67  << " overlap" << endl << Foam::abort(FatalError);
68  }
69  zoneMap[cellI] = zoneI;
70  }
71  }
72 
73  forAll(zoneMap, cellI)
74  {
75  if (zoneMap[cellI] == -1)
76  {
77  FatalErrorIn("Foam::materialsByZone::checkZoneDefinition()")
78  << "Cell zones do not include all mesh cells." << endl
79  << Foam::abort(FatalError);
80  }
81  }
82 }

143 void Foam::materialsByZone::correctThermoMechProperties()
144 {
145  if(T_ == nullptr)
146  {
147  T_ = &mesh_.lookupObject<volScalarField>("T");
148  }
149 
150  // Correct material properties
151  const scalarField& Ti = T_->internalField();
152  scalarField& rhoi = rho_.ref();
153  scalarField& Cpi = Cp_.ref();
154  scalarField& ki = k_.ref();
155  scalarField& epsi = emissivity_.ref();
156  scalarField& Ei = E_.ref();
157  scalarField& nui = nu_.ref();
158 
159  forAll(matList_, i)
160  {
161  materialModel& matI(matList_[i]);
162 
163  matI.rho(rhoi, Ti);
164  matI.Cp(Cpi, Ti);
165  matI.k(ki, Ti);
166  matI.emissivity(epsi, Ti);
167  matI.E(Ei, Ti);
168  matI.nu(nui, Ti);
169  }
170 
171  updateDensity();
172 
173  // Lame's parameters
174  lambda_ = nu_*E_/((1.0 + nu_)*(1.0 - 2.0*nu_));
175  mu_ = E_/(2.0*(1.0 + nu_));
176  threeK_ = E_/(1.0 - 2.0*nu_);
177 
178  rho_.correctBoundaryConditions();
179  Cp_.correctBoundaryConditions();
180  k_.correctBoundaryConditions();
181  emissivity_.correctBoundaryConditions();
182  E_.correctBoundaryConditions();
183  nu_.correctBoundaryConditions();
184  lambda_.correctBoundaryConditions();
185  mu_.correctBoundaryConditions();
186  threeK_.correctBoundaryConditions();
187 }

217 Foam::tmp<Foam::volSymmTensorField>
218 Foam::materialsByZone::thermalExpansion() const
219 {
220  tmp<volSymmTensorField> teps
221  (
222  new volSymmTensorField
223  (
224  IOobject
225  (
226  "threeK*alpha*T",
227  mesh_.time().timeName(),
228  mesh_
229  ),
230  mesh_,
231  dimensionedSymmTensor("threeK*alpha*T", dimless, symmTensor::zero)
232  )
233  );
234 
235  symmTensorField& epsi = teps->ref();
236 
237  const scalarField& Ti = T_->internalField();
238 
239  // We want to update the thermal expansion also on the patches (calculated
240  // BC instead of zeroGradient). Because different materials can share the
241  // same boundary (e.g. UO2 and UPuO2 in the same simulation sharing the
242  // same fuelOuter surface), the following loop is necessary.
243  // For each material, it finds the enclosing patches, and for each patch
244  // finds the appropriate list of faceIDS.
245  forAll(matList_, i)
246  {
247  const labelList& addr = mesh_.cellZones()[i];
248  labelList patchIDs_(0);
249  PtrList<List<label>> addrList(0);
250 
251  // Loop over all cells of current material
252  forAll(addr, addrI)
253  {
254  const label cellI = addr[addrI];
255 
256  // Reference to list of faces per cell
257  const cell& c = mesh_.cells()[cellI];
258 
259  // Loop over all faces of current cell
260  forAll(c, i)
261  {
262  const label patchID = mesh_.boundaryMesh().whichPatch(c[i]);
263  label indexInPatchID(0);
264 
265  if (patchID > -1)
266  {
267  bool found_ = false;
268 
269  // Check wheter patchID has been added to list of patchIDs
270  // for current material
271  forAll(patchIDs_, patchi)
272  {
273  indexInPatchID = patchi;
274  if (patchID == patchIDs_[patchi])
275  {
276  found_ = true;
277  break;
278  }
279  }
280 
281  // Add patch and face IDs only for non-empty and non-proc
282  // patches
283  if
284  (
285  !isType<processorFvPatchScalarField>
286  (
287  T_->boundaryField()[patchID]
288  )
289  and
290  !isType<emptyFvPatchScalarField>
291  (
292  T_->boundaryField()[patchID]
293  )
294  )
295  {
296 
297  // Add patchID to list for current material
298  if(!found_)
299  {
300  patchIDs_.append(patchID);
301  indexInPatchID = addrList.size();
302  addrList.append(new List<label>(0, 0));
303  }
304 
305  // Add faceID to the list for the current patch for the
306  // current material
307  label faceID =
308  mesh_.boundaryMesh()[patchID].whichFace(c[i]);
309  addrList[indexInPatchID].append(faceID);
310  }
311 
312  }
313  }
314  }
315 
316  // NOTE: necessary for implicitContact?
317  forAll(patchIDs_, patchI)
318  {
319  const label patchID = patchIDs_[patchI];
320  const scalarField& TP = T_->boundaryField()[patchID];
321 
322  symmTensorField& epsP = teps->boundaryFieldRef()[patchID];
323  matList_[i].alphaT(epsP, TP, addrList[patchI]);
324  }
325 
326  matList_[i].alphaT(epsi, Ti, addr);
327  }
328 
329  teps->correctBoundaryConditions();
330 
331  return teps;
332 }
