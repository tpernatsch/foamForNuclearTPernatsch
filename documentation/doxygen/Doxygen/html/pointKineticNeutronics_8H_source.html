44 Description
45  Run-time selectable point-kinetic based neutronics solver. It includes
46  options for a time dependent external reactivity, and an external
47  reactivity input coming from an FMU.
48 
49  The module solves the following point-kinetics equations:
50 
51  $$
52  \begin{align}
53  \frac{dP}{dt} & = \frac{\rho_{tot} - \zeta - \beta_{tot}}{\Lambda} P
54  + \sum_{i=1}^N \lambda_i C_i + \frac{S}{\Lambda} \\
55  \frac{dC_i}{dt} & = \frac{\beta_i}{\Lambda} P
56  - \lambda_i C_i \quad \forall i \in [1; N]
57  \end{align}
58  $$
59 
60  $\rho_{tot}$: the total reactivity that can be decomposed as:
61 
62  $$
63  \rho_{tot} =
64  \begin{cases}
65  \text{fast} & -\alpha_D \ln\left( \frac{T_{fuel}}{T_{fuel}^{ref}} \right) \\
66  \text{thermal} & \alpha_D \left( \sqrt{T_{fuel}} - \sqrt{T_{fuel}^{ref}} \right)
67  \end{cases}
68  + \alpha_{fuel} (T_{fuel} - T_{fuel}^{ref})
69  + \alpha_{clad} (T_{clad} - T_{clad}^{ref})
70  + \alpha_{cool}^{T} (T_{cool} - T_{cool}^{ref})
71  + \alpha_{cool}^{\rho} (\rho_{cool} - \rho_{cool}^{ref}) \\
72  + \alpha_{struct} (T_{struct} - T_{struct}^{ref})
73  + \alpha_{struct}^{mech} (T_{struct}^{mech} - T_{struct}^{mech,ref})
74  + \rho_{driveline}
75  + \rho_{GEM}
76  + \rho_{boron}
77  + \rho_{ext}
78  + \rho_{ext}^{FMU}
79  $$
80 
81 \vartable
82  P | Core power
83  C_i | Precursor power
84  S | Optional external source power
85  \beta_{tot} | Total delayed neutron fraction
86  \beta_i | Delayed neutron fraction for the ith delayed neutron precursor group
87  \lambda_i | Decay constant for the ith delayed neutron precursor group
88  \Lambda | Prompt generation lifetime
89  \zeta | Optional subcritical index
90  S | Optional external neutron source power
91  \alpha_D | Doppler feedback coefficient
92  \alpha_{fuel} | Fuel axial expansion feedback coefficient
93  \alpha_{clad} | Cladding temperature feedback coefficient
94  \alpha_{cool}^{T} | Coolant temperature feedback coefficient
95  \alpha_{cool}^{\rho} | Coolant density coefficient
96  \alpha_{struct} | Core radial expansion feedback coefficient
97  \alpha_{struct}^{mech} | Core expansion feedback coefficient from the mechanical mesh
98  \rho_{driveline} | Control rod driveline expansion reactivity contribution
99  \rho_{boron} | Boron concentration reactivity contribution
100  \rho_{ext} | External reactivity contribution
101  \rho_{ext}^{FMU} | External reactivity contribution from an FMU
102 \endvartable

105 Usage
106 
107 Example of time dependant source entries:
108 \verbatim
109  //- Optional time dependant external reactivity time profile
110  externalReactivityTimeProfile
111  {
112  type table;
113  startTime 100;
114  table table
115  (
116  ( 0 0)
117  ( 1 0.0005767131 ) //- i.e. 57.67131 pcm
118  );
119  }
120 
121  //- Optional time dependant boron reactivity time profile
122  boronReactivityTimeProfile
123  {
124  type table;
125  table table
126  (
127  ( 100 0 )
128  ( 101 1e-5 )
129  );
130  }
131 \endverbatim
132 
133 Optional external reactivity input from FMU in nuclearData file.
134 The default initial value is 0.
135 \verbatim
136  externalReactivityNameFromFMU gfExtReact;
137 \endverbatim
138 
139 Optional external neutron source file "externalSource"
140 \verbatim
141  //- If true, add the external source to the point-kinetics equations
142  isExternalSource true;
143 
144  //- External source mode
145  externalSourceMode transient; // powerMonitoring // transient
146 
147  //- Beam energy per source particle
148  beamEnergy 1.28160e-10; // [J/proton], 800 MeV/proton
149 
150  //- Neutron yield of the spallation of proton on heavy metal
151  nuSource 15.5; // [neutron/proton]
152 
153  //- Optional external source input from FMU in externalSource file.
154  //- The externalSourceMode must be set to transient. The default initial value
155  //-is 1.
156  externalSourceModulationNameFromFMU gfExternalSourceMod;
157 
158  //- External modulation time profile (time, modulation factor)
159  externalSourceModulationTimeProfile
160  {
161  type table;
162  startTime 150;
163  table table
164  (
165  (0 1)
166  (20 2)
167  );
168  }
169 \endverbatim

500  //- The deliminter is a char, so it must be provided in single-
501  // quotation marks. Type can be either word or string
502  template<class Type>
503  List<Type> split(const Type& compoundItem, char d)
504  {
505  List<Type> items(1, "");
506  int i = 0;
507  for (char const &c : compoundItem)
508  {
509  if (c == d)
510  {
511  i += 1;
512  items.append("");
513  }
514  else items[i] += c;
515  }
516  return items;
517  }
