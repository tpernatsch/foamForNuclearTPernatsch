247 void Foam::rheology::solvePressureEqn(volScalarField& sigmaHydExplicit)
248 {
249  const volScalarField& mu(mat_.mu());
250  const volScalarField& lambda(mat_.lambda());
251 
252  // Store previous iteration to allow relaxation
253  sigmaHyd_.storePrevIter();
254 
255  // Lookup the momentum equation inverse diagonal field
256  const volScalarField* ADPtr = NULL;
257 
258  if (mesh_.foundObject<volScalarField>("DEqnA"))
259  {
260  ADPtr = &mesh_.lookupObject<volScalarField>("DEqnA");
261  }
262  else if (mesh_.foundObject<volScalarField>("DDEqnA"))
263  {
264  ADPtr = &mesh_.lookupObject<volScalarField>("DDEqnA");
265  }
266  else
267  {
268  FatalErrorIn
269  (
270  "void " + type() + "updateSigmaHyd(...)\n"
271  ) << "Cannot find the DEqnA or DDEqnA field: this should be "
272  << "stored in the mechanicsSubSolver" << abort(FatalError);
273  }
274 
275  const volScalarField& AD = *ADPtr;
276 
277  // Pressure diffusivity field
278  const surfaceScalarField rDAf
279  (
280  "rDAf",
281  pressureSmoothingScaleFactor_*fvc::interpolate
282  (
283  (2*mu+lambda)/AD, "interpolate(rDAf)"
284  )
285  );
286  const dimensionedScalar one("one", dimless, 1.0);
287 
288  // Solve pressure laplacian
289  // Note: the fvm and fvc laplacian terms cancel at convergence and the
290  // laplacian - div(grad) term produce a smoothing/diffusion to quell
291  // oscillations
292  fvScalarMatrix sigmaHydEqn
293  (
294  fvm::Sp(one, sigmaHyd_)
295  - fvm::laplacian(rDAf, sigmaHyd_, "laplacian(rDA,sigmaHyd)")
296  ==
297  sigmaHydExplicit
298  - fvc::div(rDAf*fvc::interpolate(gradSigmaHyd_) & mesh_.Sf())
299  );
300 
301  //- Compute residual
302  residual_ = sigmaHydEqn.solve().max().initialResidual();
303 
304  // Relax the pressure field
305  sigmaHyd_.relax();
306 
307  gradSigmaHyd_ = fvc::grad(sigmaHyd_, "grad(sigmaHyd)");
308 }
