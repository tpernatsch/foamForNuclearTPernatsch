66  bool Foam::customPimpleControl::criteriaSatisfied()
67  {
68  // no checks on first iteration - nothing has been calculated yet
69  if
70  (
71  (corr_ == 1)
72  || residualControl_.empty()
73  || finalIter()
74  || corr_ < minNCorrPIMPLE_
75  )
76  {
77  return false;
78  }
79 
80  const bool storeIni = this->storeInitialResiduals();
81 
82  bool achieved = true;
83  bool checked = false; // safety that some checks were indeed performed
84 
85  const dictionary& solverDict = mesh_.data().solverPerformanceDict();
86  forAllConstIters(solverDict, iter)
87  {
88  const entry& solverPerfDictEntry = *iter;
89 
90  const word& fieldName = solverPerfDictEntry.keyword();
91  const label fieldi = applyToField(fieldName);
92 
93  if (fieldi != -1)
94  {
95  bool useFirstPISOInitialResidual
96  (
97  dict().subDict
98  (
99  "residualControl"
100  ).subDict
101  (
102  fieldName
103  ).lookupOrDefault<bool>("useFirstPISOInitialResidual", false)
104  );
105 
106  bool stopIfFirstResRelChangeAboveRelTol
107  (
108 
109  dict().subDict
110  (
111  "residualControl"
112  ).subDict
113  (
114  fieldName
115  ).lookupOrDefault<bool>
116  (
117  "stopIfFirstResRelChangeAboveRelTol", false
118  )
119  and useFirstPISOInitialResidual
120  );
121 
122  Pair<scalar> residuals = (useFirstPISOInitialResidual) ?
123  firstPISOPrevPIMPLEResidual(solverPerfDictEntry)
124  : maxResidual(solverPerfDictEntry);
125 
126  checked = true;
127 
128  scalar relative = 0.0;
129  bool relCheck = false;
130 
131  const bool absCheck =
132  (residuals.last() < residualControl_[fieldi].absTol);
133 
134  if (storeIni)
135  {
136  residualControl_[fieldi].initialResidual =
137  residuals.first();
138  }
139  else
140  {
141  const scalar iniRes =
142  residualControl_[fieldi].initialResidual + ROOTVSMALL;
143 
144  // Info <<"Residual first is " << residuals.first() << " and last is "<< residuals.last()<<endl;
145  relative =
146  (stopIfFirstResRelChangeAboveRelTol) ?
147  residuals.last() / residuals.first() :
148  residuals.last() / iniRes;
149 
150  relCheck =
151  (stopIfFirstResRelChangeAboveRelTol) ?
152  (relative >= residualControl_[fieldi].relTol) :
153  (relative < residualControl_[fieldi].relTol);
154 
155  if (stopIfFirstResRelChangeAboveRelTol and relCheck)
156  stopLoop_ = true;
157  }
158 
159  achieved = achieved && (absCheck || relCheck || stopLoop_);
160 
161 
162  Info<< algorithmName_ << " loop:" << endl;
163 
164  Info<< " " << fieldName
165  << " PIMPLE iter " << corr_
166  << ": ini res = "
167  << residualControl_[fieldi].initialResidual
168  << ", abs tol = " << residuals.last()
169  << " (" << residualControl_[fieldi].absTol << ")"
170  << ", rel tol = " << relative
171  << " (" << residualControl_[fieldi].relTol << ")"
172  << endl;
173 
174  }
175  }
176 
177  return checked && achieved;
178  }

269  //- This is identical to what is found in pimpleControl. However, by re-
270  // defining it here, I allow it to use the new criteriaSatisfied() functions
271  // defined by customPimpleControl
272  bool Foam::customPimpleControl::loop()
273  {
274  read();
275 
276  ++corr_;
277 
278 
279  Info<< algorithmName_ << " loop: corr = " << corr_ << endl;
280 
281  setFirstIterFlag();
282 
283 
284  if (corr_ == nCorrPIMPLE_ + 1)
285  {
286  if (!residualControl_.empty() && (nCorrPIMPLE_ != 1))
287  {
288  Info<< algorithmName_ << ": not converged within "
289  << nCorrPIMPLE_ << " iterations" << endl;
290  }
291 
292  corr_ = 0;
293  nCorrPISOInPrevPrevPIMPLE_ = 0;
294  stopLoop_ = false;
295  mesh_.data().setFinalIteration(false);
296  return false;
297  }
298 
299  bool completed = false;
300  // bool converged_=false;
301  if (converged_ || criteriaSatisfied())
302  {
303 
304  if (converged_)
305  {
306  if (!stopLoop_)
307  Info<< algorithmName_ << ": converged in " << corr_ - 1
308  << " iterations" << endl;
309  else
310  Info<< algorithmName_ << ": loop interrupted due to poor "
311  << "first PISO initialResidual convergence" << endl;
312 
313  mesh_.data().setFinalIteration(false);
314  corr_ = 0;
315  nCorrPISOInPrevPrevPIMPLE_ = 0;
316  stopLoop_ = false;
317  converged_ = false;
318 
319 
320  completed = true;
321  }
322  else
323  {
324  Info<< algorithmName_ << ": iteration " << corr_ << endl;
325  storePrevIterFields();
326 
327  mesh_.data().setFinalIteration(true);
328  converged_ = true;
329 
330  }
331  }
332  else
333  {
334 
335  if (finalIter())
336  {
337  mesh_.data().setFinalIteration(true);
338 
339  }
340 
341  if (corr_ <= nCorrPIMPLE_)
342  {
343  Info<< algorithmName_ << ": iteration " << corr_ << endl;
344  storePrevIterFields();
345  completed = false;
346 
347  }
348  }
349 
350  nCorrPISOInPrevPrevPIMPLE_ = nCorrPISOInPrevPIMPLE_;
351 
352  return !completed;
353  }
