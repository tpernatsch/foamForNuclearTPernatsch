72 void Foam::timeDependentAxialProfile::calcMinMaxZ
73 (
74  const dictionary& dict
75 )
76 {
77  // TODO - Since each point can be referenced by multiple cells, this algorithm
78  // will likely count each point multiple times. A more efficient approach
79  // is needed.
80  const fvMesh& referenceMesh =
81  (
82  mesh_.foundObject<fvMesh>("referenceMesh")
83  ) ?
84  mesh_.lookupObject<fvMesh>("referenceMesh")
85  :
86  mesh_;
87 
88  const vectorField& points = referenceMesh.points();
89  const labelListList& cellPoints = referenceMesh.cellPoints();
90  scalarField z = (points & pinDirection_);
91 
92  zMax_ = -GREAT;
93  zMin_ = GREAT;
94 
95  forAll(addr_, addrI)
96  {
97  const label cellI = addr_[addrI];
98  const labelList& ptIds = cellPoints[cellI];
99 
100  forAll(ptIds, i)
101  {
102  const label ptI = ptIds[i];
103  zMin_ = min(zMin_, z[ptI]);
104  zMax_ = max(zMax_, z[ptI]);
105  }
106  }
107  reduce(zMin_, minOp<scalar>());
108  reduce(zMax_, maxOp<scalar>());
109 
110  if(debug)
111  {
112  Info<< "Geometry calculated by timeDependentAxialProfile::calcMinMaxZ" << nl
113  << tab << "Pin direction: " << pinDirection_ << nl
114  << tab << "Z bounds: " << zMin_ << "-" << zMax_ << endl;
115  }

171 void Foam::timeDependentAxialProfile::correct()
172 {
173  if(mesh_.topoChanging())
174  {
175  calcAddressing(fastFluxOptDict_);
176  calcMinMaxZ(fastFluxOptDict_);
177  }
178 
179  if(!mesh_.foundObject<volScalarField>("fastFluxPrevIter"))
180  {
181  flux_.storePrevIter();
182  }
183 
184  flux_.prevIter().storePrevIter();
185  flux_.storePrevIter();
186 
187  const scalarField& fastFluxPrev(flux_.prevIter());
188  const scalarField& fastFluxPrevPrev(flux_.prevIter().prevIter());
189 
190  // Correct only once per time step or until flux changes
191  if
192  (
193  gMax(mag(fastFluxPrevPrev - fastFluxPrev)) > VSMALL
194  or
195  mesh_.time().value() > currentTime_
196  )
197  {
198  currentTime_ = mesh_.time().value();
199  // scalar Dt(mesh_.time().deltaT().value());
200 
201  // Interpolate fast flux
202  scalar fastFluxAv = fluxTable_
203  (mesh_.time().timeToUserTime(currentTime_));
204 
205  // Correct the axial profiles
206  axialModel_->correct();
207 
208  // Build the resulting fast flux and fluence
209  scalarField& fastFluxi = flux_.ref();
210  const scalarField& f_z = axialModel_->profile();
211 
212  forAll(addr_, addrI)
213  {
214  const label cellI = addr_[addrI];
215 
216  fastFluxi[cellI] =
217  fastFluxAv*f_z[addrI];
218  }
219 
220  flux_.correctBoundaryConditions();
221  advanceFluence();
222  }
223  else
224  {
225  currentTime_ = mesh_.time().value();
226  }
227 
228 }
