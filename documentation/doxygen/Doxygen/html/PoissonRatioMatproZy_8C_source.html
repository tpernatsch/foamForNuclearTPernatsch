109 void Foam::PoissonRatioMatproZy::correct
110 (
111  scalarField& sf,
112  const scalarField& T,
113  const labelList& addr
114 )
115 {
116  if(phi_ == nullptr or E_ == nullptr)
117  {
118  phi_ = &mesh_.lookupObject<volScalarField>(fastFluenceName_);
119  E_ = &mesh_.lookupObject<volScalarField>(YoungModulusName_);
120  }
121 
122  forAll(addr, i)
123  {
124  const label cellI = addr[i];
125 
126  const scalar Ti = T[cellI];
127  const scalar phi = phi_->internalField()[cellI];
128  const scalar E = E_->internalField()[cellI];
129 
130  //- Shear modulus
131  scalar G(0.0);
132 
133  //- Set the shear modulus, MATPRO model
134  //- The validity range is changed by one degree on each side to allow
135  //- for rounding errors
136  if (Ti < 290 || Ti > 1801)
137  {
138  WarningInFunction
139  << "Supplied temperature, " << Ti << ", is out of range 290 < T < 1800 K";
140  }
141 
142  if (Ti < 1073) //- Alpha phase
143  {
144  // Oxygen concentration effect
145  scalar K1 = (par1 - par2*Ti)*oxC_;
146 
147  // Cold work effect
148  scalar K2 = -par3*cW_;
149 
150  // Fast neutron fluence effect
151  scalar K3 = par4 + par5*exp(-phi/par6);
152 
153  G = (par7 - par8*Ti + K1 + K2) / K3;
154  }
155  else if (Ti < 1273) //- Interpolate between alpha and beta values
156  {
157  scalar TAlpha = 1073;
158  scalar TBeta = 1273;
159 
160  // Oxygen concentration effect
161  scalar K1 = (par1 - par2*TAlpha)*oxC_;
162 
163  // Cold work effect
164  scalar K2 = -par3*cW_;
165 
166  // Fast neutron fluence effect
167  scalar K3 = par4 + par5*exp(-phi/par6);
168 
169  scalar GAlpha = (par7 - par8*TAlpha + K1 + K2) / K3;
170 
171  scalar GBeta = par9 - par10*TBeta;
172 
173  G = GAlpha*(TBeta - Ti)/(TBeta - TAlpha) + GBeta*(Ti - TAlpha)/(TBeta - TAlpha);
174  }
175  else //- Beta phase
176  {
177  G = (par9 - par10*Ti);
178  }
179 
180  const scalar nominalValue = (E/(2*G) - 1);
181 
182  sf[cellI] = nominalValue*perturb;
183  }
184 }
