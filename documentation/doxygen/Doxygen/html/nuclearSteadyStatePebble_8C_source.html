567  scalar ktriso = (1/rfuel-1/rPyCout)/((1/kbuffer)*(1/rfuel-1/rbuffer)+(1/kPyC)*(1/rbuffer-1/rPyCin)+(1/kSiC)*(1/rPyCin-1/rSiC)+1/kPyC*(1/rSiC-1/rPyCout));
568  scalar PF = (4.0/3.0*pi_*pow(rPyCout,3)) * static_cast<double >(Ntriso) / (4.0/3.0*pi_*(pow(rmatrix,3)-pow(rcore,3)));

570  scalar beta_coef = (kappa_coef-1)/(kappa_coef+2);
571  scalar keffmatrix = kgraphite*((1+2*beta_coef*PF)/(1-beta_coef*PF));

576  Tmav_[celli] = Tmout_[celli] + (q_matrix/(6*keffmatrix))*(pow(rmatrix,2)-3*(pow(rmatrix,5)-pow(rcore,5))/(5*(pow(rmatrix,3)-pow(rcore,3))))+(pow(rcore,3)*q_matrix)/(3*keffmatrix)*(1/rmatrix-3*(pow(rmatrix,2)-pow(rcore,2))/(2*(pow(rmatrix,3)-pow(rcore,3))));
577  TfS_[celli] = Tmav_[celli] + Q_triso/(4*pi_) * (1/kbuffer*(1/rfuel-1/rbuffer) + 1/kPyC*(1/rbuffer-1/rPyCin) + 1/kSiC*(1/rPyCin-1/rSiC) + 1/kPyC*(1/rSiC-1/rPyCout));
578  Tfav_[celli] = TfS_[celli] + q_fuel*pow(rfuel,2) / (15*kfuel);
579  Tfmax_[celli] = TfS_[celli] + q_fuel*pow(rfuel,2) / (6*kfuel);
580  keffMatrix_[celli] = keffmatrix;

583  rhobuffer_[celli] = rhobufferCoeffs_[regioni].value((TfS_[celli]+Tmav_[celli])/2);
584  rhoSiC_[celli] = rhoSiCCoeffs_[regioni].value((TfS_[celli]+Tmav_[celli])/2);
585  rhoPyC_[celli] = rhoPyCCoeffs_[regioni].value((TfS_[celli]+Tmav_[celli])/2);
586  rhographite_[celli] = rhographiteCoeffs_[regioni].value(Tmav_[celli]);
587  rhocoolant_[celli] = rhocoolantCoeffs_[regioni].value(HTSum[celli] / HSum[celli]);
