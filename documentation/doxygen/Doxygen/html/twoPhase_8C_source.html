667 void Foam::solvers::twoPhase::correctContErrs()
668 {
669  volScalarField& cE1(fluid1_.contErr());
670  volScalarField& cE2(fluid2_.contErr());
671  volScalarField& rho1(fluid1_.rho());
672  volScalarField& rho2(fluid2_.rho());
673 
674  cE1 =
675  (
676  fvc::ddt(fluid1_, rho1)
677  + fvc::div(fluid1_.alphaRhoPhi())
678  - (fvOptions_(fluid1_, rho1) & rho1)
679  );
680  cE2 =
681  (
682  fvc::ddt(fluid2_, rho2)
683  + fvc::div(fluid2_.alphaRhoPhi())
684  - (fvOptions_(fluid2_, rho2) & rho2)
685  );
686  if (FFPair_.phaseChange())
687  {
688  cE1 += FFPair_.dmdt();
689  cE2 -= FFPair_.dmdt();
690  }
691 
692  cE1.correctBoundaryConditions();
693  cE2.correctBoundaryConditions();
694 }

709 void Foam::solvers::twoPhase::calcCumulContErrs()
710 {
711  if (pimple_.finalIter())
712  {
713  scalar& cumulContErr1(fluid1_.cumulContErr());
714  scalar& cumulContErr2(fluid2_.cumulContErr());
715 
716  const volScalarField& cE1(fluid1_.contErr());
717  const volScalarField& cE2(fluid2_.contErr());
718 
719  const scalarField& V(mesh_.V());
720  const scalar& dt(mesh_.time().deltaTValue());
721 
722  scalar totV(0);
723  scalar deltaCumulContErr1(0);
724  scalar deltaCumulContErr2(0);
725 
726  forAll(V, i)
727  {
728  const scalar& Vi(V[i]);
729  deltaCumulContErr1 += cE1[i]*Vi*dt;
730  deltaCumulContErr2 += cE2[i]*Vi*dt;
731  totV += Vi;
732  }
733  reduce(deltaCumulContErr1, sumOp<scalar>());
734  reduce(deltaCumulContErr2, sumOp<scalar>());
735  reduce(totV, sumOp<scalar>());
736 
737  cumulContErr1 += deltaCumulContErr1;
738  cumulContErr2 += deltaCumulContErr2;
739 
740  Info<< "Cumulative continuity errors ("
741  << fluid1_.name() << " " << fluid2_.name() << ") = "
742  << (cumulContErr1/totV) << " " << (cumulContErr2/totV)
743  << " kg/m3" << endl;
744  }
745 }
