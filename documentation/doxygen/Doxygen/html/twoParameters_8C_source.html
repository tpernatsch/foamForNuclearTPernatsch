309 void Foam::regimeMapModels::twoParameters::correct()
310 {
311  //- Lookup parameter field ptrs if not set
312  if (parameter1Ptr_ == nullptr)
313  {
314  parameter1Ptr_ =
315  &mesh_.lookupObjectRef<volScalarField>(parameter1Name_).field();
316  }
317  if (parameter2Ptr_ == nullptr)
318  {
319  parameter2Ptr_ =
320  &mesh_.lookupObjectRef<volScalarField>(parameter2Name_).field();
321  }
322  const scalarField& x(*parameter1Ptr_);
323  const scalarField& y(*parameter2Ptr_);
324 
325  int N(mesh_.cells().size());
326 
327  //- Loop over all mesh cell labels
328  for (int i=0; i < N; i++)
329  {
330  //- Reset regimeLabelCoeffs for the i-th mesh cell
331  DynamicList<Tuple2<label,scalar>>& regimeLabelCoeffsi
332  (
333  regimeLabelCoeffs_[i]
334  );
335  regimeLabelCoeffsi = DynamicList<Tuple2<label,scalar>>(0);
336 
337  //- Value of the tuple parameter1-parameter2 which is used to determine
338  // the existing regime in the i-th mesh cell (normalized so distances
339  // make sense for the purposes of interpolation)
340  Vector2D<scalar> p(x[i]/DX_, y[i]/DY_);
341 
342  //- Find the regime the point belongs to
343  const regimeDomain2D* regimePtr = nullptr;
344  forAllConstIter
345  (
346  regimeDomainTable,
347  regimeDomains_,
348  iter
349  )
350  {
351  //- I recall that regimeLabelCoeffs_ is a list of tuples of each
352  // mesh cell, wherein for each cell, the tuple consists of the ID
353  // of the regime that exists in said cell (first tuple element)
354  // and the coefficient weighting the contribution of said regime
355  // (which is 1.0 if only one regime exists). To check how this
356  // is factually used to interpolate the value of models of
357  // interest, go see the interpolateValue, interpolateValueCmpt
358  // template functions in regimeMapModelTemplates.C
359  if (iter().containsPoint(p))
360  {
361  regimePtr = &(iter());
362  regimeLabelCoeffsi.append
363  (
364  Tuple2<label,scalar>(regimePtr->id(), 1.0)
365  );
366  break;
367  }
368  }
369  if (regimePtr != nullptr) //- Set regimeLabelCoeffs
370  {
371  if (interpolate_)
372  {
373  const regimeDomain2D& regime(*regimePtr);
374  const List<const regimeBoundary2D*>& internalBoundaryPtrs
375  (
376  regime.internalBoundaryPtrs()
377  );
378  if (internalBoundaryPtrs.size() != 0)
379  {
380  //- In theory, a starting max distance of sqrt(2) should
381  // be sufficient as these distances are in the normalized
382  // domain, whose range for both parameters is [0, 1] (so
383  // the max possible distance is between (0,0) and (1,1)).
384  // Nonetheless, I want to play it safe so 2 it is
385  scalar distance(2);
386  scalar nbrId(-1);
387  forAll(internalBoundaryPtrs, j)
388  {
389  const regimeBoundary2D& boundaryj
390  (
391  *internalBoundaryPtrs[j]
392  );
393  scalar distancej(boundaryj.distanceTo(p));
394  if (distancej < distance)
395  {
396  distance = distancej;
397  nbrId = boundaryj.nbrId();
398  }
399  }
400 
401  //- The coefficient is calculated so that, at the boundary,
402  // distance = 0 and c = 0.5, regardless of the
403  // interpolation type
404  scalar d(distance/interpolationWidth_);
405  if (d < 1.0)
406  {
407  scalar c(0.0);
408  switch (interpolationMode_)
409  {
410  case interpolationMode::linear :
411  {
412  c = 0.5*(d + 1.0);
413  break;
414  }
415  case interpolationMode::quadratic :
416  {
417  c = 1.0-0.5*sqr(d-1.0);
418  break;
419  }
420  }
421  regimeLabelCoeffsi[0].second() = c;
422  regimeLabelCoeffsi.append
423  (
424  Tuple2<label,scalar>(nbrId, 1.0-c)
425  );
426  }
427  }
428  }
429  //- Debug & development infos
430 
431  //Info<< i << ", (" << x[i] << " " << y[i] << ") => " << p << ", "
432  // << regimePtr->id() << ", " << regimeLabelCoeffsi << endl;
433  }
434  else //- Point is outside regime map bounds, throw an error
435  {
436  FatalErrorInFunction
437  << "Point (" << x[i] << ", " << y[i] << ") in parameter space "
438  << "(" << parameter1Name_ << ", " << parameter2Name_ << ") "
439  << "is outside the bounds of " << this->name() << "!"
440  << exit(FatalError);
441  }
442  }
443 }
