100 Foam::scalar Foam::interfacialAreaModels::annular::value
101 (
102  const label& celli
103 ) const
104 {
105  scalar aSum(dispersed_[celli]+continuous_[celli]);
106  scalar a(dispersed_[celli]/aSum);
107  return
108  aSum*
109  (
110  (a < cutoffAlpha_) ?
111  4.0*a/DhDispersed_[celli] :
112  4.0*a/DhDispersed_[celli]*(1.0-sqrt(a))/(1.0-cutoffAlpha_)
113  );
114 
115 }
