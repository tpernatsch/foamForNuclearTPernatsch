72 void Foam::constantLhgr::calcReferenceDimensions
73 (
74  const dictionary& dict
75 )
76 {
77  // TODO - Since each point can be referenced by multiple cells, this algorithm
78  // will likely count each point multiple times. A more efficient approach
79  // is needed.
80  const fvMesh& referenceMesh =
81  (
82  mesh_.foundObject<fvMesh>("referenceMesh")
83  ) ?
84  mesh_.lookupObject<fvMesh>("referenceMesh")
85  :
86  mesh_;
87 
88  const vectorField& points = referenceMesh.points();
89  const labelListList& cellPoints = referenceMesh.cellPoints();
90  scalarField z = (points & pinDirection_);
91  scalarField r = mag(points - (z*pinDirection_));
92 
93  zMax_ = -GREAT;
94  zMin_ = GREAT;
95  rMax_ = 0;
96 
97  forAll(addr_, addrI)
98  {
99  const label cellI = addr_[addrI];
100  const labelList& ptIds = cellPoints[cellI];
101 
102  forAll(ptIds, i)
103  {
104  const label ptI = ptIds[i];
105  zMin_ = min(zMin_, z[ptI]);
106  zMax_ = max(zMax_, z[ptI]);
107  rMax_ = max(rMax_, r[ptI]);
108  }
109  }
110  reduce(zMin_, minOp<scalar>());
111  reduce(zMax_, maxOp<scalar>());
112  reduce(rMax_, maxOp<scalar>());
113 
114  if(debug)
115  {
116  Info<< "Geometry calculated by timeDependentLhgr::calcReferenceDimensions" << nl
117  << tab << "Pin direction: " << pinDirection_ << nl
118  << tab << "Z bounds: " << zMin_ << "-" << zMax_ << nl
119  << tab << "Radius: " << rMax_ << nl << endl;
120  }
121 }

208 void Foam::constantLhgr::correct()
209 {
210  const globalOptions& globalOpt
211  (mesh_.lookupObject<globalOptions>("globalOptions"));
212 
213  scalar angularFraction = globalOpt.angularFraction();
214 
215  if(mesh_.topoChanging())
216  {
217  calcAddressing(heatSourceOptDict_);
218  calcReferenceDimensions(heatSourceOptDict_);
219  }
220 
221  if(!mesh_.foundObject<volScalarField>("QPrevIter"))
222  {
223  Q_.storePrevIter();
224  }
225 
226  Q_.prevIter().storePrevIter();
227  Q_.storePrevIter();
228 
229  // Correct heat source only once per time step or until power changes
230  if
231  (
232  mesh_.time().value() > currentTime_
233  or
234  mesh_.foundObject<volScalarField>("porosity")
235  )
236  {
237  currentTime_ = mesh_.time().value();
238 
239  scalarField Vi(mesh_.V(), addr_);
240  scalar totalVol(gSum(Vi)/angularFraction);
241 
242  // Interpolate LHGR and calculate average power density Qavg
243  scalar Qavg = constLhgr_*(zMax_ - zMin_)/totalVol;
244 
245  // Correct the radial and axial profiles
246  radialModel_->correct();
247  azimuthalModel_->correct();
248  axialModel_->correct();
249 
250  // Build the resulting heat source
251  scalarField& Qi = Q_.ref();
252  scalarField& lhgri = lhgr_.ref();
253  const scalarField& f_r = radialModel_->profile();
254  const scalarField& h_theta = azimuthalModel_->profile();
255  const scalarField& g_z = axialModel_->profile();
256 
257  forAll(addr_, addrI)
258  {
259  const label cellI = addr_[addrI];
260 
261  Qi[cellI] = Qavg*f_r[addrI]*g_z[addrI]*h_theta[addrI];
262  lhgri[cellI] = constLhgr_*g_z[addrI];
263  }
264 
265  // Reshape power profile due to porosity redistribution
266  reshapePowerPorosity();
267 
268  scalarField Qaddr(Q_.ref(), addr_);
269  scalar powerTot = constLhgr_*(zMax_ - zMin_)*angularFraction;
270  scalar powerTotOF = gSum( Vi*Qaddr );
271  scalar error =
272  100*
273  (
274  mag(powerTotOF - powerTot)/max(powerTot, VSMALL)
275  );
276 
277  if(error > 0.01)
278  {
279  WarningIn("Foam::timeDependentLhgr::correct()") << nl
280  << " Total power does not correspond to LHGR*height, "<< nl
281  << " with an error of " << error << "%" << nl
282  << " Radial or axial profiles might be not normalized." << nl
283  << " The field Q will be scaled to preserve total power."
284  << nl << endl;
285 
286  radialModel_ -> checkNormalization();
287  azimuthalModel_ -> checkNormalization();
288  axialModel_ -> checkNormalization();
289 
290  scalar scalingFactor(powerTot/powerTotOF);
291 
292  forAll(addr_, addrI)
293  {
294  const label cellI = addr_[addrI];
295 
296  Qi[cellI] *= scalingFactor;
297  }
298  }
299  }
300  else
301  {
302  currentTime_ = mesh_.time().value();
303  }
304 
305  Q_.correctBoundaryConditions();
306  lhgr_.correctBoundaryConditions();
307 }

324 void Foam::constantLhgr::reshapePowerPorosity()
325 {
326  // Scale volumetric power according to porosity profile
327  if
328  (
329  mesh_.foundObject<volScalarField>("porosity")
330  )
331  {
332  // Ref to Q field
333  scalarField& Qi = Q_.ref();
334 
335  // Initialize avg porosity in the domain
336  scalar avgPorosity(0);
337 
338  // Local porosity field
339  scalarField porosity(
340  mesh_.lookupObject<volScalarField>("porosity").internalField(), addr_);
341 
342  // Local volume
343  scalarField volume(mesh_.V().field(), addr_);
344 
345  // Compute average porosity
346  avgPorosity = gSum(porosity*volume)/gSum(volume);
347 
348  forAll(addr_, addrI)
349  {
350  const label cellI = addr_[addrI];
351 
352  // Get local value of porosity (bounded in [0,1])
353  const scalar pI = max(min(porosity[addrI], 1.0), 0.0);
354 
355  // Scale factor for volumetric power
356  Qi[cellI] *= (1-pI)/(1-avgPorosity);
357 
358  }
359  }
360 }
