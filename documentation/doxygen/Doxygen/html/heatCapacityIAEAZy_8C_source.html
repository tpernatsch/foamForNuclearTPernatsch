100 void Foam::heatCapacityIAEAZy::correct
101 (
102  scalarField& sf,
103  const scalarField& T,
104  const labelList& addr
105 )
106 const
107 {
108  forAll(addr, i)
109  {
110  const label cellI = addr[i];
111 
112  const scalar Ti = T[cellI];
113 
114  if (Ti < 273 || Ti > 2000)
115  {
116  WarningInFunction
117  << "Supplied temperature, " << Ti << ", is outside of range 273 < T < 2000 K";
118  }
119 
120  scalar Cp1 = par1 + par2*Ti;
121  scalar Cp2 = par3 - par4*Ti + par5*Ti*Ti;
122  scalar f = par6*exp(-pow(Ti - par7,par8)/par9);
123 
124  scalar nominalValue(0.0);
125 
126  if (Ti < 1100)
127  {
128  nominalValue = Cp1;
129  }
130  else if (Ti < 1213.8)
131  {
132  nominalValue = Cp1 + f;
133  }
134  else if (Ti < 1320)
135  {
136  nominalValue = Cp2 + f;
137  }
138  else
139  {
140  nominalValue = Cp2;
141  }
142 
143  sf[cellI] = nominalValue*perturb;
144 
145  }
146 }
