82 void Foam::PARFUMEPyCCreepModel::correctCreep
83 (
84  volSymmTensorField& epsilonEl,
85  const labelList& addr
86 )
87 {
88  const volScalarField& mu(mesh_.lookupObject<volScalarField>("mu"));
89  const volScalarField& lambda(mesh_.lookupObject<volScalarField>("lambda"));
90  const volScalarField& Temp(mesh_.lookupObject<volScalarField>("T"));
91 
92  scalar tol(1.0);
93  //- Irradiation time step [s]
94  scalar deltaT(0);
95 
96  if(mesh_.time().timeIndex() > 0)
97  {
98  const volScalarField& rho = mesh_.lookupObject<volScalarField>(densityName_);
99  const volScalarField& phi = mesh_.lookupObject<volScalarField>(fastFluxName_);
100 
101  deltaT = mesh_.time().deltaTValue();
102 
103  const scalarField& epsilonCreepEqOld_ = epsilonCreepEq_.oldTime().internalField();
104  const symmTensorField& epsilonCreepOldI = epsilonCreep_.oldTime().internalField();
105 
106  //- Adjust elastic strain tensor by subtracting old creep values
107  forAll(addr, addrI)
108  {
109  const label cellI = addr[addrI];
110 
111  epsilonEl[cellI] -= epsilonCreepOldI[cellI];
112 
113  const cell& c = mesh_.cells()[cellI];
114 
115  forAll(c, faceI)
116  {
117  const label patchID = mesh_.boundaryMesh().whichPatch(c[faceI]);
118 
119  {
120  if (patchID > -1 and epsilonEl.boundaryField()[patchID].size())
121  {
122  const label faceID =
123  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
124 
125  symmTensorField& epsilonElP =
126  epsilonEl.boundaryFieldRef()[patchID];
127 
128  const symmTensorField& epsilonCreepOldP =
129  epsilonCreep_.oldTime().boundaryField()[patchID];
130 
131  epsilonElP[faceID] -= epsilonCreepOldP[faceID];
132  }
133  }
134  }
135  }
136 
137  if(mesh_.time().timeIndex() > timeIndex_)
138  {
139  forAll(addr, addrI)
140  {
141  const label cellI = addr[addrI];
142  DepsilonCreepEq_.ref()[cellI] *= 0;
143 
144  const cell& c = mesh_.cells()[cellI];
145 
146  forAll(c, faceI)
147  {
148  const label patchID = mesh_.boundaryMesh().whichPatch(c[faceI]);
149 
150  {
151  if (patchID > -1 and epsilonEl.boundaryField()[patchID].size())
152  {
153  const label faceID =
154  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
155 
156  scalarField& DepsilonCreepEqP = DepsilonCreepEq_.boundaryFieldRef()[patchID];
157 
158  DepsilonCreepEqP[faceID] *= 0;
159  }
160  }
161  }
162  }
163 
164  timeIndex_ = mesh_.time().timeIndex();
165  }
166 
167  // Do-while loop to correct creep strain
168  do
169  {
170  scalar maxTemp(0);
171  scalar maxStress(0);
172  //scalar volTot(0);
173 
174  forAll(addr, addrI)
175  {
176  const label cellI = addr[addrI];
177 
178  const symmTensor sigmaI = 2*mu[cellI]*epsilonEl[cellI] + lambda[cellI]*tr(epsilonEl[cellI])*I;
179 
180  //- Calculate the principal stress [Pa]
181  scalar p1, p2, p3;
182  bool s;
183  computePrincipalStress(sigmaI, p1, p2, p3, s);
184  //- Calculate the effective stress;
185  //- Formula taken from "D. Petti, P. Martin, M. Phelip, and R. Ballinger.
186  //- Development of improved models and designs for coated-particle gas
187  //- reactor fuels. Technical Report INL/EXT-05-02615, Idaho National
188  //- Laboratory, December 2004."
189  const scalar sigmaEff = min(p1-vc_*(p2+p3)-3*mu[cellI]*DepsilonCreepEq_.ref()[cellI], 3e10);
190  maxStress = max(sigmaEff, maxStress);
191 
192  const scalar DepsilonCreepEqPrev = DepsilonCreepEq_.ref()[cellI];
193 
194  //- Reference to temperature field [Celsius]
195  const scalar T = Temp.internalField()[cellI]-273.15;
196 
197  maxTemp = max(T, maxTemp);
198 
199  // * # * # * # * # * # * # * # * # * # * # * # * # * # * # * # * #
200  //- Irradiation-induced creep coefficient
201 
202  //- The calculation of Ks is tricky. There are two formulas generally:
203  //- Ks1 = 2.193e-29 - 4.85e-32T + 4.0147e-35T²
204  //- Ks2 = 1.996e-29 - 4.415e-32T + 3.6544e-35T²
205  //- BISON manual and PARFUME use Ks1 for both PyC and Buffer, while
206  //- according to the BISON website, Ks2 is used for PyC and Ks1 for Buffer;
207  //- It seems that Ks1 is 1.099 times of Ks2 and Ks1 is for fast neutron of E > 0.18MeV
208  //- while Ks2 is for E > 0.1MeV (common situation). So if Ks2 is used, then
209  //- we donnot need to convert the fast neutron flux/fluence of 0.1MeV.
210  //- However, because most of the correlations for Triso use fast neutron of 0.18 MeV,
211  //- Ks1 is used here with the fluxConversionFactor (fcf_).
212  const scalar Ks = C_amp*(1+par1*(par2-rho[cellI]/1e3))*(par3-par4*T+par5*T*T);
213  //- Calculate creep strain rate
214  const scalar creepRate = Ks*sigmaEff*1e-6*phi[cellI]*1e4*fcf_;
215 
216  //- Calculate creep strain
217  DepsilonCreepEq_.ref()[cellI] =
218  (1-relax_)*DepsilonCreepEq_.ref()[cellI] + relax_*creepRate*deltaT;
219 
220  // * # * # * # * # * # * # * # * # * # * # * # * # * # * # * # * #
221 
222  epsilonCreepEq_.ref()[cellI] =
223  epsilonCreepEqOld_[cellI] + DepsilonCreepEq_.ref()[cellI];
224 
225  DepsilonCreep_.ref()[cellI] =
226  (1 - relax_)*DepsilonCreep_.ref()[cellI] + relax_*1.5*DepsilonCreepEq_.ref()[cellI]*dev(sigmaI
227  - 2*mu[cellI]*DepsilonCreep_.ref()[cellI])/max(sigmaEff, SMALL);
228 
229  epsilonCreep_.ref()[cellI] =
230  epsilonCreepOldI[cellI] + DepsilonCreep_.ref()[cellI];
231 
232 
233  const scalar tolTr = min(1, mag( (DepsilonCreepEq_.ref()[cellI] - DepsilonCreepEqPrev)
234  / (max(DepsilonCreepEqPrev , SMALL))));
235 
236  tol = min(tol, tolTr );
237 
238  const cell& c = mesh_.cells()[cellI];
239 
240  forAll(c, faceI)
241  {
242  const label patchID = mesh_.boundaryMesh().whichPatch(c[faceI]);
243 
244  {
245  if (patchID > -1 and epsilonEl.boundaryField()[patchID].size())
246  {
247  const label faceID =
248  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
249 
250  const scalarField& Tp = Temp.boundaryField()[patchID];
251  const scalarField& muP = mu.boundaryField()[patchID];
252  const scalarField& lambdaP = lambda.boundaryField()[patchID];
253  const scalarField& rhoP = rho.boundaryField()[patchID];
254  const scalarField& phiP = phi.boundaryField()[patchID];
255  const symmTensorField& epsilonElP = epsilonEl.boundaryField()[patchID];
256 
257  symmTensorField& epsilonCreepP = epsilonCreep_.boundaryFieldRef()[patchID];
258  symmTensorField& DepsilonCreepP = DepsilonCreep_.boundaryFieldRef()[patchID];
259 
260  scalarField& epsilonCreepEqP = epsilonCreepEq_.boundaryFieldRef()[patchID];
261 
262  const scalarField& epsilonCreepEqOldP = epsilonCreepEq_.oldTime().boundaryField()[patchID];
263 
264  const symmTensorField& epsilonCreepOldP = epsilonCreep_.oldTime().boundaryField()[patchID];
265 
266  scalarField& DepsilonCreepEqP = DepsilonCreepEq_.boundaryFieldRef()[patchID];
267 
268  const symmTensor sigmaPI =
269  2*muP[faceID]*epsilonElP[faceID] + lambdaP[faceID]*tr(epsilonElP[faceID])*I;
270 
271  //- Calculate the principal stress [Pa]
272  scalar p1p, p2p, p3p;
273  bool sp;
274  computePrincipalStress(sigmaPI, p1p, p2p, p3p, sp);
275 
276  const scalar sigmaEff = min(p1p-vc_*(p2p+p3p)-3*muP[faceID]*DepsilonCreepEqP[faceID], 3e10);
277 
278  const scalar DepsilonCreepEqPPrev = DepsilonCreepEqP[faceID];
279  // * # * # * # * # * # * # * # * # * # * # * # * # * # * # * # * #
280  //- Reference to temperature field [Celsius]
281  const scalar T = Tp[faceID] - 273.15;
282  //- Irradiation-induced creep coefficient
283  const scalar Ks = C_amp*(1+par1*(par2-rhoP[faceID]/1e3))*(par3-par4*T+par5*T*T);
284 
285  //- Calculate creep strain rate with power law
286  const scalar creepRate = Ks*sigmaEff*1e-6*phiP[faceID]*1e4*fcf_;
287 
288  //- Calculate creep strain
289  DepsilonCreepEqP[faceID] =
290  (1-relax_)*DepsilonCreepEqP[faceID] + relax_*creepRate*deltaT;
291 
292  // * # * # * # * # * # * # * # * # * # * # * # * # * # * # * # * #
293 
294  epsilonCreepEqP[faceID] =
295  DepsilonCreepEqP[faceID] + epsilonCreepEqOldP[faceID] ;
296 
297  DepsilonCreepP[faceID] =
298  (1 - relax_)*DepsilonCreepP[faceID] + relax_*1.5*DepsilonCreepEqP[faceID]*dev(sigmaPI - 2*muP[faceID]*DepsilonCreepP[faceID])/max(sigmaEff, SMALL) ;
299 
300  epsilonCreepP[faceID] =
301  epsilonCreepOldP[faceID] + DepsilonCreepP[faceID];
302 
303  const scalar tolTr = min(1, mag( (DepsilonCreepEqP[faceID] - DepsilonCreepEqPPrev)/ (max(DepsilonCreepEqPPrev , SMALL))));
304 
305  tol = min(tol, tolTr );
306  }
307  }
308  }
309  }
310  }while(tol > 1);
311  }
312 
313  forAll(addr, addrI)
314  {
315  const label cellI = addr[addrI];
316 
317  epsilonEl[cellI] -= DepsilonCreep_[cellI];
318 
319  const cell& c = mesh_.cells()[cellI];
320 
321  forAll(c, faceI)
322  {
323  const label patchID = mesh_.boundaryMesh().whichPatch(c[faceI]);
324 
325  {
326  if (patchID > -1 and epsilonEl.boundaryField()[patchID].size())
327  {
328  const label faceID =
329  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
330 
331  symmTensorField& epsilonElP =
332  epsilonEl.boundaryFieldRef()[patchID];
333  const symmTensorField& DepsilonCreepP =
334  DepsilonCreep_.boundaryField()[patchID];
335 
336  epsilonElP[faceID] -= DepsilonCreepP[faceID];
337  }
338  }
339  }
340  }
341  timeIndex_ = mesh_.time().timeIndex();
342 }

344 //TO DO: The computation of principal stress should be coded elsewhere so that
345 //it does not need to be written again and again in different source files.
346 void Foam::PARFUMEPyCCreepModel::computePrincipalStress
347 (
348  symmTensor s, //stress tensor
349  scalar& stressPrinciple1,
350  scalar& stressPrinciple2,
351  scalar& stressPrinciple3,
352  bool& solved //whether the principle can be solved, for now it is not be used.
353 )
354 {
355  //- Parameters for the cubic equation SP³-I1*SP²+I2*SP-I3=0 (SP: Stress Principal)
356  scalar I1 = s.xx()+s.yy()+s.zz();
357  scalar I2(s.xx()*s.yy()+s.xx()*s.zz()+s.yy()*s.zz()
358  -s.xy()*s.xy()-s.yz()*s.yz()-s.xz()*s.xz());
359  scalar I3(det(s));
360 
361 
362  //--------------------------Solve the cubic equation--------------------------
363  //- Convert the cubic equation into the form y³+py+q=0, where y = SP-I1/3;
364  scalar p((3.0*I2-I1*I1)/3.0);
365  scalar q((9.0*I1*I2-27.0*I3-2.0*pow(I1, 3))/27.0);
366 
367  scalar a = pow(q/2.0, 2);
368  scalar b = pow(p/3.0, 3);
369  scalar delta = a + b;
370  //static int i = 0;
371 
372  bool noNeedtoSolve = false;
373  scalar min_ab = max(min(mag(a), mag(b)), SMALL);
374 
375  const scalar minNstress = min(min(mag(s.xx()), mag(s.yy())), mag(s.zz()));
376  const scalar maxSstress = max(max(mag(s.xy()), mag(s.xz())), mag(s.yz()));
377 
378  //- For the case when the stress tensor is already in the form of principal stress.
379  //- Then there's no need to do the calculation, if not, there will cause errors
380  //- because of the calculation error
381  if (mag(delta/min_ab) < 1e-5 || maxSstress/minNstress < 1e-3)
382  {
383  noNeedtoSolve = true;
384  delta = -1; //Just need to give a negative value
385  }
386 
387 
388  scalar SP1, SP2, SP3;
389 
390 
391  //- delta must < 0 to have 3 different real roots; However, at the beginning of
392  //- the iteration, the stress tensor might not be realistic (true?) so that the principal
393  //- could not be able to be solved. In this case, we do not calculate the creep(stress = 0.0).
394  //- Or, maybe we can use Von Mises stress instead before the principal stress
395  //- can be solved? (TO DO)
396  if (delta > 0.0)
397  {
398  stressPrinciple1 = 0.0;
399  stressPrinciple2 = 0.0;
400  stressPrinciple3 = 0.0;
401  solved = false;
402  return;
403  }
404 
405  if (noNeedtoSolve == true)
406  {
407  SP1 = s.xx();
408  SP2 = s.yy();
409  SP3 = s.zz();
410  }
411  else
412  {
413  scalar r = max(sqrt(-pow(p/3.0,3)), SMALL);
414 
415  scalar theta = 1.0/3.0*acos(-q/2.0/r);
416  //- Three real roots
417  scalar y1(2*pow(r,1.0/3.0)*cos(theta));
418  scalar y2(2*pow(r,1.0/3.0)*cos(theta+2.0/3.0*M_PI));
419  scalar y3(2*pow(r,1.0/3.0)*cos(theta+4.0/3.0*M_PI));
420  //- Three principle stress
421  SP1 = y1+I1/3.0;
422  SP2 = y2+I1/3.0;
423  SP3 = y3+I1/3.0;
424  }
425 
426  //------------ Find the max, mid, min stress; SP1 > SP2 > SP3----------------
427  scalar t;
428  if (SP3 > SP2)
429  {
430  t = SP3;
431  SP3 = SP2;
432  SP2 = t;
433  }
434  if (SP3 > SP1)
435  {
436  t = SP3;
437  SP3 = SP1;
438  SP1 = t;
439  }
440  if (SP2 > SP1)
441  {
442  t = SP2;
443  SP2 = SP1;
444  SP1 = t;
445  }
446  //- The first(max) principle stress
447  stressPrinciple1 = SP1;
448  //- The second(mid) principle stress
449  stressPrinciple2 = SP2;
450  //- The third(min) principle stress
451  stressPrinciple3 = SP3;
452  solved = true;
453 }
