56  // SpectralDiffusion
57  // Solver for the spatially averaged solution of the PDE [dy/dt = D div grad y + S]
58  // We apply a spectral approach in space, projecting the equation on the eigenfunctions of the laplacian operator.
59  // We use the first order backward Euler solver in time.
60  // The number of terms in the expansion, N, is fixed a priori.
61  double SpectralDiffusion(std::vector<double>& initial_condition, int N, double diffusion_coefficient, double domain_radius, double source_term, double time_step)
62  {
63  unsigned short int n(0);
64  unsigned short int np1(1);
65 
66  double diffusion_rate_coeff(0.0);
67  double diffusion_rate(0.0);
68  double source_rate_coeff(0.0);
69  double source_rate(0.0);
70  std::vector<double> time_coefficient(N, 0.0);
71  double projection_coeff(0.0);
72  double solution(0.0);
73 
74  diffusion_rate_coeff = pow(Pi, 2) * diffusion_coefficient / pow(domain_radius, 2);
75  projection_coeff = - 2.0 * sqrt(2.0 / Pi);
76  source_rate_coeff = projection_coeff * source_term;
77 
78  for (n = 0; n < N; n++)
79  {
80  np1 = n + 1;
81  const double n_coeff = pow(- 1.0, np1) / np1;
82 
83  diffusion_rate = diffusion_rate_coeff * pow(np1, 2);
84  source_rate = source_rate_coeff * n_coeff;
85 
86  time_coefficient[n] = Solver::Decay(initial_condition[n], diffusion_rate, source_rate, time_step);
87  initial_condition[n] = time_coefficient[n];
88 
89  solution += projection_coeff * n_coeff * time_coefficient[n] / ((4./3.) * Pi);
90  }
91 
92  return solution;
93  }

96  // Solver for the spatially average solution of the PDE [dy/dt = D div grad y + S]
97  // designed for time-varying conditions [1,2]
98  // [1] Forsberg and Massih, Journal of Nuclear Materials, 127 (1985) 141-145
99  // [2] Lassmann and Benk, Journal of Nuclear Materials, 280 (2000) 127-135
100  double FORMAS(double initial_condition[], double diffusion_coefficient, double domain_radius, double source_term, double time_step)
101  {
102  // Constants derived from a priori linear approximation of the diffusion kernel
103  // Some are not needed and reported for the sake of completeness
104  //const double a_coeff[4];
105  //a_coeff[0] = 0.2315450;
106  //a_coeff[1] = 0.0948453;
107  //a_coeff[2] = 0.0282898;
108  //a_coeff[3] = 0.6453200;
109 
110  double b_coeff[4];
111  b_coeff[0] = 72.5968;
112  b_coeff[1] = 796.773;
113  b_coeff[2] = 29083.0;
114  b_coeff[3] = 10.2469;
115 
116  double a_over_b[4];
117  a_over_b[0] = 3.18947e-03;
118  a_over_b[1] = 1.19037e-04;
119  a_over_b[2] = 9.72726e-07;
120  a_over_b[3] = 6.29771e-02;
121 
122  // Non-dimensional (diffusion) time
123  const double dtau = time_step * diffusion_coefficient / pow(domain_radius, 2);
124 
125  double term[4];
126  double t1 = 0.0;
127  for (int n = 0; n < 4; n++)
128  {
129  double arg = b_coeff[n] * dtau;
130  if (arg < 1.0e-04)
131  term[n] = - arg;
132  else if (arg > 70.0)
133  term[n] = - 1.0;
134  else
135  term[n] = exp(- arg) - 1.0;
136 
137  t1 += term[n] * a_over_b[n];
138  }
139 
140  double t2 = 0.0;
141  if (dtau < 0.1)
142  t2 = 2.25675833 * pow(dtau, 1.5) - 1.5 * pow(dtau, 2);
143  else
144  t2 = dtau - 0.06159589 * (1.0 - exp( -9.86960440 * dtau));
145 
146  // Non-dimensional source
147  if (abs(t2-t1) > 1.0e-15)
148  {
149  const double source = source_term * time_step / (t2 - t1);
150 
151  for (int n = 0; n < 4; n++)
152  {
153  initial_condition[n] += term[n] * (initial_condition[n] - source * a_over_b[n]);
154  }
155  }
156 
157  return (initial_condition[0] + initial_condition[1] + initial_condition[2] + initial_condition[3]);
158  }

168  {
169  double err(0.0);
170  double tol(1.0e-06);
171  int k,n,m;
172  const int K(50);
173  double x[N]; // solution
174  double r[N]; // residual
175  double p[N]; // direction
176  double norm(0.0); // norm of p in A-space
177  double alfa(0.0); // updater of solution
178  double beta(0.0); // updater of direction
179 
180  /*NB: discarding A[N][N] for A[N*N] means traducing every double loop on the matrix like:
181  for(n=0;n<N;n++){
182  r[n] = 0.0;
183  for(m=0;m<N;m++) r[n] += AA[n][m] * x[m];
184  }
185  into:
186  for(n=0;n<N;n++){
187  r[n] = 0.0;
188  for(m=0;m<N;m++) r[n] += AA[m+n*N] * x[m];
189  }*/

191  // Cast transposed matrix
192  double A_T[N*N];
193  for (n=0; n<N;++n) for (m=0; m<N; ++m) A_T[m+n*N]=A[n+N*m];
194 
195  // Cast new matrix AA=A^T*A and new solution vector bb=A^T*b
196  double AA[N*N];
197  double bb[N];
198 
199  for (n=0; n<N;++n){
200  for (m=0; m<N; ++m){
201  AA[m+n*N] = 0.;
202  for (unsigned short int k=0; k<N; ++k) AA[m+n*N] += A_T[k+n*N]*A[m+k*N];
203  }
204  }

206  bb[n] = 0.;
207  for (unsigned short int k=0; k<N; ++k) bb[n] += A_T[k+n*N]*b[k];
208  }
209

211  for(n=0;n<N;n++){
212  r[n] = 0.0;
213  for(m=0;m<N;m++) r[n] += AA[m+n*N] * x[m];
214  r[n] = bb[n] - r[n];
215  }
216  for(n=0;n<N;n++) p[n] = r[n];
217  for(k=0;k<K;k++){
218  norm = 0.0; alfa = 0.0; beta = 0.0; err = 0.0;
219  for(n=0;n<N;n++) for(m=0;m<N;m++) norm += p[n] * AA[m+n*N] * p[m];
220  for(n=0;n<N;n++) alfa += p[n] * r[n]; alfa /= norm;
221  for(n=0;n<N;n++) x[n] = x[n] + alfa * p[n];
222  for(n=0;n<N;n++){
223  r[n] = 0.0;
224  for(m=0;m<N;m++) r[n] += AA[m+n*N] * x[m];
225  r[n] = bb[n] - r[n];
226  }

228  for(n=0;n<N;n++) p[n] = r[n] - beta * p[n];
229  for(n=0;n<N;n++) err += r[n] * r[n]; err = sqrt(err); if(err < tol) break;
230  }
231  //cast solution
232  for(n=0;n<N;n++) b[n] = x[n];
233  }

244  void PBiCG(int N, double A[], double b[])
245  {
246  double x[N]; //solution
247  int n,m,k; //support index
248  const int K(2*N); // maximum number of iterations
249  double r[N], r_star[N]; //residuals
250  double alfa(0.), beta(0.); //projectors
251  double beta_factors[2]={0.,0.}; // support bins to store quantities useful for beta and alfa calculation
252  double p[N], p_star[N], z[N], z_star[N], q[N], q_star[N]; // directions
253  double M[N*N]; //preconditioner
254  double A_T[N*N]; //input matrix transposed
255  double err(0.), tol(1.e-20);
256 
257  //cast preconditioner (here, trivial diagonal of the input matrix)
258  for(n=0;n<N*N; ++n) M[n] = 0.; for(n=0;n<N; ++n) M[n+N*n] = A[n+N*n];
259  //cast inverse preconditioner (need to check if some diagonal entries are zero!)
260  for(n=0;n<N; ++n) M[n+N*n]=1.0/M[n+N*n];
261  //cast transposed matrix
262  for(n=0;n<N; ++n) for(m=0;m<N; ++m) A_T[N*n+m] = A[N*m+n];
263  //guess initial solution
264  for(n=0;n<N; ++n) x[n] = 1.0;//b[n];
265  //evaluate initial residuals and cnj residuals
266  for(n=0;n<N; ++n){
267  r[n] = 0.;
268  for(m=0;m<N; ++m) r[n] += A[m+N*n]*x[m];
269  r[n] = b[n] - r[n];
270  r_star[n] = r[n];
271  }
272  //main loop beginning---
273  for(k=0;k<K; ++k){
274  err = 0.;
275  beta_factors[0] = beta_factors[1]; //store previous iteration values
276  beta_factors[1] = 0.;
277  //solve systems for (preconditioned) biconjugate directions updating
278  for(n=0;n<N; ++n){
279  z[n] = M[n+N*n] * r[n];
280  z_star[n] = M[n+N*n] * r_star[n]; //with diagonal precondioner, (M and M^T) ^-1 are the same
281  }
282  // update orthogonal projector beta
283  for(n=0;n<N; ++n) beta_factors[1] += z[n]*r_star[n]; //if (std::abs(beta_factors[1]) <= 1.e-35) {std::cout<< "BREAKDOWN!"; return;}
284  //update directions
285  if (k != 0){
286  beta = beta_factors[1]/beta_factors[0];
287  for(n=0;n<N; ++n){
288  p[n] = z[n] + beta * p[n];
289  p_star[n] = z_star[n] + beta * p_star[n];
290  }
291  } else{
292  for(n=0;n<N; ++n){
293  p[n] = z[n];
294  p_star[n] = z_star[n];
295  }
296  }
297  for(n=0;n<N; ++n){
298  q[n] = 0.;
299  q_star[n] = 0.;
300  for(m=0;m<N; ++m){
301  q[n] += A[m+N*n]*p[m];
302  q_star[n] += A_T[m+N*n]*p_star[m];
303  }
304  }
305  alfa = 0.;
306  //update orthogonal projector alfa
307  for(n=0;n<N; ++n) alfa += p_star[n]*q[n];
308  alfa = beta_factors[1] / alfa;
309  //update solution and residuals
310  for(n=0;n<N; ++n){
311  x[n] += alfa * p[n];
312  r[n] -= alfa * q[n];
313  r_star[n] -= alfa * q_star[n];
314  }
315  //check for convergence
316  for(n=0;n<N; ++n) err += r[n]*r[n]; err=std::sqrt(err); if (err<tol) break;
317  } //main loop end---
318  //cast solution
319  for(n=0;n<N; ++n) b[n] = x[n];
320  }
