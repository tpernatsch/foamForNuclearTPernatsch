96 void Foam::relocationFRAPCON::correct
97 (
98  const scalarField& T,
99  const labelList& addr
100 )
101 {
102  epsilonRelocation_.storeOldTimes();
103  epsilonRecoveredRelocation_.storeOldTimes();
104 
105  // Correct only when Q, Bu and gWidth are properly initialized
106  if
107  (
108  mesh_.foundObject<volScalarField>(heatSourceName_) and
109  mesh_.foundObject<volScalarField>(burnupName_) and
110  mesh_.foundObject<volScalarField>(gapWidthName_)and
111  mesh_.foundObject<sliceMapper>("sliceMapper")
112  )
113  {
114  if( gapWidth_ == nullptr )
115  {
116  gapWidth_ = &mesh_.lookupObject<volScalarField>(gapWidthName_);
117  }
118 
119  if(mapper_ == nullptr)
120  {
121  mapper_ = &mesh_.lookupObject<sliceMapper>("sliceMapper");
122  }
123 
124  scalar nSlices(mapper_->nSlices());
125  if(nSlices<=0)
126  {
127  FatalErrorInFunction()
128  << "Number of slices cannot be less than 1." << nl
129  << "Check that a sliceMapper different than \"none\" is used "
130  << "or check the slice definition."
131  << abort(FatalError);
132  }
133 
134  // Take const references to heat source and burnup fields
135  const volScalarField& Q = mesh_.lookupObject<volScalarField>(heatSourceName_);
136  const volScalarField& Bu = mesh_.lookupObject<volScalarField>(burnupName_);
137 
138  // Calculate the slice-average values of Bu and Q
139  const scalarField& Q_1D(mapper_->sliceAverage(Q));
140  const scalarField& Bu_1D(mapper_->sliceAverage(Bu));
141 
142  scalarField gWidth_1D(nSlices, 0.0);
143 
144  const scalarField& Vi = mesh_.V();
145 
146  const fvPatchScalarField& gapWidthP =
147  gapWidth_->boundaryField()[outerPatchID_];
148 
149  // NOTE: the following loop is not optimal as each material model
150  // will calculate averages for every slice in the fuel.
151  // (only a problem for simulation with multiple fuel materials)
152  const PtrList<labelList>& sliceAddrList(mapper_->sliceAddrList());
153  const List<bool>& isFuelList(mapper_->isFuel());
154 
155  forAll(sliceAddrList, i)
156  {
157  if(isFuelList[i])
158  {
159  const labelList& sliceAddr = sliceAddrList[i];
160 
161  scalarField sliceV(Vi, sliceAddr);
162 
163  List<cell> sliceCells(mesh_.cells(), sliceAddr);
164  scalarField sliceGapWidth(sliceAddr.size(), 0.0);
165  scalarField sliceMagSf(sliceAddr.size(), 0.0);
166 
167  forAll(sliceCells, j)
168  {
169  const cell c(sliceCells[j]);
170 
171  forAll(c, faceI)
172  {
173  const label patchID =
174  mesh_.boundaryMesh().whichPatch(c[faceI]);
175 
176  if (patchID == outerPatchID_)
177  {
178  const label faceID =
179  mesh_.boundaryMesh()[outerPatchID_].whichFace(c[faceI]);
180 
181  const scalar faceSf =
182  gapWidthP.patch().magSf()[faceID];
183 
184  sliceGapWidth[j] += gapWidthP[faceID]*faceSf;
185  sliceMagSf[j] += faceSf;
186  }
187  }
188  }
189 
190  gWidth_1D[i] =
191  gSum(sliceGapWidth*sliceMagSf)/gSum(sliceMagSf);
192  }
193  }
194 
195  const scalarField& sliceIDs(mapper_->sliceID().internalField());
196 
197  // Reference to relocation, old time relocation and recovered relocation
198  scalarField& rel = epsilonRelocation_.ref();
199  const scalarField& relOld = epsilonRelocation_.oldTime().internalField();
200  scalarField& relRecover = epsilonRecoveredRelocation_.ref();
201 
202  forAll(addr, addrI)
203  {
204  const label cellI = addr[addrI];
205  const label sliceID = sliceIDs[cellI];
206 
207  // Calculate relocation. Relocation cannot decrease!
208  scalar newRelocation = setRelocation
209  (
210  Q_1D[sliceID],
211  GapCold_,
212  DiamCold_,
213  Bu_1D[sliceID]
214  );
215 
216  scalar relocation = max(relOld[cellI], newRelocation);
217 
218  // if(rel[cellI]==relOld[cellI])
219  // {
220  // relRecover[cellI] = 0;
221  // }
222 
223  // Surface penetration
224  const scalar penetration = max(-gWidth_1D[sliceID], 0.0);
225 
226  // // Relocation cannot cause stress!
227  // if(penetration >= relocation*DiamCold_/2 )
228  // {
229  // _______________________________ _ _ _ _ _ _ _ _
230  // | /|\ /|\ | /|\
231  // | |penetration |R_fuel | |relocation*R_fuel
232  // | | | _ _|_ _\|/_ _ _ _ _
233  // ________|_____|_____________|___________|_______________
234  // /|\ | | |
235  // | | | |
236  // |Ri_clad| | |
237  // | | | |
238 
239 
240  // // Relocation is recovered as much as possible to prevent penetration
241  // rel[cellI] = relocation*(1 - recFrac_);
242  // relRecover[cellI] = relocation*(recFrac_);
243  // }
244  // else
245  {
246  /* _______________________________ _ _ _ _ _ _ _ _
247  | /|\ /|\ | /|\
248  | |penetration |R_fuel | |
249  | | | | |
250  ________|_____|_____________|___________|____|__________
251  /|\ | | | |relocation*R_fuel_cold
252  | | | | |
253  |Ri_clad| | _ _|_ _\|/_ _ _ _ _
254  | | | |
255  */
256 
257  // Relocation is recovered as much as possible to prevent penetration.
258  // If there is no penetration, there is no recovery
259 
260  rel[cellI] =
261  max
262  (
263  relocation*(1 - recFrac_),
264  2/DiamCold_*(relocation*DiamCold_/2 -
265  relRecover[cellI]*DiamCold_/2 - penetration)
266  );
267 
268  relRecover[cellI] =
269  min
270  (
271  relocation*(recFrac_),
272  2/DiamCold_*penetration + relRecover[cellI]
273  );
274  }
275 
276  }
277 
278  // Check which F and delta should be used
279  const dimensionedScalar& F_relocation =
280  isInUserParameters(this->group(), "F_epsilonRelocation") ?
281  this->F_epsilonRelocation() :
282  relocationModel::F_epsilonRelocation();
283 
284  const dimensionedScalar& F_relocationRecovery =
285  isInUserParameters(this->group(), "F_epsilonRelocationRecovery") ?
286  this->F_epsilonRelocationRecovery() :
287  relocationModel::F_epsilonRelocationRecovery();
288 
289  const dimensionedScalar& delta_relocation =
290  isInUserParameters(this->group(), "delta_epsilonRelocation") ?
291  this->delta_epsilonRelocation() :
292  relocationModel::delta_epsilonRelocation();
293 
294  const dimensionedScalar& delta_relocationRecovery =
295  isInUserParameters(this->group(), "delta_epsilonRelocationRecovery") ?
296  this->delta_epsilonRelocationRecovery() :
297  relocationModel::delta_epsilonRelocationRecovery();
298 
299  // Perturb epsilon Relocation for sensitivity analysis
300  applyParameters(epsilonRelocation_.ref(), addr,
301  F_relocation, delta_relocation);
302 
303  // Perturb epsilon recovered relocation for sensitivity analysis
304  applyParameters(epsilonRecoveredRelocation_.ref(), addr,
305  F_relocationRecovery, delta_relocationRecovery);
306  }
307 }

310 Foam::scalar Foam::relocationFRAPCON::setRelocation
311 (
312  double Q,
313  double Gap,
314  double D,
315  double Buslice
316 )
317 {
318  double Bu = Buslice/1000/0.881;
319 
320  // Power in kW/m
321  scalar Qprime = Q*constant::mathematical::pi*pow(D/2.0,2.0)/1000;
322 
323  // Initialize gap change
324  scalar deltaGap(0.0);
325 
326  if(modifiedRelocationModel_)
327  {
328  if(Bu < 0.0937) // TODO : correction at low burnup or not?
329  {
330  deltaGap = 0.055*100;
331  }
332  else
333  {
334  // Inizialize reloc variable
335  scalar reloc = 0;
336 
337  if(Qprime < 20)
338  {
339  reloc = 0.345;
340  }
341  else if(Qprime <= 40)
342  {
343  reloc = 0.345 + (Qprime - 20.0)/200.0;
344  }
345  else
346  {
347  reloc = 0.445;
348  }
349 
350  deltaGap =
351  (0.055 + min(reloc, reloc*(0.5795 + 0.2447*Foam::log(Bu))))*100;
352  }
353  }
354  else
355  {
356  scalar PFACTOR = (Qprime - 20)*5/20;
357 
358  scalar FBu = 0;
359  if(Bu < 5)
360  {
361  FBu = Bu/5;
362  }
363  else
364  {
365  FBu = 1;
366  }
367 
368  if(Qprime < 20 + 1e-6 )
369  {
370  deltaGap = (30 + 10*FBu);
371  }
372  else if(Qprime < 40 + 1e-6 )
373  {
374  deltaGap = (28 + PFACTOR + (12 + PFACTOR)*FBu);
375  }
376  else
377  {
378  deltaGap = (32 + 18*FBu);
379  }
380  }
381 
382  const scalar nominalValue = deltaGap*(Gap/D)/100;
383 
384  return nominalValue*perturb;
385 
386 }
