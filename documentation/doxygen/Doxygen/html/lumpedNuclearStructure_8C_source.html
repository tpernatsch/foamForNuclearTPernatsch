287 void
288 Foam::powerModels::lumpedNuclearStructure::updateLocalTemperatureProfile
289 (
290  const label& celli,
291  const scalar& HTSumi,
292  const scalar& HSumi
293 )
294 {
295  //-
296  scalarField& T(T_[celli]);
297 
298  //- Read region values
299  const label& regioni(cellToRegion_[celli]);
300  //const word& region(regionIndexToRegionName_[regioni]);
301  const scalar& fractionOfPowerFromNeutronics(fractionOfPowerFromNeutronics_[regioni]);
302  const label& nodesNumber(nodesNumber_[regioni]);
303  const label& nodeFuel(nodeFuel_[regioni]);
304  const label& nodeClad(nodeClad_[regioni]);
305  const label& nodeMatrix(nodeMatrix_[regioni]);
306  const scalar& kappaMatrix(kappaMatrix_[regioni]);
307  const scalarList& Hs(Hs_[regioni]);
308  const scalarList& rhoCp(rhoCp_[regioni]);
309  const scalarList& volFraction(volFraction_[regioni]);
310  const scalarList& qFraction(qFraction_[regioni]);
311 
312  const scalarField& TOld = T_.oldTime()[celli];
313 
314  const scalar& iA = this->iA_[celli];
315 
316  //- Update power density
317  const scalar& qRef(structure_.powerDensityNeutronics()[celli]);
318  scalar q = qRef * fractionOfPowerFromNeutronics;
319 
320  //- Recurrent quantities
321  scalar dt(mesh_.time().deltaT().value());
322  scalar Tcool(HTSumi / max(HSumi,SMALL));
323  scalar Hcool(HSumi*iA/this->alpha_[celli]);
324 
325  scalar diffusion(0.0);
326 
327  if(nodesNumber>1)
328  {
329  //- Init matrix, source
330  SquareMatrix<scalar> M(nodesNumber, nodesNumber, Foam::zero());
331  List<scalar> S(nodesNumber, 0.0);
332 
333  //- Construct matrix, source
334  {
335  //- Set "zeroGradient" BC at innermost node
336  {
337  diffusion *= 0.0;
338  if (nodeMatrix == 0 && kappaMatrix != 0)
339  {
340  diffusion = kappaMatrix * laplacianTmatrix_[celli];
341  }
342  M[0][1] = -Hs[1];
343  M[0][0] = volFraction[0] * rhoCp[0] / dt + Hs[1];
344  S[0] = diffusion
345  + q * qFraction[0]
346  + TOld[0] * volFraction[0] * rhoCp[0] / dt;
347 
348  }
349 
350  //- Bulk
351  if(nodesNumber>2)
352  {
353  for (int i = 1; i < nodesNumber-1; i++)
354  {
355  diffusion *= 0.0;
356  if (nodeMatrix == i && kappaMatrix != 0) // TmatrixPtr_.valid())
357  {
358  // volScalarField laplacian(fvc::laplacian(TmatrixPtr_()));
359  // diffusion = kappaMatrix * laplacian[celli];
360  diffusion = kappaMatrix * laplacianTmatrix_[celli];
361  }
362  M[i][i+1] = -Hs[i+1];
363  M[i][i-1] = -Hs[i];
364  M[i][i] = volFraction[i] * rhoCp[i] / dt + Hs[i+1] + Hs[i];
365  S[i] = diffusion
366  + q * qFraction[i]
367  + TOld[i] * volFraction[i] * rhoCp[i] / dt;
368  }
369  }
370 
371  //- Outer surface, convective BC with fluid(s) wetting the pin
372  {
373  label i(nodesNumber-1);
374  diffusion *= 0.0;
375  if (nodeMatrix == i && kappaMatrix != 0)
376  {
377  // volScalarField laplacian(fvc::laplacian(TmatrixPtr_()));
378  // diffusion = kappaMatrix * laplacian[celli];
379  diffusion = kappaMatrix * laplacianTmatrix_[celli];
380  }
381 
382  scalar HtoCool(Hs[i+1]*Hcool/(Hs[i+1]+Hcool)); //total H from last node to coolant
383  M[i][i-1] = -Hs[i];
384  M[i][i] = volFraction[i] * rhoCp[i] / dt + Hs[i] + HtoCool;
385  S[i] = diffusion
386  + q * qFraction[i]
387  + TOld[i] * volFraction[i] * rhoCp[i] / dt
388  + HtoCool * Tcool;
389  }
390  }
391 
392  //- Solve linear system
393  solve(T, M, S);
394  }
395  else
396  {
397  diffusion *= 0.0;
398  if (nodeMatrix == 0 && kappaMatrix != 0)
399  {
400  diffusion = kappaMatrix * laplacianTmatrix_[celli];
401  }
402  scalar HtoCool(Hs[1]*Hcool/(Hs[1]+Hcool)); //total H from last node to coolant
403  scalar M(volFraction[0] * rhoCp[0] / dt + HtoCool);
404  scalar S
405  (
406  diffusion
407  + q * qFraction[0]
408  + TOld[0] * volFraction[0] * rhoCp[0] / dt
409  + HtoCool * Tcool
410  );
411  T = S/M;
412  }
413 
414  //- Set fields (max and outer)
415  Tmax_[celli] = T[0] + q * qFraction[0] / Hs[0];
416  Tsurface_[celli] = (Hs[nodesNumber] * T[nodesNumber-1] + Hcool * Tcool )
417  / (Hs[nodesNumber] + Hcool);
418 
419  /*
420  Info << "T " << T << endl;
421  Info << "Tmax_[celli] " << Tmax_[celli] << endl;
422  Info << "Tsurface_[celli] " << Tsurface_[celli] << endl;
423  Info << "Tcool " << Tcool << endl;
424  */
425 
426  //- Update average fuel and clad temp used for coupling
427  this->structureRef().TFuelAv()[celli] = T[nodeFuel];
428  this->structureRef().TCladAv()[celli] = T[nodeClad];
429 
430  //- Update matrix temperature
431  if (kappaMatrix != 0)
432  {
433  Tmatrix_[celli] = T[nodeMatrix];
434  }
435 }
