161 inline Foam::scalar Foam::H2::densityH2(scalar p, scalar T) const
162 {
163  // Molecular hydrogen
164  // const scalar n_(0.4986 + 1.1735*omega() + 0.4754*pow(omega(), 2.0));
165  // const scalar alpha0_(0.42748*pow(R_*Tc(), 2.0) / Pc());
166  const scalar alpha_(alpha0_ / pow(T/Tc(), n_));
167  // const scalar b_(0.08664*R_*Tc() / Pc());
168  // const scalar c_(R_*Tc() / (Pc() + alpha0_ / (Vc() * (Vc()+b_))) + b_ - Vc());
169 
170  // A*rho**3 + B*rho**2 + C*rho + D = 0
171  const scalar A_(alpha_ * (c_-b_));
172  const scalar B_(W()*(alpha_ + p*b_*(c_-b_) - R_*T*b_ ));
173  const scalar C_(sqr(W()) * (p*c_ - R_*T));
174  const scalar D_(pow(W(), 3.0) * p);
175 
176  // Reduced form
177  const scalar p_((3.0*A_*C_ - sqr(B_)) / (3.0*sqr(A_)));
178  const scalar q_
179  (
180  (2.0*pow(B_, 3.0) - 9.0*A_*B_*C_ + 27.0*sqr(A_)*D_) / (27.0*pow(A_, 3.0))
181  );
182 
183  // https://en.wikipedia.org/wiki/Cubic_equation//Trigonometric_and_hyperbolic_solutions
184  const scalar discriminant(pow(p_, 3.0)/27.0 + sqr(q_)/4.0);
185 
186  if (discriminant < 0)
187  {
188  return(
189  2.0*sqrt(-p_/3.0) * cos(acos(3.0*q_/(2.0*p_)*sqrt(-3.0/p_))/3.0)
190  - B_/(3.0*A_)
191  );
192  }
193  else
194  {
195  const scalar u1(-q_/2.0 + sqrt(discriminant));
196  const scalar u2(-q_/2.0 - sqrt(discriminant));
197  return cbrt(u1) + cbrt(u2) - B_/(3.0*A_);
198  }
199 }

660 inline Foam::scalar Foam::H2::kappa(scalar p, scalar T) const
661 {
662  const scalar x_H_(x_H(p, T));
663  const scalar W_H_H2_(x_H_*W_H_ + (1.0-x_H_)*W());
664  const scalar A12(A_12(T));
665  const scalar B12(B_12(T));
666  const scalar M_12(2.0*W_H_*W()/(W_H_+W()));
667  const scalar lambdaH2(thermalConductivityH2(p, T));
668 
669  const scalar lambda_H(15.0/4.0 * R_*viscosityH(T)/W_H_);
670 
671  const scalar lambda_12(15.0/4.0 * R_/W_H_H2_ * 2.6693e-4*sqrt(M_12*1e-3*T)/omega22_12(T));
672  const scalar U_y
673  (
674  4.0/15.0 * A12 * W_H_*W()/sqr(M_12) * sqr(lambda_12)/(lambda_H*lambdaH2)
675  - (B12/5.0+1.0/12.0)
676  - 1.0/A12*(3.0*B12/8.0-25.0/32.0) * sqr(W_H_-W())/(W_H_*W())
677  );
678  const scalar U_z
679  (
680  4.0/15.0 * A12 * (
681  lambda_12*(1.0/lambda_H + 1.0/lambdaH2)*W_H_*W()/sqr(M_12) -1.0
682  )
683  - (B12/5.0+1.0/12.0)
684  );
685  const scalar U_1
686  (
687  4.0/15.0 * A12 - (B12/5.0+1.0/12.0)*W_H_/W()
688  + sqr(W_H_-W())/(2.0*W_H_*W())
689  );
690  const scalar U_2
691  (
692  4.0/15.0 * A12 - (B12/5.0+1.0/12.0)*W()/W_H_
693  + sqr(W()-W_H_)/(2.0*W_H_*W())
694  );
695 
696  const scalar X_lambda
697  (
698  sqr(x_H_)/lambda_H + 2.0*x_H_*(1.0-x_H_)/lambda_12
699  + sqr(1.0-x_H_)/lambdaH2
700  );
701  const scalar Y_lambda
702  (
703  sqr(x_H_)/lambda_H*U_1 + 2.0*x_H_*(1.0-x_H_)/lambda_12*U_y
704  + sqr(1.0-x_H_)/lambdaH2*U_2
705  );
706  const scalar Z_lambda
707  (
708  sqr(x_H_)*U_1 + 2.0*x_H_*(1.0-x_H_)*U_z + sqr(1.0-x_H_)*U_2
709  );
710 
711  const scalar lambda_f((1.0+Z_lambda)/(X_lambda+Y_lambda));
712 
713  const scalar delta_H
714  (
715  2.0*W_H_*specificEnthalpyH(T) - W()*specificEnthalpyH2(p, T)
716  );
717 
718  const scalar lambda_r
719  (
720  p*D(p, T)/T * sqr(delta_H/(R_*T))
721  * x_H_*(1.0-x_H_)/sqr(2.0-x_H_)
722  );
723 
724  return lambda_f + lambda_r;
725 }
