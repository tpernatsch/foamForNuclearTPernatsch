153 void GFTractionDisplacementFvPatchVectorField::updateCoeffs()
154 {
155  if (updated())
156  {
157  return;
158  }
159 
160  const dictionary& thermoMechanicalProperties =
161  db().lookupObject<IOdictionary>("thermoMechanicalProperties");
162 
163  //const dictionary& thermalProperties =
164  // db().lookupObject<IOdictionary>("thermalProperties");
165 
166 
167  const fvPatchField<scalar>& rho =
168  patch().lookupPatchField<volScalarField, scalar>("rho");
169 
170  const fvPatchField<scalar>& rhoE =
171  patch().lookupPatchField<volScalarField, scalar>("E");
172 
173  const fvPatchField<scalar>& nu =
174  patch().lookupPatchField<volScalarField, scalar>("nu");
175 
176  scalarField E(rhoE/rho);
177  scalarField mu(E/(2.0*(1.0 + nu)));
178  scalarField lambda(nu*E/((1.0 + nu)*(1.0 - 2.0*nu)));
179  scalarField threeK(E/(1.0 - 2.0*nu));
180 
181  // Switch planeStress(thermoMechanicalProperties.lookup("planeStress"));
182 
183  Switch planeStress
184  (
185  thermoMechanicalProperties.found("rheologyOptions") ?
186  thermoMechanicalProperties.subDict("rhelogyOptions").get<bool>("planeStress")
187  : false
188  );
189 
190  if (planeStress)
191  {
192  lambda = nu*E/((1.0 + nu)*(1.0 - nu));
193  threeK = E/(1.0 - nu);
194  }
195 
196  scalarField twoMuLambda(2*mu + lambda);
197 
198  vectorField n(patch().nf());
199 
200  const fvPatchField<symmTensor>& sigmaD =
201  patch().lookupPatchField<volSymmTensorField, symmTensor>("sigmaD");
202 
203  gradient() =
204  (
205  (traction_ - pressure_*n)/rho
206  + twoMuLambda*fvPatchField<vector>::snGrad() - (n & sigmaD)
207  )/twoMuLambda;
208 
209  //Switch thermalStress(thermalProperties.lookup("thermalStress"));
210 
211 
212  //if (thermalStress)
213  {
214  const fvPatchField<scalar>& threeKalpha=
215  patch().lookupPatchField<volScalarField, scalar>("threeKalpha");
216 
217  const fvPatchField<scalar>& Tmech =
218  patch().lookupPatchField<volScalarField, scalar>("TStruct");
219 
220  const fvPatchField<scalar>& TrefStructure =
221  patch().lookupPatchField<volScalarField, scalar>("TStructRef"); //modified
222  //fvPatchField<scalar> TrefStructurePatch =
223  //patch().lookupPatchField<volScalarField, scalar>("Tmech");
224  //fvPatchField<scalar> TrefStructurePatch =
225  //(Tmech/Tmech)*TrefStructure.value();
226 
227  gradient() += n*threeKalpha*(Tmech-TrefStructure)/twoMuLambda; //modified
228  //gradient() += n*threeKalpha*(Tmech)/twoMuLambda;
229  }
230 
231  fixedGradientFvPatchVectorField::updateCoeffs();
232 }
