217 void Foam::powerModels::fixedPower::powerUpdate()
218 {
219  forAll(this->toc(), regioni)
220  {
221  word region(this->toc()[regioni]);
222 
223  //- Setup cellToRegion_ mapping
224  const labelList& regionCells
225  (
226  structure_.cellLists()[region]
227  );
228 
229  // if(timeDependent_[regioni])
230  if (timeDependent_[regioni] && timeProfile_[regioni].valid())
231  {
232  // scalar t(mesh_.time().timeOutputValue()-t0_[regioni]);
233  scalar t(mesh_.time().timeOutputValue());
234  scalar timeDependentPowerDensity(timeProfile_[regioni].value(t));
235 
236  // The following assumes that each coefficient in the timeProfile_
237  // array represents a scaling factor relative to the initial power
238  // density value. For instance, a coefficient of 1 at position 'i'
239  // implies that the power at time step 'i' is equal to the initial
240  // value.
241 
242  // The power density at each cell and at time t is calculated using:
243  // powerDensity(t - deltat) =
244  // initialPowerDensity * timeCoefficient(t - deltat)
245  //
246  // powerDensity(t) = initialPowerDensity * timeCoefficient(t)
247 
248  // Using the previous relations, we can eliminate the initial
249  // power value and derive the updated power density.
250 
251  const volScalarField& powerDensityOld = powerDensity_.oldTime();
252 
253  // scalar tOld(mesh_.time().timeOutputValue()-t0_[regioni] -
254  // mesh_.time().deltaT().value());
255  scalar tOld
256  (
257  mesh_.time().timeOutputValue() - mesh_.time().deltaT().value()
258  );
259 
260  scalar timeDependentPowerDensityOld(
261  timeProfile_[regioni].value(tOld));
262 
263  forAll(regionCells, i)
264  {
265  label celli(regionCells[i]);
266 
267  powerDensity_[celli] = timeDependentPowerDensity * (
268  powerDensityOld[celli]/timeDependentPowerDensityOld);
269  }
270  }
271  }

274 void Foam::powerModels::fixedPower::correct
275 (
276  const volScalarField& HTSum, // == SUM_j [htc_j*T_j*frac_j]
277  const volScalarField& HSum // == SUM_j [htc_j*frac_j]
278 )
279 {
280  this->powerUpdate();
281  scalar dt(mesh_.time().deltaT().value());
282  volScalarField& T0(T_.oldTime());
283 
284  forAll(cellList_, i)
285  {
286  label celli(cellList_[i]);
287  scalar alphaRhoCpByDt(alphaRhoCp_[celli]/dt);
288  const scalar& iA(iA_[celli]);
289  T_[celli] =
290  (
291  iA*HTSum[celli]
292  + alpha_[celli]*powerDensity_[celli]
293  + alphaRhoCpByDt*T0[celli]
294  )/
295  (alphaRhoCpByDt + iA*HSum[celli]);
296  }
297  T_.correctBoundaryConditions();
298 
299  /*
300  What I did above is equivalent to doing this, but possibly faster, given
301  that I might not have to solve the equation for the entire mesh, but only
302  over the cells over which the fixedPower models exists. The time
303  derivative was hardcoded to be forward Euler
304 
305  fvScalarMatrix EEqn
306  (
307  fvm::ddt(alphaRhoCp_, T_)
308  ==
309  iA_*HTSum
310  - fvm::Sp(iA_*HSum, T_)
311  + alpha_*powerDensity_
312  );
313  EEqn.solve();
314  T_.correctBoundaryConditions();
315  */
316 }
