59 template<class BasicTurbulenceModel>
60 void porousKEpsilon<BasicTurbulenceModel>::correctNut()
61 {
62  this->nut_ = Cmu_*sqr(k_)/epsilon_;
63  this->nut_.correctBoundaryConditions();
64 
65  //- Correct alphat before nut is stabilized
66  this->Prt_ = dimensioned<scalar>::lookupOrDefault
67  (
68  "Prt",
69  this->coeffDict(),
70  1.0
71  );
72  this->alphat_ = this->rho_*this->nut_/this->Prt_;
73  this->alphat_.correctBoundaryConditions();
74 
75  if (nutStabilization_)
76  {
77  this->nut_ +=
78  pos(structure_)*FSPair_.fluidRef().magU()*DhStructPtr_()/
79  laminarReStructPtr_();
80  this->nut_.correctBoundaryConditions();
81  }
82 
83  fv::options::New(this->mesh_).correct(this->nut_);

521 template<class BasicTurbulenceModel>
522 void porousKEpsilon<BasicTurbulenceModel>::correct()
523 {
524  if (!this->turbulence_)
525  {
526  return;
527  }
528 
529  // Local references
530  const alphaField& alpha = this->alpha_;
531  const rhoField& rho = this->rho_;
532  const surfaceScalarField& alphaRhoPhi = this->alphaRhoPhi_;
533  const volVectorField& U = this->U_;
534  volScalarField& nut = this->nut_;
535  fv::options& fvOptions(fv::options::New(this->mesh_));
536 
537  eddyViscosity<RASModel<BasicTurbulenceModel>>::correct();
538 
539  volScalarField::Internal divU
540  (
541  fvc::div(fvc::absolute(this->phi(), U))().v()
542  );
543 
544  tmp<volTensorField> tgradU = fvc::grad(U);
545  volScalarField::Internal G
546  (
547  this->GName(),
548  nut.v()*(dev(twoSymm(tgradU().v())) && tgradU().v())
549  );
550  tgradU.clear();
551 
552  // Update epsilon and G at the wall
553  epsilon_.boundaryFieldRef().updateCoeffs();
554 
555  //- Structure marker fields
556  tmp<volScalarField> noStructure(neg0(structure_));
557  tmp<volScalarField> yesStructure(pos(structure_));
558 
559  //- Compute equilibrium k, epsilon values
560  equilibriumK_ =
561  (
562  1.5*sqr
563  (
564  mag(U)*
565  turbulenceIntensityCoeff_*pow
566  (
567  FSPair_.Re(), turbulenceIntensityExp_
568  )
569  )
570  );
571  //equilibriumEpsilon_.correctBoundaryConditions();
572  equilibriumEpsilon_ =
573  (
574  Cmu3by4_*pow(equilibriumK_, 1.5)
575  /
576  (
577  max
578  (
579  turbulenceLengthScaleCoeff_*structure_.Dh(),
580  dimensionedScalar("", dimLength, SMALL)
581  )
582  )
583  );
584  //equilibriumK_.correctBoundaryConditions();
585 
586  /*
587  Info<< "eqEpsilon (avg min max) = "
588  << equilibriumEpsilon_.weightedAverage(this->mesh_.V()).value()
589  << " " << min(equilibriumEpsilon_).value()
590  << " " << max(equilibriumEpsilon_).value()
591  << " " << equilibriumEpsilon_.dimensions() << endl;
592  Info<< "eqK (avg min max) = "
593  << equilibriumK_.weightedAverage(this->mesh_.V()).value()
594  << " " << min(equilibriumK_).value()
595  << " " << max(equilibriumK_).value()
596  << " " << equilibriumK_.dimensions() << endl;
597  */
598 
599  //- magU/convergenceLength gives the inverse of the convergence time scale,
600  // i.e. what was previously know/provided as convergenceRate
601  tmp<volScalarField> alphaRhoConv
602  (
603  alpha*rho*(mag(U)/convergenceLength_)*yesStructure()
604  );
605 
606  // Dissipation equation
607  tmp<fvScalarMatrix> epsEqn
608  (
609  fvm::ddt(alpha, rho, epsilon_)
610  + fvm::div(alphaRhoPhi, epsilon_)
611  - fvm::laplacian(alpha*rho*DepsilonEff(), epsilon_)
612  ==
613  C1_*alpha()*rho()*G*epsilon_()/k_()*noStructure()
614  - fvm::SuSp
615  (
616  ((2.0/3.0)*C1_ - C3_)*alpha()*rho()*divU*noStructure(),
617  epsilon_
618  )
619  - fvm::Sp(C2_*alpha()*rho()*epsilon_()/k_()*noStructure(), epsilon_)
620  + epsilonSource()
621  - fvm::Sp(alphaRhoConv(), epsilon_)
622  + alphaRhoConv()*equilibriumEpsilon_
623  + fvOptions(alpha, rho, epsilon_)
624  );
625 
626  epsEqn.ref().relax();
627  fvOptions.constrain(epsEqn.ref());
628  epsEqn.ref().boundaryManipulate(epsilon_.boundaryFieldRef());
629  solve(epsEqn);
630  fvOptions.correct(epsilon_);
631  bound(epsilon_, this->epsilonMin_);
632 
633  // Turbulent kinetic energy equation
634  tmp<fvScalarMatrix> kEqn
635  (
636  fvm::ddt(alpha, rho, k_)
637  + fvm::div(alphaRhoPhi, k_)
638  - fvm::laplacian(alpha*rho*DkEff(), k_)
639  ==
640  alpha()*rho()*G*noStructure()
641  - fvm::SuSp((2.0/3.0)*alpha()*rho()*divU*noStructure(), k_)
642  - fvm::Sp(alpha()*rho()*epsilon_()/k_()*noStructure(), k_)
643  + kSource()
644  - fvm::Sp(alphaRhoConv(), k_)
645  + alphaRhoConv()*equilibriumK_
646  + fvOptions(alpha, rho, k_)
647  );
648 
649  kEqn.ref().relax();
650  fvOptions.constrain(kEqn.ref());
651  solve(kEqn);
652  fvOptions.correct(k_);
653  bound(k_, this->kMin_);
654 
655  correctNut();
656 }
