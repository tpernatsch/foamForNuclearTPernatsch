133 void Foam::creepModel::correctCreep
134 (
135  volSymmTensorField& epsilonEl,
136  const labelList& addr
137 )
138 {
139  if(mesh_.time().timeIndex() > 0)
140  {
141  const symmTensorField& epsilonCreepOldI =
142  epsilonCreep_.oldTime().internalField();
143 
144  //- Adjust elastic strain tensor by subtracting old creep values
145  forAll(addr, addrI)
146  {
147  const label cellI = addr[addrI];
148 
149  epsilonEl[cellI] -= epsilonCreepOldI[cellI];
150 
151  const cell& c = mesh_.cells()[cellI];
152 
153  forAll(c, faceI)
154  {
155  const label patchID(mesh_.boundaryMesh().whichPatch(c[faceI]));
156 
157  if (patchID > -1 and epsilonEl.boundaryField()[patchID].size())
158  {
159  const label faceID =
160  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
161 
162  symmTensorField& epsilonElP =
163  epsilonEl.boundaryFieldRef()[patchID];
164  const symmTensorField& epsilonCreepOldP =
165  epsilonCreep_.oldTime().boundaryField()[patchID];
166 
167  epsilonElP[faceID] -= epsilonCreepOldP[faceID];

228 Foam::scalar Foam::creepModel::nextDeltaT
229 (
230  const labelList& addr
231 )
232 {
233  //- Store old value of deltaT
234  scalar oldDeltaT = mesh_.time().deltaT().value();
235 
236  //- Prepare new deltaT
237  // scalar deltaT(0);
238 
239  //- Prepare average and maximum DepsilonCreepEq from last time step
240  scalar averageDepsilonCreepEq(0);
241  scalar maxDepsilonCreepEq(0);
242  scalar volTot(0);
243 
244  const scalarField& DepsilonCreepEqI(DepsilonCreepEq_());
245  const scalarField& V(mesh_.V());
246 
247  //- Find maximum and average DepsilonCreepEq
248  forAll(addr, addrI)
249  {
250  const label cellI = addr[addrI];
251  averageDepsilonCreepEq += DepsilonCreepEqI[cellI]*V[cellI];
252  volTot += V[cellI];
253 
254  maxDepsilonCreepEq = max(maxDepsilonCreepEq, DepsilonCreepEqI[cellI]);
255  }
256 
257  reduce(volTot, sumOp<scalar>());
258 
259  reduce(averageDepsilonCreepEq, sumOp<scalar>());
260 
261  reduce(maxDepsilonCreepEq, maxOp<scalar>());
262 
263  averageDepsilonCreepEq /= volTot;
264 
265  //- Calculate average and maximum creep rate from last time step
266  scalar averageCreepRate(averageDepsilonCreepEq/oldDeltaT);
267  scalar maxCreepRate(maxDepsilonCreepEq/oldDeltaT);
268 
269  Info << "deltaT: averageDepsilonCreepEq " << averageDepsilonCreepEq << endl;
270  Info << "deltaT: averageCreepRate " << averageCreepRate << endl;
271  Info << "deltaT: maxDepsilonCreepEq " << maxDepsilonCreepEq << endl;
272  Info << "deltaT: maxCreepRate " << maxCreepRate << endl;
273 
274  return min
275  (
276  maxAverageCreep_/max(averageCreepRate, SMALL),
277  maxMaximumCreep_/max(maxCreepRate, SMALL)
278  );
279
