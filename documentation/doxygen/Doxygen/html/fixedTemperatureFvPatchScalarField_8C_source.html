37 void Foam::fixedTemperatureFvPatchScalarField::updateSurfaceTemperature()
38 {
39  if
40  (
41  patch().boundaryMesh().mesh().foundObject<surfaceScalarField>
42  ("oxideThickness")
43  )
44  {
45  //- Current patch temperature
46  scalarField Tp(*this);
47 
48  //- Oxide thicnkess
49  const scalarField& S
50  = patch().lookupPatchField<surfaceScalarField, scalar>("oxideThickness");
51 
52  //- Internal field
53  const scalarField Ti(patchInternalField());
54 
55  //- Material conductivity
56  const scalarField& k
57  = patch().lookupPatchField<volScalarField, scalar>(kappaName_);
58 
59  //- Compute oxide conductivity (MATPRO)
60  const scalarField kox = 0.835 + 1.81e-4 * Tp;
61 
62  //- Inverse cell to patch face distances
63  const scalarField d = patch().deltaCoeffs();
64 
65  //- Thermal resistances
66  scalarField R(1/d/k);
67  scalarField Rox(S/kox);
68  scalarField Reff(R + Rox);
69 
70  //- Loop over all the faces of the boundary patch
71  forAll(patch(), i)
72  {
73  //- Compute Fourier heat flux
74  const scalar q = -(T_[i] - Ti[i])/Reff[i];
75 
76  //- Compute outer cladding or metal/oxide interface temperature
77  Tp[i] = (T_[i] + q*S[i]/kox[i]);
78  }
79 
80  fvPatchField<scalar>::operator==(Tp);
81  }
82  else
83  {
84 
85  fvPatchField<scalar>::operator==(T_);
86  }
87 }
