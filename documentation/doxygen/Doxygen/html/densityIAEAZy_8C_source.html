90 void Foam::densityIAEAZy::correct
91 (
92  scalarField& sf,
93  const scalarField& T,
94  const labelList& addr
95 )
96 const
97 {
98  forAll(addr, i)
99  {
100  const label cellI = addr[i];
101 
102  const scalar Ti = T[cellI];
103 
104  scalar rho1 = par1 - par2*Ti;
105  scalar rho2 = par3 - par4*Ti;
106 
107  scalar nominalValue(0.0);
108 
109  if (Ti < 1083)
110  {
111  nominalValue = rho1;
112  }
113  else if (Ti < 1144)
114  {
115  nominalValue = rho1 + (rho2-rho1)*(Ti-1083)/(1144-1083);
116  }
117  else if (Ti < 1800)
118  {
119  nominalValue = rho2;
120  }
121  else
122  {
123  WarningInFunction
124  << "Supplied temperature, " << Ti << ", above maximum of 1800 K";
125 
126  nominalValue = 0;
127  }
128 
129  sf[cellI] = nominalValue*perturb;
130  }
131 }
