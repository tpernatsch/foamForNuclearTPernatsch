38 Class
39  Foam::structure
40 
41 Description
42  Class to handle the description of a mesh-unresolved structure. It is
43  derived from phaseBase, to represent the volume fraction of the structure.
44  The key member for the understanding of this class is cellFields_.
45  The structure can be in fact though of as being composed by different
46  regions, with properties defined only on certain cells of the mesh.
47  However, as these region are meant NOT to spatially overlap, the
48  overall structure can be described by a single set of fields (Dh_, iApow_,
49  etc.) that are initialized/set/modified on a cell-by-cell basis.
50  The cellFields_ member is thus a HashTable of scalarFields indexed by
51  region name. These scalarFields are = 1.0 where the given
52  region exists, and 0 elsewhere. What the region names are, that
53  is set by the actual run-time-selected instance of the structure
54  class.
55  Currently, the only available structure is byZone, in which
56  the regions are effectively cellZones, and the cellFields_ is
57  thus a dicts of scalarFields representing where each cellZone exists.

435 void structure::setRegionField
436 (
437  const dictionary& dicts,
438  GeometricField<Type, fvPatchField, volMesh>& field,
439  word key
440 ) const
441 {
442  typedef GeometricField<Type, fvPatchField, volMesh> fieldType;
443 
444  IOobject fieldHeader
445  (
446  field.name(),
447  mesh_.time().timeName(),
448  mesh_,
449  IOobject::NO_READ
450  );
451 
452  //- field is a reference to a field that already exists. If it was
453  // created with a READ_IF_PRESENT or MUST_READ option, do nothing
454  if
455  (
456  ((
457  field.readOpt() == IOobject::MUST_READ
458  or field.readOpt() == IOobject::READ_IF_PRESENT
459  )
460  and fieldHeader.typeHeaderOk<fieldType>(true))
461  )
462  {
463  // DO NOTHING
464  /*
465  // This part used to spatially limit the field only over the cells
466  // overwhich it should be defined.
467  // Removed. This creates problems to setting powerDensityNeutronics
468  // since each powerModel will set it to zero outside its cells
469  scalarField cellField(mesh_.cells().size(), 0);
470  forAll(dicts.toc(), i)
471  {
472  word region(dicts.toc()[i]);
473  const labelList& cells(cellLists_[region]);
474  forAll(cells, j)
475  {
476  cellField[cells[j]] = 1.0;
477  }
478  }
479  field.primitiveFieldRef() *= cellField;
480  */
481  }
482  //- Otherwise, read the value the field should have on a
483  // region-by-region basis
484  else
485  {
486  forAll(dicts.toc(), i)
487  {
488  word region(dicts.toc()[i]);
489  const labelList& cells(cellLists_[region]);
490  Type value(dicts.subDict(region).get<Type>(key));
491  forAll(cells, j)
492  {
493  field[cells[j]] = value;
494  }
495  }
496  }
497  field.correctBoundaryConditions();
498 }
