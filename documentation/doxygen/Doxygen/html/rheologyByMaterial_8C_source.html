138 void Foam::rheologyByMaterial::correct
139 (
140  volSymmTensorField& sigma,
141  volSymmTensorField& epsilon,
142  volVectorField& D
143 )
144 {
145  const labelListList& matAddrList(mat_.matAddrList());
146 
147  if(thermalExpansion_)
148  {
149  epsTh_ = mat_.thermalExpansion();
150  }
151 
152  // For planeStress or modifiedPlaneStrain, correct strain in the out of
153  // plane direction
154  if (planeStress_)
155  {
156  if (mesh_.solutionD()[vector::Z] > -1)
157  {
158  FatalErrorInFunction()
159  << "For planeStress, this material law assumes the empty "
160  << "direction is the Z direction!" << exit(FatalError);
161  }
162 
163  correctPlaneStress(epsilon);
164  }
165  else if(modifiedPlaneStrain_)
166  {
167  if (mesh_.solutionD()[vector::Z] > -1)
168  {
169 
170  FatalErrorInFunction()
171  << "For modifiedPlaneStrain, this material law assumes the empty "
172  << "direction is the Z direction!" << exit(FatalError);
173  }
174 
175  correctModifiedPlaneStrain(epsilon, D);
176  }
177 
178  // Set elastic strain tensor
179  epsEl_ = epsilon - additionalStrain_;
180 
181  // Subtract thermal strain
182  if(thermalExpansion_)
183  {
184  epsEl_ -= epsTh_;
185  }
186 
187  volScalarField sigmaHydExplicit(sigmaHyd_);
188 
189  // Correct the elastic strain tesnor removing oldTime copmonents of plastic
190  // strains
191  forAll(mat_.materialsList(), i)
192  {
193  const labelList& addr(matAddrList[i]);
194  lawsList_[i].correctEpsilonEl(epsEl_, addr);
195  }
196 
197  // Correct material rheology laws. It calculates the hydrostatic and deviatoric
198  // component of the stress and it updates the elastic strain component
199  forAll(mat_.materialsList(), i)
200  {
201  const labelList& addr(matAddrList[i]);
202  lawsList_[i].correct(sigmaHydExplicit, sigmaDev_, epsEl_, addr);
203  }
204 
205  if(solvePressureEqn_)
206  {
207  solvePressureEqn(sigmaHydExplicit);
208  }
209  else
210  {
211  sigmaHyd_ = sigmaHydExplicit;
212  }
213 
214  // Calculate sigma from combination of sigmaHyd and sigmaDev.
215  // Apply damage if applicable.
216  forAll(mat_.materialsList(), i)
217  {
218  const labelList& addr(matAddrList[i]);
219  lawsList_[i].updateStress(sigma, sigmaHyd_, sigmaDev_, addr);
220  }
221 
222  sigma.correctBoundaryConditions();
223 }

321 void Foam::rheologyByMaterial::correctModifiedPlaneStrain
322 (
323  volSymmTensorField& epsilon,
324  volVectorField& D
325 )
326 {
327  if(mesh_.foundObject<sliceMapper>("sliceMapper"))
328  {
329  if(mapper_ == nullptr)
330  {
331  mapper_ = &mesh_.lookupObject<sliceMapper>("sliceMapper");
332  }
333 
334  // Include the gap gas pressure
335  const gapGasModel& gapGas
336  = mesh_.lookupObject<gapGasModel>("gapGas");
337 
338  // Pin direction assumed to be the z-axis
339  vector pinDirection_(0, 0, 1);
340 
341  const globalOptions& globalOpt
342  (mesh_.lookupObject<globalOptions>("globalOptions"));
343 
344  // Obtain model angular fraction to calculate the slice quantities for the
345  // corresponding 360 degrees 3-D rod
346  scalar angularFraction = globalOpt.angularFraction();
347 
348  // Get number of slices. Slices are equivalent to cellZones.
349  int nSlices(mapper_->nSlices());
350 
351  if(nSlices<=0)
352  {
353  FatalErrorInFunction()
354  << "When the option \'modifiedPlaneStrain\' is activated, the number "
355  << "of slices cannot be less than 1, "<< nl << "as the slices are needed "
356  << "to perform the axial force balance." << nl
357  << "Check that a sliceMapper different than \"none\" is used "
358  << "or check the slice definition."
359  << abort(FatalError);
360  }
361 
362  if(!springForce_.valid())
363  {
364 #ifdef OPENFOAMFOUNDATION
365  springForce_.set(new scalarField(nSlices, 0.0));
366 #elif OPENFOAMESI
367  springForce_.reset(new scalarField(nSlices, 0.0));
368 #endif
369  }
370 
371  if(!pressureForce_.valid())
372  {
373 #ifdef OPENFOAMFOUNDATION
374  pressureForce_.set(new scalarField(nSlices, 0.0));
375 #elif OPENFOAMESI
376  pressureForce_.reset(new scalarField(nSlices, 0.0));
377 #endif
378  }
379 
380  // Reference to material properties
381  const volScalarField& mu(mat_.mu());
382  const volScalarField& lambda(mat_.lambda());
383 
384  // Assume that:
385  // sigmaZ =
386  // (2mu+lambda)*epsZ - (2mu+lambda)*epsZadd + lambda*(epsElx+epsElY)
387  //
388  // integral(sigmaZ*dA) = -k_spring*D_top -p_gas*Atop
389  //
390  // Substituting sigmaZ in the second equaition (and assuming epsZ costant
391  // in each slice):
392  // epsZ_slice*int((2mu+lambda)*dA) =
393  // int((2mu+lambda)*epsZadd*dA) - int(lambda*(epsElx+epsElY)*dA) -
394  // k_spring*D_top -p_gas*Atop
395  //
396  // therefore:
397  // epsZ_slice = 1/(int((2mu+lambda)*dA))*
398  // [int((2mu+lambda)*epsZadd*dA) - int(lambda*(epsElx+epsElY)*dA)
399  // -k_spring*D_top -p_gas*Atop]
400  //
401  // Note that the integral terms are substituded with the volume average
402  // times the area, eg:
403  // int((2mu+lambda)*dA) = (2mu+lambda)_avg*A
404  //
405  // Therefore:
406  // epsZ_slice = 1/((2mu+lambda)_avg*A)*
407  // [ ((2mu+lambda)*epsZadd)_avg*A - (lambda*(epsElx+epsElY))_avg*A
408  // -k_spring*D_top -p_gas*Atop]
409  //
410  // Simplifing and dropping the avg subscript:
411  // epsZ_slice = 1/(2mu+lambda)*
412  // [ (2mu+lambda)*epsZadd - lambda*(epsElx+epsElY)
413  // -k_spring*D_top/A -p_gas*Atop/A]
414  //
415  // where the area of the slice A can be expressed as:
416  // A = V/(alpha*H)
417  //
418  // where V is the slice volume, H is the slice Height and alpha is the
419  // angular fraction (e.g. 1.0 for a 360degree 3-D model)
420 
421  // Prepare twoMuLambda
422  volScalarField twoMuLambda("twoMuLambda", 2*mu + lambda);
423 
424  // Prepare lambda*(epsElX+epsElY) field
425 
426  volScalarField lambdaEpsElXEpsElY
427  (
428  "lambdaEpsElXEpsElY",
429  lambda*
430  (
431  epsEl_.component(symmTensor::XX) +
432  epsEl_.component(symmTensor::YY)
433  )
434  );
435 
436  // Prepare (2mu+lambda)*epsZadd
437 
438  // Get additional strains defined by the constitutive mechanical laws
439  volSymmTensorField lawsAdditionalStrain
440  (
441  getLawsAdditionalStrain()
442  );
443 
444  volScalarField epsilonAddZ
445  (
446  additionalStrain_.component(symmTensor::ZZ)
447  + lawsAdditionalStrain.component(symmTensor::ZZ)
448  );
449 
450  if(thermalExpansion_)
451  {
452  epsilonAddZ += epsTh_.component(symmTensor::ZZ);
453  }
454 
455  volScalarField twoMuLambdaEpsAddZ
456  (
457  "twoMuLambdaEpsAddZ",
458  twoMuLambda*epsilonAddZ
459  );
460 
461  // Old time epsZ (for incremental solvers)
462  volScalarField epsilonOldZ
463  (
464  "epsilonOldZ",
465  epsilon.oldTime().component(symmTensor::ZZ)
466  );
467 
468  // Convert 3D fields to slice-average
469  const scalarField& twoMuLambda_1D(mapper_->sliceAverage(twoMuLambda));
470  const scalarField& lambdaEpsElXEpsElY_1D(mapper_->sliceAverage(lambdaEpsElXEpsElY));
471  const scalarField& twoMuLambdaEpsAddZ_1D(mapper_->sliceAverage(twoMuLambdaEpsAddZ));
472  const scalarField& epsilonOldZ_1D(mapper_->sliceAverage(epsilonOldZ));
473 
474  // Reference to mesh volumes
475  const scalarField& V(mesh_.V());
476 
477  // Prepare list of bool indicating if cellZone is fuel type or not
478  labelList isFuel(nSlices, 0);
479 
480  // Prepare slice volumes list
481  scalarField volumePerSlice(nSlices, 0.0);
482 
483  const PtrList<labelList>& sliceAddrList(mapper_->sliceAddrList());
484 
485  forAll(sliceAddrList, i)
486  {
487  const labelList& addr(sliceAddrList[i]);
488  // if(addr.size())
489  {
490  isFuel[i] = mapper_->isFuel()[i];
491 
492  scalarField Vaddr(V, addr);
493  volumePerSlice[i] = gSum(Vaddr);
494  }
495  }
496 
497  // Prepare 1D epsilonZ field
498  scalarField epsilonZ_1D(nSlices, 0);
499 
500  // Prepare slice heights.
501  // It is assumed that the slices for fuel and cladding are given in order
502  // from bottom to top. It is also assumed that the cladding caps are not
503  // modeled, i.e. there might be a bottom or top plenum with an additional
504  // portion of the cladding outside the bounds of the fuel column, but the
505  // cladding model is truncated
506  scalarField sliceHeights(nSlices, 0.0);
507 
508  const vectorField& points = mesh_.points();
509  const labelListList& cellPoints = mesh_.cellPoints();
510  scalarField z = (points & pinDirection_);
511  scalarField r = mag(points - (z*pinDirection_));
512 
513  scalar fuelTopMaxR(0);
514  scalar fuelTopMinR(0);
515  scalar cladTopMaxR(0);
516  scalar cladTopMinR(0);
517 
518  const PtrList<labelList>& sliceAddr(mapper_->sliceAddrList());
519 
520  // Loop over slices
521  forAll(sliceAddr, sliceID)
522  {
523  const labelList& addr(mapper_->sliceAddrList()[sliceID]);
524 
525  scalar zMax = -GREAT;
526  scalar zMin = GREAT;
527 
528  scalar rMax = -GREAT;
529  scalar rMin = GREAT;
530 
531  forAll(addr, addrI)
532  {
533  const label cellI = addr[addrI];
534  const labelList& ptIds = cellPoints[cellI];
535 
536  forAll(ptIds, i)
537  {
538  const label ptI = ptIds[i];
539  zMin = min(zMin, z[ptI]);
540  zMax = max(zMax, z[ptI]);
541 
542  rMax = max(rMax, r[ptI]);
543  rMin = min(rMin, r[ptI]);
544  }
545  }
546 
547  reduce(zMin, minOp<scalar>());
548  reduce(zMax, maxOp<scalar>());
549 
550  reduce(rMin, minOp<scalar>());
551  reduce(rMax, maxOp<scalar>());
552 
553  if(isFuel[sliceID])
554  {
555  fuelTopMaxR = rMax;
556  fuelTopMinR = rMin;
557  }
558  else
559  {
560  cladTopMaxR = rMax;
561  cladTopMinR = rMin;
562  }
563 
564  sliceHeights[sliceID] = zMax - zMin;
565  }
566 
567  // Prepare spring force quantities
568  scalar forceAverage(0);
569  scalar forceAveragePrev(GREAT);
570 
571  // The axial strains are a function of the spring force, which in turn is
572  // calculated as a function of the axial strain. Iterate until convergence.
573  do
574  {
575  forceAveragePrev =
576  gSum((springForce_()+pressureForce_())*volumePerSlice)/gSum(volumePerSlice);
577 
578  // Calculate slice-average axial strain
579  forAll(springForce_(), sliceID)
580  {
581  scalar sliceA =
582  volumePerSlice[sliceID]/(angularFraction*sliceHeights[sliceID]);
583 
584  epsilonZ_1D[sliceID] =
585  (
586  springForce_()[sliceID]/sliceA +
587  pressureForce_()[sliceID]/sliceA -
588  lambdaEpsElXEpsElY_1D[sliceID] +
589  twoMuLambdaEpsAddZ_1D[sliceID]
590  )/(max(twoMuLambda_1D[sliceID], SMALL));
591  }
592 
593  scalar totalAxialDispFuel(0);
594  scalar totalAxialDispClad(0);
595  scalar totalAxialDispFuelOld(0);
596  scalar totalAxialDispCladOld(0);
597 
598  // Redistribute average axial strain in cells and faces
599  forAll(mapper_->sliceAddrList(), sliceID)
600  {
601  const labelList addr(mapper_->sliceAddrList()[sliceID]);
602 
603  scalar totalAxialDisp(0);
604  scalar totalAxialDispOld(0);
605 
606  if(isFuel[sliceID])
607  {
608  totalAxialDispFuel +=
609  epsilonZ_1D[sliceID]*sliceHeights[sliceID];
610 
611  totalAxialDispFuelOld +=
612  epsilonOldZ_1D[sliceID]*sliceHeights[sliceID];
613 
614  totalAxialDisp = totalAxialDispFuel;
615  totalAxialDispOld = totalAxialDispFuelOld;
616  }
617  else
618  {
619  totalAxialDispClad +=
620  epsilonZ_1D[sliceID]*sliceHeights[sliceID];
621 
622  totalAxialDispCladOld +=
623  epsilonOldZ_1D[sliceID]*sliceHeights[sliceID];
624 
625  totalAxialDisp = totalAxialDispClad;
626  totalAxialDispOld = totalAxialDispCladOld;
627  }
628 
629  forAll(addr, addrI)
630  {
631  const label& cellI(addr[addrI]);
632 
633  epsilon[cellI].zz() = epsilonZ_1D[sliceID];
634  if(mesh_.foundObject<fvMesh>("referenceMesh"))
635  {
636  D[cellI].z() = totalAxialDisp - totalAxialDispOld;
637  }
638  else
639  {
640  D[cellI].z() = totalAxialDisp;
641  }
642 
643  // Correct z component of the strain on the patch face
644  // if current cell is adjacent to boundary patch.
645  // Cannot use correctBoundaryConditions as it alters also
646  // the strain components along x and y
647 
648  // Reference to list of cell faces
649  const cell& c = mesh_.cells()[cellI];
650 
651  forAll(c, faceI)
652  {
653  const label patchID =
654  mesh_.boundaryMesh().whichPatch(c[faceI]);
655 
656  if (patchID > -1)
657  {
658  const label faceID =
659  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
660 
661  if
662  (
663  !isType<emptyFvPatchSymmTensorField>
664  (
665  epsilon.boundaryField()[patchID]
666  )
667  )
668  {
669  epsilon.boundaryFieldRef()[patchID][faceID].zz() =
670  epsilon[cellI].zz();
671 
672  D.boundaryFieldRef()[patchID][faceID].z() =
673  D[cellI].z();
674  }
675  }
676  }
677  }
678  }
679 
680  // Update value of spring and pressure force
681 
682  scalar pFluid(0);
683 
684 #ifdef OPENFOAMESIORFOUNDATION
685  if(fluidPressureList_.valid())
686  {
687  fluidPressureList_->value
688  (
689  mesh_.time().timeToUserTime
690  (
691  mesh_.time().value()
692  )
693  );
694  }
695 #endif
696 
697  scalar pi = Foam::constant::mathematical::pi;
698 
699  // Total spring elongation
700  scalar springDisp =
701  max(totalAxialDispFuel - totalAxialDispClad + springPreCompression_, 0.0);
702 
703  // Calculate spring force for the fuel column
704  forAll(mapper_->sliceAddrList(), sliceID)
705  {
706  if(isFuel[sliceID])
707  {
708  springForce_()[sliceID] = -springModulus_*springDisp;
709 
710  scalar rMax = fuelTopMaxR;
711  scalar rMin = fuelTopMinR;
712 
713  pressureForce_()[sliceID] =
714  -gapGas.p()*pi*(pow(rMax, 2.0) - pow(rMin, 2.0));
715  }
716  else
717  {
718  springForce_()[sliceID]= springModulus_*springDisp;
719 
720  scalar rMax = cladTopMaxR;
721  scalar rMin = cladTopMinR;
722 
723  pressureForce_()[sliceID] =
724  gapGas.p()*pi*(pow(rMin, 2.0)) - pFluid*pi*(pow(rMax, 2.0));
725  }
726  }
727 
728  forceAverage =
729  gSum((springForce_()+pressureForce_())*volumePerSlice)/gSum(volumePerSlice);
730  }
731  while
732  (
733  (mag(forceAveragePrev - forceAverage)) >
734  (precision_*mag(forceAveragePrev))
735  );
736  }
737 }
