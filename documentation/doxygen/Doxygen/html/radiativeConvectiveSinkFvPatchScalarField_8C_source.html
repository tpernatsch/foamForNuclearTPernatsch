152 void radiativeConvectiveSinkFvPatchScalarField::updateCoeffs()
153 {
154  if (this->updated())
155  {
156  return;
157  }
158 
159  const fvMesh& mesh = patch().boundaryMesh().mesh();
160  const scalarField& Tb = *this;
161  const fvPatch& patch = this->patch();
162  const scalarField& deltaCoeffs = patch.deltaCoeffs();
163 
164  if (mesh.foundObject<volTensorField>(kappaName_))
165  {
166  const fvPatchField<tensor>& k = patch.lookupPatchField<volTensorField, tensor>
167  (
168  kappaName_
169  );
170 
171  vectorField n = (patch.Sf() / patch.magSf());
172  scalarField alphaP = ((n & k) & n)*deltaCoeffs;
173  scalarField alphaN = (H_ + 5.67e-8*emissivity_*(Tb*Tb + T0_*T0_)*(Tb+T0_));
174 
175  alpha_ = alphaP / (alphaP + alphaN);
176  }
177  else
178  {
179  const fvPatchField<scalar>& k = patch.lookupPatchField<volScalarField, scalar>
180  (
181  kappaName_
182  );
183 
184  scalarField alphaP = k*deltaCoeffs;
185  scalarField alphaN = (H_ + 5.67e-8*emissivity_*(Tb*Tb + T0_*T0_)*(Tb+T0_));
186 
187  alpha_ = alphaP / (alphaP + alphaN);
188  }
189 
190  fvPatchField<scalar>::updateCoeffs();
191 }
