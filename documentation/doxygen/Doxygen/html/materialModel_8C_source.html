87 Foam::materialModel::New
88 (
89  const fvMesh& mesh,
90  const dictionary& materialModelDict,
91  const labelList& addr
92 )
93 {
94  word materialName;
95  materialModelDict.lookup("material") >> materialName;
96 
97  auto cstrIter
98  = dictionaryConstructorTablePtr_->find(materialName);
99 
100  if (cstrIter == dictionaryConstructorTablePtr_->end())
101  {
102  FatalErrorIn("materialModel::New(const fvMesh&, const dictionary&)")
103  << "Unknown materialModel type "
104  << materialName << nl << nl
105  << "Valid types are:" << endl
106  << dictionaryConstructorTablePtr_->toc()
107  << exit(FatalError);
108  }
109 
110  Info<< " Selecting material model --> "
111  << materialName << endl;
112 
113  if (debug)
114  {
115  Info<< "Selecting materialModel type "
116  << materialName << endl;
117  }
118 
119  autoPtr<Foam::materialModel>
120  materialModelPtr(cstrIter()(mesh, materialModelDict, addr));
121 
122  return materialModelPtr;
123 }
