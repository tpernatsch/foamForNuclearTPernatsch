96 void Foam::thermalExpansionPARFUMEPyC::correct
97 (
98  symmTensorField& sf,
99  const scalarField& T,
100  const labelList& addr
101 )
102 const
103 {
104  //- Orientation parameters
105  const scalar Rr = 2.0 / (2.0 + BAF_);
106  const scalar Rt = (1.0 + BAF_) / (2.0 + BAF_);
107 
108  forAll(addr, i)
109  {
110  const label cellI = addr[i];
111 
112  // Temperature must be given in K
113  const scalar Ti = T[cellI];
114  const scalar Tref = Tref_.value();
115  //- radial thermal expansion coef
116  const scalar alpha_r = (par1-par2*Rr)*(1.0+par3*(Ti-par4)/par5)*1e-6;
117  //- tagential thermal expansion coef
118  const scalar alpha_t = (par6*pow(Rt-1.0,2.0)+1.0)*(1.0+par3*(Ti-par4)/par5)*1e-6;
119 
120  //-Thermal strain in spherical coordinate. Radial thermal strain is firstly given
121  symmTensor nominalValue_sphere = alpha_r*(Ti-Tref)*I;
122  symmTensor nominalValue;
123 
124  //- Transverse thermal strain replace the value of raidal strain at yy and zz.
125  nominalValue_sphere.yy() = alpha_t*(Ti-Tref);
126  nominalValue_sphere.zz() = alpha_t*(Ti-Tref);
127 
128  // NOTE: avoids instabilities when trying to simulate a material at
129  // constant temperature equal to Tref
130  if
131  (
132  nominalValue_sphere.xx() < 1e-7 &&
133  nominalValue_sphere.yy() < 1e-7 &&
134  nominalValue_sphere.zz() < 1e-7
135  )
136  {
137  nominalValue_sphere *= 0;
138  }
139 
140  if (sp_ == true)
141  {
142  sf[cellI] = nominalValue_sphere*perturb;
143  }
144  else //in case of non-1D case
145  {
146  //*************************coordinate conversion***********************
147  //- obtain the coordinate of elements
148  const scalar x_coord = mesh_.C()[cellI].x();
149  const scalar y_coord = mesh_.C()[cellI].y();
150  const scalar z_coord = mesh_.C()[cellI].z();
151  //- Radius
152  const scalar R = sqrt(pow(x_coord,2)+pow(y_coord,2)+pow(z_coord,2));
153 
154  //- Calculate the angels theta and phi in sphere coordinate
155  const scalar theta = acos(z_coord/R);
156  const scalar phi = atan(y_coord/x_coord);
157  //- Transformation matrix. According to :
158  //- https://www.brown.edu/Departments/Engineering/Courses/En221/Notes/Polar_Coords/Polar_Coords.htm
159  const tensor T1(sin(theta)*cos(phi), cos(theta)*cos(phi), -sin(phi),
160  sin(theta)*sin(phi), cos(theta)*sin(phi), cos(phi),
161  cos(theta), -sin(theta), 0.0);
162  const tensor T2(sin(theta)*cos(phi), sin(theta)*sin(phi), cos(theta),
163  cos(theta)*cos(phi), cos(theta)*sin(phi), -sin(theta),
164  -sin(phi), cos(phi), 0.0);
165 
166  //inner product
167  nominalValue = symm(T1 & nominalValue_sphere & T2);
168 
169  sf[cellI] = nominalValue*perturb;
170  }
171 
172  }
173 }
