312 void Foam::solvers::thermalHydraulicsModel::correctBaffleLessFields()
313 {
314  if (mesh_.time().controlDict().found("removeBaffles"))
315  {
316  const dictionary& removeBafflesDict = mesh_.time().controlDict().subDict("removeBaffles");
317  if (removeBafflesDict.getOrDefault<bool>(mesh_.name(), false))
318  {
319 
320  const IOdictionary couplingDict
321  (
322  IOobject
323  (
324  "multiRegionCouplingDict",
325  runTime.time().constant(),
326  runTime.db(),
327  IOobject::READ_IF_PRESENT,
328  IOobject::NO_WRITE
329  )
330  );
331 
332  //Lookup for the fields that need to be mapped FROM this mesh
333  const dictionary mappingDict(couplingDict.subDict("mappings"));
334  //Loop on every region that is not this one and look for the fields in "sourceFields"
335  const wordList regions(mappingDict.toc());
336 
337  forAll(regions, regioni)
338  {
339  if (regions[regioni] != mesh_.name()) //look for other regions
340  {
341  const dictionary regionFromDict(mappingDict.subDict(regions[regioni]));
342  const wordList regionsFrom(regionFromDict.toc());
343  forAll(regionsFrom, regionFromi)
344  {
345  if (regionsFrom[regionFromi] == mesh_.name())
346  {
347  const wordList fieldsList(regionFromDict.subDict(regionsFrom[regionFromi]).get<wordList>("sourceFields")); // list of fields to create
348  fvMesh& baffleLessMesh = const_cast<fvMesh&>(mesh_.time().lookupObject<fvMesh>(mesh_.name()+".baffleLess"));
349  forAll(fieldsList, fieldi)
350  {
351  correctBaffleLessField<scalar>(fieldsList[fieldi], baffleLessMesh);
352  correctBaffleLessField<vector>(fieldsList[fieldi], baffleLessMesh);
353  correctBaffleLessField<tensor>(fieldsList[fieldi], baffleLessMesh);
354  correctBaffleLessField<symmTensor>(fieldsList[fieldi], baffleLessMesh);
355  correctBaffleLessField<sphericalTensor>(fieldsList[fieldi], baffleLessMesh);
356 
357  }
358  }
359  }
360  }
361  }
362  }
363  }
