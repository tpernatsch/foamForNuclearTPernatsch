579 void Foam::powerModels::nuclearFuelFMU::correctHeatFluxInputsFromFMUs
580 (
581  const volScalarField& HTSum, // == SUM_j [htc_j*T_j*frac_j]
582  const volScalarField& HSum, // == SUM_j [htc_j*frac_j]
583  label regioni
584 )
585 {
586  // Communicating with the FMU, extract default values stored in
587  // commDataLayer
588  const Time& runTime = this->db().time();
589  commDataLayer& data = commDataLayer::New(runTime);
590 
591  // Volume of the cells
592  const scalarList& V(mesh_.V());
593 
594  // Get dict in the cellZone power model
595  word region(this->toc()[regioni]);
596  const labelList& regionCells(structure_.cellLists()[region]);
597  const dictionary& dict(this->subDict(region));
598 
599  const bool isHeatFluxInput(isHeatFluxInput_[regioni]);
600  const bool isFuelTemperatureFieldFromFMU(isFuelTemperatureFieldFromFMU_[regioni]);
601  const word radialBasisFunctionMethod(radialBasisFunctionMethod_[regioni]);
602 
603  const scalarField axialLoc(axialLoc_[regioni]);
604 
605  // Get field name for heat flux reconstruction
606  const wordList fieldNameFromFMU(dict.get<wordList>
607  (
608  isHeatFluxInput ? "heatFluxNameFromFMU" : "rhoCpdTdtNameFromFMU"
609  ));
610 
611  // Get field name for the fuel pin temperature (how to generalize it for
612  // pebble bed?)
613  wordList TFuelFieldNameFromFMU(0);
614  wordList TCladFieldNameFromFMU(0);
615  if (isFuelTemperatureFieldFromFMU)
616  {
617  TFuelFieldNameFromFMU = dict.get<wordList>("TFuelNameFromFMU");
618  TCladFieldNameFromFMU = dict.get<wordList>("TCladNameFromFMU");
619  }
620 
621  // List of positions and values for the interpolation algorithm
622  const label nFMIports(fieldNameFromFMU.size());
623  const label nAxialLoc(axialLoc.size());
624  scalarList xPosList(nFMIports * nAxialLoc);
625  scalarList yPosList(nFMIports * nAxialLoc);
626  scalarList zPosList(nFMIports * nAxialLoc);
627  scalarList fieldFromFMUList(nFMIports * nAxialLoc);
628  scalarList TFuelFieldFromFMUList(nFMIports * nAxialLoc);
629  scalarList TCladFieldFromFMUList(nFMIports * nAxialLoc);
630 
631  // Extract x, y, z, heat flux (or enthalpy) into list
632  forAll(fieldNameFromFMU, nameI)
633  {
634  const scalar& xPos(xPos_[regioni][nameI]);
635  const scalar& yPos(yPos_[regioni][nameI]);
636 
637  // x, y, z positions
638  forAll(axialLoc, locI)
639  {
640  xPosList[nAxialLoc * nameI + locI] = xPos;
641  yPosList[nAxialLoc * nameI + locI] = yPos;
642  zPosList[nAxialLoc * nameI + locI] = axialLoc[locI];
643  }
644 
645  // Extract heat flux or enthalpy
646  word& fieldFromFMU = data.getObj<word>
647  (
648  fieldNameFromFMU[nameI],
649  commDataLayer::causality::in
650  );
651  word& TFuelFieldFromFMU = data.getObj<word>
652  (
653  isFuelTemperatureFieldFromFMU
654  ? TFuelFieldNameFromFMU[nameI]
655  : fieldNameFromFMU[nameI],
656  commDataLayer::causality::in
657  );
658  word& TCladFieldFromFMU = data.getObj<word>
659  (
660  isFuelTemperatureFieldFromFMU
661  ? TCladFieldNameFromFMU[nameI]
662  : fieldNameFromFMU[nameI],
663  commDataLayer::causality::in
664  );
665 
666  // Split string to list of scalar
667  word value;
668  scalarList fieldFromFMUListTemp(0);
669  scalarList TFuelFieldFromFMUListTemp(0);
670  scalarList TCladFieldFromFMUListTemp(0);
671 
672  std::stringstream ss(fieldFromFMU);
673  while (getline(ss, value, ' '))
674  {
675  fieldFromFMUListTemp.append(std::stod(value));
676  }
677 
678  if (isFuelTemperatureFieldFromFMU)
679  {
680  std::stringstream ssTFuel(TFuelFieldFromFMU);
681  while (getline(ssTFuel, value, ' '))
682  {
683  TFuelFieldFromFMUListTemp.append(std::stod(value));
684  }
