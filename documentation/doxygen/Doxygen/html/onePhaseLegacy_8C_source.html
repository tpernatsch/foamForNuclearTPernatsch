63 Foam::thermalHydraulicsModels::onePhaseLegacy::onePhaseLegacy
64 (
65  Time& time,
66  fvMesh& mesh,
67  customPimpleControl& pimple,
68  fv::options& fvOptions
69 )
70 :
71  thermalHydraulicsModel
72  (
73  time,
74  mesh,
75  pimple,
76  fvOptions
77  ),
78  //- The structure model needs to be created before the fluids because
79  // the turbulence models created by the fluids might require a reference
80  // to the structure (obtained via objectRegistry lookup in the specific
81  // turbulence model class)
82  structure_
83  (
84  this->subDict("structureProperties"),
85  mesh,
86  this->powerDensityNeutronics_
87  ),
88  fluid_
89  (
90  (this->found("fluidProperties"))
91  ? this->subDict("fluidProperties") : *this,
92  mesh,
93  word(""), //- This is the phase name, setting it to "" signals a
94  // onePhase solver to the rest of the FFS library
95  this->powerDensityNeutronicsToLiquid_,
96  false //- No need to read or write the fluid phaseFraction in
97  // onePhase, it is tied to the structure phaseFraction
98  ),
99  FSPair_(fluid_, structure_, *this),
100  fixedRho_(fluid_.thermo().rho()),
101  rhok_
102  (
103  volScalarField
104  (
105  IOobject
106  (
107  "rhok",
108  mesh_.time().timeName(),
109  mesh_,
110  IOobject::READ_IF_PRESENT,
111  IOobject::NO_WRITE
112  ),
113  mesh_,
114  dimensionedScalar("", dimless, 1.0),
115  zeroGradientFvPatchScalarField::typeName
116  )
117  )
118 {
119  fixedRho_.correctBoundaryConditions();
120  rhok_.correctBoundaryConditions();
121 
122  //- checking if the incompressible treatment is activated. In that case
123  //- the solver becomes a porous version of buoyantBoussinesqPIMPLFoam.
124  //- If this is activated, the only equation of state accepted is
125  //- rhoConst.
126  dictionary phaseDict = fluid_.dict();
127  incompressibleTreatment_ = bool
128  (
129  phaseDict.lookup
130  (
131  "incompressibleBuoyantBoussinesqTreatment"
132  )
133  );
134 
135  //- Checking if heRhoThermo has the correct equationOfState for the
136  //- current solver.
137  dictionary thermoDict = fluid_.thermo().subDict("thermoType");
138  if(incompressibleTreatment_
139  and word(thermoDict.lookup("equationOfState")) != "rhoConst")
140  {
141  Foam::error e("The equation of state is not rhoConst (constant density)! "
142  "If 'incompressibleBuoyantBoussinesqTreatment' is activated,"
143  "one is only allowed to use rhoConst!");
144  e.exit(100);
145  }
146 
147  //- Reading reference temperature and thermal expansion coefficient
148  //- for the Boussinesq approximation
149  dictionary eosDict = fluid_.thermo()
150  .subDict("mixture")
151  .subDict("equationOfState");
152 
153  beta_.reset
154  (
155  new dimensionedScalar
156  (
157  "beta",
158  pow(dimTemperature,-1),
159  eosDict.lookupOrDefault<scalar>("beta", 0.0)
160  )
161  );
162 
163  Tref_.reset
164  (
165  new dimensionedScalar
166  (
167  "Tref",
168  dimTemperature,
169  eosDict.lookupOrDefault<scalar>("T0", 0.0)
170  )
171  );
172 
173  //- Create turbulence model
174  fluid_.constructTurbulenceModel();
175 
176  //- Set phase fraction fields (constant in time), structure has priority
177  fluid_.volScalarField::operator=(1.0-structure_);
178 
179  //- The normalized field is non-trivial (i.e. different than 1) only in the
180  // twoPhase solver. However, it is used by some models in the shared
181  // thermal-hydraulics library, so it should be set nonetheless! The most
182  // important quantity that relies on this is the Reynolds computed by
183  // the FSPair object
184  fluid_.normalized() = fluid_/(1.0-structure_);
185 
186  //- Calculating rhok value for boussinesq approximation if incompressible flow.
187  //- also, update thermo.tho() to make sure the neutronics solver has access to
188  //- the density feedback.
189  if(incompressibleTreatment_)
190  {
191  rhok_ = 1.0 - beta_()*(fluid_.thermo().T() - Tref_());
192  rhok_.correctBoundaryConditions();
193  fluid_.thermo().rho() = fixedRho_*rhok_;
194  fluid_.thermo().rho().correctBoundaryConditions();
195  }
196 
197  //- Set fluid characteristic dimension to structure hydraulic diameter.
198  // This is handled by the fluidGeometry class in the twoPhase solver
199  // (as the fluid characteristic dimension will depend on the regime)
200  // so here I need to do it manually. Since I assume that the structure is
201  // immutable, this is done only once
202  fluid_.Dh() = structure_.Dh();
203 
204  //- Initialize fluid-intensive fluxes (i.e. that depend on the phase
205  // fraction, namely alphaPhi and alphaRhoPhi, which are the REAL
206  // volumetric flux in m3/s and the REAL mass flux in kg/s. By REAL I mean
207  // not superficial). This is done after the phaseFraction normalization
208  // step to ensure consistency. This step has an effect ONLY IF the
209  // alphaPhi, alphaRhoPhi fields were NOT found on disk
210  fluid_.initAlphaPhis();
211 
212  //- The total volumetric flux is the REAL fluid volumetric flux. Might
213  // as well remove phi_ entirely as a field, I know... Maybe in the future
214  phi_ = fluid_.alphaPhi();
215 
216  //- Compute initialFluidMass
217  initialFluidMass_ = fvc::domainIntegrate(fluid_.thermo().rho()*fluid_);
218 
219  //- Setting up the initial Darcy velocity and flux
220  UDarcy_.reset
221  (
222  new volVectorField
223  (
224  IOobject
225  (
226  "UDarcy",
227  mesh_.time().timeName(),
228  mesh_,
229  IOobject::NO_READ,
230  IOobject::AUTO_WRITE
231  ),
232  fluid_.U()*fluid_
233  )
234  );
235 
236  //- Making sure that the division of boundary conditions does not result in
237  //- a "calculated" boundary field. So we set the same boundary conditions as
238  //- the ones given for the real velocity. Exceptions are wedge and empty BCs,
239  //- because those are not touched.
240 
241  forAll(fluid_.U().boundaryField(), bcInd)
242  {
243  if(fluid_.U().boundaryField()[bcInd].type() == "empty" or
244  fluid_.U().boundaryField()[bcInd].type() == "wedge")
245  {
246  Info << "Skipping boundary type assignment for UDarcy"
247  << "because it is empty/wedge." << endl;
248  continue;
249  }
250 
251  word bcType = fluid_.U().boundaryField()[bcInd].type();
252 
253  tmp<fvPatchField<vector>> originalPatch(UDarcy_().boundaryField()[bcInd]);
254 
255  UDarcy_().boundaryFieldRef().set
256  (
257  bcInd,
258  fvPatchField<vector>::New
259  (
260  bcType,
261  UDarcy_().mesh().boundary()[bcInd],
262  UDarcy_()
263  )
264  );
265 
266  forAll(UDarcy_().boundaryFieldRef()[bcInd], faceI)
267  {
268  UDarcy_().boundaryFieldRef()[bcInd][faceI] = originalPatch()[faceI];
269  }
270  }
271  UDarcy_().write();
272 
273  phiDarcy_.reset
274  (
275  new surfaceScalarField
276  (
277  IOobject
278  (
279  "phiDarcy",
280  mesh_.time().timeName(),
281  mesh_,
282  IOobject::NO_READ,
283  IOobject::NO_WRITE
284  ),
285  fvc::interpolate(fixedRho_)*fvc::interpolate(fluid_)*fluid_.phi()
286  )
287  );
288 
289  //- Initialize continuity errors. It's important to do it here or, if not
290  // solving for fluidMechanics, these would never get corrected
291  correctContErr();
292 
293  Info << endl;
294 }

345 {
346  CoNum_ = 0.0;

348 
349  scalarField sumPhi
350  (
351  fvc::surfaceSum

356  )

358  );
359 
360  CoNum_ = 0.5*gMax(sumPhi/mesh_.V().field())*runTime_.deltaTValue();

369 void Foam::thermalHydraulicsModels::onePhaseLegacy::correctContErr()
370 {
371  if(incompressibleTreatment_)
372  {
373  fluid_.contErr() =
374  (
375  fvc::div(phiDarcy_())
376  );
377  fluid_.contErr().correctBoundaryConditions();
378  }
379  else
380  {
381  volScalarField& rho(fluid_.thermo().rho());
382  fluid_.contErr() =
383  (
384  fvc::ddt(fluid_, rho)
385  + fvc::div(phiDarcy_())
386  - (fvOptions_(fluid_, fixedRho_) & rho)
387  );
388  fluid_.contErr().correctBoundaryConditions();
389  }
