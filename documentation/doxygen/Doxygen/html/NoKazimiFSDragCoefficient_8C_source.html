70  (

73  objReg
74  )
75 {
76  scalar Dp(dict.get<scalar>("pinDiameter"));
77  scalar Dw(dict.get<scalar>("wireDiameter"));
78  scalar H(dict.get<scalar>("wireLeadLen"));
79  scalar Pt(Dp+1.0444*Dw);
80  A_ = (32/sqrt(100.0*H))*pow(Pt/Dp, 1.5);
81  B_ = 1.034/pow(Pt/Dp, 0.124);
82  C_ = 29.7*pow(Pt/Dp, 6.9)/pow(H/(Dp+Dw), 2.239);
83 }
