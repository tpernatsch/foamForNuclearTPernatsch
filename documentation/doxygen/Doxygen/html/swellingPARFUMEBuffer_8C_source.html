96  //- Correspond the Polynomial coefficients data with temperature.Values at

100  //- Radial components
101  table_a1_r_.set(new scalarFieldInterpolateTable(tempValues_, a1_r_, Method_, obMethod_ex_));
102  table_a2_r_.set(new scalarFieldInterpolateTable(tempValues_, a2_r_, Method_, obMethod_ex_));
103  table_a3_r_.set(new scalarFieldInterpolateTable(tempValues_, a3_r_, Method_, obMethod_ex_));

107  600 | (-1.24080 -1.10640 -0.94333 -0.78045 -0.15714 0.40265)

113  //- Swelling strain at current density of 1100 C and fast fluence of 3.7x1025 n/m2 throng the Interpolation

126 void Foam::swellingPARFUMEBuffer::correct
127 (
128  const scalarField& T,
129  const labelList& addr
130 )
131 {
132  const fvMesh& mesh_ = epsilonSwelling_.mesh();
133 
134  if(fastFluence_ == nullptr or rho_==nullptr)
135  {
136  fastFluence_ = &mesh_.lookupObject<volScalarField>(fastFluenceName_);
137  fastFlux_ = &mesh_.lookupObject<volScalarField>(fastFluxName_);
138  rho_ = &mesh_.lookupObject<volScalarField>(densityName_);
139  }
140  const symmTensorField& swellingIOld = epsilonSwelling_.oldTime().internalField();
141  symmTensorField& swellingI = epsilonSwelling_.ref();
142 
143  const volScalarField& phii = *fastFluence_;
144  const volScalarField& rhooo = *rho_;
145  const volScalarField& fluxx = *fastFlux_;
146 
147  //Only read the inital density
148  static scalarField rhoo = rhooo;
149 
150  //- Reference to the bulit table
151  const scalarFieldInterpolateTable& table_a1_r_ref(table_a1_r_());
152  const scalarFieldInterpolateTable& table_a2_r_ref(table_a2_r_());
153  const scalarFieldInterpolateTable& table_a3_r_ref(table_a3_r_());
154  const scalarFieldInterpolateTable& table_a4_r_ref(table_a4_r_());
155 
156  const scalarInterpolateTable& table_epsilon_iso_ref(table_epsilon_iso_());
157 
158  forAll(addr, addrI)
159  {
160  const label cellI = addr[addrI];
161  //- The model requires 1e25 n/m^2, it is bounded at 3.96e25 n/m^2
162  const scalar phi = min(phii[cellI]*1e4/1e25*fcf_, 3.96);
163 
164  //- Fast flux in unit 1e25n/m²/s
165  const scalar flux = fluxx[cellI]*1e4/1e25*fcf_;
166 
167  //- Density in g/cm3
168  const scalar rho = rhoo[cellI]/1e3;
169  //- Temperature in Celsius
170  const scalar Ti = T[cellI] - 273.15;
171  //- Do the interpolation for the temperature
172  scalarField a1_r_T(table_a1_r_ref(Ti));
173  scalarField a2_r_T(table_a2_r_ref(Ti));
174  scalarField a3_r_T(table_a3_r_ref(Ti));
175  scalarField a4_r_T(table_a4_r_ref(Ti));
176 
177  //- Correspond the scalarField above with the BAF data.
178  //- BAFvalue outside validity range are clamped at the bounds.
179  scalarInterpolateTable a1_r_T_BAF(BAFValues_r_, a1_r_T, Method_, obMethod_fix_);
180  scalarInterpolateTable a2_r_T_BAF(BAFValues_r_, a2_r_T, Method_, obMethod_fix_);
181  scalarInterpolateTable a3_r_T_BAF(BAFValues_r_, a3_r_T, Method_, obMethod_fix_);
182  scalarInterpolateTable a4_r_T_BAF(BAFValues_r_, a4_r_T, Method_, obMethod_fix_);
183 
184  //- Interpolation for "a"
185  scalar a1_iso, a2_iso, a3_iso, a4_iso;
186  a1_iso = a1_r_T_BAF(1.0);
187  a2_iso = a2_r_T_BAF(1.0);
188  a3_iso = a3_r_T_BAF(1.0);
189  a4_iso = a4_r_T_BAF(1.0);
190 
191  //- At density 1.96 g/cm3
192  const scalar epsilon_ref = a1_iso+2.0*a2_iso*phi+3.0*a3_iso*pow(phi,2)+4.0*a4_iso*pow(phi,3);
193 
194 
195  //- Isotropic scaling factor μ.
196  scalar miu = table_epsilon_iso_ref(rho) / table_epsilon_iso_ref(1.96);
197 
198  scalar nominalValue(epsilon_ref*miu*flux*mesh_.time().deltaT().value());
199  //- Here the strain is in unit of %
200  swellingI[cellI] = swellingIOld[cellI] + (nominalValue/1e2*perturb)*I;
201  }
202 
203  // Check which F and delta should be used
204  const dimensionedScalar& F =
205  isInUserParameters(this->group(), "F_epsilonSwelling") ?
206  this->F_epsilonSwelling() :
207  swellingModel::F_epsilonSwelling();
208 
209  const dimensionedScalar& delta =
210  isInUserParameters(this->group(), "delta_epsilonSwelling") ?
211  this->delta_epsilonSwelling() :
212  swellingModel::delta_epsilonSwelling();
213 
214  // Perturb epsilon swelling for sensitivity analysis
215  applyParameters(epsilonSwelling_.ref(), addr, F, delta);
216 
217 }
