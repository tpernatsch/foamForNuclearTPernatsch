98  if (p_ == nullptr)
99  {
100  if ( mesh_.foundObject<volScalarField>("porosity") )
101  {
102  p_ = &mesh_.lookupObject<volScalarField>("porosity");
103  }
104  }
105 
106  //- Correct conductivity for every cell in addr
107  forAll(addr, i)
108  {
109  const label cellI = addr[i];
110 
111  // Temperature
112  const scalar Ti = T[cellI];
113 
114  // Porosity
115  if (p_ != nullptr)
116  {
117  Pi = p_->internalField()[cellI];
118  }
119 
120  // Maximum limit for porosity in this correlation, otherwise k->0 !
121  if( !mesh_.foundObject<fvMesh>("referenceMesh") )
122  {
123  PiLim = min(Pi, 0.95);
124  }
125 
126  // Compute correction factor f(p) - Noascone
127  const scalar fp = (1-PiLim)/(1+0.5*PiLim);
128 
129  // Compute denominator
130  const scalar den = 1.595e-2 + 2.713*x_ + 3.583e-1*Am_
131  + 6.317e-2 *Np_ + (-2.625*x_ + 2.493)*1e-4*Ti;
132 
133  // Compute K(T)
134  const scalar k = (1/(den)+1.541e11/pow(Ti,2.5)*exp(-1.522e4/Ti));
135 
136  // Correct k - Noascone method
