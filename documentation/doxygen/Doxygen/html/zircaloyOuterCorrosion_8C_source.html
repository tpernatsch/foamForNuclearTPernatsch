74 void Foam::zircaloyOuterCorrosion::correct
75 (
76  surfaceScalarField& oxideThickness,
77  surfaceScalarField& DOxideThickness,
78  const label& patchID
79 )
80 {
81  // Legend for the following lines:
82  // - Ti cell-center temperature for cells attached to patch
83  // - Tf face-center temperature, interface between metal and oxide
84  // - To outer face-center temperature, oxide surface
85 
86  const fvPatch& patch(mesh_.boundary()[patchID]);
87 
88  // Current and oldTime oxide layer thickness
89  scalarField& S(oxideThickness.boundaryFieldRef()[patchID]);
90  const scalarField& Sold(oxideThickness.oldTime().boundaryField()[patchID]);
91  const scalarField& Sprev(oxideThickness.prevIter().boundaryField()[patchID]);
92 
93  // Oxide layer increase
94  scalarField& DS(DOxideThickness.boundaryFieldRef()[patchID]);
95 
96  // Delta time
97  scalar deltaT = mesh_.time().deltaTValue();
98 
99  // Inverse cell to patch face distances
100  const scalarField d = patch.deltaCoeffs();
101 
102  // Temperature on the patch (= metal/oxide interface temperature)
103  const scalarField& Tf
104  = patch.lookupPatchField<volScalarField, scalar>("T");
105 
106  // Find temperatures of cells adjacents to patch
107  const volScalarField& T = mesh_.lookupObject<volScalarField>("T");
108  const labelList faceCells = patch.faceCells();
109  const scalarField Ti(T.internalField(), faceCells);
110 
111  // Fast flux on the patch
112  const scalarField& phi
113  = patch.lookupPatchField<volScalarField, scalar>("fastFlux");
114 
115  // Loop over all the faces of the boundary patch
116  forAll(patch, i)
117  {
118  // Normal operating condition model (EPRI KWU CE)
119  if ( Tf[i] < 673 )
120  {
121  // Correlation parameters
122  const scalar C1(6.3e-9);
123  const scalar C2((8.04e7+2.59*1e8*pow(7.46e-15*phi[i],0.25))*1e-6);
124  const scalar Q1R(16266);
125  const scalar Q2R(13775);
126 
127  // Transition thickness
128  scalar sTransition = 2e-6 ;
129 
130  // Compute oxide thickness
131  if( S[i] <= sTransition )
132  {
133  S[i] = pow
134  (
135  C1*exp(-Q1R/Tf[i])*deltaT/3600.0/24.0
136  + pow(Sold[i], 3.0)
137  , 1.0/3.0
138  );
139  }
140  // If S is crossing sTransition in this timeStep
141  else if( Sold[i] < sTransition )
142  {
143  // Calculate S continuing with pre-transition law
144  scalar S_preTransition = pow
145  (
146  C1*exp(-Q1R/Tf[i])*deltaT/3600.0/24.0
147  + pow(Sold[i], 3.0)
148  , 1.0/3.0
149  );
150 
151  // Calculate the deltaT up to sTransition
152  scalar deltaTpreTransition = deltaT*
153  (sTransition - Sold[i])/(S_preTransition - Sold[i]);
154 
155  // Calculate the deltaT post sTransition
156  scalar deltaTpostTransition = deltaT - deltaTpreTransition;
157 
158  S[i] = sTransition
159  + C2*exp(-Q2R/Tf[i])*deltaTpostTransition/3600.0/24.0;
160  }
161  else
162  {
163  S[i] = Sold[i] + C2*exp(-Q2R/Tf[i])*deltaT/3600.0/24.0;
164  }
165  }
166  // High temperature model:
167  else
168  {
169  // Correlation parameters
170  scalar As;
171  scalar QsR;
172 
173  // Leistikov correlation
174  if ( Tf[i] < 1800 )
175  {
176  As = 7.82e-6;
177  QsR = 20214;
178  }
179  // Interpolation (Procedure from G. Schanz - 2003)
180  else if (Tf[i] < 1900)
181  {
182  QsR = 1900*1800/(1900-1800)
183  * log
184  (
185  2.98e-3*Foam::exp(-28420.0/1900.0)
186  /
187  7.82e-6*Foam::exp(-20214.0/1800.0)
188  );
189 
190  As = 7.82e-6*exp(QsR/1900.0);
191  }
192  // Prater-Courtright correlation
193  else
194  {
195  As = 2.98e-3;
196  QsR = 28420;
197  }
198 
199  S[i] = sqrt( As*exp(-QsR/Tf[i])*deltaT + pow(Sold[i], 2.0) );
200  }
201 
202  DS[i] = S[i] - Sold[i];
203  }
204 
205  updateMesh_ = false;
206 
207  // Check absolute change and decide whether to update the mesh or not
208  scalar oxideChange(gMax(mag(S - Sprev)));
209  if(oxideChange > 1e-10)
210  {
211  updateMesh_ = true;
212  }
213 };
