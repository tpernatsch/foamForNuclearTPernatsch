478  //- Lambda function to rotate a vector around an axis by a certain
479  // angle in radians, counter-clockwise
480  auto rotateCCWAroundAxisByAngle = []
481  (
482  vector& v,
483  const vector& axis,
484  scalar angle
485  )
486  {
487  scalar c(Foam::cos(angle));
488  scalar s(Foam::sin(angle));
489  scalar x(axis[0]);
490  scalar y(axis[1]);
491  scalar z(axis[2]);
492  tensor R
493  (
494  c+sqr(x)*(1-c), x*y*(1-c)-z*s, x*z*(1-c)+y*s,
495 
496  y*x*(1-c)+z*s, c+sqr(y)*(1-c), y*z*(1-c)-x*s,
497 
498  z*x*(1-c)-y*s, z*y*(1-c)+x*s, c+sqr(z)*(1-c)
499 
500  );
501  v = R & v;
502  };
