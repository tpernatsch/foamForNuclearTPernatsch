91 void Foam::FFDragFactor::correctField(volScalarField& Kd) const
92 {
93  const volScalarField& a1(pair_.fluid1());
94  const volScalarField& a2(pair_.fluid2());
95  forAll(pair_.mesh().cells(), i)
96  {
97  Kd[i] =
98  (a1[i]+a2[i])*(0.5/Dhd_[i])*rhoc_[i]*magUr_[i]*fdPtr_->value(i);
99  }
