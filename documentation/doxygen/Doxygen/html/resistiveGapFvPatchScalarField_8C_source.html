37 Foam::tmp<Foam::scalarField>
38 Foam::resistiveGapFvPatchScalarField::weights() const
39 {
40  if (coupled_)
41  {
42  const fvPatch& patch = this->patch();
43  const fvPatch& nbrPatch = regionCoupledPatch_.neighbFvPatch();
44 
45  const scalarField deltas
46  (
47  patch.nf() & patch.delta()
48  );
49 
50  const scalarField nbrKappa
51  (
52  regionCoupledPatch_.regionCoupledPatch().interpolate
53  (
54  nbrPatch.lookupPatchField<volScalarField, scalar>(kappaName_)
55  )
56  );
57 
58  const scalarField nbrDeltas
59  (
60  regionCoupledPatch_.regionCoupledPatch().interpolate
61  (
62  nbrPatch.nf() & nbrPatch.delta()
63  )
64  );
65 
66  const scalarField alphaP(kappa()/deltas);
67 
68  const scalarField alphaN(nbrKappa/max(nbrDeltas, SMALL));
69 
70  scalarField alphaEff = 1/(1/alphaP + 1/max(alphaN, SMALL) + 1/max(alpha_gap_, SMALL));
71 
72  return 1 - alphaEff / alphaP;
73  }
74  else
75  {
76  return patch().weights();
77  }
78 }

184 void Foam::resistiveGapFvPatchScalarField::updateCoeffs()
185 {
186  if (updated())
187  {
188  return;
189  }
190 
191  if (!regionCoupledPatch_.owner())
192  {
193  // Update the heat transfer coefficient from the owner patchField
194  const fvPatch& nbrPatch = regionCoupledPatch_.neighbFvPatch();
195 
196  alpha_gap_ = regionCoupledPatch_.regionCoupledPatch().interpolate
197  (
198  refCast<const resistiveGapFvPatchScalarField>
199  (
200  nbrPatch.lookupPatchField<volScalarField, scalar>
201  (
202  internalField().name()
203  )
204  ).alpha()
205  );
206  }
207 
208  const_cast<fvPatchScalarField&>
209  (
210  patch().lookupPatchField<volScalarField, scalar>("hGap")
211  ) = alpha_gap_;
212 
213  // The if statement should be performed for the same patch (owner or nbr)
214  // where the hGap is directly updated (not mapped from the other side)
215  if (regionCoupledPatch_.owner())
216  {
217  if( gMax(alpha_gap_) <= 0 )
218  {
219  FatalErrorInFunction
220  << patch().name() << " has no positive gap conductance."
221  << exit(FatalError);
222  }
223  }
224 
225  temperatureCoupledFvPatchScalarField::updateCoeffs();
226 }
