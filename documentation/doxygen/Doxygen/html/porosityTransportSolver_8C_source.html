95 void Foam::porosityTransportSolver::updateRestructuringFields()
96 {
97  // Update central void field
98  centralVoid_ *= 0.0;
99  forAll(porosity_, i)
100  {
101  if(porosity_[i] >= voidThreshold_)
102  {
103  centralVoid_[i] = 1;
104  }
105  }
106 
107  // Update restructured region field
108  restructuredRegion_ *= 0.0;
109 
110  // Compute porosity gradient
111  volScalarField gradP = mag(fvc::grad(porosity_));
112 
113  forAll(gradP, i)
114  {
115  const scalar porosityChange =
116  mag(porosity_[i] - asFabricatedP_[i])/max(asFabricatedP_[i],VSMALL);
117  if
118  (
119  porosityChange > 0.01
120  or
121  gradP[i] > 1
122  )
123  {
124  restructuredRegion_[i] = 1.0;
125  }
126  }
127 }

236 void Foam::porosityTransportSolver::correct()
237 {
238 
239  const dictionary& stressControl =
240  mesh_.solutionDict().subDict("stressAnalysis");
241 
242  const int nCorr =
243  stressControl.lookupOrDefault<int>("nCorrectors", 1);
244 
245  int nInnerIter = 0;
246 
247  bool convergedInner(false);
248 
249  do
250  {
251  // Store prev iter values for rel residual
252  porosity_.storePrevIter();
253 
254  // Flux for advective term
255  surfaceScalarField flux = fvc::interpolate( poreVelocity_ ) & mesh_.Sf();
256 
257  // Equation for pore transport (only advection contribution)
258  fvScalarMatrix poreEqn
259  (
260  fvm::ddt(porosity_)
261  + fvm::div(flux, porosity_, "div(flux,P)")
262  );
263 
264  // Add diffusive term to the equation
265  if( diffusiveTerm_ )
266  {
267  poreEqn -= fvm::laplacian(d_, porosity_);
268  }
269 
270  // Multiplies the porosity in div term by (1-p)
271  if ( boundedAdvectionTerm_ )
272  {
273  poreEqn -= fvc::div(flux, pow(porosity_,2.0));
274  }
275 
276  // Relax equation
277  poreEqn.relax();
278 
279  // Compute residual
280  residual_ = poreEqn.solve().max().initialResidual();
281 
282  // Store initial residual if in first inner iteration
283  if(nInnerIter==0)
284  {
285  initialResidual_ = residual_;
286  }
287 
288  // Relax fields
289  porosity_.relax();
290 
291  // Calculate a different residual based on the relative change
292  scalar denom =
293  gMax(mag(porosity_.primitiveField() - porosity_.oldTime().primitiveField()));
294 
295  scalar num =
296  gMax(mag(porosity_.primitiveField() - porosity_.prevIter().internalField()));
297 
298  if (denom < SMALL)
299  {
300  denom = max(gMax(mag(porosity_.primitiveField())), SMALL);
301  }
302 
303  relResidual_ = num/denom;
304 
305  if(denom < 1e-5)
306  {
307  relResidual_ = VSMALL;
308  }
309 
310  Info << "relResidualPorosity " << relResidual_ << endl;
311 
312  convergedInner = converged(porosity_.name());
313 
314  } while
315  (
316  not(convergedInner)
317  && ++nInnerIter < nCorr
318  );
319 
320  // Update centralVoid and restructuredRegion fields
321  updateRestructuringFields();
322 
323  Info << endl;
324 }

326 Foam::scalar Foam::porosityTransportSolver::nextDeltaT()
327 {
328  // Flux for Courant number
329  const surfaceScalarField phi =
330  fvc::interpolate((poreVelocity_)*porosity_) & mesh_.Sf();
331 
332  // Define tmp field of Courant numbers
333  tmp<volScalarField> tCourantNo
334  (
335  volScalarField::New
336  (
337  "CourantNo",
338  mesh_,
339  dimensionedScalar(dimless, 0),
340  "zeroGradient"
341  )
342  );
343 
344  // Compute Courant number (according to OpenFOAM method)
345  tCourantNo->ref() = (0.5*mesh_.time().deltaT())*fvc::surfaceSum(mag(phi))()()/mesh_.V();
346  tCourantNo->correctBoundaryConditions();
347 
348  // Compute maximum Courant number
349  const scalar maxCo(gMax(tCourantNo->field()));
350 
351  scalar currentDeltaT = mesh_.time().deltaT().value();
352 
353  // Compute maximum allowed deltaT multiplier
354  scalar maxDtMultiplier = maxCoAllowed_/max(maxCo, SMALL);
355 
356  // Compute maximum allowed next deltaT
357  scalar nextDeltaT( currentDeltaT * maxDtMultiplier );
358 
359  Info << "Maximum deltaT calculated by porosityTransportSolver: "
360  << mesh_.time().timeToUserTime(nextDeltaT)
361  << " with a maximum Courant No of " << maxCo
362  << endl;
363 
364  return min(nextDeltaT, 1e10);
365 }
