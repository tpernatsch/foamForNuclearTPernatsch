121 scalar polyharmonicSplineFunction
122 (
123  const scalar rSquare,
124  const label mode
125 )
126 {
127  switch (mode) {
128  case 1:
129  return(sqrt(rSquare));
130  case 2:
131  return(rSquare * 0.5 * log(rSquare));
132  case 3:
133  return(sqrt(rSquare) * rSquare);
134  case 4:
135  return(sqr(rSquare) * 0.5 * log(rSquare));
136  default:
137  Info<< "Polyharmonic spline mode " << mode << " not in range [1; 4], return 0"
138  << endl;
139  break;
140  }
141  return(0);
142 }

212 scalarList solvePolyharmonicSpline
213 (
214  const List<scalarList>& xList,
215  const scalarList& vList,
216  SquareMatrix<scalar>& invRBFmatrix,
217  const label mode
218 )
219 {
220  const label nx(xList.first().size());
221  const label nParameters(xList.size());
222 
223  scalarList w(nx+nParameters+1, 0.0);
224 
225  if (invRBFmatrix.m() != w.size())
226  {
227  invRBFmatrix.resize(w.size());
228 
229  SquareMatrix<scalar> A(w.size(), 0.0);
230 
231  forAll(xList.first(), dataI)
232  {
233  forAll(xList.first(), dataJ)
234  {
235  if (dataI != dataJ)
236  {
237  scalar rSquare(0);
238  forAll(xList, paramI)
239  {
240  rSquare += sqr(xList[paramI][dataI] - xList[paramI][dataJ]);
241  }
242  A[dataI][dataJ] = polyharmonicSplineFunction(rSquare, mode);
243  }
244  else
245  {
246  A[dataI][dataJ] = 0.0;
247  }
248  }
249  // Polynomial correction
250  A[dataI][nx] = 1.0;
251  A[nx][dataI] = 1.0;
252  forAll(xList, paramI)
253  {
254  A[dataI][nx+paramI+1] = xList[paramI][dataI];
255  A[nx+paramI+1][dataI] = xList[paramI][dataI];
256  }
257  }
258 
259  // Inverse the matrix once and store it for later iterations
260  LUscalarMatrix Atemp(A);
261 
262  Atemp.inv(invRBFmatrix);
263 
264  // solve(w, A, vListTemp);
265  }
266 
267  scalarList vListTemp(vList);
268  forAll(xList, paramI)
269  {
270  vListTemp.append(0);
271  }
272 
273  w = invRBFmatrix * vListTemp;
274 
275  return(w);
276 }

305 scalar polyharmonicSpline
306 (
307  const scalarList& w,
308  const List<scalarList>& xList,
309  const scalarList& xInput,
310  const label mode
311 )
312 {
313  const label nx(xList.first().size());
314 
315  // Polynomial Correction
316  scalar res(w[nx]);
317  forAll(xInput, paramI)
318  {
319  res += w[nx+paramI+1] * xInput[paramI];
320  }
321 
322  scalar rSquare(0);
323  forAll(xList.first(), dataI)
324  {
325  rSquare = 0;
326  forAll(xInput, paramI)
327  {
328  rSquare += sqr(xInput[paramI] - xList[paramI][dataI]);
329  }
330  if (rSquare > 0)
331  {
332  res += w[dataI] * polyharmonicSplineFunction(rSquare, mode);
333  }
334  }
335  return(res);
336 }

339 scalarList solvePolyharmonicSplineDerivative
340 (
341  const scalarList xList,
342  const scalarList yList,
343  const scalarList zList,
344  const scalarList vList,
345  const scalarList dvxList,
346  const scalarList dvyList
347 )
348 {
349  const label nx(xList.size());
350  SquareMatrix<scalar> A(3*nx+4, 0.0);
351  scalarList w(3*nx+4, 0.0);
352 
353  forAll(xList, i)
354  {
355  const scalar xi(xList[i]);
356  const scalar yi(yList[i]);
357  const scalar zi(zList[i]);
358  forAll(xList, j)
359  {
360  const scalar xj(xList[j]);
361  const scalar yj(yList[j]);
362  const scalar dx(xi-xj);
363  const scalar dy(yi-yj);
364  const scalar r(sqrt(
365  sqr(dx) + sqr(dy) + sqr(zi-zList[j])
366  ));
367  const scalar rd(sqrt(sqr(dx) + sqr(dy)));
368  const scalar phir(sqr(r) * log(r));
369  if (r > 0)
370  {
371 
372  // wi with fi (A)
373  A[i][j] = phir;
374  // vix with fi (C)
375  A[i][nx+j] = dx * phir;
376  // viy with fi (H)
377  A[i][2*nx+j] = dy * phir;
378  }
379  if (rd > 0)
380  {
381  const scalar dphix(2.0 * dx * log(rd) + dx);
382  const scalar dphiy(2.0 * dy * log(rd) + dy);
383  // wi with df/dx (D)
384  A[nx+i][j] = dphix;
385  // vix with df/dx (E)
386  A[nx+i][nx+j] = phir + dx * dphix;
387  // viy with df/dx (F)
388  A[nx+i][2*nx+j] = dy * dphix;
389 
390  // wi with df/dy (I)
391  A[2*nx+i][j] = dphiy;
392  // vix with df/dy (K)
393  A[2*nx+i][nx+j] = dx * dphiy;
394  // viy with df/dy (J)
395  A[2*nx+i][2*nx+j] = phir + dy * dphiy;
396  }
397  }
398  // Polynomial correction (B)
399  A[i][3*nx] = 1;
400  A[i][3*nx+1] = xi;
401  A[i][3*nx+2] = yi;
402  A[i][3*nx+3] = zi;
403  A[3*nx][i] = 1;
404  A[3*nx+1][i] = xi;
405  A[3*nx+2][i] = yi;
406  A[3*nx+3][i] = zi;
407 
408  // Polynome for df/dx (G)
409  A[nx+i][3*nx+1] = 1;
410  // Polynome for df/dy (L)
411  A[2*nx+i][3*nx+2] = 1;
412  }
413 
414  scalarList vListTemp(vList);
415  forAll(dvxList, i)
416  {
417  vListTemp.append(dvxList[i]);
418  }
419  forAll(dvyList, i)
420  {
421  vListTemp.append(dvyList[i]);
422  }
423  for (int i = 0; i < 4; i++)
424  {
425  vListTemp.append(0);
426  }
427 
428  solve(w, A, vListTemp);
429 
430  return(w);
431 }

434 scalar polyharmonicSplineDerivative
435 (
436  const scalarList w,
437  const scalarList xList,
438  const scalarList yList,
439  const scalarList zList,
440  const scalar x,
441  const scalar y,
442  const scalar z
443 )
444 {
445  const label nx(xList.size());
446  scalar res(0);
447  scalar rSquare(0);
448  forAll(xList, i)
449  {
450  rSquare = sqr(x-xList[i])
451  + sqr(y-yList[i])
452  + sqr(z-zList[i]);
453  if (rSquare > 0)
454  {
455  res += (
456  w[i] + w[i+nx] * (x-xList[i]) + w[i+2*nx] * (y-yList[i])
457  ) * rSquare * 0.5 * log(rSquare);
458  }
459  }
460  res += w[3*nx] + w[3*nx+1]*x + w[3*nx+2]*y + w[3*nx+3]*z;
461  return(res > 0 ? res : 0);
462 }

581 void solveKriging
582 (
583  const scalarList xList,
584  const scalarList yList,
585  const scalarList zList,
586  const variogramType type,
587  const scalar a,
588  const scalar c,
589  const scalar c0,
590  SquareMatrix<scalar>& invRBFmatrix
591 )
592 {
593  const label nx(xList.size());
594 
595  if (invRBFmatrix.m() != nx+1)
596  {
597  invRBFmatrix.resize(nx+1);
598 
599  SquareMatrix<scalar> A(nx+1, 0.0);
600 
601  scalar distance(0);
602  switch (type) {
603  case spherical:
604  forAll(xList, i)
605  {
606  forAll(xList, j)
607  {
608  distance = sqrt(
609  sqr(xList[i]-xList[j]) +
610  sqr(yList[i]-yList[j]) +
611  sqr(zList[i]-zList[j])
612  ) / a;
613  if (distance <= 1) {
614  A[i][j] = c0 + c*(1.5*distance - 0.5*pow(distance, 3.0));
615  } else {
616  A[i][j] = c0 + c;
617  }
618  }
619  A[i][nx] = 1.0;
620  A[nx][i] = 1.0;
621  }
622  break;
623  case gaussian:
624  forAll(xList, i)
625  {
626  forAll(xList, j)
627  {
628  distance = sqr(xList[i]-xList[j]) +
629  sqr(yList[i]-yList[j]) +
630  sqr(zList[i]-zList[j]);
631  A[i][j] = c0 + c * (1.0 - exp(-distance / sqr(a)));
632  }
633  A[i][nx] = 1.0;
634  A[nx][i] = 1.0;
635  }
636  break;
637  case exponential:
638  forAll(xList, i)
639  {
640  forAll(xList, j)
641  {
642  distance = sqrt(
643  sqr(xList[i]-xList[j]) +
644  sqr(yList[i]-yList[j]) +
645  sqr(zList[i]-zList[j])
646  );
647  A[i][j] = c0 + c * (1.0 - exp(-distance/a));
648  }
649  A[i][nx] = 1.0;
650  A[nx][i] = 1.0;
651  }
652  break;
653  }
654 
655  LUscalarMatrix Atemp(A);
656 
657  Atemp.inv(invRBFmatrix);
658  }
659 }
