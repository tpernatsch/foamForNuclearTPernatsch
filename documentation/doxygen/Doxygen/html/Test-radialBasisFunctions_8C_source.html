76 int test_PHSRBFlin1D(const scalar threshold)
77 {
78  Info<< "Test 1D polyharmonic spline RBF with " << threshold*100 << "% threshold ... ";
79 
80  List<scalarList> posList(1);
81  scalarList xList(2);
82  xList[0] = 1.0;
83  xList[1] = 2.0;
84  posList[0] = xList;
85 
86  scalarList values(2);
87  values[0] = 10.0;
88  values[1] = 20.0;
89 
90  SquareMatrix<scalar> invRBFmatrix(0);
91 
92  scalarList weights = solvePolyharmonicSpline(posList, values, invRBFmatrix);
93 
94  //- Reference values
95  std::vector<scalar> xRef{0.1, 1.0, 2.0, 3.0};
96  std::vector<scalar> yRef{1, 10.0, 20.0, 30.0};
97 
98  //- Loop over the tests
99  int nFailed(0);
100  int nCorrect(0);
101  for (size_t i = 0; i < xRef.size(); i++)
102  {
103  scalarList x_i(1);
104  x_i[0] = xRef[i];
105  const scalar y_i(yRef[i]);
106  const scalar y_e(polyharmonicSpline(weights, posList, x_i));
107 
108  //- Relative error test
109  const scalar relErr(relativeError(y_i, y_e));
110  if (relErr > threshold)
111  {
112  Info<< endl;
113  Info<< " **** \033[33mWARNING\033[0m: Foam::polyharmonicSpline("
114  << weights << ", "
115  << posList << ", "
116  << x_i
117  << ") = " << y_e << " (exp: " << y_i
118  << ") -> rel error = " << relErr;
119  nFailed++;
120  }
121  else
122  {
123  nCorrect++;
124  }
125  }
126 
127  //- Resume all tests
128  printSummary(nFailed, nCorrect);
129 
130  return nFailed;
131 }
