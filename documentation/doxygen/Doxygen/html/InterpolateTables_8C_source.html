88 template<>
89 scalar scalarInterpolateTable::integral(scalar k) const
90 {
91  scalar res = 0;
92 
93  for (int i=1; i < xValues_.size(); i++)
94  {
95  scalar dx = pow(xValues_[i], k) - pow(xValues_[i-1], k);
96 
97  if (method_ == STEP)
98  {
99  res += data_[i-1]*dx;
100  }
101  else if (method_ == LINEAR)
102  {
103  // Trapezoidal integration
104  res += 0.5*(data_[i-1] + data_[i])*dx;
105  }
106  }
107 
108  return res;
109 }
