171 Foam::scalar Foam::heatSource::nextDeltaT()
172 {
173  // Calculate deltaT to next time marker
174  scalar deltaTtoMarker = this->nextTimeMarker()
175  - mesh_.time().value();
176 
177  // Starting deltaT for time steppin algorithm cannot be larger than
178  // deltaTtoMarker
179  scalar deltaT = min(mesh_.time().deltaT().value(), deltaTtoMarker);
180 
181  // TODO: decide if relative change must be use or rate
182  // Current limiting rate of change
183  scalar currentQ(this->averagePowerDensity());
184  scalar predictedNewQ(this->predictNextPowerDensity());
185 
186  // Calculate relative rate of change
187  scalar deltaQ = (predictedNewQ - currentQ);
188  scalar relDeltaQ = deltaQ/max(currentQ,SMALL);
189 
190  if( currentQ < SMALL)
191  {
192  // Current heat source is 0. Relative change does not work
193  deltaT = this->nextDeltaTAbsoluteChange(deltaT);
194  }
195  else
196  {
197  // Current heat source is not 0
198  // Check wheter relDeltaQ is negative (power is decreasing)
199  if (relDeltaQ < 0)
200  {
201  // Scale deltaT to allow a maximum relative decrease equal to
202  // maxRelativePowerDecrease
203  deltaT = deltaT*maxRelativePowerDecrease_ / max(-relDeltaQ, SMALL);
204 
205  Info<< "Maximum deltaT calculated by heatSource model: "
206  << mesh_.time().timeToUserTime(deltaT)
207  << " with a relative power decrease of " << relDeltaQ
208  << endl;
209  }
210  else // Power is increasing or staying constant
211  {
212  // Scale deltaT to allow a maximum relative increase equal to
213  // maxRelativePowerIncrease
214  deltaT = deltaT*maxRelativePowerIncrease_ / max(relDeltaQ, SMALL);
215 
216  Info<< "Maximum deltaT calculated by heatSource model: "
217  << mesh_.time().timeToUserTime(deltaT)
218  << " with a relative power increase of " << relDeltaQ
219  << endl;
220  }
221  }
222 
223  if (deltaT > deltaTtoMarker)
224  {
225  deltaT = deltaTtoMarker;
226 
227  Info<< "Maximum deltaT re-calculated by heatSource model: "
228  << mesh_.time().timeToUserTime(deltaT)
229  << " so that deltaT is equal to the deltaT to next time point in heat source history"
230  << endl;
231  }
232  else if (deltaTtoMarker / deltaT < 5)
233  {
234  // We are approaching the next time marker and adapt the
235  // time step to avoid large changes in time step as we approach.
236  deltaT = deltaTtoMarker / ceil(deltaTtoMarker/deltaT);
237 
238  Info<< "Maximum deltaT re-calculated by heatSource model: "
239  << mesh_.time().timeToUserTime(deltaT)
240  << " as: deltaTtoMarker < 5*deltaT , i.e. next time point is approaching in heat source history, so time-steps are decreased"
241  << endl;
242  }
243 
244  return deltaT;
245 }

248 Foam::scalar Foam::heatSource::nextDeltaTAbsoluteChange(scalar deltaT) const
249 {
250  scalar currentQ(this->averagePowerDensity());
251  scalar predictedNewQ(this->predictNextPowerDensity());
252 
253  // Calculate absolute rate of change
254  scalar deltaQ = (predictedNewQ - currentQ);
255 
256  // Set next deltaT
257  scalar nextDeltaT = deltaT*maxAbsolutePowerIncrease_/max(deltaQ, SMALL);
258 
259  Info<< "Maximum deltaT calculated by heatSource model: "
260  << mesh_.time().timeToUserTime(nextDeltaT)
261  << " with current heat source " << currentQ <<" W/m3, and an absolute change of " << deltaQ << " W/m3"
262  << endl;
263 
264  return nextDeltaT;
265 }
