180 void Foam::FSDragFactor::correctField(volTensorField& Kd) const
181 {
182  scalar minMagU
183  (
184  pair_.mesh().solutionDict().subDict("PIMPLE").lookupOrDefault<scalar>
185  (
186  "minMagU",
187  0
188  )
189  );
190  if (isotropic_)
191  {
192  forAll(cells_, i)
193  {
194  const label& celli(cells_[i]);
195  tensor& Kdi(Kd[celli]);
196  scalar value
197  (
198  0.5/Dh_[celli]*
199  (1.0-alphas_[celli])*
200  rho_[celli]*
201  max(magU_[celli], minMagU)
202  *fdPtr_->value(celli)
203  );
204  Kdi[0] = value;
205  Kdi[4] = value;
206  Kdi[8] = value;
207  }
208  }
209  else
210  {
211  forAll(cells_, i)
212  {
213  const label& celli(cells_[i]);
214  halfAlphaRhoMagUByDh_[i] =
215  0.5/Dh_[celli]*
216  (1.0-alphas_[celli])*
217  rho_[celli]*
218  max(magU_[celli], minMagU);
219  }
220  if (validX_)
221  {
222  const autoPtr<FSDragCoefficientModel>& fdXPtr
223  (
224  fdPtrs_["localX"]
225  );
226  forAll(cells_, i)
227  {
228  const label& celli(cells_[i]);
229  tensor& Kdi(Kd[celli]);
230  Kdi[0] = halfAlphaRhoMagUByDh_[i]*fdXPtr->value(celli);
231  }
232  }
233  if (validY_)
234  {
235  const autoPtr<FSDragCoefficientModel>& fdYPtr
236  (
237  fdPtrs_["localY"]
238  );
239  forAll(cells_, i)
240  {
241  const label& celli(cells_[i]);
242  tensor& Kdi(Kd[celli]);
243  Kdi[4] = halfAlphaRhoMagUByDh_[i]*fdYPtr->value(celli);
244  }
245  }
246  if (validZ_)
247  {
248  const autoPtr<FSDragCoefficientModel>& fdZPtr
249  (
250  fdPtrs_["localZ"]
251  );
252  forAll(cells_, i)
253  {
254  const label& celli(cells_[i]);
255  tensor& Kdi(Kd[celli]);
256  Kdi[8] = halfAlphaRhoMagUByDh_[i]*fdZPtr->value(celli);
257  }
258  }
259  }
260 }
