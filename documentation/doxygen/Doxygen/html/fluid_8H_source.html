113  //- Dilation term of the phase.
114  // It encompasses the "compressible" part of the continuity
115  // equation, including mass sources/sinks, if any.
116  // Example given in meta-code:
117  /*
118  ddt(rho*alpha) + div(alphaRhoPhi) = dmdt
119 
120  -> (expand the derivatives)
121 
122  rho1*ddt(alpha) + rho1*div(alphaPhi) +
123  alpha1*ddt(rho) + alphaPhi1&grad(rho) = dmdt
124 
125  -> (re-arrange)
126 
127  ddt(alpha) + div(alphaPhi)
128  =
129  (1/rho)*
130  (
131  alpha*ddt(rho) + alphaPhil&grad(rho) + dmdt
132  )
133  =
134  dgdt
135  */
136  volScalarField dgdt_;
137 
138  //- A field that represents the sign of phase change in each cell, i.e.
139  // 1 if the phase is appearning, -1 if disappearing, 0 else. This field
140  // is currently only used by the nucleateBoiling heatTransfer model
141  mutable autoPtr<scalarField> phaseChangeSignPtr_;
142 
143  //- Flow quality of the phase. While these are not intrinsic
144  // properties to the fluid itself (they depend on the properties of the
145  // other fluid too), their are placed here for programming convenience.
146  // Handled by an autoPtr as it is non-trivial only in two-phase scenarios
147  mutable autoPtr<volScalarField> flowQualityPtr_;
148 
149  //- Place-holder for the Lockhart-Martinelli parameter. Logically, just as
150  // flow quality, it should belong to FFPair. However, if I place it there
151  // I loose the ability to compute it differently cell-by-cell based on
152  // the fluid multiplier that was selected for the regime that exists in
153  // said cell. Conversely, the twoPhaseDragMultiplierModels access a phase
154  // directly, so by placing it here, I avoid a lot of issues. I generally
155  // favour coding simplicity over philosophical consistency.
156  // Handled by an autoPtr as it is useless for the onePhase solver
157  mutable autoPtr<volScalarField> XLMPtr_;
