197 Foam::scalar Foam::mazars2015_RIA::triaxialFactor
198 (
199  const symmTensor& epsilon,
200  const scalar& mu,
201  const scalar& lambda,
202  const scalar& tr
203 )
204 {
205  tensor eigenVectors = tensor::zero;
206  vector eigenVal = vector::zero;
207 
208  // Compute eigen values of the strain tensor
209  eig3().eigen_decomposition(epsilon, eigenVectors, eigenVal);
210  scalar& e1 = eigenVal.component(0);
211  scalar& e2 = eigenVal.component(1);
212  scalar& e3 = eigenVal.component(2);
213 
214  // Compute principal stresses
215  const scalar s1 = 2*mu*e1 + lambda*tr;
216  const scalar s2 = 2*mu*e2 + lambda*tr;
217  const scalar s3 = 2*mu*e3 + lambda*tr;
218 
219  // positive part
220  const scalar pps1 = max(0, s1);
221  const scalar pps2 = max(0, s2);
222  const scalar pps3 = max(0, s3);
223 
224  const scalar sn = max(mag(s1), max(mag(s2), mag(s3)));
225  const scalar seps = 1e-12 * 190e9; // check stress state of the material
226  // (190e9 --> 190 gigapascal, UO2 young modulus)
227  const scalar r = (sn > seps) ?
228  min((pps1+pps2+pps3) / (mag(s1)+mag(s2)+mag(s3)), 1)
229  : 0 ;
230  return r;
231 }

235 (
236  const scalar& r,
237  const scalar& At,
238  const scalar& Ac,
239  const scalar& k
240 )
241 {
242  const scalar r2 = r*r;
243  return At * (2*r2 * (1 - 2*k) - r*(1 - 4*k)) +
244  Ac * (2*r2 - 3*r + 1);

298 void Foam::mazars2015_RIA::correct
299 (
300  const labelList& addr
301 )
302 {
303  // Material properties
304  const scalarField& nu(mesh_.lookupObject<volScalarField>("nu"));
305  const scalarField& mu(mesh_.lookupObject<volScalarField>("mu"));
306  const scalarField& lambda(mesh_.lookupObject<volScalarField>("lambda"));
307 
308  // Regerence to T, epsEl and epsEl (for ddtEpsEl)
309  const scalarField& T(mesh_.lookupObject<volScalarField>("T"));
310  const symmTensorField& epsilonEl(
311  mesh_.lookupObject<volSymmTensorField>("epsEl"));
312  const symmTensorField& epsilonEl0(
313  mesh_.lookupObject<volSymmTensorField>("epsEl").oldTime());
314 
315  // Old values of Yt and Yc
316  const scalarField& Yt0(Yt_.oldTime().internalField());
317  const scalarField& Yc0(Yc_.oldTime().internalField());
318 
319  if(mesh_.time().value() > currentTime_)
320  {
321  forAll(addr, addrI)
322  {
323  const label cellI = addr[addrI];
324 
325  // mu-model Parameters
326  const scalar At = this->At(T[cellI]);
327  const scalar Ac = this->Ac(T[cellI]);
328  const scalar Bt = this->Bt(T[cellI]);
329  const scalar Bc = this->Bc(T[cellI]);
330  const scalar epsilonEqt0 = this->epsilonEqt0(T[cellI]);
331  const scalar epsilonEqc0 = this->epsilonEqc0(T[cellI]);
332 
333  // Triaxial factor calculation
334  r_[cellI] = triaxialFactor(epsilonEl[cellI], mu[cellI],
335  lambda[cellI], tr(epsilonEl[cellI]));
336 
337  // Calculate strain eigenvalues
338  tensor eigenVectors = tensor::zero;
339  vector eigenVal = vector::zero;
340 
341  eig3().eigen_decomposition(epsilonEl[cellI], eigenVectors, eigenVal);
342  scalar& e1(eigenVal.component(0));
343  scalar& e2(eigenVal.component(1));
344  scalar& e3(eigenVal.component(2));
345 
346  // 1st & 2nd invariant
347  scalar inv1 = e1 + e2 + e3;
348  scalar inv2 = 0.5 * (pow(e1-e2,2) + pow(e2-e3, 2) + pow(e3-e1, 2));
349 
350  // Current equivalent strain - traction
351  scalar epsilonEqt(inv1 > 0 ?
352  inv1/(2*(1-2*nu[cellI])) + sqrt(inv2)/(2*(1+nu[cellI]))
353  : 0);
354 
355  // Update Yt
356  Yt_[cellI] = max(max(Yt0[cellI], epsilonEqt0), epsilonEqt);
357 
358  // Current equivalent strain - compression
359  scalar epsilonEqc(inv1 < 0 ?
360  inv1/(5*(1-2*nu[cellI])) + 6*sqrt(inv2)/(5*(1+nu[cellI]))
361  : 0);
362 
363  // Update Yc
364  if(equivalentStrainRateThreshold_)
365  {
366  // Calculate ddtEpsEl
367  scalar deltaT(mesh_.time().deltaT().value());
368  symmTensor ddtEpsEl((epsilonEl[cellI] - epsilonEl0[cellI])/deltaT);
369 
370  scalar ddtEpsElEq = sqrt(
371  pow(ddtEpsEl.component(symmTensor::XX), 2)
372  + pow(ddtEpsEl.component(symmTensor::YY), 2)
373  + pow(ddtEpsEl.component(symmTensor::ZZ), 2));
374 
375  if(T[cellI]>temperatureThreshold_ && ddtEpsElEq >= strainRateThreshold_)
376  {
377  Yc_[cellI] = max(max(Yc0[cellI], epsilonEqc0), epsilonEqc);
378  }
379  else
380  {
381  Yc_[cellI] = max(Yc0[cellI], epsilonEqc0);
382  }
383  }
384  else
385  {
386  if(T[cellI]>temperatureThreshold_)
387  {
388  Yc_[cellI] = max(max(Yc0[cellI], epsilonEqc0), epsilonEqc);
389  }
390  else
391  {
392  Yc_[cellI] = max(Yc0[cellI], epsilonEqc0);
393  }
394  }
395 
396  // thermodynamics variables computation
397  const scalar A = this->A(r_[cellI], At, Ac, k_);
398  const scalar B = this->B(r_[cellI], Bt, Bc);
399  const scalar Y0 = r_[cellI]*epsilonEqt0 + (1-r_[cellI])*epsilonEqc0;
400  Y_[cellI] = r_[cellI]*Yt_[cellI] + (1-r_[cellI])*Yc_[cellI];
401 
402  // Updated damage value
403  scalar d_new(
404  1 - (1-A)*Y0/max(Y_[cellI], SMALL) - A*exp(-B*(Y_[cellI] - Y0)));
405 
406  // Relax damage and modify E
407  d_[cellI] = relax_*min(max(0, d_new), d_max_) + (1 - relax_)*d_[cellI];
408 
409  if(d_[cellI] < SMALL)
410  {
411  d_[cellI] = 0;
412  }
413  }
414  }
415 }
