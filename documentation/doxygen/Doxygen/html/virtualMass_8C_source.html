116 void Foam::virtualMass::correct()
117 {
118  //- Store Vm in alphaRhoVm_
119  VmPtr_->correctField(Vm_);
120 
121  //- Actually update alphaRhoVm
122  bool virtualMassUsesMixtureDensity
123  (
124  pair_.pimple().dict().lookupOrDefault<bool>
125  (
126  "virtualMassUsesMixtureDensity",
127  false
128  )
129  );
130  volScalarField alphaRhoVm(Vm_*pair_.alphaDispersed());
131  if (virtualMassUsesMixtureDensity)
132  alphaRhoVm *=
133  (
134  fluid1_.rho()*fluid1_.normalized()
135  + fluid2_.rho()*fluid2_.normalized()
136  );
137  else
138  alphaRhoVm *= pair_.rhoContinuous();
139 
140  //-
141  VmForces_[U1_.name()] =
142  alphaRhoVm*
143  (
144  fvm::ddt(U1_)
145  + fvm::div(phi1_, U1_)
146  + fvm::SuSp(-fvc::div(phi1_), U1_)
147  - (
148  fvc::ddt(U2_)
149  + fvc::div(phi2_, U2_)
150  - fvc::div(phi2_)*U2_
151  )
152  );
153 
154  VmForces_[U2_.name()] =
155  alphaRhoVm*
156  (
157  fvm::ddt(U2_)
158  + fvm::div(phi2_, U2_)
159  + fvm::SuSp(-fvc::div(phi2_), U2_)
160  - (
161  fvc::ddt(U1_)
162  + fvc::div(phi1_, U1_)
163  - fvc::div(phi1_)*U1_
164  )
165  );
166 }
