44 void Foam::myMULES::limitSum(UPtrList<scalarField>& phiPsiCorrs)
45 {
46  forAll(phiPsiCorrs[0], facei)
47  {
48  scalar sumPos = 0;
49  scalar sumNeg = 0;
50 
51  for (int phasei=0; phasei<phiPsiCorrs.size(); phasei++)
52  {
53  if (phiPsiCorrs[phasei][facei] > 0)
54  {
55  sumPos += phiPsiCorrs[phasei][facei];
56  }
57  else
58  {
59  sumNeg += phiPsiCorrs[phasei][facei];
60  }
61  }
62 
63  scalar sum = sumPos + sumNeg;
64 
65  if (sum > 0 && sumPos > vSmall)
66  {
67  scalar lambda = -sumNeg/sumPos;
68 
69  for (int phasei=0; phasei<phiPsiCorrs.size(); phasei++)
70  {
71  if (phiPsiCorrs[phasei][facei] > 0)
72  {
73  phiPsiCorrs[phasei][facei] *= lambda;
74  }
75  }
76  }
77  else if (sum < 0 && sumNeg < -vSmall)
78  {
79  scalar lambda = -sumPos/sumNeg;
80 
81  for (int phasei=0; phasei<phiPsiCorrs.size(); phasei++)
82  {
83  if (phiPsiCorrs[phasei][facei] < 0)
84  {
85  phiPsiCorrs[phasei][facei] *= lambda;
86  }
87  }
88  }
89  }
90 }
