74 void Foam::constitutiveLaw::updateStress
75 (
76  volSymmTensorField& sigma,
77  volScalarField& sigmaHyd,
78  volSymmTensorField& sigmaDev,
79  const labelList& addr
80 )
81 {
82  damageModel_->correct(addr);
83  const volScalarField& d(damageModel_->d());
84  bool applyDamage(!isNull(d));
85 
86  forAll(addr, addrI)
87  {
88  const label cellI = addr[addrI];
89 
90  // Apply damage only when applicable
91  if(applyDamage)
92  {
93  sigmaHyd[cellI] *= (1 - d[cellI]);
94  sigmaDev[cellI] *= (1 - d[cellI]);
95  }
96 
97  sigma[cellI] = sigmaHyd[cellI]*I + sigmaDev[cellI];
98 
99  const cell& c = mesh_.cells()[cellI];
100 
101  forAll(c, faceI)
102  {
103  const label patchID = mesh_.boundaryMesh().whichPatch(c[faceI]);
104 
105  if (patchID > -1 and sigma.boundaryField()[patchID].size())
106  {
107  const label faceID =
108  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
109 
110  scalarField& sigmaHydF =
111  sigmaHyd.boundaryFieldRef()[patchID];
112 
113  symmTensorField& sigmaDevF =
114  sigmaDev.boundaryFieldRef()[patchID];
115 
116  // Apply damage only when applicable
117  if(applyDamage)
118  {
119  const scalarField& dF =
120  d.boundaryField()[patchID];
121 
122  sigmaHydF[faceID] *= (1 - dF[faceID]);
123  sigmaDevF[faceID] *= (1 - dF[faceID]);
124  }
125 
126  symmTensorField& sigmaF = sigma.boundaryFieldRef()[patchID];
127 
128  sigmaF[faceID] = sigmaHydF[faceID]*I + sigmaDevF[faceID];
129  }
130  }
131  }
132 }
