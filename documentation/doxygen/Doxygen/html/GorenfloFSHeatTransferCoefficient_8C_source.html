110 /*
111 IMPLEMENTATION NOTES
112 
113 The original pool boiling heat transfer coefficient by Gorenflo is the following
114 form:
115 
116 htcPB = h0_*F*((q/q0_)^n)*(R_/R0_)^0.133
117 
118 with F being a function of the reduced pressure, i.e. p/pCtri_. Needless to
119 say, this form is not particularly suitable for a numerical implementation in
120 its current form as htcPB is required to compute q in the first place. In
121 general, the pool boiling heat flux is given by:
122 
123 q = htc*(Twall-Tsat)
124 
125 with htc being the total heat transfer coefficient between wall and fluid,
126 Twall being the wall temperature and Tsat the saturation temperature. By
127 assuming that htc ~ htcPB (i.e. that most of the heat flux is due to the pool
128 boiling mechanism), one can tha equate:
129 
130 htcPB = h0_*F*((htcPB*(Twall-Tsat)/q0_)^n)*(R_/R0_)^0.133
131 
132 which can be re-arranged to isolate htcPB on the left hand side as:
133 
134 htcPB = (h0_*F*((Twall-Tsat)^n)*(R_/R0_)^0.133/(q0_^n))^(1.0/(1.0-n))
135 
136 Which is how the correlation is implemented here.
137 The assumption of htc ~ htcPB can be rather crude but it is widely employed
138 to yield a feasable numerical implementation in other computer codes too (e.g.
139 TRACE)
140 */

142 Foam::scalar Foam::FSHeatTransferCoefficientModels::Gorenflo::value
143 (
144  const label& celli
145 ) const
146 {
147  if (!useExplicitHeatFlux_)
148  {
149  scalar deltaT((*Twall_)[celli]-(*Tsat_)[celli]);
150  if (deltaT > 0.0)
151  {
152  scalar pR(p_[celli]/pCrit_);
153  scalar F(1.73*pow(pR, 0.27) + (6.1+0.68/(1.0-pR))*sqr(pR));
154  scalar n(0.9-0.3*pow(pR, 0.15));
155  return pow(h0_*A_*F*pow(deltaT/q0_,n), (1.0/(1.0-n)));
156  }
157  else
158  return 0.0;
159  }
160  else
161  {
162  const scalar& q((*q_)[celli]);
163  if (q > 0.0)
164  {
165  //- Reduced pressure
166  scalar pR(p_[celli]/pCrit_);
167  scalar F(1.73*pow(pR, 0.27) + (6.1+0.68/(1.0-pR))*sqr(pR));
168  scalar n(0.9-0.3*pow(pR, 0.15));
169  return h0_*F*pow(q/q0_, n)*A_;
170  }
171  else
172  return 0.0;
173  }
174 }
