131 template<class BasicTurbulenceModel>
132 void LaheyKEpsilon<BasicTurbulenceModel>::setFluidNames() const
133 {
134  const fvMesh& mesh(this->mesh_);
135  HashTable<const fluid*> fluids(mesh.lookupClass<const fluid>());
136  const fluid& fluid1 = *(fluids[fluids.toc()[0]]);
137  const fluid& fluid2 = *(fluids[fluids.toc()[1]]);
138  if
139  (
140  !(fluid1.isLiquid() and fluid2.isGas()) and
141  !(fluid2.isLiquid() and fluid1.isGas())
142  )
143  {
144  FatalErrorInFunction
145  << "The LaheyKEpsilon turbulence model only works for "
146  << "liquid-gas systems. Set the the stateOfMatter entry in "
147  << "phaseProperties." << fluid1.name() << "Properties and/or "
148  << "phaseProperties." << fluid2.name() << "Properties) to "
149  << "distinguish between gas and liquid"
150  << exit(FatalError);
151  }
152  liquidName_ = (fluid1.isLiquid()) ? fluid1.name() : fluid2.name();
153  gasName_ = (fluid1.isGas()) ? fluid1.name() : fluid2.name();
154 }

201 template<class BasicTurbulenceModel>
202 const Foam::tmp<Foam::volScalarField> LaheyKEpsilon<BasicTurbulenceModel>::Cd()
203 const
204 {
205  tmp<volScalarField> tCd
206  (
207  new volScalarField
208  (
209  IOobject
210  (
211  "Cd",
212  this->mesh_.time().timeName(),
213  this->mesh_
214  ),
215  this->mesh_,
216  dimensionedScalar("", dimless, 1e-3),
217  zeroGradientFvPatchScalarField::typeName
218  )
219  );
220  volScalarField& Cd = tCd.ref();
221 
222  const FFPair& p(pair());
223  const volScalarField& l(liquid());
224  const volScalarField& g(gas());
225 
226  //- Compute Cd from Kd, cell-by-cell as it's faster (I don't really care
227  // about BCs)
228  forAll(Cd, i)
229  {
230  const scalar& li(l[i]);
231  const scalar& gi(g[i]);
232  Cd[i] =
233  (2.0)*p.Kd()[i]*p.DhDispersed()[i]/p.rhoContinuous()[i]/
234  max
235  (
236  (li*gi)/(li+gi)*p.magUr()[i], 1e-3
237  );
238  }
239  Cd.correctBoundaryConditions();
240 
241  return tCd;
242 }

277 template<class BasicTurbulenceModel>
278 tmp<volScalarField> LaheyKEpsilon<BasicTurbulenceModel>::bubbleG() const
279 {
280  tmp<volScalarField> bubbleG
281  (
282  Cp_*
283  (
284  1.0
285  + pow(Cd(), 4.0/3.0)
286  )*
287  gas()*pow3(pair().magUr())/
288  max(gas().Dh(), dimensionedScalar("", dimLength, 1e-3))
289  );
290 
291  return bubbleG;

296 tmp<volScalarField>
297 LaheyKEpsilon<BasicTurbulenceModel>::phaseTransferCoeff() const
298 {
299  const volVectorField& U = this->U_;

301  const rhoField& rho = this->rho_;
302 
303  const phaseCompressibleTurbulenceModel& gasTurbulence = gas().turbulence();
304
