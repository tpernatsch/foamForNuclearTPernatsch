42 void Foam::AMIMethodOFFBEAT::checkPatches() const
43 {
44  if (debug && (!srcPatch_.size() || !tgtPatch_.size()))
45  {
46  Pout<< "AMI: Patches not on processor: Source faces = "
47  << srcPatch_.size() << ", target faces = " << tgtPatch_.size()
48  << endl;
49  }
50 
51 
52  if (conformal())
53  {
54  const scalar maxBoundsError = 0.05;
55 
56  // check bounds of source and target
57  boundBox bbSrc(srcPatch_.points(), srcPatch_.meshPoints(), true);
58  boundBox bbTgt(tgtPatch_.points(), tgtPatch_.meshPoints(), true);
59 
60  boundBox bbTgtInf(bbTgt);
61  bbTgtInf.inflate(maxBoundsError);
62 
63  if (!bbTgtInf.contains(bbSrc))
64  {
65  WarningInFunction
66  << "Source and target patch bounding boxes are not similar"
67  << nl
68  << " source box span : " << bbSrc.span() << nl
69  << " target box span : " << bbTgt.span() << nl
70  << " source box : " << bbSrc << nl
71  << " target box : " << bbTgt << nl
72  << " inflated target box : " << bbTgtInf << endl;
73  }
74  }
75 }

78 bool Foam::AMIMethodOFFBEAT::initialise
79 (
80  labelListList& srcAddress,
81  scalarListList& srcWeights,
82  labelListList& tgtAddress,
83  scalarListList& tgtWeights,
84  label& srcFacei,
85  label& tgtFacei
86 )
87 {
88  checkPatches();
89 
90  // set initial sizes for weights and addressing - must be done even if
91  // returns false below
92  srcAddress.setSize(srcPatch_.size());
93  srcWeights.setSize(srcPatch_.size());
94  tgtAddress.setSize(tgtPatch_.size());
95  tgtWeights.setSize(tgtPatch_.size());
96 
97  // check that patch sizes are valid
98  if (!srcPatch_.size())
99  {
100  return false;
101  }
102  else if (!tgtPatch_.size())
103  {
104  WarningInFunction
105  << srcPatch_.size() << " source faces but no target faces" << endl;
106 
107  return false;
108  }
109 
110  // reset the octree
111  resetTree();
112 
113  // find initial face match using brute force/octree search
114  if ((srcFacei == -1) || (tgtFacei == -1))
115  {
116  srcFacei = 0;
117  tgtFacei = 0;
118  bool foundFace = false;
119  forAll(srcPatch_, facei)
120  {
121  tgtFacei = findTargetFace(facei);
122  if (tgtFacei >= 0)
123  {
124  srcFacei = facei;
125  foundFace = true;
126  break;
127  }
128  }
129 
130  if (!foundFace)
131  {
132  if (requireMatch_)
133  {
134  FatalErrorInFunction
135  << "Unable to find initial target face"
136  << abort(FatalError);
137  }
138 
139  return false;
140  }
141  }
142 
143  if (debug)
144  {
145  Pout<< "AMI: initial target face = " << tgtFacei << endl;
146  }
147 
148  return true;
149 }

203 void Foam::AMIMethodOFFBEAT::resetTree()
204 {
205  // Clear the old octree
206  treePtr_.clear();
207 
208  treeBoundBox bb(tgtPatch_.points(), tgtPatch_.meshPoints());
209  // NOTE: different than normal AMI method
210  // necessary for large gaps
211  bb.inflate(0.1);
212 
213  if (!treePtr_.valid())
214  {
215  treePtr_.reset
216  (
217  new indexedOctree<treeType>
218  (
219  treeType
220  (
221  false,
222  tgtPatch_,
223  indexedOctree<treeType>::perturbTol()
224  ),
225  bb, // overall search domain
226  8, // maxLevel
227  10, // leaf size
228  3.0 // duplicity
229  )
230  );
231  }
232 }

235 Foam::label Foam::AMIMethodOFFBEAT::findTargetFace
236 (
237  const label srcFacei
238 ) const
239 {
240  label targetFacei = -1;
241 
242  const pointField& srcPts = srcPatch_.points();
243  const face& srcFace = srcPatch_[srcFacei];
244  const point srcPt = srcFace.centre(srcPts);
245  const scalar srcFaceArea = srcMagSf_[srcFacei];
246 
247  pointIndexHit sample = treePtr_->findNearest(srcPt, 10.0*srcFaceArea);
248 
249  if (sample.hit())
250  {
251  targetFacei = sample.index();
252 
253  if (debug)
254  {
255  Pout<< "Source point = " << srcPt << ", Sample point = "
256  << sample.hitPoint() << ", Sample index = " << sample.index()
257  << endl;
258  }
259  }
260 
261  return targetFacei;
262 }

265 void Foam::AMIMethodOFFBEAT::appendNbrFaces
266 (
267  const label facei,
268  const primitivePatch& patch,
269  const DynamicList<label>& visitedFaces,
270  DynamicList<label>& faceIDs
271 ) const
272 {
273  const labelList& nbrFaces = patch.faceFaces()[facei];
274 
275  // filter out faces already visited from face neighbours
276  forAll(nbrFaces, i)
277  {
278  label nbrFacei = nbrFaces[i];
279  bool valid = true;
280  forAll(visitedFaces, j)
281  {
282  if (nbrFacei == visitedFaces[j])
283  {
284  valid = false;
285  break;
286  }
287  }
288 
289  if (valid)
290  {
291  forAll(faceIDs, j)
292  {
293  if (nbrFacei == faceIDs[j])
294  {
295  valid = false;
296  break;
297  }
298  }
299  }
300 
301  // prevent addition of face if it is not on the same plane-ish
302  if (valid)
303  {
304  const vector& n1 = patch.faceNormals()[facei];
305  const vector& n2 = patch.faceNormals()[nbrFacei];
306 
307  scalar cosI = n1 & n2;
308 
309  if (cosI > cos(maxWalkAngle()))
310  {
311  faceIDs.append(nbrFacei);
312  }
313  }
314  }
315 }
