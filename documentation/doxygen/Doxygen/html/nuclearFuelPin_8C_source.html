643 void Foam::powerModels::nuclearFuelPin::updateLocalAvgGlobalMinMaxT
644 (
645  const label& starti,
646  const label& endi,
647  const scalarList& r,
648  const scalar& dr,
649  const scalarField& Trad,
650  scalar& Tavi,
651  scalar& Tmin,
652  scalar& Tmax
653 )
654 {
655  scalar intr(0);
656  scalar intTr(0);
657  for(int j = starti; j < endi; j++)
658  {
659  const scalar& T(Trad[j]);
660  scalar rdr(r[j]*dr);
661 
662  //- Cells at the mesh ends are only half as wide (the other half
663  // belongs to the ghost node). Thus, weigh temperatures at the extrema
664  // by a factor 0.5
665  if (j == starti or j == endi-1)
666  {
667  intr += rdr/2.0;
668  intTr += T*rdr/2.0;
669  }
670  else
671  {
672  intr += rdr;
673  intTr += T*rdr;
674  }
675  if (T > Tmax) Tmax = T;
676  if (T < Tmin) Tmin = T;
677  }
678  Tavi = intTr/intr;
679 }

782  const scalar& dA(dARegion[i]);

785  scalar XdA(Xc*dA);
786  M[i][i+1] = -C;
787  M[i][i-1] = -D;
788  M[i][i] = C+D+XdA;
789  S[i] = TOld[i]*XdA;
790  }
791 
792  //- Cladding bulk
793  for (int i = fuelMeshSize+1; i < meshSize-1; i++)
794  {
795  const scalar& r(rRegion[i]);
796  const scalar& dA(dARegion[i]);
797  scalar B(twoPkByDrc*(r+drhc));
798  scalar C(twoPkByDrc*(r-drhc));
799  scalar XdA(Xc*dA);
800  M[i][i+1] = -B;
801  M[i][i-1] = -C;
802  M[i][i] = B+C+XdA;
803  S[i] = TOld[i]*XdA;
804  }
805 
806  //- Cladding outer surface, convective BC with fluid(s) wetting the pin
807  {
808  label i(meshSize-1);
809  const scalar& r(rRegion[i]);
810  const scalar& dA(dARegion[i]);
811  scalar C(twoPkByDrc*(r-drhc));
812  scalar D(2.0*pi_*r);
813  scalar XdA(Xc*dA);
814  M[i][i-1] = -C;
815  M[i][i] = C+D*HSumi+XdA;
816  S[i] = TOld[i]*XdA + D*HTSumi;
817  }
818  }

821  solve(Trad, M, S);
822

825  Tfo_[celli] = Trad[fuelMeshSize-1];
826  Tci_[celli] = Trad[fuelMeshSize];
827  Tco_[celli] = Trad[meshSize-1];
