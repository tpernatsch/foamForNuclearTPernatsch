150 void Foam::largeStrainTotLag::correct()
151 {
152  const dictionary& stressControl =
153  mesh_.solutionDict().subDict("stressAnalysis");
154 
155  // Maximum inner iterations
156  const int nCorr = stressControl.lookupOrDefault<int>("nCorrectors", 1);
157 
158  if (multiMaterialCorrection_)
159  {
160  materialInterface_->correct();
161  }
162 
163  List<Tuple2<label, vector> > refPairs;
164  if (stressControl.found("referencePairs"))
165  {
166  refPairs = List<Tuple2<label, vector> >(stressControl.lookup("referencePairs"));
167  }
168 
169  // Intermediate fields
170  const volScalarField& mu(mat_.mu());
171  const volScalarField& lambda(mat_.lambda());
172  volScalarField twoMuLambda("2mu+lambda", 2*mu + lambda);
173  surfaceScalarField twoMuLambda_f
174  = mechanicsSubSolver::correctedTwoMuLambdaF(twoMuLambda);
175 
176  // Reference to old time
177  const volVectorField& DOld(D_.oldTime());
178 
179  // Inner loop convergence variables
180  int nInnerIter(0);
181  bool convergedInner(false);
182 
183  do
184  {
185  // Store fields for under-relaxation and residual calculation
186  D_.storePrevIter();
187 
188  // Update explicit component of the stress
189  sigmaExp_ =
190  mechanicsSubSolver::correctedSigmaExp(
191  (J_*(Finv_ & sigma_))(), twoMuLambda, gradD_);
192 
193  // Update the divergence of the explicit component of the stress
194  divSigmaExp_ =
195  mechanicsSubSolver::correctedDivSigmaExp
196  (sigmaExp_, (J_*(Finv_ & sigma_))(), twoMuLambda, twoMuLambda_f, D_, gradD_);
197 
198  // Couple implicit contact boundaries (if present)
199  coupleBoundaries(D_, true);
200 
201  // Define displacement equation
202  fvVectorMatrix DEqn
203  (
204  fvm::d2dt2(mat_.rho(), D_)
205  ==
206  fvmLaplacianD(twoMuLambda_f, D_, "laplacian(DD,D)")
207  + divSigmaExp_
208  );
209 
210  // Manipulate matrix based on BC rule (necessary for implicit contact)
211  DEqn.boundaryManipulate(D_.boundaryFieldRef());
212 
213  forAll(refPairs, i)
214  {
215  DEqn.setReference(refPairs[i].first(), refPairs[i].second());
216  }
217 
218  // Relax matrix
219  DEqn.relax();
220 
221  // Solve equation for displacement
222  residual_ = DEqn.solve().initialResidual();
223 
224  // Un-couple implicit contact boundaries (if present)
225  coupleBoundaries(D_, false);
226 
227  // Store initialResidual when in first inner iteration
228  if(nInnerIter==0)
229  {
230  initialResidual_ = residual_;
231  }
232 
233  // Relax field
234  D_.relax();
235 
236  // Update strain and stress
237  gradD_ =
238  mechanicsSubSolver::correctedGradD(twoMuLambda, D_, gradD_, sigmaExp_);
239 
240  // Total deformation gradient
241  F_ = I + gradD_.T();
242 
243  // Inverse of the deformation gradient
244  Finv_ = inv(F_);
245 
246  // Relative deformation gradient
247  relF_ = F_ & inv(F_.oldTime());
248 
249  // Jacobian of the deformation gradient
250  J_ = det(F_);
251 
252  // Calculate epsilon
253  if(strainTensorName_ == "linear")
254  {
255  epsilon_ = symm(gradD_);
256  }
257  else if(strainTensorName_ == "EulerAlmansi")
258  {
259  epsilon_ = 0.5*(I - symm(Finv_.T() & Finv_));
260  }
261  else if(strainTensorName_ == "Hencky")
262  {
263  epsilon_ = 0.5*log(symm(F_.T() & F_));
264  }
265  else if(strainTensorName_ == "GreenLagrange")
266  {
267  epsilon_ = 0.5*(symm(F_.T() & F_) - I);
268  }
269  else
270  {
271  // Error for strain tensor
272  FatalErrorInFunction() << nl
273  << "Possible strain tensors are:" << nl
274  << tab << "- linear" << nl
275  << tab << "- EulerAlmansi" << nl
276  << tab << "- Hencky" << nl
277  << tab << "- GreenLagrange" << nl
278  << exit(FatalError);
279  }
280 
281  // Save matrix coefficient
282  const volScalarField DEqnA("DEqnA", DEqn.A());
283 
284  // Correct constitutive material laws
285  rheo_.correct(sigma(), epsilon(), D());
286 
287  // Update explicit component of the stress
288  sigmaExp_ =
289  mechanicsSubSolver::correctedSigmaExp(
290  (J_*(Finv_ & sigma_))(), twoMuLambda, gradD_);
291  sigma_ = symm( ((sigmaExp_ + twoMuLambda*gradD_) & F_)/J_) ;
292 
293  // Calculate a different residual based on the relative change of D
294  scalar denom = gMax
295  (
296  mag(D_.primitiveField() - DOld.primitiveField())
297  );
298 
299  if (denom < SMALL)
300  {
301  denom = max(gMax(mag(D_.primitiveField())), SMALL);
302  }
303 
304  absErr_ = gMax
305  (
306  mag(D_.primitiveField() - D_.prevIter().internalField())
307  );
308 
309  relResidual_ = absErr_/denom;
310 
311  Info << "relResidualD is " << relResidual_ << endl;
312  Info << "absErrD is " << absErr_ << endl;
313 
314  nInnerIter++;
315 
316  convergedInner = converged(D(), residual_);
317 
318  } while
319  (
320  not(convergedInner)
321  && nInnerIter < nCorr
322  );
323 
324  // Transform stress and displacement in cyl. coord. (for output)
325  correctCylindrical();
326  // Transform stress and displacement in sph. coord. (for output)
327  correctSpherical();
328 
329  // Print total force on each face
330  forceSummary();
331 
332  Info<< "Domain average displacement: "
333  << fvc::domainIntegrate(D_).value()/gSum(mesh_.V())
334  << endl << endl;
335 }
