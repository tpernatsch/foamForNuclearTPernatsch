25  Foam::InterpolateTable

27 Description
28  Class for interpolation in tabulated data.
29  This class doesn't actually store any fields, but is intended as
30  a quick helper class to perform the actual interpolation for
31  different data types, i.e. Fields, FieldFields, etc. effectively
32  facilitating multi-dimensional interpolation.
33  Currently provides two interpolation methods, step and linear (see )
34 
35  This is an alternative to OpenFOAM's built in interpolateTable and
36  interpolateTable2D classes, which partially combines the two using
37  templates. It was created largely for 2D tables where we want to look up
38  one dimension first and then interpolate many times in the resulting
39  data, i.e. create a 1D table by interpolating in a 2D table.

153  //- Check the table bounds, etc.
154  void check() const
155  {
156  //- Check that table bounds match
157  if ((xValues_.size() < 2) || (xValues_.size() != data_.size()))
158  {
159  //- TODO Extended error message
160  FatalErrorInFunction()
161  << "Tabulated data has unexpected/mismatched dimensions:"
162  << endl << tab << "ordinates: " << xValues_.size()
163  << endl << tab << "data: " << data_.size()
164  << abort(FatalError);
165  }
166 
167  //- Check that supplied x values are in ascending order

169  {
170  if (xValues_[i] <= xValues_[i-1])

213  //- Interpolation function
214  inline ReturnType operator()(scalar x) const
215  {
216  if (x >= xValues_[0])
217  {
218  for (int i=1; i < xValues_.size(); i++)
219  {
220  if(x <= xValues_[i])
221  {
222  if (method_ == STEP)
223  {
224  // TODO i-1 or i?
225  return data_[i-1];
226  }
227  else if (method_ == LINEAR)
228  {
229  scalar x0 = xValues_[i-1];
230  scalar x1 = xValues_[i];
231  scalar alpha = (x-x0)/(x1-x0);
232 
233  return (1-alpha)*data_[i-1] + alpha*data_[i];
234  }
235  }
236  }
237  }
238 
239  //- To keep the main body of the code unchanged, the modification for
240  //- the out-of-bound case is done below. -F.X.
241  switch (ob_method_ )
242  {
243  case ERROR:// Out of bounds error
244  {
245  FatalErrorInFunction()
246  << "Supplied value out of bounds." << abort(FatalError);
247 
248  return 1.0*data_[0]; // Dummy return value
249  break;
250  }
251 
252  case FIXED:
253  {
254  if (x < xValues_[0])
255  {
256  return data_[0];
257  }
258  else
259  {
260  return data_[xValues_.size()-1];
261  }
262  break;
263  }
264 
265  case EXTRAPOLATION:
266  {
267  if (x < xValues_[0])
268  {
269  scalar x0 = xValues_[0];
270  scalar x1 = xValues_[1];
271  return (data_[1]-data_[0])/(x1-x0)*(x-x0)+data_[0];
272  }
273  else
274  {
275  scalar x0 = xValues_[xValues_.size()-2];
276  scalar x1 = xValues_[xValues_.size()-1];
277  return (data_[xValues_.size()-1]-data_[xValues_.size()-2])/(x1-x0)*(x-x0)
278  +data_[xValues_.size()-2];
279  }
280 
281  break;
282  }
283 
284  default:
285  FatalErrorInFunction() << "error in interpolateTable.H"<< abort(FatalError);
286  return 1.0*data_[0];
287  break;
288  }
289  }
