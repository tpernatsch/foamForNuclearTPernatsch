94 void Foam::timeDependentLhgr::correct()
95 {
96  const globalOptions& globalOpt
97  (mesh_.lookupObject<globalOptions>("globalOptions"));
98 
99  scalar angularFraction = globalOpt.angularFraction();
100 
101  if(mesh_.topoChanging())
102  {
103  calcAddressing(heatSourceOptDict_);
104  calcReferenceDimensions(heatSourceOptDict_);
105  }
106 
107  if(!mesh_.foundObject<volScalarField>("QPrevIter"))
108  {
109  Q_.storePrevIter();
110  }
111 
112  Q_.prevIter().storePrevIter();
113  Q_.storePrevIter();
114 
115  const scalarField& Qprev(Q_.prevIter());
116  const scalarField& QprevPrev(Q_.prevIter().prevIter());
117 
118  // Correct heat source only once per time step or until power changes
119  if
120  (
121  gMax(mag(QprevPrev - Qprev)) > 1e-3
122  or
123  mesh_.time().value() > currentTime_
124  or
125  mesh_.foundObject<volScalarField>("porosity")
126  )
127  {
128  currentTime_ = mesh_.time().value();
129 
130  scalarField Vi(mesh_.V(), addr_);
131  scalar totalVol(gSum(Vi)/angularFraction);
132 
133  // Interpolate LHGR and calculate average power density Qavg
134  scalar lhgr = lookupLHGR(currentTime_);
135  scalar Qavg = lhgr*(zMax_ - zMin_)/totalVol;
136 
137  // Correct the radial and axial profiles
138  radialModel_->correct();
139  azimuthalModel_->correct();
140  axialModel_->correct();
141 
142  // Build the resulting heat source
143  scalarField& Qi = Q_.ref();
144  scalarField& lhgri = lhgr_.ref();
145  const scalarField& f_r = radialModel_->profile();
146  const scalarField& h_theta = azimuthalModel_->profile();
147  const scalarField& g_z = axialModel_->profile();
148 
149  forAll(addr_, addrI)
150  {
151  const label cellI = addr_[addrI];
152 
153  Qi[cellI] = Qavg*f_r[addrI]*g_z[addrI]*h_theta[addrI];
154  lhgri[cellI] = lhgr*g_z[addrI];
155  }
156 
157  // Reshape power profile due to porosity redistribution
158  reshapePowerPorosity();
159 
160  scalarField Qaddr(Q_.ref(), addr_);
161  scalar powerTot = lhgr*(zMax_ - zMin_)*angularFraction;
162  scalar powerTotOF = gSum( Vi*Qaddr );
163 
164  scalar error =
165  100*
166  (
167  mag(powerTotOF - powerTot)/max(powerTot, VSMALL)
168  );
169 
170  if(error > 0.01)
171  {
172  WarningIn("Foam::timeDependentLhgr::correct()") << nl
173  << " Total power does not correspond to LHGR*height, "<< nl
174  << " with an error of " << error << "%" << nl
175  << " Radial or axial profiles might be not normalized." << nl
176  << " The field Q will be scaled to preserve total power."
177  << nl << endl;
178 
179  radialModel_ -> checkNormalization();
180  azimuthalModel_ -> checkNormalization();
181  axialModel_ -> checkNormalization();
182 
183  scalar scalingFactor(powerTot/powerTotOF);
184 
185  forAll(addr_, addrI)
186  {
187  const label cellI = addr_[addrI];
188 
189  Qi[cellI] *= scalingFactor;
190  }
191  }
192  }
193  else
194  {
195  currentTime_ = mesh_.time().value();
196  }
197 
198  Q_.correctBoundaryConditions();
199  lhgr_.correctBoundaryConditions();
200 }

307 Foam::scalar Foam::timeDependentLhgr::nextDeltaTAbsoluteChange(scalar deltaT) const
308 {
309  // Assume that deltaT does not change or that it is limited by lhgr history
310  scalar predictedNextTime = min
311  (
312  currentTime_ + mesh_.time().deltaT().value(),
313  mesh_.time().userTimeToTime
314  (
315  lhgrTable_.getNextPoint(mesh_.time().timeToUserTime(currentTime_))
316  )
317  );
318 
319  scalar currentLhgr(lookupLHGR(currentTime_));
320  scalar predictedNewLhgr(lookupLHGR(predictedNextTime));
321 
322  // Calculate absolute rate of change
323  scalar deltaLhgr = (predictedNewLhgr - currentLhgr);
324 
325  // Set next deltaT
326  scalar nextDeltaT = deltaT*maxAbsoluteLhgrIncrease_/max(deltaLhgr, SMALL);
327 
328  Info<< "Maximum deltaT calculated by heatSource model: "
329  << mesh_.time().timeToUserTime(nextDeltaT)
330  << " with current lhgr " << currentLhgr <<" W/m, and an absolute change of " << deltaLhgr << " W/m"
331  << endl;
332 
333  return nextDeltaT;
334 }
