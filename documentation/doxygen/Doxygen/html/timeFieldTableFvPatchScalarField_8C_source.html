136 void Foam::timeFieldTableFvPatchScalarField::updateField()
137 {
138  //- Simulation time
139  const scalar t = db().time().timeOutputValue();
140 
141  if (t > tStart_)
142  {
143  scalar t0(0);
144  scalar t1(0);
145  if (currentIndex_ < (table_.size()-1))
146  {
147  t0 = table_[currentIndex_].first();
148  t1 = table_[currentIndex_+1].first();
149  while (true)
150  {
151  if ((t >= t0 and t <= t1))
152  break;
153  currentIndex_++;
154  t0 = table_[currentIndex_].first();
155  t1 = table_[currentIndex_+1].first();
156  }
157  }
158 
159  if (currentIndex_ < (table_.size()-1))
160  {
161  //- Read table values at correct index
162  scalarField& f0(table_[currentIndex_].second());
163  scalarField& f1(table_[currentIndex_+1].second());
164 
165  //- Interpolation coefficient
166  scalar c((t1-t)/(t1-t0));
167 
168  //- Linear interpolation between the fields at the provided times
169  // if the time falls in between two time bins
170  this->operator==
171  (
172  c*f0+(1.0-c)*f1
173  );
174  }
175  else //- i.e. t > tLast
176  {
177  this->operator==(table_[table_.size()-1].second());
178  }
179  }
180  else //- i.e. t <= tStart
181  {
182  this->operator==(table_[0].second());
183  }
184 
185  fixedValueFvPatchScalarField::updateCoeffs();
186 }
