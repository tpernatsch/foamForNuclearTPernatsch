97 void Foam::solvers::extendedThermoMechanics::correctPhysics()
98 {
99 
100  Info << "Solving region: " << mesh_.name()<<nl<<endl;
101  // True when time step is converged
102  bool converged_ = false;
103 
104  // Read from fvSolution dict the maximum number of outer iterations
105  const dictionary& stressControl =
106  mesh_.solutionDict().subDict("stressAnalysis");
107  int maxOuterIter(readInt(stressControl.lookup("maxOuterIter")));
108 
109  // Track number of outer iterations (per time-step)
110  int nOuterIter = 0;
111 
112  // Update mesh if necessary
113  if
114  (
115  mesh_.foundObject<fvMesh>("referenceMesh")
116  )
117  {
118  mechanics_->updateMesh();
119  }
120 
121  // Update incremental fields (DD and gradDD = 0) if necessary
122  mechanics_->updateIncrementalFields();
123 
124  do
125  {
126  Info << "OuterIteration n. " << nOuterIter << nl << endl;
127 
128  storeGlobalFieldsPrevIter();
129 
130  // First update corrosion, only then update AMI. Necessary because
131  // if corrosion moves the mesh, the AMI is also updated given that
132  // moving the mesh clears the geometry and the AMIPtr. However this
133  // first update, does not use the updated mesh location
134  // TODO: find a way to update mesh without AMI?
135 
136  forAll(mesh_.boundary(), patchi)
137  {
138  if(isType<regionCoupledOFFBEATFvPatch>(mesh_.boundary()[patchi]))
139  {
140  const regionCoupledOFFBEATFvPatch& patch
141  = refCast<const regionCoupledOFFBEATFvPatch>(mesh_.boundary()[patchi]);
142 
143  patch.updateAMI();
144  }
145  }
146 
147  bool isCouplingDict(solverDict_.found("couplingOptions"));
148 
149  bool getTFromTH
150  (
151  isCouplingDict ?
152  solverDict_.subDict("couplingOptions").getOrDefault<bool>("correctTFromTH", false)
153  : false
154  );
155  bool computeNeutroMeshDisplacement
156  (
157  isCouplingDict ?
158  solverDict_.subDict("couplingOptions").getOrDefault<bool>("correctDispForNeutro", false)
159  : false
160  );
161 
162  heatSrc_->correct();
163  thermal_->correct();
164  if (getTFromTH)
165  {
166  #include "correctT.H"
167  }
168  mat_->correctBehavioralModels();
169  rheo_->getAdditionalStrain();
170  mechanics_->correct();
171  if (computeNeutroMeshDisplacement)
172  {
173  #include "correctMeshDisp.H"
174  }
175  correctBCsGlobalFields();
176  relaxGlobalFields();
177 
178  nOuterIter++;
179 
180  // Check convergence
181  converged_ =
182  (
183  mechanics_->converged()
184  and
185  thermal_->converged()
186  and
187  rheo_->converged()
188  );
189  }
190  while
191  (
192  (nOuterIter < 2)
193  ||
194  (not(converged_) && nOuterIter < maxOuterIter)
195  );
196 
197  mechanics_->updateTotalFields();
198 
199  // Check if failure occurred
200  mat_->checkFailure();
201 
202 }
