89 void Foam::solvers::multiPhysicsSolver::createSolvers(word name)
90 {
91  multiPhysicsDict_ = this->subDict("multiPhysicsSolvers").subDict(name);
92  solverNames_ = multiPhysicsDict_.subDict("solvers").toc();
93  singlePhysicsSolverNames_.setSize(0);
94  minResidual_ = multiPhysicsDict_.get<scalar>("minResidual");
95  maxIterations_ = multiPhysicsDict_.get<label>("maxIterations");
96 
97  // First create the new solvers
98 
99  solvers_.setSize(solverNames_.size());
100 
101  correctOnlyEnergy_.setSize(solverNames_.size());
102  forAll(correctOnlyEnergy_, boolI)
103  {
104  correctOnlyEnergy_[boolI]=true;
105  }
106 
107  wordList energyOnlyList = multiPhysicsDict_.getOrDefault<wordList>("includeFluidMechanicsInLoop", wordList());
108 
109  forAll(solverNames_, nameI)
110  {
111 
112  if(meshHandler_->contains(energyOnlyList, solverNames_[nameI]))
113  correctOnlyEnergy_[nameI]=false;
114 
115  Info<< "Creating sub-scale solver " << solverNames_[nameI] << nl
116  << endl;
117 
118  word solverType
119  (
120  multiPhysicsDict_
121  .subDict("solvers")
122  .get<word>(solverNames_[nameI])
123  );
124 
125  // This replicates the structure of regionSolvers. This allows to create
126  // subSolvers which are multiPhysicsSolvers themselves and so on,
127  // creating a tree-like structure
128 
129  if (solverType != "multiPhysicsSolver")
130  {
131  solvers_.set
132  (
133  nameI,
134  solver::New(solverType, meshHandler_->returnMesh(solverNames_[nameI]))
135  );
136  singlePhysicsSolverNames_.append(solverNames_[nameI]);
137  }
138  else
139  {
140  solvers_.set
141  (
142  nameI,
143  solver::New(solverType, meshHandler_->returnMesh("dummy"))
144  );
145  Foam::solvers::multiPhysicsSolver* multiPhysicsSolverPtr = dynamic_cast<Foam::solvers::multiPhysicsSolver*>(&solvers_[nameI]);
146  multiPhysicsSolverPtr->getMapper(*meshHandler_);
147  multiPhysicsSolverPtr->createSolvers(solverNames_[nameI]);
148  multiPhysicsSolverPtr = nullptr;
149  }
150  }
151 }

172 void Foam::solvers::multiPhysicsSolver::correctPhysics()
173 {
174  scalar iterN(0);
175  scalar residual(0);
176 
177  const Time& runTime(meshHandler_->returnMesh(singlePhysicsSolverNames_[0]).time());
178 
179  do
180  {
181  meshHandler_->mapTheseFields(runTime, singlePhysicsSolverNames_ );
182  forAll(solvers_, solvI)
183  {
184  solvers_[solvI].deformMesh();
185  (iterN>0 and correctOnlyEnergy_[solvI])?
186  solvers_[solvI].correctTightlyCoupledPhysics():
187  solvers_[solvI].correctPhysics();
188  solvers_[solvI].correctBaffleLessFields();
189  }
190 
191  forAll(solvers_, solvI)
192  {
193  residual = max(residual, solvers_[solvI].getResidual());
194  }
195 
196  ++iterN;
197 
198  if(residual<minResidual_)
199  Info << nl<<"Multiphysics loop converged after " << iterN <<" iterations"<<endl<<nl;
200  }
201  while(residual>minResidual_ && iterN < maxIterations_);
202 }
