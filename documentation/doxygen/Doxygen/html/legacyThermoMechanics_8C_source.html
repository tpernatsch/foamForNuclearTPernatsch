407  forAll(dictEntries,zoneI)

410 
411  // For compatibility with OFFBEAT material card, just read

413

426  nuList.set(zoneI,new scalar(nu.value()));

506  divSigmaExp_ = fvc::div(sigmaD_);

512  fvc::laplacian(2*mu_ + lambda_, disp_, "laplacian(DD,D)");
513  }

518  }
519  mesh_.setFluxRequired(disp_.name());
