56 scalar scalarInterpolateTableGF::integral(scalar k) const
57 {
58  scalar res = 0;
59 
60  for (int i=1; i < xValues_.size(); i++)
61  {
62  scalar dx = pow(xValues_[i], k) - pow(xValues_[i-1], k);
63 
64  if (method_ == STEP)
65  {
66  res += data_[i-1]*dx;
67  }
68  else if (method_ == LINEAR)
69  {
70  // Trapezoidal integration
71  res += 0.5*(data_[i-1] + data_[i])*dx;
72  }
73  }
74 
75  return res;
76 }
