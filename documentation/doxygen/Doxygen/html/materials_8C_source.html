261 void Foam::materials::updateDensity()
262 {
263  if(mesh_.foundObject<fvMesh>("referenceMesh"))
264  {
265  //- Constant reference to reference volumes
266  const fvMesh& refMesh = mesh_.lookupObject<fvMesh>("referenceMesh");
267  const scalarField& Vref = refMesh.V();
268 
269  forAll(mesh_.cellZones(), zoneI)
270  {
271  const labelList& addr = mesh_.cellZones()[zoneI];
272 
273  forAll(addr, i)
274  {
275  const label cellI = addr[i];
276 
277  //- Rescale the cell density with the ref/new volume ratio
278  rho_[cellI] = rho_[cellI]
279  * Vref[cellI]
280  / mesh_.V()[cellI];
281  }
282  }
283 
284  rho_.correctBoundaryConditions();
285  }
286 }

288 Foam::scalar Foam::materials::nextDeltaT()
289 {
290  scalar nextDeltaT(0.0);
291  scalar currentDeltaT = mesh_.time().deltaT().value();
292  scalar deltaTMultiplier(1.0);
293 
294  const scalarField& rhoi = rho_.internalField();
295  const scalarField& rhoOld = rho_.oldTime().internalField();
296  const scalarField& Cpi = Cp_.internalField();
297  const scalarField& CpOld = Cp_.oldTime().internalField();
298  const scalarField& ki = k_.internalField();
299  const scalarField& k_Old = k_.oldTime().internalField();
300 
301  const scalar dRhoPct = max((rhoi - rhoOld) / rhoOld * 100);
302  const scalar dCpPct = max((Cpi - CpOld) / rhoOld * 100);
303  const scalar dKPct = max((ki - k_Old) / k_Old * 100);
304 
305  scalar maxDelta = max(dRhoPct, max(dCpPct, dKPct));
306  deltaTMultiplier = maxDeltaMatTol_ / (maxDelta + SMALL);
307  nextDeltaT = deltaTMultiplier * currentDeltaT;
308 
309 
310  Info<< "Maximum deltaT calculated by relative material properties increase: "
311  << mesh_.time().timeToUserTime(nextDeltaT)
312  << " with a maximum material properties increase of " << maxDeltaMatTol_*100 << "%"
313  << endl;
314 
315  return nextDeltaT;
316 }
