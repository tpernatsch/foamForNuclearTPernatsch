50 Foam::latticeMap::latticeMap
51 (
52  const dictionary& dict
53 )
54 :
55  dict_(dict),
56  type_(dict.get<word>("type")),
57  originX_(dict.get<scalarList>("origin")[0]),
58  originY_(dict.get<scalarList>("origin")[1]),
59  pitch_(dict.get<scalar>("pitch")),
60  rotationAngle_(dict.lookupOrDefault<scalar>("rotationAngle", 0)),
61  nx_(dict.get<labelList>("nElements")[0]),
62  ny_(dict.get<labelList>("nElements")[1]),
63  lattice_(dict.get<wordList>("lattice")),
64  xPos_(0),
65  yPos_(0)
66 {
67  if (nx_ * ny_ != lattice_.size())
68  {
69  FatalErrorInFunction
70  << "nElements and lattice size are different"
71  << exit(FatalError);
72  }
73 
74  const scalar summitToSummit(pitch_ * 2.0/sqrt(3.0));
75  const word nullElement("e");
76  for (int i = 0; i < nx_; i++)
77  {
78  for (int j = 0; j < ny_; j++)
79  {
80  if (lattice_[ny_*i+j] != nullElement)
81  {
82  scalar x(0), y(0);
83  if (type_ == "square")
84  {
85  x = pitch_ * (i - (nx_-1.0)/2.0);
86  y = pitch_ * (j - (ny_-1.0)/2.0);
87  }
88  else if (type_ == "hexagonal")
89  {
90  x = 0.75 * summitToSummit * (i - (nx_-1.0)/2.0);
91  y = pitch_ * ((j - (ny_-1.0)/2.0) + 0.5 * (i - (nx_-1.0)/2.0));
92  }
93  else
94  {
95  FatalErrorInFunction
96  << "Lattice type " << type_ << " is not correct. Available lattice types: "
97  << "hexagonal or square."
98  << exit(FatalError);
99  }
100  xPos_.append(
101  originX_ + x*cos(rotationAngle_) - y*sin(rotationAngle_)
102  );
103  yPos_.append(
104  originY_ + x*sin(rotationAngle_) + y*cos(rotationAngle_)
105  );
106  }
107  }
108  }
109 }
