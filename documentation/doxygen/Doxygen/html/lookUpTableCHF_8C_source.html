105 Foam::scalar Foam::criticalHeatFluxModels::lookUpTableCHF::value
106 (
107  const label& celli
108 ) const
109 {
110 
111  // - Pointers breaking encapsulation - needed
112  if (FFPairPtr_ == nullptr)
113  {
114  HashTable<const FFPair*> FFPairs(pair_.mesh().lookupClass<FFPair>());
115  FFPairPtr_ = FFPairs[FFPairs.toc()[0]];
116  }
117 
118  // ---------------------------------- //
119  //--- Refs --//
120  const scalar& xi(mass_flow_quality_[celli]);
121  const scalar& pi(p_[celli]);
122  const scalar& rhoLi(rhoL_[celli]);
123  const scalar& rhoVi(rhoV_[celli]);
124  const scalar& uLi(uL_[celli]);
125  const scalar& uVi(uV_[celli]);
126  const scalar& aLi(normalizedL_[celli]);
127  const scalar& Dhi(Dh_[celli]);
128 
129  // Note that in this case, the fluid1 and fluid2 must be liquid and vapor, respectively.
130  const fluid& fluid1(FFPairPtr_->fluid1());
131  const fluid& fluid2(FFPairPtr_->fluid2());
132  // Calculate the saturated variables field.
133  const volScalarField& Tsat(FFPairPtr_->iT());
134  volScalarField hLsat(fluid1.thermo().he(p_, Tsat));
135  // Calculate the latent heat in this cell
136  const scalar Li(mag(FFPairPtr_->L()[celli]));
137  // Calculate the liquid and vapor saturated enthalpy in this cell.
138  scalar hLsati(hLsat[celli]);
139  scalar hVsati(hLsat[celli] + Li);
140  // Calculate the mass flow rate in this cell
141  scalar massFlowi(rhoLi*uLi*aLi+rhoVi*uVi*(1-aLi));
142  // Calculate the liquid and vapor enthalpy in this cell.
143  scalar hLi(fluid1.thermo().he()[celli]);
144  scalar hVi(fluid2.thermo().he()[celli]);
145  // Calculate the mixture enthalpy in this cell.
146  // Since the enthalpy of vapour phase is always similar to saturated properties, we assume the enthalpy of vapour phase equal to saturated property.
147 // scalar hMixi(xi*hLi+(1.0-xi)*hVi);
148  scalar hMixi(xi*hLi+(1.0-xi)*hVsati);
149  // Calculate the equilibrium quality in this cell
150  scalar EquilibriumQualityi((hMixi-hLsati)/Li);
151 
152  //convert pressure to KPa, which is used in CHF look-up table
153  scalar pk(pi/1000.0);
154 
155  // Interpolate 3D table with pressure
156  FieldField<Field, scalar> pData(pTable_(pk));
157 
158  // Create 2D table (x is massFlowRate, y is quality)
159  scalarFieldInterpolateTableGF gTable(massFlowRateValues_, pData, gMethod_);
160 
161  // Interpolate 2D table with mass flow rate
162  scalarField gData(gTable(massFlowi));
163 
164  // Create 1D table (main coordinate is quality)
165  scalarInterpolateTableGF xeTable(qualityValues_, gData, xeMethod_);
166 
167  // Interpolate 1D table and get final value of critical heat flux.
168  // Then convert kW/m2 ----> W/m2
169  scalar QCHF(xeTable(EquilibriumQualityi)*1e3);
170 
171  // Implement the correction
172  // K1 for large diameter
173  // K2 for rod bundle geometries
174  if (PitchToDiameter_ != 1.0)
175  {
176  scalar K2((2.0*PitchToDiameter_ - 1.5)*exp(-pow(abs(EquilibriumQualityi),3.0)/2.0));
177  QCHF = QCHF * K2;
178  }
179  else
180  {
181  scalar K1(max(0.6,sqrt(0.008/Dhi)));
182  QCHF = QCHF * K1;
183  }
184 
185  return QCHF;
186 }
