126 void Foam::thermalSubSolver::calcEnthalpy()
127 {
128  if(!enthalpy_.valid())
129  {
130  enthalpy_.set
131  (
132  new volScalarField
133  (
134  IOobject
135  (
136  "enthalpy",
137  mesh_.time().timeName(),
138  mesh_,
139  IOobject::NO_READ,
140  IOobject::AUTO_WRITE
141  ),
142  mesh_,
143  dimensionedScalar("enthalpy", dimArea/(dimTime*dimTime), 0),
144  zeroGradientFvPatchField<scalar>::typeName
145  )
146  );
147  }
148 
149  //Internal field of T, Cp
150  const scalarField& TiOld = T_.oldTime().internalField();
151  const scalarField& Ti = T_.internalField();
152 
153  const scalarField& CpiOld = mat_.Cp().oldTime().internalField();
154  const scalarField& Cpi = mat_.Cp().internalField();
155 
156  // Enthalpy computation
157  const labelListList& matAddrList(mat_.matAddrList());
158  forAll(mat_.materialsList(), i)// iterate over materials
159  {
160  if(isA<fuelMaterial>(mat_.materialsList()[i]))
161  {
162  const labelList& addr(matAddrList[i]);
163 
164  //Loop over cells of fuelMaterial
165  forAll(addr, j)
166  {
167  const label cellI = addr[j];
168  //- dT
169  double dT = Ti[cellI] - TiOld[cellI];
170  //- Mean Cp times dT = enthalpy variation.
171  double dH = (CpiOld[cellI] + Cpi[cellI]) / 2 * dT;
172  //- Add to current time the enthalpy variation
173  enthalpy_()[cellI] += dH;
174  }
175  }
176  }
177 
178  enthalpy_->correctBoundaryConditions();
179 }

182 void Foam::thermalSubSolver::computeGradT()
183 {
184  // This function computes gradT correcting the mistake done by OpenFOAM at
185  // the regionCoupled boundaries. Problem is: when T is interpolated at the
186  // boundaries in case of a regionCoupled boundary the mesh weight are used
187  // rather than the correct weights (that account for hGap).
188  // Workaround: correct the boundary values of Tsurf with the one of T_ that
189  // have been correctly calculated with the coupled boundary condition, and
190  // then compute the grad with Tsurf.
191 
192  surfaceScalarField Tsurf(fvc::interpolate(T_));
193 
194  forAll(mesh_.boundary(), patchID)
195  {
196  if(isType<regionCoupledOFFBEATFvPatch>(mesh_.boundary()[patchID]))
197  {
198  forAll(mesh_.boundary()[patchID], faceID)
199  {
200  Tsurf.boundaryFieldRef()[patchID][faceID] =
201  T_.boundaryField()[patchID][faceID];
202  }
203  }
204  }
205 
206  gradT_ = fvc::grad(Tsurf);
207 }

341 {
342 
343  bool converged(false);
344 
345  const dictionary& stressControl =
346  mesh_.solutionDict().subDict("stressAnalysis");
347 
348  bool useRelRes(stressControl.lookupOrDefault("useRelResT", true));
349 
350  scalar convergenceTolerance
351  (
352  stressControl.lookupOrDefault("T", 1e-6)
353  );
354 
355  scalar absErrTolerance
356  (
357  stressControl.lookupOrDefault("absErrT", 0.0)
358  );
359 
360  if(absErr_ > absErrTolerance)
361  {
362  if ( useRelRes )
363  {
364  scalar relConvergenceTolerance
365  (
366  readScalar(stressControl.lookup("relT"))
367  );
368 
369  // Consider converged if both initialResidual and relResidual are
370  // below threshold or if one of the two is 10 times below threshold.
371  converged =
372  (
373  initialResidual_ < convergenceTolerance
374  &&
375  relResidual_ < relConvergenceTolerance
376  )
377  or
378  (
379  initialResidual_ < convergenceTolerance/10
380  )
381  or
382  (
383  relResidual_ < relConvergenceTolerance/10
384  )
385  ;
386  }
387  else
388  {
389  // In this case only the residual computed by OpenFOAM is accounted for
390  // the convergence monitoring.
391  converged =
392  (
393  initialResidual_ < convergenceTolerance
394  );
395  }
396  }
397  else
398  {
399  converged = true;
400  }
401 
402  return converged;
403 }
