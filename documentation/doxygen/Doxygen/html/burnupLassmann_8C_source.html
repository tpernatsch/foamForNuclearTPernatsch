386 void Foam::burnupLassmann::correct()
387 {
388  if(mapper_ == nullptr)
389  {
390  mapper_ = &mesh_.lookupObject<sliceMapper>("sliceMapper");
391  }
392 
393  // Update average burnup
394  burnupFromPower::correct();
395 
396  // Update inner/outer slice dimensions in case of moving mesh
397  if(mesh_.foundObject<fvMesh>("referenceMesh"))
398  {
399  const labelListList& matAddrList(mat_.matAddrList());
400  forAll(mat_.materialsList(), i)
401  {
402  const labelList& addr(matAddrList[i]);
403 
404  // Compute inner and outer radii
405  calcRinRout(addr);
406  }
407  }
408 
409  // Number of cells
410  scalar Ncells = mesh_.nCells();
411 
412  // Get number of slices
413  const scalar nSlices(mapper_->nSlices());
414 
415  scalarField Bu_av(nSlices, 0.0);
416  scalarField Bu_av_old(nSlices, 0.0);
417  scalarField volPerSlice(nSlices, 0.0);
418 
419  // Burnup in MWd/MTUO2
420  scalarField& Bui = Bu_.ref();
421  const scalarField& BuOldi = Bu_.oldTime().internalField();
422 
423  // Reference to volumes
424  const scalarField& V(mesh_.V());
425 
426  const PtrList<labelList>& sliceAddr(mapper_->sliceAddrList());
427 
428  forAll(sliceAddr, sliceID)
429  {
430  const labelList& addr(sliceAddr[sliceID]);
431 
432  scalarField Vslice(V, addr);
433  scalarField Buslice(Bui, addr);
434  scalarField BuOldislice(BuOldi, addr);
435 
436  Bu_av[sliceID] = gSum(Buslice*Vslice)/gSum(Vslice)/0.8815;
437  Bu_av_old[sliceID] = gSum(BuOldislice*Vslice)/gSum(Vslice)/0.8815;
438  volPerSlice[sliceID] = gSum(Vslice);
439  }
440 
441  // Prepare burnup average increment field
442  scalarField dBu(Ncells, 0.0);
443  forAll(sliceAddr, sliceID)
444  {
445  const labelList& addr(sliceAddr[sliceID]);
446 
447  forAll(addr, addrI)
448  {
449  const label cellI(addr[addrI]);
450 
451  dBu[cellI] =
452  (Bu_av[sliceID] - Bu_av_old[sliceID])/1e3;
453  }
454  }
455 
456  scalar averageBu(0);
457 
458 // READ INPUT AND OLD TIME CONCENTRATIONS
459 
460  const List<bool>& isFuelList(mapper_->isFuel());
461  forAll(sliceAddr, sliceID)
462  {
463  if(isFuelList[sliceID])
464  {
465  //TOD0 enrichment and density fraction initials or current?
466  scalar enrichmentAverage(0.0);
467  scalar densityFractionAverage(0.0);
468 
469  const List<label>& addrSlice_(sliceAddr[sliceID]);
470  scalar localSize(addrSlice_.size());
471 
472  scalarField V_local(V, addrSlice_);
473 
474  averageBu *= 0;
475  forAll(addrSlice_, i)
476  {
477  const label cellI = addrSlice_[i];
478  averageBu += Bui[cellI]*V[cellI]; // cm³
479  enrichmentAverage += enrichment_[cellI]*V[cellI];
480  densityFractionAverage += densityFraction_[cellI]*V[cellI];
481  }
482 
483  reduce(averageBu, sumOp<scalar>());
484  reduce(enrichmentAverage, sumOp<scalar>());
485  reduce(densityFractionAverage, sumOp<scalar>());
486 
487  averageBu /= volPerSlice[sliceID];
488  enrichmentAverage /= volPerSlice[sliceID];
489  densityFractionAverage /= volPerSlice[sliceID];
490 
491  // Old concentrations and total volume
492  scalarRectangularMatrix N_Old_(Nelements_,localSize);
493  scalarField N_average_Old_(Nelements_,0.0);
494  scalarField N_average_(Nelements_,0.0);
495 
496  forAll(addrSlice_, i)
497  {
498  const label cellI = addrSlice_[i];
499 
500  N_Old_[iU235_][i] = N_U235_.oldTime().internalField()[cellI];
501  N_Old_[iU236_][i] = N_U236_.oldTime().internalField()[cellI];
502  N_Old_[iNp237_][i] = N_Np237_.oldTime().internalField()[cellI];
503  N_Old_[iPu238_][i] = N_Pu238_.oldTime().internalField()[cellI];
504  N_Old_[iU238_][i] = N_U238_.oldTime().internalField()[cellI];
505  N_Old_[iPu239_][i] = N_Pu239_.oldTime().internalField()[cellI];
506  N_Old_[iPu240_][i] = N_Pu240_.oldTime().internalField()[cellI];
507  N_Old_[iPu241_][i] = N_Pu241_.oldTime().internalField()[cellI];
508  N_Old_[iPu242_][i] = N_Pu242_.oldTime().internalField()[cellI];
509  N_Old_[iAm241_][i] = N_Am241_.oldTime().internalField()[cellI];
510  N_Old_[iAm243_][i] = N_Am243_.oldTime().internalField()[cellI];
511  N_Old_[iCm242_][i] = N_Cm242_.oldTime().internalField()[cellI];
512  N_Old_[iCm243_][i] = N_Cm243_.oldTime().internalField()[cellI];
513  N_Old_[iCm244_][i] = N_Cm244_.oldTime().internalField()[cellI];
514 
515  for (int isotope = 0; isotope < Nelements_; isotope ++)
516  {
517  N_average_Old_[isotope] += N_Old_[isotope][i]*V_local[i];
518  }
519 
520  }
521 
522  reduce(N_average_Old_, sumOp<scalarField>());
523 
524  N_average_Old_ /= volPerSlice[sliceID];
525 
526  N_average_ = N_average_Old_;
527 
528  // NUCLEAR DATA
529 
530  /*
531  TODO :
532  Here the xs of Pu-240 (Bu-dependent) should be corrected !
533  */
534 
535  // Compute absorption xs
536  scalarField sa_(sc_+sf_);
537 
538  // Decay constants
539  scalarField lambda_(Nelements_,0.0);
540  lambda_[iPu238_] = log(2.) / (8.77e+01 * 365.);
541  lambda_[iPu241_] = log(2.) / (1.43e+01 * 365.);
542  lambda_[iPu242_] = log(2.) / (3.73e+05 * 365.);
543  lambda_[iAm241_] = log(2.) / (4.32e+02 * 365.);
544  lambda_[iAm243_] = log(2.) / (7.37e+03 * 365.);
545  lambda_[iCm242_] = log(2.) / (1.63e+02 );
546  lambda_[iCm244_] = log(2.) / (1.81e+01 * 365.);
547 
548  // Branching ratios for the Am242
549  const scalar BR_EC_Am242 = 0.173;
550  const scalar BR_ng_Am242 = 0.88;
551  const scalar BR_beta_Am242 = 0.827;
552  const scalar BRPu2 = BR_EC_Am242 * BR_ng_Am242;
553  const scalar BRCm2 = BR_beta_Am242 * BR_ng_Am242;
554 
555  // RADIAL SHAPE FUNCTIONS
556 
557  scalarField frU_(localSize, 0.0);
558  scalarField frPu_(localSize, 0.0);
559 
560  scalar normU_ = 0;
561  scalar normPu_ = 0;
562 
563  const volVectorField& C(mesh_.C());
564 
565  // Calculate volume-averaged radial shape functions
566  forAll(addrSlice_, i)
567  {
568  const label cellI = addrSlice_[i];
569 
570  scalar r = pow(pow(C[cellI][0],2)+pow(C[cellI][1],2),0.5);
571 
572  frU_[i] = (1 + 3.45*exp(-3.0*pow((rOut_[cellI] - r)*1e3, 0.45 )));
573  frPu_[i] = (1 + 0.94*exp(-3.0*pow((rOut_[cellI] - r)*1e3, 0.45 )));
574 
575  // Calculate average frU and frPu
576  normU_ += frU_[i]*V_local[i];
577  normPu_ += frPu_[i]*V_local[i];
578  }
579 
580  reduce(normU_, sumOp<scalar>());
581  reduce(normPu_, sumOp<scalar>());
582 
583  normU_ /= volPerSlice[sliceID];
584  normPu_ /= volPerSlice[sliceID];
585 
586  frU_ = frU_/normU_;
587  frPu_ = frPu_/normPu_;
588 
589  // CONVERGENCE LOOP
590 
591  scalarRectangularMatrix N_old_iteration_(Nelements_, localSize);
592  scalarRectangularMatrix N_(Nelements_, localSize);
593  scalarField Qprofile(localSize, 0.0);
594  N_ = N_Old_;
595 
596  scalarField dBu_local_profile(dBu, addrSlice_);
597  scalarField dBu_local_no_profile(dBu, addrSlice_);
598 
599  const scalar alpha = 3.8e-22; // conversion factor
600 
601  scalar E_fis = 3.6156e-22; // MWd
602  scalar SigmaF_tot_ = 0;
603  scalar SigmaF_tot_old = 0;
604  scalarField SigmaF_(localSize,0.0);
605 
606  for (int isotope = 0; isotope < Nelements_; isotope ++)
607  {
608  SigmaF_tot_ += sf_[isotope] * N_average_[isotope]; // cm^-1
609  SigmaF_tot_old += sf_[isotope] * N_average_Old_[isotope]; // cm^-1
610 
611  forAll(addrSlice_, i)
612  {
613  SigmaF_[i] += sf_[isotope]*N_[isotope][i];
614  }
615  }
616 
617  SigmaF_tot_=SigmaF_tot_old;
618 
619  scalarField phi(localSize, 2.4e14*3600*24); // n/cm^2/d
620  scalarField delta_t(localSize, 0);
621  scalar difference = VGREAT;
622 
623  while (difference > precision_)
624  {
625  scalar rho_hm = 10960*densityFractionAverage*0.8815;
626  scalarField phiBu_dbu = rho_hm*1e-6*dBu_local_no_profile/(E_fis*SigmaF_tot_); // kg/cm^3 * MWd/kg / MWd / cm^-1 = 1/cm^2
627 
628  N_old_iteration_ = N_;
629  scalarField S(Nelements_,0.0);
630 
631  // Matrix creation and new compositions calculation
632  forAll(addrSlice_, i)
633  {
634  // const label cellI = addrSlice_[i];
635 
636  delta_t[i] = max(phiBu_dbu[i]/phi[i], SMALL);
637 
638  scalar A = 10960*densityFractionAverage*0.8815/1e6/(alpha*SigmaF_[i]);
639 
640  // Matrix creation
641  scalarSquareMatrix M_(Nelements_,0.0);
642  scalarSquareMatrix Absorption_(Nelements_,0.0);
643  scalarSquareMatrix Capture_(Nelements_,0.0);
644  scalarSquareMatrix Decay_loss_(Nelements_,0.0);
645  scalarSquareMatrix Decay_creation_(Nelements_,0.0);
646  List<double> B_(Nelements_);
647  List<double> X_(Nelements_);
648 
649  for (int isotope = 0; isotope < Nelements_; isotope ++)
650  {
651  Decay_loss_[isotope][isotope]=lambda_[isotope]*delta_t[i];
652  B_[isotope]=N_Old_[isotope][i];
653  if (isotope == iU238_)
654  {
655  Absorption_[isotope][isotope]=1+(frU_[i]*sa_[isotope]+sn2n_)*A*dBu_local_profile[i];
656  }
657  else if (isotope == iPu240_)
658  {
659  Absorption_[isotope][isotope]=1+frPu_[i]*sa_[isotope]*A*dBu_local_profile[i];
660  }
661  else
662  {
663  Absorption_[isotope][isotope]=1+1*sa_[isotope]*A*dBu_local_profile[i];
664  }
665  }
666 
667  Capture_[ iU236_ ][ iU235_ ] = -1 * sc_[iU235_] * A * dBu_local_profile[i];
668  Capture_[ iNp237_ ][ iU236_ ] = -1 * sc_[iU236_] * A * dBu_local_profile[i];
669  Capture_[ iPu238_ ][ iNp237_ ] = -1 * sc_[iNp237_] * A * dBu_local_profile[i];
670  Capture_[ iPu239_ ][ iPu238_ ] = -1 * sc_[iPu238_] * A * dBu_local_profile[i];
671  Capture_[ iPu239_ ][ iU238_ ] = -frU_[i] * sc_[iU238_] * A * dBu_local_profile[i];
672  Capture_[ iPu240_ ][ iPu239_ ] = -1 * sc_[iPu239_] * A * dBu_local_profile[i];
673  Capture_[ iPu241_ ][ iPu240_ ] = -frPu_[i] * sc_[iPu240_] * A * dBu_local_profile[i];
674  Capture_[ iPu242_ ][ iPu241_ ] = -1 * sc_[iPu241_] * A * dBu_local_profile[i];
675  Capture_[ iPu242_ ][ iAm241_ ] = -BRPu2 * sc_[iAm241_] * A * dBu_local_profile[i];
676  Capture_[ iAm243_ ][ iPu242_ ] = -1 * sc_[iPu242_] * A * dBu_local_profile[i];
677  Capture_[ iCm242_ ][ iAm241_ ] = -BRCm2 * sc_[iAm241_] * A * dBu_local_profile[i];
678  Capture_[ iCm243_ ][ iCm242_ ] = -1 * sc_[iCm242_] * A * dBu_local_profile[i];
679  Capture_[ iCm244_ ][ iAm243_ ] = -1 * sc_[iAm243_] * A * dBu_local_profile[i];
680  Capture_[ iCm244_ ][ iCm243_ ] = -1 * sc_[iCm243_] * A * dBu_local_profile[i];
681

685  Decay_creation_[iPu238_][iCm242_] = -lambda_[iCm242_] * delta_t[i];

687  Decay_creation_[iAm241_][iPu241_] = -lambda_[iPu241_] * delta_t[i];

691  solve(X_,M_,B_);

694  for (int isotope = 0; isotope < Nelements_; isotope ++)
695  {
696  N_[isotope][i] = X_[isotope];
697  S[isotope] += N_[isotope][i]*V_local[i];
698  SigmaF_[i] += sf_[isotope]*N_[isotope][i];
699  }
700  }
701 
702  reduce(S, sumOp<scalarField>());
703  S /= volPerSlice[sliceID];
704 
705  N_average_ = max(S, SMALL);
706 
707  // Flux, burnup and power profiles calculation
708  SigmaF_tot_ = 0;
709  for (int isotope = 0; isotope < Nelements_; isotope ++)
710  {
711  SigmaF_tot_ += sf_[isotope] * N_average_[isotope];
712  }
713 
714  scalarField phi_old = rho_hm*1e-6 * dBu_local_no_profile / max((E_fis * SigmaF_tot_old * delta_t), SMALL);
715  scalarField phi_new = rho_hm*1e-6 * dBu_local_no_profile / max((E_fis * SigmaF_tot_ * delta_t), SMALL);
716 
717  phi = (phi_old + phi_new)/2;
718 
719  // Get flux normalized distribution from registry
720  scalarField fluxProfile;
721  if( mesh_.foundObject<volScalarField>("neutronFlux0") )
722  {
723  fluxProfile =
724  scalarField(mesh_.lookupObject<volScalarField>("neutronFlux0")(), addrSlice_);
725  }
726  else
727  {

729  << "This burnup model needs a neutronics subsolver to "
730  << "be set. " << exit(FatalError);
731  }
732

734  Qprofile = fluxProfile * SigmaF_;
735  Qprofile /= gSum(Qprofile * V_local)/volPerSlice[sliceID];
736 
737  // Apply Q profile to local bu
738  dBu_local_profile = Qprofile*dBu_local_no_profile;
739 
740  // Difference calculation for convergence
741  scalar dif(0);
742  scalar maxdif(0);
743  for (int isotope = 0; isotope < Nelements_; isotope ++)
744  {
745  forAll(addrSlice_, i)
746  {
747  // const label cellI = addrSlice_[i];
748 
749  if (N_old_iteration_[isotope][i]>SMALL)
750  {
751  dif=mag((N_[isotope][i]-N_old_iteration_[isotope][i])/N_old_iteration_[isotope][i]);
752 
753  if (dif>maxdif)
754  {
755  maxdif=dif;
756  }
757  }
758  }
759  }
760 
761  reduce(maxdif, maxOp<scalar>());
762 
763  difference = maxdif;
764  }
765 
766  // COMPOSITIONS WRITING
767 
768  forAll(addrSlice_, i)
769  {
770  const label cellI = addrSlice_[i];
771 
772  N_U235_[cellI] = N_[iU235_][i] ;
773  N_U236_[cellI] = N_[iU236_][i] ;
774  N_Np237_[cellI] = N_[iNp237_][i];
775  N_Pu238_[cellI] = N_[iPu238_][i];
776  N_U238_[cellI] = N_[iU238_][i] ;
777  N_Pu239_[cellI] = N_[iPu239_][i];
778  N_Pu240_[cellI] = N_[iPu240_][i];
779  N_Pu241_[cellI] = N_[iPu241_][i];
780  N_Pu242_[cellI] = N_[iPu242_][i];
781  N_Am241_[cellI] = N_[iAm241_][i];
782  N_Am243_[cellI] = N_[iAm243_][i];
783  N_Cm242_[cellI] = N_[iCm242_][i];
784  N_Cm243_[cellI] = N_[iCm243_][i];
785  N_Cm244_[cellI] = N_[iCm244_][i];
786 
787  formFactor_[cellI] = Qprofile[i];
788 
789  // Calculate Bu as MWd/MTUO2 and MWd/MTU
790  // It might duplicate the field Bu, but it can be useful for example
791  // when using "fromList" as "burnupType". In that case the field Bu
792  // represents only the average burnup
793  const scalar Buold_ = Bu_av_old[sliceID]*0.8815;
794  const scalar Bunew_ = Bu_av[sliceID]*0.8815;
795  const scalar BukgUO2_old = Bu_.oldTime().internalField()[cellI];
796

798  Bui[cellI] = newBu;
799  BukgU_[cellI] = newBu/0.8815;
800  }
801  }
802  }
