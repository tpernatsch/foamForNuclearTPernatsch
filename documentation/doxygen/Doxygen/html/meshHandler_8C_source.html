53 Foam::meshHandler::meshHandler(const Time& runTime)
54 {
55  // Read meshHandler dict. Will be used later
56  IOdictionary multiRegionCouplingDict
57  (
58  IOobject
59  (
60  "multiRegionCouplingDict",
61  runTime.time().constant(),
62  runTime.db(),
63  IOobject::MUST_READ,
64  IOobject::NO_WRITE
65  )
66  );
67 
68  IOdictionary controlDict
69  (
70  IOobject
71  (
72  "controlDict",
73  runTime.system(),
74  runTime.db(),
75  IOobject::MUST_READ,
76  IOobject::NO_WRITE
77  )
78  );
79 
80  mappingDict_ = multiRegionCouplingDict.subDict("mappings");
81 
82  // First create all the meshes -> they are created and handled by meshHandler
83 
84  wordList allRegions(controlDict.subDict("regionSolvers").toc());
85  wordList meshNames(0);
86 
87  forAll(allRegions, regioni)
88  {
89  word solverName(controlDict.subDict("regionSolvers").get<word>(allRegions[regioni]));
90  if (solverName != "multiPhysicsSolver")
91  {
92  meshNames.append(allRegions[regioni]);
93  }
94  }
95 
96  // If multiPhysicsSolvers are present, also add those extra solvers
97 
98  if (multiRegionCouplingDict.found("multiPhysicsSolvers"))
99  {
100  forAll(multiRegionCouplingDict.subDict("multiPhysicsSolvers").toc(), MPsolvI) // loop over all multiphysics solvers
101  {
102  wordList subSolvers // get list of solvers belonging to multiphysics solver i
103  (
104  multiRegionCouplingDict
105  .subDict("multiPhysicsSolvers")
106  .subDict(multiRegionCouplingDict.subDict("multiPhysicsSolvers").toc()[MPsolvI])
107  .subDict("solvers").toc()
108  );
109 
110  forAll(subSolvers, solvI) // loop ovcer subsolvers and add them to list of meshes to create only if they are not sub-multiphysicssolvers
111  {
112  if (multiRegionCouplingDict
113  .subDict("multiPhysicsSolvers")
114  .subDict(multiRegionCouplingDict.subDict("multiPhysicsSolvers").toc()[MPsolvI])
115  .subDict("solvers")
116  .get<word>(subSolvers[solvI]) != "multiPhysicsSolver"
117  )
118  {
119  meshNames.append(subSolvers[solvI]);
120  }
121  }
122  }
123  }
124 
125  meshes_.setSize(meshNames.size());
126 
127  forAll(meshNames, meshI)
128  {
129  meshes_.set
130  (
131  meshI,
132  Foam::dynamicFvMesh::New
133  (
134  IOobject
135  (
136  meshNames[meshI],
137  runTime.name(),
138  runTime,
139  IOobject::MUST_READ
140  )
141  )
142  );
143  }
144 
145  // Set size of list of (possibly) baffleLess meshes
146  mappingMeshes_.setSize(meshes_.size());
147  // Set size of list of couplingFields
148  scalarCouplingFields_.setSize(meshes_.size());
149  vectorCouplingFields_.setSize(meshes_.size());
150  couplingFields_.setSize(meshes_.size());
151 
152  createBaffleLessMeshes(runTime);
153  //createCouplingFields(runTime);
154 
155 
156  // Create mappings
157  mappingList_.setSize(meshes_.size());
158 
159  forAll(meshes_, i)
160  {
161  mappingList_.set
162  (
163  i,
164  new PtrList<meshToMesh>
165  (
166  meshes_.size()
167  )
168  );
169  }
170 
171  // Loop over all regions
172  forAll(meshes_, i)
173  {
174  // Check if i find entry in the meshHandler dict
175  if (mappingDict_.found(meshes_[i].name()))
176  {
177  // Loop over all other regions
178  forAll(meshes_, j)
179  {
180  // Check if i find entry in the meshHandler subDict
181  if (mappingDict_.subDict(meshes_[i].name()).found(meshes_[j].name()))
182  {
183  bool removeBaffles(false);
184  if(runTime.controlDict().found("removeBaffles"))
185  {
186  removeBaffles= runTime.controlDict().
187  subDict("removeBaffles").
188  getOrDefault<bool>(meshes_[j].name(),false);
189  }
190  mappingList_[i].set
191  (
192  j,
193  new meshToMesh
194  (
195  meshes_[i],
196  removeBaffles
197  ? mappingMeshes_[j]
198  : meshes_[j],
199  Foam::meshToMesh::interpolationMethod::imCellVolumeWeight, // for now hard-coded, possibly implement meshHandler type via dict
200  Foam::meshToMesh::procMapMethod::pmAABB,
201  false
202  )
203  );
204  }
205  }
206  }
207  }
208 }

229  List<word> targetFields(iTojRegionmappingDict_.getOrDefault<List<word>>("targetFields", List<word>::null()));
230  List<word> sourceFields(iTojRegionmappingDict_.getOrDefault<List<word>>("sourceFields",List<word>::null()));
231  // List<word> fieldTypes(iTojRegionmappingDict_.getOrDefault<List<word>>("fieldTypes",List<word>::null()));

233  bool removeBaffles(false);
234 
235  if (runTime.controlDict().found("removeBaffles"))
236  {
237  const dictionary& removeBafflesDict(runTime.controlDict().subDict("removeBaffles"));
238 
239  if (removeBafflesDict.getOrDefault<bool>(meshes_[j].name(), false))
240  {
241  forAll(sourceFields, sourcei)
242  {
243  sourceFields[sourcei]+=".baffleLess";
244  }
245  removeBaffles = true;
246  }
247  }
248 
249  if (targetFields.size()!=sourceFields.size())
250  {
251  FatalErrorInFunction

256  << exit(FatalError);

259  {
260  forAll(sourceFields, fieldi)
261  {
262  map<scalar>(sourceFields[fieldi], targetFields[fieldi],i, j, removeBaffles);
263  map<vector>(sourceFields[fieldi], targetFields[fieldi],i, j, removeBaffles);
264  map<tensor>(sourceFields[fieldi], targetFields[fieldi],i, j, removeBaffles);

386 void Foam::meshHandler::createBaffleLessMeshes(const Time& runTime)
387 {
388  forAll(meshes_, regioni)
389  {
390  autoPtr<dynamicFvMesh> baffleLessMesh;
391 
392  bool removeBaffleBool(false);
393 
394  if (runTime.controlDict().found("removeBaffles"))
395  {
396  const dictionary& removeBafflesDict = runTime.controlDict().subDict("removeBaffles");
397  removeBaffleBool = removeBafflesDict.getOrDefault<bool>(meshes_[regioni].name(), "false");
398  }
399 
400  if (removeBaffleBool)
401  {
402 
403  Info<< "Creating baffleless mesh for region " << meshes_[regioni].name()
404  << endl;
405  if (UPstream::nProcs() > 1)
406  {
407  Info<< "WARNING: the removeBaffles feature is guaranteed to work only "
408  << "if each processor contains complete pairs of master and slave "
409  << "faces of baffe patches! Otherwise, projection artifacts can "
410  << " still occur" << nl << endl;
411  }
412 
413 
414  //- Copy internal geometric data from fluidMesh
415  pointField points(meshes_[regioni].points());
416  faceList faces(meshes_[regioni].faces());
417  cellList cells(meshes_[regioni].cells());
418 
419  //- Copy boundary geometric data from fluidMesh
420  const polyBoundaryMesh& patches = meshes_[regioni].boundaryMesh();
421  wordList patchNames(patches.names());
422  List<polyPatch*> pList;
423  forAll(patchNames, i)
424  {
425  pList.append
426  (
427  new polyPatch
428  (
429  patches[patchNames[i]],
430  patches
431  )
432  );
433  }
434 
435  //- Assemble mesh as copy from points, faces, cells with no boundary
436  baffleLessMesh.reset
437  (
438  Foam::dynamicFvMesh::New
439  (
440  Foam::IOobject
441  (
442  IOobject::groupName(meshes_[regioni].name(), "baffleLess"),
443  runTime.timeName(),
444  runTime,
445  Foam::IOobject::NO_READ
446  )
447  // std::move(points),
448  // std::move(faces),
449  // std::move(cells)
450  )
451  );
452 
453  //- Add boundary data
454  baffleLessMesh().addFvPatches(pList);
455 
456  //- One of the main issues when passing the actual runTime to removeBaffles
457  // was that the functionObjects execution flags were getting reset to
458  // false. Probably other aspects of runTime were getting modified too. To
459  // avoid all these issues, let removeBaffles operate with a dummy runTime
460  // created on the fly, which is deleted afterwards
461 
462  autoPtr<Time> dummyRunTimePtr
463  (
464  new Time
465  (
466  Foam::Time::controlDictName,
467  meshes_[regioni].time().rootPath(),
468  meshes_[regioni].time().caseName()
469  )
470  );
471  removeBaffles(baffleLessMesh(), dummyRunTimePtr());
472  dummyRunTimePtr.clear();
473 
474  //- Copy mesh zones
475  List<pointZone*> pointZonesNB(0);
476  List<faceZone*> faceZonesNB(0);
477  List<cellZone*> cellZonesNB(0);
478  forAll(meshes_[regioni].pointZones().names(), i)
479  {
480  word name(meshes_[regioni].pointZones().names()[i]);
481  const pointZone& origZone(meshes_[regioni].pointZones()[name]);
482  pointZonesNB.append
483  (
484  new pointZone
485  (
486  name,
487  origZone,
488  i,
489  meshes_[regioni].pointZones()
490  )
491  );
492  }
493  forAll(meshes_[regioni].faceZones().names(), i)
494  {
495  word name(meshes_[regioni].faceZones().names()[i]);
496  const faceZone& origZone(meshes_[regioni].faceZones()[name]);
497  faceZonesNB.append
498  (
499  new faceZone
500  (
501  name,
502  origZone,
503  origZone.flipMap(),
504  i,
505  meshes_[regioni].faceZones()
506  )
507  );
508  }
509  forAll(meshes_[regioni].cellZones().names(), i)
510  {
511  word name(meshes_[regioni].cellZones().names()[i]);
512  const cellZone& origZone(meshes_[regioni].cellZones()[name]);
513  cellZonesNB.append
514  (
515  new cellZone
516  (
517  name,
518  origZone,
519  i,
520  meshes_[regioni].cellZones()
521  )
522  );
523  }
524  baffleLessMesh().addZones(pointZonesNB, faceZonesNB, cellZonesNB);
525  }
526  else
527  {
528  baffleLessMesh = nullptr;
529  }
530 
531  mappingMeshes_.set(regioni,baffleLessMesh);
532  }
533 }

537 {
538 
539  const IOdictionary couplingDict
540  (
541  IOobject
542  (
543  "multiRegionCouplingDict",
544  runTime.time().constant(),
545  runTime.db(),
546  IOobject::MUST_READ,
547  IOobject::NO_WRITE
548  )

550
