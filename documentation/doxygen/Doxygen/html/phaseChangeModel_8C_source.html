271 void Foam::phaseChangeModel::limitMassTransfer()
272 {
273  //- Prevent removing mass from phases that are not present in a cell
274  dmdt_ = posPart(dmdt_)*pos(fluid1_) + negPart(dmdt_)*pos(fluid2_);
275 
276  /*
277  Given that the continuity equations are:
278 
279  ddt(alpha1*rho1) + div(alphaRhoPhi1) = -dmdt
280  ddt(alpha2*rho2) + div(alphaRhoPhi2) = dmdt
281 
282  and that they are solved explicitly (albeit with the whole MULES limiter
283  step before the explicit solution), there should be a maximum allowable
284  limit on the value of the mass transfer term. This can be computed from
285  the continuity equations assuming the maximum allowable ddt within a
286  single time-step (i.e. that ddt that would make my phase fraction change
287  from its current value to 0 in one time step). Due to the way the
288  equations are implemented, a maximum on dmdt is derived from the
289  continuity equation of phase 1 while a minimum is derived from that of
290  phase 2. Continuity errors are (for now) not considered.
291  */
292  if (pimple_.dict().found("massTransferSafetyFactor"))
293  {
294  scalar SF(pimple_.dict().get<scalar>("massTransferSafetyFactor"));
295 
296  scalar dt(mesh_.time().deltaT().value());
297  const scalarField& V(mesh_.V());
298  scalar totV(0);
299  forAll(V, i)
300  {
301  totV += V[i];
302  }
303 
304  scalarField div1 = fvc::div(fluid1_.alphaRhoPhi())().primitiveField();
305  scalarField div2 = fvc::div(fluid2_.alphaRhoPhi())().primitiveField();
306 
307  forAll(dmdt_, i)
308  {
309  scalar& dmdt(dmdt_[i]);
310  if (dmdt > 0.0)
311  {
312  const scalar& Vi(V[i]);
313  scalar maxDmdt
314  (
315  max(SF*fluid1_[i]*fluid1_.rho()[i]/dt-div1[i], 0.0)
316  );
317  scalar dmdt0(dmdt);
318  dmdt = min(dmdt, maxDmdt);
319  if (pimple_.finalIter())
320  dmLostToLimiter_ += (dmdt0-dmdt)*Vi*dt;
321  }
322  else if (dmdt < 0.0)
323  {
324  const scalar& Vi(V[i]);
325  scalar minDmdt
326  (
327  min(-SF*fluid2_[i]*fluid2_.rho()[i]/dt+div2[i], 0.0)
328  );
329  scalar dmdt0(dmdt);
330  dmdt = max(dmdt, minDmdt);
331  if (pimple_.finalIter())
332  dmLostToLimiter_ += (dmdt0-dmdt)*Vi*dt;
333  }
334  }
335 
336  /*if (pimple_.finalIter())
337  {
338  Info<< "Cumulative dm lost to limiter = "
339  << (dmLostToLimiter_)
340  << " kg/m3" << endl;
341  }*/
342  }
343 
344  dmdt_.correctBoundaryConditions();
345 }

347 void Foam::phaseChangeModel::correctInterfacialTemperature()
348 {
349  if (mesh_.relaxField("T.interface"))
350  iT_.storePrevIter();
351  if (pimple_.dict().found("maxTInterfaceDdt"))
352  {
353  scalar maxDTdt
354  (
355  pimple_.dict().get<scalar>
356  (
357  "maxTInterfaceDdt"
358  )
359  );
360 
361  scalar dt(mesh_.time().deltaTValue());
362  saturationPtr_->correctField(iT_, "TSat");
363 
364  forAll(iT_, i)
365  {
366  const scalar& iT0(iT_.oldTime()[i]);
367  scalar& iT(iT_[i]);
368  scalar dTdt(min(max((iT-iT0)/dt, -maxDTdt), maxDTdt));
369  iT = iT0 + dTdt*dt;
370  }
371  }
372  else
373  {
374  saturationPtr_->correctField(iT_, "TSat");
375  }
376 
377  iT_.relax();
378 }

382 void Foam::phaseChangeModel::correct()
383 {
384  //- Limit interfacial area so boiling can start
385  // (very crude, it's the best I have for now)
386  limitInterfacialArea();
387 
388  //- Update saturation temperature
389  correctInterfacialTemperature();
390 
391  //- Update latent heat
392  myOps::storePrevIterIfRelax(L_);
393  latentHeatPtr_->correctField(L_);
394  L_.relax();
395 
396  //- Calculate dmdtI <- depends on the actual run-time selected phaseChange
397  // model
398  correctInterfacialDmdt();
399 
400  //- Update total dmdt and relax
401  dmdt_.storePrevIter();
402  dmdt_ = dmdtI_ + dmdtW_;
403  limitMassTransfer();
404  dmdt_.relax();
405 
406  //- The rest of this function is to set heSources_ in an
407  // energy-conservative way
408  //- Refs and fields
409  const volScalarField& he1(fluid1_.thermo().he());
410  const volScalarField& he2(fluid2_.thermo().he());
411  const volScalarField& T1(fluid1_.thermo().T());
412  const volScalarField& T2(fluid2_.thermo().T());
413  volScalarField& htc1(pair_.htc(fluid1_.name()));
414  volScalarField& htc2(pair_.htc(fluid2_.name()));
415  volScalarField he1I(fluid1_.thermo().he(p_, iT_));
416  volScalarField he2I(fluid2_.thermo().he(p_, iT_));
417 
418  /*-----------------------------------------------------------------------*/
419 
420  volScalarField c1
421  (
422  IOobject
423  (
424  "heDmdtSourceCoeff."+fluid1_.name(),
425  mesh_.time().timeName(),
426  mesh_
427  ),
428  mesh_,
429  dimensionedScalar("", dimDensity/dimTime, 0),
430  zeroGradientFvPatchScalarField::typeName
431  );
432  volScalarField c2
433  (
434  IOobject
435  (
436  "heDmdtSourceCoeff."+fluid2_.name(),
437  mesh_.time().timeName(),
438  mesh_
439  ),
440  mesh_,
441  dimensionedScalar("", dimDensity/dimTime, 0),
442  zeroGradientFvPatchScalarField::typeName
443  );
444 
445  //- Fraction of total dmdt that is due to wall boiling (can be negative,
446  // e.g. vapour at saturation but sub-cooled boiling implies that dmdtI
447  // and dmdt (=dmdtI+dmdtW) have opposing signs)
448  scalarField fw(mesh_.cells().size(), 0.0);
449  scalar dmdtMin(1e-2);
450  forAll(mesh_.cells(), i)
451  {
452  const scalar& dmdti(dmdt_[i]);
453  if (mag(dmdti) > dmdtMin)
454  {
455  fw[i] = dmdtW_[i]/dmdti;
456  }
457  }
458 
459  //- The htcs are set to 0 in the phase-change region and the mass
460  // transfer enthalpy contribution are accounted for via a SuSp(c, he) term
461  // for both phases. The next for loop is used to calculate these terms.
462  // Nonetheless, for visual purposes, I still want the htcs to have their
463  // model-computed values in the boiling/condensing region just so that I
464  // can have an idea of what is going on in paraView. This storePrevIter is
465  // thus used for chaching purposes (the htcs are restored only in FFPair.C
466  // correct() after the interfacial non-mass-transfer
467  // enthalpy contributions are added).
468  htc1.storePrevIter();
469  htc2.storePrevIter();
470  forAll(mesh_.cells(), i)
471  {
472  const scalar& dmdti(dmdt_[i]);
473  if (mag(dmdti) > dmdtMin)
474  {
475  const scalar& he1i(he1[i]);
476  const scalar& he2i(he2[i]);
477  const scalar& iTi(iT_[i]);
478  const scalar& iAi(iA_[i]);
479  const scalar& Li(L_[i]);
480  const scalar& fwi(fw[i]);
481  scalar& htc1i(htc1[i]);
482  scalar& htc2i(htc2[i]);
483  scalar& c1i(c1[i]);
484  scalar& c2i(c2[i]);
485  scalar dmdtIi((1.0-fwi)*dmdti);
486  scalar dmdtWi(fwi*dmdti);
487  scalar q1i(iAi*htc1i*(T1[i]-iTi));
488  scalar q2i(iAi*htc2i*(T2[i]-iTi));
489 
490  //- Interfacial contribution
491  /*
492  Nothing guarantees that the computed dmdtI
493  already satisfies energy conservation in the sense that
494  dmdtIi = (q1i+q2i)/Li, as it might have: 1) been computed
495  via other approaches (e.g. gas kinetic theory); 2) been
496  under-relaxed after its calculation, even if it was done
497  via energy conservative approaches (e.g. heat conduction
498  limited). Thus, to be consistent, I need to adjust the
499  interfacial heat fluxes by the ratio of the original
500  dmdt had it been computed via a perfectly
501  energy-conservative approach (i.e. dmdtIi0) to the actual
502  dmdti which is not assured to satisfy dmdtiI = (q1i+q2i)/Li.
503  In this way I assure energy conservation when accounting
504  for dmdti in the energy equations, regardless of how dmdti
505  was actually computed (note, there is nothing un-physical
506  about computing dmdt with approaches that are not based
507  on energy conservation, as long as the dmdtI is added to
508  the energy equations in a way that is energy conservative).
509  However, I am not sure that this energy-conservativity
510  necessarily lead to meaningful temperature profiles IF
511  the dmdtI is not computed via approaches other than the
512  heat-conduction limited one. Oh, well!
513  */
514  scalar dmdtIi0((q1i+q2i)/Li);
515  dmdtIi0 =
516  (dmdtIi0 >= 0.0) ?
517  max(dmdtMin, dmdtIi0) :
518  min(-dmdtMin, dmdtIi0);
519  scalar f(dmdtIi/dmdtIi0);
520  scalar dmdtILi(dmdtIi*Li);
521  c1i = (dmdtILi-f*q2i)/he1i;
522  c2i = (dmdtILi-f*q1i)/he2i;
523 
524  //- Wall contribution
525  if (dmdtWi > 0.0)
526  c1i += (dmdtWi*Li)/he1i;
527  else if (dmdtWi < 0.0)
528  c2i += (dmdtWi*Li)/he2i;
529 
530  //- Set htcs to 0 so that the interfacial contribtuion is not
531  // counted twice (added in FFPair.C) in cells that are boiling
532  htc1i = 0;
533  htc2i = 0;
534  }
535  }
536 
537  /*-----------------------------------------------------------------------*/
538 
539  //- This is longer to write but faster than using posPart/negPart
540  // because I do not care about boundary values
541  volScalarField dmdt12
542  (
543  IOobject
544  (
545  "posPartDmdt."+pair_.name(),
546  mesh_.time().timeName(),
547  mesh_
548  ),
549  mesh_,
550  dimensionedScalar("", dimDensity/dimTime, 0),
551  zeroGradientFvPatchScalarField::typeName
552  );
553  volScalarField dmdt21
554  (
555  IOobject
556  (
557  "negPartDmdt."+pair_.name(),
558  mesh_.time().timeName(),
559  mesh_
560  ),
561  mesh_,
562  dimensionedScalar("", dimDensity/dimTime, 0),
563  zeroGradientFvPatchScalarField::typeName
564  );
565  forAll(mesh_.cells(), i)
566  {
567  const scalar& dmdti(dmdt_[i]);
568  if (dmdti > 0.0)
569  dmdt12[i] = dmdti;
570  else if (dmdti < 0.0)
571  dmdt21[i] = -dmdti;
572  }
573  dmdt12.correctBoundaryConditions();
574  dmdt21.correctBoundaryConditions();
575 
576  //- If phase 1 gains enthalpy from the disappearance of phase 2 (e.g. if
577  // phase 1 is liquid, 2 is vapour and the vapour is condensing) then the
578  // enthalpy added to phase 1 is at saturation (i.e. the interfacial
579  // enthalpy, he1I). If phase 1 loses enthalpy from its own disappearance
580  // (e.g. if phase 1 is liquid, 2 is vapour and the liquid is boiling) then
581  // the enthalpy is removed from phase 1 at its current enthalpy (he1).
582  // This strategy is to prevent possible thermal run-aways. Note that
583  // beacuse of this formulation, the enthalpy gain term can be treated
584  // implicitly but the loss term must be treated explcitly. The same logic
585  // is applied to the he source of phase 2. It works regardless of which
586  // phase is liquid and which phase is vapour
587  *(heSources_[he1.name()]) =
588  fvm::SuSp(c1, he1) //- Contrib. from interf. and wall mass transf.
589  + fvm::Sp(dmdt12, he1) //- Contrib. from intrinsic phase change (imp.)
590  - dmdt21*he1I; //- Contrib. from intrinsic phase change (exp.)
591  *(heSources_[he2.name()]) =
592  fvm::SuSp(c2, he2)
593  + fvm::Sp(dmdt21, he2)
594  - dmdt12*he2I;
595 
596  //- Store prev iters for under-relaxation (at the next iteration)
597  dmdtI_.storePrevIter();
598  dmdtW_.storePrevIter();
599 
600  //- Reset wall contribution (it is computed by some
601  // FSHeatTransferCoefficient models but due to how said models work, it
602  // cannot be reset from within them)
603  forAll(mesh_.cells(), i)
604  {
605  dmdtW_[i] = 0;
606  }
607  dmdtW_.correctBoundaryConditions();
608 
609  /*
610  forAll(mesh_.cells(), i)
611  {
612  scalar deltai
613  (
614  he1I[i]+L_[i]-he2I[i]
615  );
616 
617  scalar Xi
618  (
619  (
620  (
621  fluid1_.normalized()[i]*fluid1_.rho()[i]*he1[i]
622  + fluid2_.normalized()[i]*fluid2_.rho()[i]*(he2[i]+deltai)
623  )/
624  (
625  fluid1_.normalized()[i]*fluid1_.rho()[i]
626  + fluid2_.normalized()[i]*fluid2_.rho()[i]
627  ) - he1I[i]
628  )/(he2I[i]+deltai-he1I[i])
629  );
630  Info<< he1[i] << " " << he1I[i] << " " << (he2[i]+deltai) << " "
631  << (he2I[i]+deltai) << " " << L_[i] << " | X = " << Xi << endl;
632  }*/
633 }
