145 void Foam::timeProfile::initializeFMI()
146 {
147 #ifdef isCommDataLayerIncluded
148 
149  word keyFromFMU("nameFromFMU");
150  if (dict_.found(keyFromFMU))
151  {
152  nameFromFMU_ = dict_.get<word>(keyFromFMU);
153 
154  Info<< dict_.dictName()
155  << " FMI input name: " << nameFromFMU_
156  << endl;
157 
158  // Communicating with the FMU
159  commDataLayer& data = commDataLayer::New(runTime_);
160  // Store in data layer and set its initial value to 0 or user defined
161  currentValue_ = dict_.lookupOrDefault<scalar>("initialValue", 0.0);
162  oldValue_ = currentValue_;
163  data.storeObj(
164  currentValue_,
165  nameFromFMU_,
166  commDataLayer::causality::in
167  );
168 
169  isfmiPortSet_ = true;
170  }
171 
172 #else // !isCommDataLayerIncluded
173 
174  FatalErrorInFunction
175  << "FMI mode selected but FMI lib not built or linked to GeN-Foam."
176  << exit(FatalError);
177 
178 #endif // isCommDataLayerIncluded
179 }

194 scalar Foam::timeProfile::value(scalar time)
195 {
196 #ifdef isCommDataLayerIncluded
197  if (isfmiPortSet_)
198  {
199  if (time < runTime_.timeOutputValue()) // Ask for old time value
200  {
201  return(oldValue_);
202  }
203 
204  if (oldTime_ < runTime_.timeOutputValue() - runTime_.deltaT().value()) // Old time is too old
205  {
206  oldTime_ = runTime_.timeOutputValue() - runTime_.deltaT().value();
207  oldValue_ = currentValue_;
208  }
209 
210  commDataLayer& data = commDataLayer::New(runTime_);
211  currentValue_ = data.getObj<scalar>(
212  nameFromFMU_,
213  commDataLayer::causality::in
214  );
215  return(currentValue_);
216  }
217 #endif // isCommDataLayerIncluded
218 
219  return
220  (
221  functionPtr_.valid()
222  ? functionPtr_->value(time - startTime_)
223  : 0.0
224  );
225 }
