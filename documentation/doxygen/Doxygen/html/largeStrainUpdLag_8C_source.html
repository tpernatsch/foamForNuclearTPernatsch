233 void Foam::largeStrainUpdLag::correct()
234 {
235  const dictionary& stressControl =
236  mesh_.solutionDict().subDict("stressAnalysis");
237 
238  // Maximum inner iterations
239  const int nCorr = stressControl.lookupOrDefault<int>("nCorrectors", 1);
240 
241  if (multiMaterialCorrection_)
242  {
243  materialInterface_->correct();
244  }
245 
246  List<Tuple2<label, vector> > refPairs;
247 
248  if (stressControl.found("referencePairs"))
249  {
250  refPairs = List<Tuple2<label, vector> >(stressControl.lookup("referencePairs"));
251  }
252 
253  // Intermediate fields
254  const volScalarField& mu(mat_.mu());
255  const volScalarField& lambda(mat_.lambda());
256  volScalarField twoMuLambda("2mu+lambda", 2*mu + lambda);
257  surfaceScalarField twoMuLambda_f
258  = mechanicsSubSolver::correctedTwoMuLambdaF(twoMuLambda);
259 
260  // Reference to old time
261  // const volVectorField& DDOld(DD_.oldTime());
262 
263  // Inner loop convergence variables
264  int nInnerIter(0);
265  bool convergedInner(false);
266 
267  do
268  {
269  // Store DD field for under-relaxation and residual calculation
270  D_.storePrevIter();
271 
272  DispPrevPrev = DispPrev;
273  DD_.storePrevIter();
274  DispPrev = DD_.prevIter();
275 
276  // Update explicit component of the stress
277  sigmaExp_ =
278  mechanicsSubSolver::correctedSigmaExp(
279  (relJ_*(relFinv_ & sigma_))(), twoMuLambda, gradDD_);
280 
281  // Update the divergence of the explicit component of the stress
282  divSigmaExp_ =
283  mechanicsSubSolver::correctedDivSigmaExp
284  (sigmaExp_, (relJ_*(relFinv_ & sigma_))(), twoMuLambda, twoMuLambda_f, DD_, gradDD_);
285 
286  // Couple implicit contact boundaries (if present)
287  coupleBoundaries(DD_, true);
288 
289  // Define incremental displacement equation
290  fvVectorMatrix DDEqn
291  (
292  fvm::d2dt2(mat_.rho(), DD_)
293  + fvc::d2dt2(mat_.rho().oldTime(), D_.oldTime())
294  ==
295  fvmLaplacianD(twoMuLambda_f, DD_, "laplacian(DD,D)")
296  + divSigmaExp_
297  );
298 
299  // Manipulate matrix based on BC rule (necessary for implicit contact)
300  DDEqn.boundaryManipulate(DD_.boundaryFieldRef());
301 
302  forAll(refPairs, i)
303  {
304  DDEqn.setReference(refPairs[i].first(), refPairs[i].second());
305  }
306 
307  // //-Possible relaxation technique derived from ddt2
308  // {
309  // scalar deltaT = 1e-5;
310  // scalar deltaT0 = 1e-5;
311 
312  // scalar coefft = (deltaT + deltaT0)/(2*deltaT);
313  // scalar coefft00 = (deltaT + deltaT0)/(2*deltaT0);
314  // scalar coefft0 = coefft + coefft00;
315 
316  // scalar rDeltaT2 = 4.0/sqr(deltaT + deltaT0);
317 
318  // {
319  // DDEqn.diag() += (coefft*rDeltaT2)*mesh_.V()*mat_.rho().primitiveField();
320 
321  // DDEqn.source() += rDeltaT2*mesh_.V()*mat_.rho().primitiveField()*
322  // (
323  // coefft0*DispPrev
324  // - coefft00*DispPrevPrev
325  // );
326  // }
327  // }
328 
329  // Relax matrix
330  DDEqn.relax();
331 
332  // Solve equation for DD
333  residual_ = DDEqn.solve().initialResidual();
334 
335  // Un-couple implicit contact boundaries (if present)
336  coupleBoundaries(DD_, false);
337 
338  // Relax field
339  DD_.relax();
340 
341  // Update the total displacement
342  D_ = D_.oldTime() + DD_;
343 
344  // Store initialResidual when in first inner iteration
345  if(nInnerIter==0)
346  {
347  // Rescale residuals:
348  initialResidual_ = residual_;
349  }
350 
351  // Calculate gradient of incremental displacement
352  gradDD_ = mechanicsSubSolver::correctedGradD(twoMuLambda, DD_, gradDD_, sigmaExp_);
353 
354  // Calculate gradient of total displacement
355  gradD_ = gradD_.oldTime() + gradDD_;
356 
357  // Relative deformation gradient
358  relF_ = I + gradDD_.T();
359 
360  // Inverse relative deformation gradient
361  relFinv_ = inv(relF_);
362 
363  // Total deformation gradient
364  F_ = relF_ & F_.oldTime();
365 
366  // Inverse of total deformation gradient
367  Finv_ = inv(F_);
368 
369  // Relative Jacobian (Jacobian of relative deformation gradient)
370  relJ_ = det(relF_);
371 
372  // Jacobian of deformation gradient
373  J_ = relJ_*J_.oldTime();
374 
375  // Calculate epsilon
376  if(strainTensorName_ == "linear")
377  {
378  epsilon_ = symm(gradD_);
379  }
380  else if(strainTensorName_ == "EulerAlmansi")
381  {
382  epsilon_ = 0.5*(I - symm(Finv_.T() & Finv_));
383  }
384  else if(strainTensorName_ == "Hencky")
385  {
386  epsilon_ = 0.5*log(symm(F_.T() & F_));
387  }
388  else if(strainTensorName_ == "GreenLagrange")
389  {
390  epsilon_ = 0.5*(symm(F_.T() & F_) - I);
391  }
392  else
393  {
394  // Error for strain tensor
395  FatalErrorInFunction() << nl
396  << "Possible strain tensors are:" << nl
397  << tab << "- linear" << nl
398  << tab << "- EulerAlmansi" << nl
399  << tab << "- Hencky" << nl
400  << tab << "- GreenLagrange" << nl
401  << exit(FatalError);
402  }
403 
404  // Save matrix coefficient
405  const volScalarField DDEqnA("DDEqnA", DDEqn.A());
406 
407  // Correct constitutive material laws
408  rheo_.correct(sigma(), epsilon(), DD());
409 
410  // Update explicit component of the stress
411  sigmaExp_ =
412  mechanicsSubSolver::correctedSigmaExp(
413  (relJ_*(relFinv_ & sigma_))(), twoMuLambda, gradDD_);
414  sigma_ = symm( ((sigmaExp_ + twoMuLambda*gradDD_) & relF_)/relJ_) ;
415 
416  // Calculate a different residual based on the relative change of D
417  scalar denom = max(gMax(mag(DD_.primitiveField())), SMALL);
418 
419  absErr_ = gMax
420  (
421  mag(DD_.primitiveField() - DD_.prevIter().internalField())
422  );
423 
424  relResidual_ = absErr_/denom;
425 
426  Info << "relResidualD is " << relResidual_ << endl;
427  Info << "absErrD is " << absErr_ << endl;
428 
429  nInnerIter++;
430 
431  convergedInner = mechanicsSubSolver::converged(DD(), residual_);
432 
433  } while
434  (
435  not(convergedInner)
436  && nInnerIter < nCorr
437  );
438 
439  // Transform stress and displacement in cyl. coord. (for output)
440  correctCylindrical();
441  // Transform stress and displacement in sph. coord. (for output)
442  correctSpherical();
443 
444  // Print total force on each face
445  forceSummary();
446 
447  Info<< "Domain average displacement: "
448  << fvc::domainIntegrate(D_).value()/gSum(mesh_.V())
449  << endl << endl;
450 }
