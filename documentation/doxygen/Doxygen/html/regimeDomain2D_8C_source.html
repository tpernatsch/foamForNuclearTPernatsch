69  pointPtrs_(pointPtrs),

73  //Info << " constructing regimeDomain2D ID " << id_ << endl;
74  int N(pointPtrs_.size());
75  forAll(pointPtrs_, i)
76  {
77  const Vector2D<scalar>* p0Ptr(nullptr);
78  const Vector2D<scalar>* p1Ptr(nullptr);
79  if (i < N-1)
80  {
81  p0Ptr = pointPtrs_[i];
82  p1Ptr = pointPtrs_[i+1];
83  }
84  else
85  {
86  p0Ptr = pointPtrs_[N-1];
87  p1Ptr = pointPtrs_[0];
88  }
89  boundaries_.append
90  (
91  regimeBoundary2D
92  (
93  p0Ptr,
94  p1Ptr
95  )
96  );
97  //Info<< " added boundary " << boundaries_[i].p0() << " "
98  // << boundaries_[i].p1() << endl;
99  }
100 
101  //- Determine polygon sign (1 = convex counter-clockwise, -1 = convex
102  // clockwise, 0 = concave (ordering is inconsequential in that scenario))
103  forAll(boundaries_, i)
104  {
105  const regimeBoundary2D& b0(boundaries_[i]);
106  const regimeBoundary2D* b1Ptr(nullptr);
107  if (i == boundaries_.size()-1)
108  b1Ptr = &(boundaries_[0]);
109  else
110  b1Ptr = &(boundaries_[i+1]);
111  const regimeBoundary2D& b1(*b1Ptr);
112  const Vector2D<scalar>& v0(b0.vNorm());
113  const Vector2D<scalar>& v1(b1.vNorm());
114  scalar x(v0[0]*v1[1] - v0[1]*v1[0]);
115  int sign(int(x/max(mag(x), 1e-9)));
116  if (i == 0)
117  {
118  if (sign >= 0)
119  sign_ = 1;
120  else
121  sign_ = -1;
122  }
123  else
124  {
125  if (sign_*sign < 0)
126  {
127  //- Then domain is concave
128  sign_ = 0;
129  break;
130  }
131  }
132  }
133 
134  //- Construct the ray to be used for intercept testing in a manner so
135  // that is not parallel to any boundary in the domain
136  Random rng(-(1922-1991)*(id+1917));
137  bool flag = true;
138  while (flag)
139  {
140  forAll(boundaries_, i)
141  {
142  rayNorm_ = Vector2D<scalar>
143  (
144  rng.sample01<scalar>(),
145  rng.sample01<scalar>()
146  ).normalise();

148  Info << dot << endl;

150  if ((dot < tol) or (dot > 1.0-tol))

152  flag = true;
153  break;
154  }

157  }
158  }

212 bool Foam::regimeMapModels::regimeDomain2D::convexContainsPoint
213 (
214  const Vector2D<scalar>& p
215 ) const
216 {
217  //- The logic here is to perform a cross-product (wherein the thrid cmpt of
218  // the involved vectors is always 0, as these objects are 2-D) between
219  // each of the vectors that can be constructed between each boundary
220  // starting poing p0 and the point under exam p with the corresponding
221  // p0-p1 vector for each boundary, for all boundaries. If the sign of
222  // these products is the same for all boundaries, then the point lies in
223  // the domain. This only works if the domain is convex (i.e. sign_ != 0)
224  forAll(boundaries_, i)
225  {
226  const regimeBoundary2D& b(boundaries_[i]);
227  const Vector2D<scalar> v01(b.v());
228  Vector2D<scalar> v0p(p-b.p0());
229  scalar x(v01[0]*v0p[1] - v01[1]*v0p[0]);
230  if (sign_*x < 0)
231  return false;
232  }
233  return true;
234 }

237 bool Foam::regimeMapModels::regimeDomain2D::concaveContainsPoint
238 (
239  const Vector2D<scalar>& p
240 ) const
241 {
242  //- The idea is to count the number of times a test ray that starts at
243  // p and extends towards infinity intersects domain boundaries. If said
244  // number is odd, the point is inside the domain, otherwise it is outside.
245  // The test ray is rayNorm_. This approax works for both concave and
246  // convex polygons, but it is farily more expensive, so it is used only
247  // for concave shapes in containsPoint(p)
248  int numberOfIntersections= 0;
249  forAll(boundaries_, i)
250  {
251  const regimeBoundary2D& b(boundaries_[i]);
252  const Vector2D<scalar> v01(b.v());
253  Vector2D<scalar> vp0(b.p0()-p);
254  Vector2D<scalar> vp1(b.p1()-p);
255  scalar vxvp0(rayNorm_[0]*vp0[1]-rayNorm_[1]*vp0[0]);
256  scalar vxvp1(rayNorm_[0]*vp1[1]-rayNorm_[1]*vp1[0]);
257  scalar vp0xv01(vp0[0]*v01[1]-vp0[1]*v01[0]);
258 
259  //- The condition on vxvp1 is strictly greater or strictly lesser so to
260  // not consider the intersections with end-points in the count. Only
261  // intersections with start points are considered. The condition on
262  // vp0xv010 is striclty greater than 0 so that points on the edge are
263  // considered outtside of the domain. In my experience with typical
264  // regime maps, this is the best option
265  if ((vp0xv01 > 0.0 and vxvp0 <= 0.0 and vxvp1 > 0.0) or
266  (vp0xv01 < 0.0 and vxvp0 >= 0.0 and vxvp1 < 0.0))
267  numberOfIntersections += 1.0;
268  }
269  return (numberOfIntersections%2 == 1);
270 }
