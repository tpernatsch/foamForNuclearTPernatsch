436 void Foam::FFPair::correct
437 (
438  const bool& correctFluidDynamics,
439  const bool& correctEnergy
440 )
441 {
442  //- Init refs and dispersion marker fields
443  const volVectorField& U1(fluid1_.U());
444  const volVectorField& U2(fluid2_.U());
445  const volScalarField& Dh1(fluid1_.Dh());
446  const volScalarField& Dh2(fluid2_.Dh());
447 
448  //- Update dispersion/continuity
449  scalarField oldDispersion(fluid1_.dispersion());
450  dispersion1Ptr_->correctField(fluid1_.dispersion());
451  scalar f = myOps::relaxationFactor(mesh_, "dispersion");
452  if (f != 1.0)
453  fluid1_.dispersion() = (1-f)*oldDispersion + f*fluid1_.dispersion();
454  fluid2_.dispersion() = 1.0-fluid1_.dispersion();
455  const scalarField& continuity1(fluid2_.dispersion());
456  const scalarField& continuity2(fluid1_.dispersion());
457 
458  //- Update dipsersed/continuous phase fractions
459  alphaDispersed_.primitiveFieldRef() =
460  fluid1_.primitiveField()*fluid1_.dispersion()
461  + fluid2_.primitiveField()*fluid2_.dispersion();
462  alphaDispersed_.correctBoundaryConditions();
463  alphaContinuous_.primitiveFieldRef() =
464  fluid1_.primitiveField()*continuity1
465  + fluid2_.primitiveField()*continuity2;
466  alphaContinuous_.correctBoundaryConditions();
467 
468  //- Update fluid diameter (needs to be done AFTER dispersion update)
469  fluid1_.correctDiameter();
470  fluid2_.correctDiameter();
471 
472  DhDispersed_.primitiveFieldRef() =
473  Dh1.primitiveField()*fluid1_.dispersion()
474  + Dh2.primitiveField()*fluid2_.dispersion();
475  DhDispersed_.correctBoundaryConditions();
476  DhContinuous_.primitiveFieldRef() =
477  Dh1.primitiveField()*continuity1
478  + Dh2.primitiveField()*continuity2;
479  DhContinuous_.correctBoundaryConditions();
480 
481  //- Correct interfacial area density
482  iAPtr_->correctField(iA_);
483 
484  rhoContinuous_.primitiveFieldRef() =
485  fluid1_.rho().primitiveField()*continuity1
486  + fluid2_.rho().primitiveField()*continuity2;
487  rhoContinuous_.correctBoundaryConditions();
488 
489  nuContinuous_.primitiveFieldRef() =
490  fluid1_.thermo().nu()()*continuity1
491  + fluid2_.thermo().nu()()*continuity2;
492  nuContinuous_.correctBoundaryConditions();
493 
494  PrContinuous_.primitiveFieldRef() =
495  fluid1_.Pr()*continuity1
496  + fluid2_.Pr()*continuity2;
497  PrContinuous_.correctBoundaryConditions();
498 
499  //-
500  magUr_ = mag(U1-U2);
501  Re_ = max(magUr_*DhDispersed_/nuContinuous_, minRe_);
502 
503  //- Correct two-phase quantities. I do this on a by-cell basis to
504  // spare myself the need to do annoying pow operations in cells that
505  // are basically single-phase
506  volScalarField& X1(fluid1_.flowQuality());
507  volScalarField& X2(fluid2_.flowQuality());
508  volScalarField& XLM1(fluid1_.XLM());
509  volScalarField& XLM2(fluid2_.XLM());
510  const volScalarField& rho1(fluid1_.rho());
511  const volScalarField& rho2(fluid2_.rho());
512  const volScalarField& mu1(fluid1_.thermo().mu());
513  const volScalarField& mu2(fluid2_.thermo().mu());
514  const volScalarField& magU1(fluid1_.magU());
515  const volScalarField& magU2(fluid2_.magU());
516  forAll(mesh_.cells(), i)
517  {
518  scalar& X1i(X1[i]);
519  scalar& X2i(X2[i]);
520  scalar& XLM1i(XLM1[i]);
521  scalar& XLM2i(XLM2[i]);
522 
523  scalar mDot1i(fluid1_[i]*rho1[i]*magU1[i]);
524  scalar mDot2i(fluid2_[i]*rho2[i]*magU2[i]);
525  scalar mDotSum(max(mDot1i+mDot2i, 1e-69));
526  X1i = mDot1i/mDotSum;
527  X2i = mDot2i/mDotSum;
528  if (X1i < 1e-6)
529  {
530  XLM1i = fluid1_.minXLM();
531  }
532  else if (X2i < 1e-6)
533  {
534  XLM1i = fluid1_.maxXLM();
535  }
536  else
537  {
538  XLM1i =
539  min
540  (
541  max
542  (
543  pow(mu1[i]/mu2[i], 0.1)*
544  pow(X1i/X2i, 0.9)*
545  sqrt(rho2[i]/rho1[i]),
546  fluid1_.minXLM()
547  ),
548  fluid1_.maxXLM()
549  );
550  }
551  if (X2i < 1e-6)
552  {
553  XLM2i = fluid2_.minXLM();
554  }
555  else if (X1i < 1e-6)
556  {
557  XLM2i = fluid2_.maxXLM();
558  }
559  else
560  {
561  XLM2i =
562  min
563  (
564  max
565  (
566  pow(mu2[i]/mu1[i], 0.1)*
567  pow(X2i/X1i, 0.9)*
568  sqrt(rho1[i]/rho2[i]),
569  fluid2_.minXLM()
570  ),
571  fluid2_.maxXLM()
572  );
573  }
574  //- Normalize to account for the fact that the clipping for XLM1 and
575  // XLM2 might differ, so that XML1*XLM2 = 1.0 could be violated. So,
576  // restore it via a normalization
577  scalar c(sqrt(1.0/(XLM1i*XLM2i)));
578  XLM1i *= c;
579  XLM2i *= c;
580  }
581  X1.correctBoundaryConditions();
582  X2.correctBoundaryConditions();
583  XLM1.correctBoundaryConditions();
584  XLM2.correctBoundaryConditions();
585 
586  if (correctFluidDynamics)
587  {
588  //- Correct drag coefficient
589  if (KdPtr_.valid())
590  {
591  myOps::storePrevIterIfRelax(Kd_);
592  KdPtr_->correctField(Kd_);
593  }
594  //- Limit drag coefficient
595  scalar KdFF0
596  (
597  pimple_.dict().lookupOrDefault<scalar>("minKdFF", 1)
598  );
599  scalar KdFF1
600  (
601  pimple_.dict().lookupOrDefault<scalar>("maxKdFF", KdFF0)
602  );
603  scalar a1
604  (
605  pimple_.dict().lookupOrDefault<scalar>("maxAlphaKdFF", 0)
606  );
607  if (KdFF1 != KdFF0 and a1 != 0)
608  {
609  forAll(mesh_.C(), i)
610  {
611  scalar& Kdi(Kd_[i]);
612  scalar Kdlim(KdFF0);
613  scalar a(min(fluid1_.normalized()[i], fluid2_.normalized()[i]));
614  if (a < a1)
615  {
616  scalar c((a1-a)/a1);
617  Kdlim = c*KdFF1 + (1.0-c)*KdFF0;
618  }
619  Kdi = max(Kdi, Kdlim);
620  }
621  }
622  else if (KdFF0 != 0)
623  {
624  forAll(mesh_.C(), i)
625  {
626  scalar& Kdi(Kd_[i]);
627  Kdi = max(Kdi, KdFF0);
628  }
629  }
630  Kd_.correctBoundaryConditions();
631  Kd_.relax();
632 
633  //- Correct virtual mass forces
634  if (virtualMassPtr_.valid())
635  {
636  virtualMassPtr_->correct();
637  }
638  }
639 
640  if (correctEnergy)
641  {
642  //- Correct heat transfer coefficient
643  volScalarField& htc1(*htcs_[fluid1_.name()]);
644  volScalarField& htc2(*htcs_[fluid2_.name()]);
645  if (htc1Ptr_.valid())
646  {
647  myOps::storePrevIterIfRelax(htc1);
648  htc1Ptr_->correctField(htc1);
649  htc1.relax();
650  }
651  if (htc2Ptr_.valid())
652  {
653  myOps::storePrevIterIfRelax(htc2);
654  htc2Ptr_->correctField(htc2);
655  htc2.relax();
656  }
657 
658  //- Correct interfacial temperature and/or mass transfer
659  fluid1_.correctThermoResidualMarkers();
660  fluid2_.correctThermoResidualMarkers();
661  if (phaseChangePtr_.valid())
662  {
663  //- Limits interfacial area, updates interfacial temperature,
664  // latent heat and mass transfer
665  phaseChangePtr_->correct();
666  }
667  else
668  {
669  const volScalarField& pos1(fluid1_.aboveThermoResidualAlpha());
670  const volScalarField& pos2(fluid2_.aboveThermoResidualAlpha());
671  const volScalarField& neg1(fluid1_.belowThermoResidualAlpha());
672  const volScalarField& neg2(fluid2_.belowThermoResidualAlpha());
673  myOps::storePrevIterIfRelax(iT_);
674  iT_ =
675  pos1*pos2*
676  (
677  htc1*fluid1_.thermo().T() + htc2*fluid2_.thermo().T()
678  )/
679  (
680  max
681  (
682  htc1+htc2,
683  dimensionedScalar
684  (
685  "",
686  dimPower/dimArea/dimTemperature,
687  1e-6
688  )
689  )
690  )
691  + pos1*neg2*fluid1_.thermo().T()
692  + neg1*pos2*fluid2_.thermo().T();
693  if (!firstTimeStepAndIter_)
694  iT_.relax();
695  else
696  firstTimeStepAndIter_ = false;
697  }
698 
699  //- Add interfacial and mass transfer enthalpy contributions to
700  // heSources
701  const volScalarField& he1(fluid1_.thermo().he());
702  const volScalarField& he2(fluid2_.thermo().he());
703  const volScalarField& Cp1(fluid1_.Cp());
704  const volScalarField& Cp2(fluid2_.Cp());
705  *(heSources_[he1.name()]) =
706  - (
707  htc1*iA_*(he1/Cp1 + iT_ - fluid1_.thermo().T())
708  - fvm::Sp(htc1*iA_/Cp1, he1)
709  )();
710  *(heSources_[he2.name()]) =
711  - (
712  htc2*iA_*(he2/Cp2 + iT_ - fluid2_.thermo().T())
713  - fvm::Sp(htc2*iA_/Cp2, he2)
714  )();
715 
716  if (phaseChangePtr_.valid())
717  {
718  *heSources_[he1.name()] += phaseChangePtr_->heSource(he1.name());
719  *heSources_[he2.name()] += phaseChangePtr_->heSource(he2.name());
720 
721  //- Re-set htc to their values before their were modified by
722  // phaseChangePtr_->correctHeSources() (which caches them in
723  // prevIter)
724  htc1 = htc1.prevIter();
725  htc2 = htc2.prevIter();
726  }
727  }
728 }
