454 scalar Foam::solvers::interFoam::maxDeltaT()
455 {
456  scalar newDeltaT = mesh.time().controlDict().lookupOrDefault<scalar>("maxDeltaT", GREAT);
457 
458  // adjustTimeStep = runTime.controlDict().getOrDefault("adjustTimeStep", false);
459 
460  scalar maxCo = mesh.time().controlDict().getOrDefault<scalar>("maxCo", 1);
461 
462  if (mesh.time().value() > mesh.time().controlDict().get<scalar>("deltaT"))
463  {
464  scalar CoNum = 0.0;
465  scalar meanCoNum = 0.0;
466 
467  if (mesh.nInternalFaces())
468  {
469  scalarField sumPhi
470  (
471  mixture.nearInterface()().primitiveField()
472  *fvc::surfaceSum(mag(phi))().primitiveField()
473  );
474 
475  CoNum = 0.5*gMax(sumPhi/mesh.V().field())*mesh.time().deltaTValue();
476 
477  meanCoNum =
478  0.5*(gSum(sumPhi)/gSum(mesh.V().field()))*mesh.time().deltaTValue();
479  }
480 
481  Info<< "Courant Number mean: " << meanCoNum
482  << " max: " << CoNum << endl;
483 
484 
485  scalar maxAlphaCo
486  (
487  mesh.time().controlDict().get<scalar>("maxAlphaCo")
488  );
489 
490  scalar alphaCoNum = 0.0;
491  scalar meanAlphaCoNum = 0.0;
492 
493  if (mesh.nInternalFaces())
494  {
495  scalarField sumPhi
496  (
497  mixture.nearInterface()().primitiveField()
498  *fvc::surfaceSum(mag(phi))().primitiveField()
499  );
500 
501  alphaCoNum = 0.5*gMax(sumPhi/mesh.V().field())*mesh.time().deltaTValue();
502 
503  meanAlphaCoNum =
504  0.5*(gSum(sumPhi)/gSum(mesh.V().field()))*mesh.time().deltaTValue();
505  }
506 
507  Info<< "Interface Courant Number mean: " << meanAlphaCoNum
508  << " max: " << alphaCoNum << endl;
509 
510  scalar maxDeltaTFact =
511  min(maxCo/(CoNum + SMALL), maxAlphaCo/(alphaCoNum + SMALL));
512 
513  scalar deltaTFact = min(min(maxDeltaTFact, 1.0 + 0.1*maxDeltaTFact), 1.2);
514 
515  newDeltaT =
516  (
517  min
518  (
519  deltaTFact*mesh.time().deltaTValue(),
520  newDeltaT
521  )
522  );
523  }
524 
525  return newDeltaT;
526 }
