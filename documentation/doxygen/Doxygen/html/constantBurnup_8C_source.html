103 Foam::scalar Foam::constantBurnup::nextDeltaT()
104 {
105  scalar nextDeltaT(0);
106  scalar currentDeltaT = mesh_.time().deltaT().value();
107 
108  const scalarField& oldBui(Bu_.oldTime().internalField());
109  const scalarField& currentBui(Bu_.internalField());
110 
111  // Calculate deltaT multiplier based on maximum allowed burnup increase
112  scalar burnupIncrement(gMax(currentBui - oldBui));
113  scalar deltaTMultiplier = maxBuIncrease_ / (burnupIncrement + SMALL);
114 
115  // Set deltaT
116  nextDeltaT = deltaTMultiplier*currentDeltaT;
117 
118  Info<< "Maximum deltaT calculated by burnup model: "
119  << mesh_.time().timeToUserTime(nextDeltaT)
120  << " with a maximum burnup increment of " << burnupIncrement
121  << endl;
122 
123  return nextDeltaT;
124 }
