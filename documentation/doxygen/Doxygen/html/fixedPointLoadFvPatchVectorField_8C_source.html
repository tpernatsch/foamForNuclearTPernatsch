125 void fixedPointLoadFvPatchVectorField::updateTraction()
126 {
127  vectorField n(patch().nf());
128 
129  const polyPatch& thisPatch = patch().patch();
130 
131  // List of patch points
132  labelList labelPointsPatch( thisPatch.meshPoints() );
133 
134  // List of meshPoints
135  pointField meshPoints(patch().boundaryMesh().mesh().points());
136 
137  scalar totalArea(0);
138 
139  // Find faces attached or containing the point given in point_.
140  // Distribute the input force over these faces as a stress.
141  forAll(thisPatch, faceI)
142  {
143  face patchFace = thisPatch[faceI];
144 
145  forAll(patchFace , pointFaceI)
146  {
147  if(mag(meshPoints[patchFace[pointFaceI]] - point_) < 1e-6)
148  {
149  totalArea += patch().magSf()[faceI];
150  vector forceDir = force_/mag(force_);
151  scalar cosine = forceDir&n[faceI];
152  pressure_[faceI] = -(mag(force_)/cosine);
153 
154  break;
155  }
156  else
157  {
158  pressure_[faceI] = 0;
159  }
160  }
161  }
162 
163  reduce(totalArea, sumOp<scalar>());
164 
165  pressure_ /= totalArea;
166  traction_ = vector::zero;
167 }
