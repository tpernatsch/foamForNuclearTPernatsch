109 void Foam::timeDependentVhgr::correct()
110 {
111  if(mesh_.topoChanging())
112  {
113  calcAddressing(heatSourceOptDict_);
114  }
115 
116  if(!mesh_.foundObject<volScalarField>("QPrevIter"))
117  {
118  Q_.storePrevIter();
119  }
120 
121  Q_.prevIter().storePrevIter();
122  Q_.storePrevIter();
123 
124  const scalarField& Qprev(Q_.prevIter());
125  const scalarField& QprevPrev(Q_.prevIter().prevIter());
126 
127  //-Correct heat source only once per time step or until power changes
128  if
129  (
130  gMax(mag(QprevPrev - Qprev)) > 1e-3
131  or
132  mesh_.time().value() > currentTime_
133  )
134  {
135  currentTime_ = mesh_.time().value();
136 
137  scalarField Vi(mesh_.V(), addr_);
138 
139  // Interpolate vhgr and calculate average power density Qavg
140  scalar vhgr = lookupVHGR(currentTime_);
141  scalar Qavg = vhgr;
142 
143  // Build the resulting heat source
144  scalarField& Qi = Q_.ref();
145  forAll(addr_, addrI)
146  {
147  const label cellI = addr_[addrI];
148  Qi[cellI] = Qavg;
149  }
150  }
151  else
152  {
153  currentTime_ = mesh_.time().value();
154  }
155 
156  Q_.correctBoundaryConditions();
157 }

184 Foam::scalar Foam::timeDependentVhgr::nextDeltaTAbsoluteChange(scalar deltaT) const
185 {
186  // Assume that deltaT does not change or that it is limited by vhgr history
187  scalar predictedNextTime = min
188  (
189  currentTime_ + mesh_.time().deltaT().value(),
190  mesh_.time().userTimeToTime
191  (
192  vhgrTable_.getNextPoint(mesh_.time().timeToUserTime(currentTime_))
193  )
194  );
195 
196  scalar currentVhgr(lookupVHGR(currentTime_));
197  scalar predictedNewVhgr(lookupVHGR(predictedNextTime));
198 
199  // Calculate absolute rate of change
200  scalar deltaVhgr = (predictedNewVhgr - currentVhgr);
201 
202  // Set next deltaT
203  scalar nextDeltaT = deltaT*maxAbsolutePowerIncrease_/max(deltaVhgr, SMALL);
204 
205  Info<< "Maximum deltaT calculated by heatSource model: "
206  << mesh_.time().timeToUserTime(nextDeltaT)
207  << " with current vhgr " << currentVhgr <<" W/m3, and an absolute change of " << deltaVhgr << " W/m3"
208  << endl;
209 
210  return nextDeltaT;
211 }
