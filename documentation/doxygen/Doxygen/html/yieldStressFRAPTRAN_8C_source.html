175  //- strain hardening exponent n
176  forAll(addr, addrI)
177  {
178  const label cellI = addr[addrI];
179  const scalar T = Temp.internalField()[cellI];
180 
181  if(T < 419.4)

183  n_[cellI] = 0.11405;
184  }
185  else if(419.4 <= T && T <= 1099.0722)
186  {
187  n_[cellI] = -9.490e-2 + 1.165e-3*T - 1.992e-6*pow(T, 2) + 9.558e-10*pow(T, 3);
188  }
189  else if(1099.0722 <= T && T <= 1600)
190  {
191  n_[cellI] = -0.22655119 + 2.5e-4*T;
192  }
193  else if(T > 1600)
194  {
195  n_[cellI] = 0.17344880;
196  }
197 
198  if(phiValue_ < 0.1e25){
199  n_[cellI] *= 1.321 + 0.48e-25*phiValue_;
200  }
201  else if(0.1e25 <= phiValue_ && phiValue_ < 2e25){
202  n_[cellI] *= 1.369 + 0.096e-25*phiValue_;
203  }
204  else if(2e25 <= phiValue_ && phiValue_ < 7.5e25){
205  n_[cellI] *= 1.5435 + 0.008727e-25*phiValue_;
206  }
207  else if(phiValue_ >= 7.5e25){
208  n_[cellI] *= 1.608953;
209  }
210  }
211 
212  //- Strength coefficient
213  const scalar KCW = 0.546 * CWValue_;
214  scalar Kphi = 0;
215  forAll(addr, addrI)
216  {
217  const label cellI = addr[addrI];
218  const scalar T = Temp.internalField()[cellI];
219 
220  if(T < 750)
221  {
222  Kcoeff_[cellI] = 1.17628e9 + 4.54859e5*T - 3.28185e3*pow(T, 2) + 1.72752*pow(T,3);
223  }
224  else if(750 <= T && T < 1090)
225  {
226  Kcoeff_[cellI] = 2.522488e6 * exp(2.8500027e6/pow(T, 2));
227  }
228  else if(1090 <= T && T < 1255)
229  {
230  Kcoeff_[cellI] = 1.841376039e8 - 1.4345448e5*T;
231  }
232  else if(1255 <= T && T < 2100)
233  {
234  Kcoeff_[cellI] = 4.330e7 - 6.685e4*T + 37.579*pow(T, 2) - 7.33e-3*pow(T, 3);
235  }
236 
237  if(phiValue_ < 0.1e25){
238  Kphi = -0.1461 + 1.464e-25*phiValue_ * ((2.25 * exp(-20*CWValue_) * min(1, exp((T-550)/10))) + 1);
239  Kcoeff_[cellI] *= 1 + KCW + Kphi;
240  }
241  else if(phiValue_ <= 0.1e25 && phiValue_ < 2e25){
242  Kphi = 2.928e-26*phiValue_;
243  Kcoeff_[cellI] *= 1 + KCW + Kphi;
244  }
245  else if(phiValue_ <= 2e25 && phiValue_ < 12e25){
246  Kphi = 0.53236 + 2.6618e-27*phiValue_;
247  Kcoeff_[cellI] *= 1 + KCW + Kphi;
248  }

250  }

259  const scalar EpsDotZZi = EpsDot_ZZ.internalField()[cellI];*/
