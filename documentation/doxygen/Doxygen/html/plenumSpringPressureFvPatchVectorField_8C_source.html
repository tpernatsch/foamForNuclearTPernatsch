106 void plenumSpringPressureFvPatchVectorField::updateTraction()
107 {
108  const globalOptions& globalOpt
109  (db().lookupObject<globalOptions>("globalOptions"));
110 
111  scalar angularFraction = globalOpt.angularFraction();
112 
113  // Include the gap gas pressure
114  const gapGasModel& gapGas
115  = db().lookupObject<gapGasModel>("gapGas");
116 
117  // Calculate total spring elongation (positive if spring contracts)
118  scalar Dtot(plenumSpringBase::springElongation());
119 
120  scalar Atot = gSum(patch().magSf());
121 
122  scalarField springPressure
123  (patch().size(), plenumSpringBase::springModulus()*Dtot/(Atot/angularFraction));
124 
125  // Update total pressure
126  pressure_ = springPressure + gapGas.p();
127 }
