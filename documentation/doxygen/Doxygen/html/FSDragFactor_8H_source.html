41 Description
42  Base class to represent a FSDragFactor with basing name access
43  functionality.
44 
45 
46 Usage
47  The following sub-dictionary should be included as a powerModel
48  in phaseProperties.
49 
50  Example for isotropic:
51  \verbatim
52  physicsModels
53  {
54  dragModels
55  {
56  [nameOfCellZone]
57  {
58  type ReynoldsPower;
59  coeff 1.84; //
60  exp -0.2;
61  }
62  }
63  }
64 
65  \endverbatim
66 
67  Example for anisotropic:
68  \verbatim
69  physicsModels
70  {
71  dragModels
72  {
73  [nameOfCellZone]
74  {
75  localX
76  {
77  type ReynoldsPower;
78  coeff 1.84; //
79  exp -0.2;
80  }
81  localY
82  {
83  type ReynoldsPower;
84  coeff 1.84; //
85  exp -0.2;
86  }
87  localZ
88  {
89  type ReynoldsPower;
90  coeff 0.184; //
91  exp -0.2;
92  }
93 
94  }
95  }
96  }
97  \endverbatim
