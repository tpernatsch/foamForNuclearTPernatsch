173 // sigma Equivalent function
174 Foam::scalar Foam::fastNeutronGPLS::sigmaEq_func
175 (
176  const symmTensor& sigmaEffCyl,
177  const scalar& T,
178  const scalar& phi
179 
180 )
181 {
182  const scalar Hshear = 1.5;
183 
184  const scalar sigmaEq = sqrt(
185  Hrr(T, phi) * pow(sigmaEffCyl.yy() - sigmaEffCyl.zz(),2 )
186  + Htt(T, phi) * pow(sigmaEffCyl.zz() - sigmaEffCyl.xx(), 2)
187  + Hzz(T, phi) * pow(sigmaEffCyl.xx() - sigmaEffCyl.yy(), 2)
188  + 2 * Hshear * pow(sigmaEffCyl.xy(), 2)
189  + 2 * Hshear * pow(sigmaEffCyl.xz(), 2)
190  + 2 * Hshear * pow(sigmaEffCyl.yz(), 2));
191  return sigmaEq;
192 }

194 // cartesian -> cylindrical coordinates
195 Foam::symmTensor Foam::fastNeutronGPLS::cylSig
196 (
197  const symmTensor& sigma,
198  const vector& Ci
199 )
200 {
201  symmTensor sigmaEffCyl = sigma * scalar(0);
202 
203  // Transform the sigmaEffCyl tensor in cylindrical coordinates
204  tensor R = tensor::zero;
205 
206  const scalar radius = sqrt(Ci[0]*Ci[0] + Ci[1]*Ci[1]);
207 
208  R.xx() = Ci[0]/radius;
209  R.xy() = Ci[1]/radius;
210  R.yy() = Ci[0]/radius;
211  R.yx() = -Ci[1]/radius;
212  R.zz() = 1;
213  sigmaEffCyl = symm( R & sigma & R.T() );
214 
215  const scalar theta = atan(Ci[1]/Ci[0]);
216  return sigmaEffCyl;
217 }

248 // Derivative of F
249 Foam::scalar Foam::fastNeutronGPLS::dF
250 (
251  const scalar& sigmaEq,
252  const scalar& T,
253  const scalar& phi,
254  const scalar& zeta,
255  const scalar& p,
256  const scalar& dEpsilonPEq,
257  const scalar& deltaT,
258  const scalar& theta
259 )
260 {
261  const scalar one_m = 1 / m(T, phi);
262  const scalar pDot = G(sigmaEq, T, phi, zeta, p + theta*dEpsilonPEq);
263 
264  const scalar term1 = deltaT * one_m * pDot / L(T, phi, p + theta*dEpsilonPEq);
265 
266  const scalar Lp = p + theta*dEpsilonPEq + 1e-4;
267  const scalar Ln0 = pow(Lp, n0(T, phi));
268  const scalar Lninf = pow(Lp, ninf(T, phi));
269  const scalar Lalpha = exp(-alpha_n(T) * (p + theta*dEpsilonPEq));
270 
271  const scalar dL = Ln0 * Lalpha * (n0(T, phi) / Lp - alpha_n(T)) +
272  Lninf * (1-Lalpha) * (ninf(T, phi) / Lp - alpha_n(T)) +
273  alpha_n(T)*Lninf;
274 
275 
276  return 1 + term1 * dL * theta;
277 }

279 /*
280 ------ Hill Tensor nota -----
281 - Ideally we should modifiy OpenFOAM to add a new class of field...
282  (see FOAM extend with the symmTensor4thOrder class)
283 - We dont really need a full 4th order tensor class
284 - We just need the double dot product!
285 - We use 6 (shear component are equals to 1.5) scalars
286 - The symmetric 4th order Hill Tensor can be visualized as:
287  "Hxxxx", "Hxxyy", "Hxxzz",
288  "Hyyyy", "Hyyzz",
289  "Hzzzz",
290  "Hxyxy",
291  "Hyzyz",
292  "Hzxzx"
293 
294 Its components are expressed with 6 anisotropy coefficients: F, G, H, L, M, N.
295 L, M, N are the shear components that are here equal to 1.5 (isotropic case)
296 We have (Lemaître and Chaboche, Mechanics of solid materials, 1994):
297 Hxxxx = F + H Hyyzz = - G
298 Hxxyy = - F Hzzzz = H + G
299 Fxxzz = -H
300 Hyyyy = G + F
301 
302 F, G, H are equal to Hzz, Hrr or Htt (model parameters)
303 
304 ------ Double dot product of a symm4thOrderTensor and a symmTensor -----
305 t4th = symm 4th order tensor
306 st = symm tensor
307 (
308  t4th.xxxx()*st.xx() + t4th.xxyy()*st.yy() + t4th.xxzz()*st.zz(),
309  t4th.xyxy()*st.xy(),
310  t4th.zxzx()*st.xz(),
311 
312  t4th.xxyy()*st.xx() + t4th.yyyy()*st.yy() + t4th.yyzz()*st.zz(),
313  t4th.yzyz()*st.yz(),
314  t4th.xxzz()*st.xx() + t4th.yyzz()*st.yy() + t4th.zzzz()*st.zz()
315 );
316 
317 
318 */
319 Foam::symmTensor Foam::fastNeutronGPLS::normal_tensor
320 (
321  const scalar& T,
322  const scalar& phi,
323  const scalar& sigmaEq,
324  const symmTensor& sigmaEff
325 
326 )
327 {
328  //- SigmaEff/sigmaEq
329  const symmTensor sig_ratio = sigmaEff / sigmaEq;
330 
331  // - Scalar parameters Hill tensor (Lemaitre - Caboche formalism)
332  const scalar F = Hzz(T, phi);
333  const scalar G = Hrr(T, phi);
334  const scalar H = Htt(T, phi);
335 
336  const scalar Hxxxx = F + H;
337  const scalar Hxxyy = -F;
338  const scalar Hxxzz = -H;
339  const scalar Hyyyy = G + F;
340  const scalar Hyyzz = - G;
341  const scalar Hzzzz = H + G;
342  const scalar Hshear = 1.5;
343 
344  return symmTensor(
345  Hxxxx * sig_ratio.xx() + Hxxyy*sig_ratio.yy() + Hxxzz*sig_ratio.zz(),
346 
347  Hshear * sqrt(2.0) * sig_ratio.xy(),
348  Hshear * sqrt(2.0) * sig_ratio.xz(),
349 
350  Hxxyy * sig_ratio.xx() + Hyyyy*sig_ratio.yy() + Hyyzz*sig_ratio.zz(),
351 
352  Hshear * sqrt(2.0)*sig_ratio.yz(),
353 
354  Hxxzz * sig_ratio.xx() + Hyyzz*sig_ratio.yy() + Hzzzz*sig_ratio.zz()
355  );
356 }

473 void Foam::fastNeutronGPLS::correct
474 (
475  volScalarField& sigmaHyd,
476  volSymmTensorField& sigmaDev,
477  volSymmTensorField& epsilonEl,
478  const labelList& addr
479 )
480 {
481  creepModel_->correctCreep(epsilonEl, addr);
482 
483  //- TODO: Instead of looping over cells one might store the addressing and
484  //- work with openfoam field syntax.
485  const volScalarField& mu_(mesh_.lookupObject<volScalarField>("mu"));
486  const volScalarField& K_(mesh_.lookupObject<volScalarField>("threeK"));
487  const volScalarField& lambda_(mesh_.lookupObject<volScalarField>("lambda"));
488 
489  const scalarField& muI = mu_.internalField();
490  const scalarField& lambdaI = lambda_.internalField();
491 
492  symmTensorField& plasticNI = plasticN_.ref();
493  scalarField& DEpsilonPEqI = DEpsilonPEq_.ref();
494  symmTensorField& DEpsilonPI = DEpsilonP_.ref();
495  symmTensorField& epsilonI = epsilonEl.ref();
496 
497  symmTensorField& epsilonPI = epsilonP_.ref();
498  scalarField& epsilonPEqI = epsilonPEq_.ref();
499 
500  const symmTensorField& epsilonPOldTimeI = epsilonP_.oldTime().internalField();
501  const scalarField& epsilonPEqOldTimeI = epsilonPEq_.oldTime().internalField();
502  const scalarField& DEpsilonPEqOldTimeI = DEpsilonPEq_.oldTime().internalField();
503 
504  const scalar& deltaT = mesh_.time().deltaT().value();
505  const volScalarField& Temp(mesh_.lookupObject<volScalarField>("T"));
506 
507  //- Check if the simulation is large strain
508  bool largeStrain = mesh_.foundObject<volTensorField>("F") ?
509  true : false;
510 
511  forAll(addr, addrI)
512  {
513  //- Cell adrress. Act only on element of addr
514  const label cellI = addr[addrI];
515 
516  //- Cell temperature
517  const scalar TOld = Temp.oldTime().internalField()[cellI];
518  const scalar T = Temp.internalField()[cellI];
519  const scalar Tm = TOld + theta_*(T-TOld);
520 
522  m_[cellI] = m(T, phiValue_);
523  Kmod_[cellI] = K(T, phiValue_, zetaValue_);
524  //L at the "begining" of the time step is calculated based on the old value of the equivalent strain
525  L_[cellI] = L(T, phiValue_, epsilonPEqOldTimeI[cellI]);
526  etaVis_[cellI] = Kmod_[cellI] * L_[cellI];
527 
528  // Calculate trial stress
529  const symmTensor sigmaEff = 2.0*muI[cellI]*epsilonI[cellI]
530  + lambdaI[cellI]*tr(epsilonI[cellI])*I;
531 
532  //convert to cylindrical coordinates
533  const vector& Ci = mesh_.C()[cellI];
534  const symmTensor sigmaEffCyl = cylSig(sigmaEff, Ci);
535 
536  // Calculate Sigma Hill Equivalent
537  sigmaEqH_[cellI] = sigmaEq_func(sigmaEffCyl, Tm, phiValue_);
538 
539  //Init plastic vector at 0
540  plasticNI[cellI] = symmTensor::zero;
541 
542  //Init the equivalent strain increment over the current time step
543  scalar DEpsilonPEq_n = 0;
544  DEpsilonPEqI[cellI] = 0.0;
545 
546  //- Flow ---> Newton-Raphson iteration
547  if(sigmaEqH_[cellI] > 1.){
548 
549  // Calculate normal vector / plastic multiplier N = H : (sigma/sigmaEq)
550  plasticNI[cellI] += normal_tensor(Tm, phiValue_, sigmaEqH_[cellI], sigmaEff);
551 
552  // Initialize Newton-Raphson
553  // If we assume that the time step is small enough
554  // ---> DEpsilonPEq(t) ~ DEpsilonPEq(t+dt) and DEpsilonPEq(t) is a good first estimation for the Newton-Raphson method
555  //DEpsilonPdEqI[cellI] = DEpsilonPEqOldTimeI[cellI];
556  DEpsilonPEq_n = DEpsilonPEqI[cellI];
557 
558  scalar i = 0;
559  scalar error = 10;
560  // Newton Raphson iterations
561  do{
562  //F
563  scalar F_n = F(sigmaEqH_[cellI], Tm, phiValue_, zetaValue_,
564  epsilonPEqOldTimeI[cellI], DEpsilonPEq_n, deltaT, theta_);
565 
566  //dF
567  scalar dF_n = dF(sigmaEqH_[cellI], Tm, phiValue_, zetaValue_,
568  epsilonPEqOldTimeI[cellI], DEpsilonPEq_n, deltaT, theta_);
569 
570  //Newton Raphson estimation
571  // np1 --> n + 1 with n the iteration of the newton rahpson method
572  scalar DEpsilonPEq_np1 = DEpsilonPEq_n - F_n/dF_n;
573 
574  // Error d
575  error = mag(DEpsilonPEq_np1 - DEpsilonPEq_n);
576 
577  //Update NR estimation with a relaxation factor
578  DEpsilonPEq_n = relax_ * DEpsilonPEq_np1 + (1-relax_) * DEpsilonPEq_n;
579 
580  i++;
581  }while(error > tol_ && i < maxIter_);
582 
583  if(i >= maxIter_)
584  {
585  Info << "WARNING GPLS INTEGRATOR" << endl;
586  Info << "max iterations for the NR method reached (cell computation)" << endl;
587  Info << "With a value error of: " << error << endl;
588  Info << "Number of iterations: " << i << endl;
589  Info << "Value of the increment: " << DEpsilonPEq_n << endl;
590  Info << "" << endl;
591  }
592 
593  //Update DEpsilonPEq with value obtained with the Newton-Rahpson method
594  DEpsilonPEqI[cellI] = DEpsilonPEq_n;
595  }
596 
597  // Update hardening and viscosity with the new epsilon viscoplastic equivalent strain
598  L_[cellI] = L(T, phiValue_, epsilonPEqOldTimeI[cellI]+DEpsilonPEqI[cellI]);
599  etaVis_[cellI] = Kmod_[cellI] * L_[cellI];
600 
601  // Update epsilonPEq
602  epsilonPEqI[cellI] = epsilonPEqOldTimeI[cellI] + DEpsilonPEqI[cellI];
603 
604  // Update DEpsilonP
605  DEpsilonPI[cellI] = plasticNI[cellI] * DEpsilonPEqI[cellI];
606 
607  // Update total plastic strain
608  epsilonPI[cellI] = epsilonPOldTimeI[cellI] + DEpsilonPI[cellI];
609 
610  // Update elastic strain
611  epsilonI[cellI] -= DEpsilonPI[cellI];
612 
613  // Calculate deviatoric stress
614  sigmaDev[cellI] = dev(sigmaEff) - 2*muI[cellI]*DEpsilonPI[cellI];
615 
616  // Calculate the hydrostatic pressure
617  const scalar trEpsilon = tr(epsilonI[cellI]);
618 
619  sigmaHyd[cellI] = 1.0/3.0*K_[cellI]*trEpsilon;
620 
621  const cell& c = mesh_.cells()[cellI];
622  forAll(c, faceI)
623  {
624  const label patchID = mesh_.boundaryMesh().whichPatch(c[faceI]);
625 
626  // forAll(patchIDs, registeredID)
627  {
628  if (patchID > -1 and epsilonEl.boundaryField()[patchID].size())
629  {
630 
631  const label faceID =
632  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
633 
634  // Take references to the boundary patch fields for efficiency
635  symmTensorField& sigmaDevP =
636  sigmaDev.boundaryFieldRef()[patchID];
637 
638  scalarField& sigmaHydP =
639  sigmaHyd.boundaryFieldRef()[patchID];
640 
641  symmTensorField& plasticNP =
642  plasticN_.boundaryFieldRef()[patchID];
643 
644  const scalarField& muP =
