117 void Foam::twoPhaseDragFactor::correct()
118 {
119  bool notFirstTimeStep
120  (
121  (mesh_.time().timeIndex() != mesh_.time().startTimeIndex()+1)
122  );
123  if (notFirstTimeStep)
124  myOps::storePrevIterIfRelax(KdTotU_);
125  twoPhaseDragMultiplierPtr_->correctField(KdTotU_);
126  if (notFirstTimeStep)
127  KdTotU_.relax();
128  dimensionedScalar minMagU("", dimVelocity, 1e-9);
129  F1SPair_.Kd() =
130  F1SPair_.f()*KdTotU_/max(F1SPair_.fluidRef().magU(), minMagU);
131  F2SPair_.Kd() =
132  F2SPair_.f()*KdTotU_/max(F2SPair_.fluidRef().magU(), minMagU);
133 }
