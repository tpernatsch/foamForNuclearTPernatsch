174  {
175  word region(this->toc()[regioni]);
176  const dictionary& dict(this->subDict(region));
177 
178  //- Setup cellToRegion_ mapping
179  const labelList& regionCells
180  (
181  structure_.cellLists()[region]
182  );
183  forAll(regionCells, i)
184  {
185  label celli(regionCells[i]);

187  }

189  //- Read region dict entries
190  scalar ri(dict.get<scalar>("innerRadius"));
191  scalar ro(dict.get<scalar>("outerRadius"));
192  label meshSize(dict.get<label>("meshSize"));
193  scalar dr((ro-ri)/(meshSize-1));
194  scalar rhoCp(0);
195  if (dict.found("rho") and dict.found("Cp"))
196  {
197  rhoCp =
198  dict.get<scalar>("rho")*
199  dict.get<scalar>("Cp");
200  }

214  bool hollow((ri >= 1e-5) ? true : false);

216  if (!foundBoundaryTemperatures and !foundTrad)

222  scalarList r(0);
223  r.append(ri);
224  for (int i = 0; i < meshSize-1; i++)
225  {

233  {

235  }

239  meshSize_.append(meshSize);
240  r_.append(r);
241  ri_.append(ri);
242  ro_.append(ro);
243  dr_.append(dr);
244  rhoCp_.append(rhoCp);
245  k_.append(k);
246  hollow_.append(hollow);
247  dA_.append(dA);
248  }

250  //- If Trad not found, init it from either the boundary temperatures
251  // (I mean boundary in a mathematical sense, i.e. inner/outer fuel/clad
252  // temperature) or from dictionary values Tf0, Tc0 read previously
253  if (!foundTrad)
254  {
255  forAll(mesh_.cells(), i)
256  {
257  Trad_.set(i, new Field<scalar>(0, 0));
258  }
259 
260  //- If the files are present, reconstruct initial Trad_ profile
261  // analytically. The analytical form is:
262  // T(r) = -(1/4)*powerDensity_(r)*r^2/k + C*ln(r)/k + D
263  // with C and D coming from imposing fixedValue BC on all sides,
264  // equal to the starting temperatures found in the files.
265  if (foundBoundaryTemperatures)
266  {
267  Info<< "Found heatedPin temperatures "
268  << "reconstructing profiles " << endl;
269  forAll(this->cellList_, i)
270  {
271  label celli(this->cellList_[i]);
272 
273  label regioni(cellToRegion_[celli]);
274  const scalarList& rList(r_[regioni]);
275  scalar k(k_[regioni]);
276  scalar ri(ri_[regioni]);
277  scalar ro(ro_[regioni]);
278  bool hollow(hollow_[regioni]);
279 
280  Trad_.set(celli, new Field<scalar>(meshSize_[regioni], 0));
281  scalar q = powerDensity_[celli];
282  scalar ti = Ti_[celli];
283  scalar to = To_[celli];
284  scalar C;
285  scalar D;
286 
287  if (hollow)
288  {
289  C =
290  (k*(ti-to)-0.25*q*(sqr(ro)-sqr(ri)))/
291  log(ri/ro);
292  }
293  else
294  {
295  C = 0.0;
296  }
297  D = to+(0.25*q*sqr(ro)-C*log(ro))/k;
298 
299  forAll(Trad_[celli], j)
300  {
301  scalar r(rList[j]);
302  Trad_[celli][j] = -0.25*q*sqr(r)/k + D;
303  Trad_[celli][j] +=
304  (hollow) ? C*log(r)/k : 0.0;
305  }
306  }
307  }
308  else //- Otherwise, read from dict
309  {

311  << "dictionary" << endl;
312

314  {

317  Trad_.set(celli, new Field<scalar>(meshSize_[regioni], 0));

322  }
323  }
324  }

329  }

334  const scalarList& V(mesh_.V());
335  forAll(this->cellList_, i)

356  Tmax_
357  );
358 
359  //- This is for updating the global averages, not the local cell ones!
360  const scalar& dV(V[celli]);
361  totV += dV;
362  Tavav += Tavi*dV;
363  }
364

366  reduce(totV, sumOp<scalar>());

404 void Foam::powerModels::heatedPin::updateLocalAvgGlobalMinMaxT
405 (
406  const label& starti,
407  const label& endi,
408  const scalarList& r,
409  const scalar& dr,
410  const scalarField& Trad,
411  scalar& Tavi,
412  scalar& Tmin,
413  scalar& Tmax
414 )
415 {
416  scalar intr(0);
417  scalar intTr(0);
418  for(int j = starti; j < endi; j++)
419  {
420  const scalar& T(Trad[j]);
421  scalar rdr(r[j]*dr);
422 
423  //- Cells at the mesh ends are only half as wide (the other half
424  // belongs to the ghost node). Thus, weigh temperatures at the extrema
425  // by a factor 0.5
426  if (j == starti or j == endi-1)
427  {
428  intr += rdr/2.0;
429  intTr += T*rdr/2.0;
430  }
431  else
432  {
433  intr += rdr;
434  intTr += T*rdr;
435  }
436  if (T > Tmax) Tmax = T;
437  if (T < Tmin) Tmin = T;
438  }
439  Tavi = intTr/intr;
440 }

442 void
443 Foam::powerModels::heatedPin::updateLocalTemperatureProfile
444 (
445  const label& celli,
446  const scalar& HTSumi,
447  const scalar& HSumi
448 )
449 {
450  //-
451  scalarField& Trad(Trad_[celli]);
452 
453  //- Read region values
454  const label& regioni(cellToRegion_[celli]);
455  const label& meshSize(meshSize_[regioni]);
456  const scalarList& rRegion(r_[regioni]);
457  const scalarList& dARegion(dA_[regioni]);
458  const scalar& dr(dr_[regioni]);
459  const scalar& k(k_[regioni]);
460  const scalar& q(powerDensity_[celli]);
461  const scalarField& TOld = Trad_.oldTime()[celli];
462 
463  //- Recurrent quantities
464  scalar drh(dr/2.0);
465  scalar dt(mesh_.time().deltaT().value());
466  scalar twoPkByDr(2.0*pi_*k/dr);
467  scalar X(rhoCp_[regioni]/dt);
468 
469  //- Init matrix, source
470  SquareMatrix<scalar> M(meshSize, meshSize, Foam::zero());
471  List<scalar> S(meshSize, 0.0);
472 
473  //- Fill in matrix, source coeffs
474  {
475  //- Set zeroGradient BC at pin inner surface
476  {
477  const scalar& r(rRegion[0]);
478  const scalar& dA(dARegion[0]);
479  scalar B(twoPkByDr*(r+drh));
480  scalar XdA(X*dA);
481  M[0][1] = -B;
482  M[0][0] = B+XdA;
483  S[0] = q*dA+TOld[0]*XdA;
484  }
485 
486  //- Set bulk
487  for (int i = 1; i < meshSize-1; i++)
488  {
489  const scalar& r(rRegion[i]);
490  const scalar& dA(dARegion[i]);
491  scalar B(twoPkByDr*(r+drh));
492  scalar C(twoPkByDr*(r-drh));
493  scalar XdA(X*dA);
494  M[i][i+1] = -B;
495  M[i][i-1] = -C;
496  M[i][i] = B+C+XdA;
497  S[i] = q*dA+TOld[i]*XdA;
498  }
499 
500  //- Set convective BC at pin outer surface
501  {
502  label i(meshSize-1);
503  const scalar& r(rRegion[i]);
504  const scalar& dA(dARegion[i]);
505  scalar C(twoPkByDr*(r-drh));
506  scalar D(2.0*pi_*r);
507  scalar XdA(X*dA);
508  M[i][i-1] = -C;
509  M[i][i] = C+D*HSumi+XdA;
510  S[i] = q*dA+TOld[i]*XdA + D*HTSumi;
511  }
512  }
513 
514  //- Solve linear system
515  solve(Trad, M, S);
516 
517  //- Set inner/outer pin temperature fields
518  Ti_[celli] = Trad[0];
519  To_[celli] = Trad[meshSize-1];
520 
521  //- Check energy conservation via linear power
522  /*
523  scalar aLP(q*pi_*(sqr(rRegion[0])-sqr(rRegion[meshSize-1])));
524  scalar nLP((HSumi*To_[celli]-HTSumi)*2.0*pi_*rRegion[meshSize-1]);
525  Info<< celli << " " << nLP << " " << aLP << " W/m" << endl;
526  */
527 
528  //- Update local T averages and local min/max
529  scalar& Tavi(Tav_[celli]);
530  updateLocalAvgGlobalMinMaxT
531  (
532  0,
533  meshSize,
534  rRegion,
535  dr,
536  Trad,
537  Tavi,
538  Tmin_,
539  Tmax_
540  );
541 }

544 void Foam::powerModels::heatedPin::correct
545 (
546  const volScalarField& HTSum, // == SUM_j [htc_j*T_j*frac_j]
547  const volScalarField& HSum // == SUM_j [htc_j*frac_j]
548 )
549 {
550  //- Reset min, max, fuel, clad temperatures
551  Tmax_ = 0.0;
552  Tmin_ = 1e69;
553 
554  //- Update temperatures cell-by-cell and compute averages over the entire
555  // spatial extent of the heatedPin model (what I call global
556  // averages, opposed to local averages, which are the average temperature
557  // values, fuel and clad, of the local cell radial pin temperature
558  // profile)
559  const scalarField& V(mesh_.V());
560  scalar totV(0);
561  scalar Tavav(0);
562  forAll(this->cellList_, i)
563  {
564  label celli(this->cellList_[i]);
565  updateLocalTemperatureProfile(celli, HTSum[celli], HSum[celli]);
566  const scalar& dV(V[celli]);
567  totV += dV;
568  Tavav += Tav_[celli]*dV;
569  }
570  reduce(totV, sumOp<scalar>());
571  reduce(Tavav, sumOp<scalar>());
572  Tavav /= totV;
573 
574  reduce(Tmax_, maxOp<scalar>());
575  reduce(Tmin_, minOp<scalar>());
576 
577  Info<< "T.heatedPin (avg min max) = "
578  << Tavav << " " << Tmin_ << " " << Tmax_ << " K" << endl;
579 
580  //- Save these to the dictionary
581  this->IOdictionary::set("Tavav", Tavav);
582  this->IOdictionary::set("Tmax", Tmax_);
583  this->IOdictionary::set("Tmin", Tmin_);
584 }
