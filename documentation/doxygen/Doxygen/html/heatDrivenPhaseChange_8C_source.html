126 void Foam::phaseChangeModels::heatDrivenPhaseChange::correctInterfacialDmdt()
127 {
128  //- Interfacial mass transfers for each side
129  volScalarField dmdtI1i
130  (
131  htc1_*iA_*(T1_-iT_)/L_
132  );
133  volScalarField dmdtI2i
134  (
135  htc2_*iA_*(T2_-iT_)/L_
136  );
137 
138  switch (mode_)
139  {
140  case heatDrivenPhaseChange::mode::conductionLimited :
141  dmdtI_ = dmdtI1i + dmdtI2i;
142  break;
143 
144  case heatDrivenPhaseChange::mode::twoPhaseDriven :
145  dmdtI_ = posPart(dmdtI1i) + negPart(dmdtI2i);
146  break;
147 
148  case heatDrivenPhaseChange::mode::onePhaseDriven :
149  dmdtI_ =
150  (fluid1_.name() == drivingPhaseName_) ?
151  dmdtI1i : dmdtI2i;
152  break;
153 
154  case heatDrivenPhaseChange::mode::mixedDriven :
155  dmdtI_ =
156  (fluid1_.name() == drivingPhaseName_) ?
157  dmdtI1i + negPart(dmdtI2i) :
158  posPart(dmdtI1i) + dmdtI2i;
159  break;
160 
161  default : break;
162  }
163 }
