112 {
113  // Set maximum fgr allowed per timestep (used in case of adjustable time
114  // step)
115  const dictionary& controlDict(mesh_.time().controlDict());
116 
117  bool adjustTime = controlDict.lookupOrDefault<bool>("adjustableTimeStep", false);
118 
119  if(adjustTime)
120  {
121  if(controlDict.found("maxFGR"))
122  {
123  // This method is deprecated (left temporarily to be consistent
124  // with old cases)
125  maxFGR_ = (readScalar(controlDict.lookup("maxFGR")));
126  timeSteppingCriterion_ = "maxFGR";
127  }
128  else
129  {
130  maxLocalDeltaFgr_ = controlDict.lookupOrDefault<scalar>("maxLocalFgrChange", GREAT);
131  maxTotalDeltaFgr_ = controlDict.lookupOrDefault<scalar>("maxTotalFgrChange", GREAT);
132  timeSteppingCriterion_ = "maxFgrChange";
133  }
134  }
135 
136  const labelListList& matAddrList(mat_.matAddrList());
137 
138  // Set initial grain radius (only if current value is == 0)
139  forAll(mat_.materialsList(), i)
140  {
141  const materialModel& materialI(mat_.materialsList()[i]);
142  const labelList& addr(matAddrList[i]);
143 
144  const word& materialName(materialI.name());
145 
146  if(isA<fuelMaterial>(mat_.materialsList()[i]))
147  {
148  const fuelMaterial& fuelMatI(refCast<const fuelMaterial>(materialI));
149 
150  // Take initial grain radius
151  scalar rGrain = fuelMatI.rGrain();
152 
153  // Take initial O/M
154  scalar oxygenMetalRatio = fuelMatI.oxygenMetalRatio();
155 
156  scalar averageGrainRadius(0);
157  scalar averageOxygenMetal(0);
158  scalar totalVol(0);
159  const scalarField& volumes(mesh_.V());
160 
161  forAll(addr, j)
162  {
163  const label cellI = addr[j];
164 
165  if(grainRadius_[cellI] < VSMALL)
166  {
167  grainRadius_[cellI] = rGrain;
168  }
169 
170  if(oxygenMetalRatio_[cellI] < VSMALL)
171  {
172  oxygenMetalRatio_[cellI] = oxygenMetalRatio;
173  }
174 
175  averageGrainRadius = grainRadius_[cellI]*volumes[cellI];
176  averageOxygenMetal = oxygenMetalRatio_[cellI]*volumes[cellI];
177  totalVol = volumes[cellI];
178  }
179 
180  reduce(averageGrainRadius, sumOp<scalar>());
181  reduce(averageOxygenMetal, sumOp<scalar>());
182  reduce(totalVol, sumOp<scalar>());
183 
184  Info << "Average grain radius for material \"" << materialName
185  << "\": " << averageGrainRadius/max(totalVol,SMALL) << "m" << nl
186  << endl;
187 
188  Info << "Average grain O/M for material \"" << materialName
189  << "\": " << averageOxygenMetal/max(totalVol,SMALL) << "m" << nl
190  << endl;
191  }
192  }
193 
194  grainRadius_.correctBoundaryConditions();
195  oxygenMetalRatio_.correctBoundaryConditions();
196 
197  intragranularGasSwelling_.oldTime();
198  intergranularGasSwelling_.oldTime();
199 
200  intergranularGasSwelling_.correctBoundaryConditions();
201  intragranularGasSwelling_.correctBoundaryConditions();
202 }

265 Foam::scalar Foam::fissionGasRelease::nextDeltaT()
266 {
267  if(timeSteppingCriterion_ == "maxFgrChange")
268  {
269  // Current deltaT
270  scalar oldDeltaT = mesh_.time().deltaT().value();
271 
272  // DeltaFgr field
273  scalarField deltaFgr(fgr_ - fgr0_);
274 
275  // Reference to cell volumes
276  const scalarField& Vi = mesh_.V();
277 
278  // Calculate total volume and weighted average total fgr
279  scalar totalVol(0);
280  scalar totalDeltaFgr(0);
281 
282  const labelListList& matAddrList(mat_.matAddrList());
283 
284  forAll(mat_.materialsList(), i)
285  {
286  if(isA<fuelMaterial>(mat_.materialsList()[i]))
287  {
288  const labelList& addr(matAddrList[i]);
289 
290  scalarField Vaddr(Vi, addr);
291  scalarField deltaFgrAddr(deltaFgr, addr);
292 
293  totalDeltaFgr += gSum(Vaddr*deltaFgrAddr);
294  totalVol += gSum(Vaddr);
295  }
296  }
297 
298  totalDeltaFgr = totalDeltaFgr/max(totalVol, VSMALL);
299 
300  // Current maximum local change in fgr percent
301  scalar localMaxDeltaFgr = gMax(mag(fgr_ - fgr0_));
302 
303  // Scale time step based on desired change in local fgr per cent
304  scalar nextDeltaTLocal(oldDeltaT*maxLocalDeltaFgr_/max(localMaxDeltaFgr, SMALL));
305 
306  // Scale time step based on desired change in total fgr per cent
307  scalar nextDeltaTTotal(oldDeltaT*maxTotalDeltaFgr_/max(totalDeltaFgr, SMALL));
308 
309  Info<< "Maximum deltaT calculated by fgr model: "
310  << mesh_.time().timeToUserTime(min(nextDeltaTLocal, nextDeltaTTotal))
311  << " with a current maximum local change in fgr% of " << localMaxDeltaFgr
312  << " and a current total change in fgr% of " << totalDeltaFgr
313  << endl;
314 
315  // // Possible criterion based on swelling?
316  // scalarField swellingI = tr(intergranularGasSwelling_.internalField())/3.0;
317  // scalarField swellingOldI = tr(intergranularGasSwelling_.oldTime().internalField())/3.0;
318  // scalarField deltaSwelling(mag(100.0*(swellingI - swellingOldI)/max(swellingOldI, SMALL)));
319  // // Scale time step based on desired change in fgr per cent
320  // nextDeltaT = (oldDeltaT*maxDeltaFgr_/max(gMax(deltaSwelling), SMALL));
321 
322  return min(nextDeltaTLocal, nextDeltaTTotal);
323  }
324  else
325  {
326  // This method is deprecated. Left temporarily only to be consistent
327  // with old cases. To be removed in the future
328 
329  // Store old value of deltaT
330  scalar oldDeltaT = mesh_.time().deltaT().value();
331 
332  // Calculate the current rate of mol released
333  scalar fgrRate(molFGR_/max(oldDeltaT, SMALL));
334 
335  scalar nextDeltaT(maxFGR_/max(fgrRate, SMALL));
336 
337  Info<< "Maximum deltaT calculated by fgr model: "
338  << mesh_.time().timeToUserTime(nextDeltaT)
339  << " with a release rate of " << fgrRate << " mol/s"
340  << endl;
341 
342  return nextDeltaT;
343  }
344 }
