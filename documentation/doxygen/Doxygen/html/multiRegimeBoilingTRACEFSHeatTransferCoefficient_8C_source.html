161 Foam::scalar
162 Foam::FSHeatTransferCoefficientModels::multiRegimeBoilingTRACE::value
163 (
164  const label& celli
165 ) const
166 {
167  if (FFPairPtr_ == nullptr)
168  {
169  HashTable<const FFPair*> FFPairs(pair_.mesh().lookupClass<FFPair>());
170  FFPairPtr_ = FFPairs[FFPairs.toc()[0]];
171  }
172  if (dmdtWPtr_ == nullptr)
173  {
174  //- Ptr to wall mass transfer term (i.e. due to subcooled boiling)
175  dmdtWPtr_ =
176  &
177  (
178  pair_.mesh().lookupObjectRef<volScalarField>
179  (
180  "dmdtW."+FFPairPtr_->name()
181  )
182  );
183  }
184 
185  //- Refs
186  const scalar& alphai(1.0-this->pair_.fluidRef().normalized()[celli]);
187  const scalar& Twi(Tw_[celli]);
188  const scalar& Tfi(Tf_[celli]);
189  const scalar& Tsati(FFPairPtr_->iT()[celli]);
190 
191  //- Two-phase forced-convection heat transfer coefficient
192  // and explicit heat flux, used later throughout the map
193  scalar htc2pFCi(htcFCPtr_->value(celli)*FPtr_->value(celli));
194  scalar qFCi(htc2pFCi*(Twi-Tfi));
195 
196  /*-----------------------------------------------------------------------*/
197  /*- HTC MAP TAKEN FROM TRACE, BUT THIS SUPPORTS USER-SELECTABLE MODELS --*/
198  /*-----------------------------------------------------------------------*/
199  if (Twi < Tsati) //- Wall below saturation, either condensing or
200  // nothing special happens
201  {
202  if (htcCndPtr_.valid()) //- If a film condensation model has been
203  // specified
204  {
205  if (alphai <= 0.8) //- Nothing special
206  return htc2pFCi;
207  else //- Film condensation
208  {
209  //- Film condensation heat transfer coefficient
210  scalar htcCndi(htcCndPtr_->value(celli));
211 
212  //- Linear interpolation if 0.8 < alphai < 0.9
213  if (alphai < 0.9)
214  {
215  scalar f(0.9-alphai/0.1);
216  return f*htc2pFCi+(1.0-f)*htcCndi;
217  }
218  else //- Film condensation only
219  return htcCndi;
220  }
221  }
222  else
223  return htc2pFCi;
224  }
225  else
226  {
227  //- If a TONB model has not been provided, no sub-cooled boiling
228  // and set it equal to Tsati
229  scalar TONBi
230  (
231  (TONBPtr_.valid()) ?
232  TONBPtr_->value(celli, htc2pFCi) :
233  Tsati
234  );
235  if (Twi < TONBi) //- Wall above saturation but below onset of
236  // nucleate boiling, nothing special
237  {
238  return htc2pFCi;
239  }
240  else //- Wall above onset of nucleate boiling
241  {
242  scalar TCHFi(1e69); //- Needs dedicated model for its setting
243  if (Twi < TCHFi) //- Below CHF, i.e. either nucleate boiling
244  // or subcooled boiling
245  {
246  //- Calc hPB at boiling onset, i.e. the hPB when the wall
247  // temperature is TONBi. This is done by caching Twi,
248  // setting it to TONB, using the htcPB model to calc the
249  // hPB with Tw=TONB, then re-setting the Twi
250  scalar Twi0(Twi);
251  const_cast<scalar&>(Twi) = TONBi;
252  scalar hPBONBi(htcPBPtr_->value(celli));
253  const_cast<scalar&>(Twi) = Twi0;
254  scalar hPBi(htcPBPtr_->value(celli));
255 
256  //- Explicit heat fluxes
257  scalar qBIi(hPBONBi*(TONBi-Tsati));
258  scalar qPBi(hPBi*(Twi-Tsati));
259  scalar qNBi = pow(pow(qFCi, 3)+pow(qPBi-qBIi, 3), 1.0/3.0);
260 
261  //- Please note that only the dmdtW is set in this scope as
262  // the subcooled boiling heat transfer coefficient and the
263  // nucleate boiling heat transfer coefficient are computed
264  // in the same way, outside and after this scope
265  scalar f(1.0);
266  if (SCBFPtr_.valid())
267  {
268  f = SCBFPtr_->value(celli, qNBi);
269  }
270  scalar qSCDmdti(f*(qNBi-qFCi));
271 
272  //- If fluid1 is liquid and 2 is vapour then L > 0 and
273  // this sub-cooled boiling term is also > 0. If fluid2
274  // is liquid and fluid1 is vapour then L < 0 and
275  // everything still works out in terms of the dmdtW
276  // sign, as I recall that it is positive for phase
277  // changes from fluid1 to fluid2 and negative
278  // vice-versa
279  (*dmdtWPtr_)[celli] =
280  pair_.structureRef().iAact()[celli]*qSCDmdti/
281  mag(FFPairPtr_->L()[celli]);
282 
283  scalar dT(Twi-Tfi);
284  dT = (dT >= 0.0) ? max(dT, 1e-3) : min(-dT, -1e-3);
285  return qNBi/dT;
286  }
287  else //- Post-CHF heat transfer, currently missing
288  // implementation
289  {
290  return 1.0;
291  }
292  }
293  }
294 }
