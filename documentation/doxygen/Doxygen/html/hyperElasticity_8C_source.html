73 void Foam::hyperElasticity::correct
74 (
75  volScalarField& sigmaHyd,
76  volSymmTensorField& sigmaDev,
77  volSymmTensorField& epsilonEl,
78  const labelList& addr
79 )
80 {
81  const volScalarField& mu(mesh_.lookupObject<volScalarField>("mu"));
82  const volScalarField& lambda(mesh_.lookupObject<volScalarField>("lambda"));
83  const volTensorField& F = mesh_.lookupObject<volTensorField>("F");
84 
85  forAll(addr, addrI)
86  {
87  const label cellI = addr[addrI];
88 
89  // Calculate the right Cauchy–Green deformation tensor
90  symmTensor C = symm(F[cellI].T() & F[cellI]);
91 
92  // Calculate the Green strain tensor
93  symmTensor E = 0.5*(C - I);
94 
95  // Calculate the 2nd Piola Kirchhoff stress
96  symmTensor S = 2.0*mu[cellI]*E + lambda[cellI]*tr(E)*I;
97 
98  // Calculate the Jacobian of the deformation gradient
99  scalar J = det(F[cellI]);
100 
101  // Convert the 2nd Piola Kirchhoff stress to the Cauchy stress
102  // sigma = (1.0/J)*symm(F() & S & F().T());
103  symmTensor sigma = (1.0/J)*transform(F[cellI], S);
104 
105  // hyperElasticity laws does not split between hydrostatic and
106  // deviatoric component.
107  // NOTE: maybe sigmaHyd and sigmaDev could be obtained directly from
108  // the 2nd Piola Kirchoff stress tensor
109  sigmaHyd[cellI] = tr(sigma)/3.0;
110  sigmaDev[cellI] = sigma - sigmaHyd[cellI]*I;
111 
112  const cell& c = mesh_.cells()[cellI];
113 
114  forAll(c, faceI)
115  {
116  const label patchID = mesh_.boundaryMesh().whichPatch(c[faceI]);
117 
118  if (patchID > -1 and F.boundaryField()[patchID].size())
119  {
120  const label faceID =
121  mesh_.boundaryMesh()[patchID].whichFace(c[faceI]);
122 
123  const scalarField& muF = mu.boundaryField()[patchID];
124  const scalarField& lambdaF = lambda.boundaryField()[patchID];
125  const tensorField& FF = F.boundaryField()[patchID];
126 
127  symmTensorField& sigmaDevF =
128  sigmaDev.boundaryFieldRef()[patchID];
129  scalarField& sigmaHydF =
130  sigmaHyd.boundaryFieldRef()[patchID];
131 
132  symmTensor C = symm(FF[faceID].T() & FF[faceID]);
133  symmTensor E = 0.5*(C - I);
134  symmTensor S = 2.0*muF[faceID]*E + lambdaF[faceID]*tr(E)*I;
135  scalar J = det(FF[faceID]);
136  symmTensor sigma = (1.0/J)*transform(FF[faceID], S);
137 
138  sigmaHydF[faceID] = tr(sigma)/3.0;
139  sigmaDevF[faceID] = sigma - sigmaHydF[faceID]*I;
140  }
141  }
142  }
143 }
