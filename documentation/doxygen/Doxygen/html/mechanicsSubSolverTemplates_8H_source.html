42 Foam::mechanicsSubSolver::correctedSigmaExp
43 (

46  const volTensorField& gradD
47 ) const

90 template<class Type>
91 Foam::tmp<Foam::volVectorField>
92 Foam::mechanicsSubSolver::correctedDivSigmaExp
93 (
94  const volTensorField& sigmaExp,
95  const GeometricField<Type, fvPatchField, volMesh>& sigma,
96  const volScalarField& twoMuLambda,
97  const surfaceScalarField& twoMuLambda_f,
98  const volVectorField& D,
99  const volTensorField& gradD
100 ) const
101 {
102  tmp<volVectorField> tdivSigmaExp
103  (
104  new volVectorField
105  (
106  IOobject
107  (
108  "tdivSigmaExp",
109  mesh_.time().timeName(),
110  mesh_,
111  IOobject::NO_READ,
112  IOobject::NO_WRITE
113  ),
114  mesh_,
115  dimensionedVector
116  (
117  "tdivSigmaExp",
118  sigma.dimensions()/dimLength,
119  Foam::vector::zero
120  )
121  )
122  );
123 
124  volVectorField& divSigmaExp = tdivSigmaExp.ref();
125 
126  // Multi-material derivation imposes a RhieChow approach
127  if (multiMaterialCorrection_)
128  {
129  // Normal vectors
130  surfaceVectorField nf = mesh().Sf()/mesh().magSf();
131 
132  // Delta vectors
133  surfaceVectorField deltaf = mesh().delta();
134 
135  // Non-orthogonal correction vectors
136  surfaceVectorField k = (nf - deltaf/(nf&deltaf));
137 
138  // Reverse-linear interpolated surface gradient
139  surfaceTensorField gradDf =
140  reverseLinear<tensor>(mesh()).interpolate(gradD);
141 
142  // Reverse-linear interpolated explicit stress component
143  // (the sigmaExp BCs have been set to zeroGradient)
144  surfaceTensorField sigma_f =
145  twoMuLambda_f
146  *reverseLinear<tensor>(mesh()).interpolate(sigmaExp/twoMuLambda);
147 
148  divSigmaExp = fvc::div( mesh().Sf() & sigma_f)
149  + fvc::div(twoMuLambda_f*mesh().magSf()* k & gradDf);
150  }
151  else if (RhieChowCorrection_)
152  {
153  const scalar scaleFactor
154  (
155  mechanicsDict_.lookupOrDefault<scalar>("RhieChowScaleFactor", 1.0)
156  );
157 
158  word laplacianName("laplacian(D" + D.name() + "," + D.name() + ")");
159 
160  surfaceTensorField gradDf =
161  fvc::interpolate(gradD, "interpolate(grad("+ D.name() + "))");
162 
163  // Divergence of the explicit component with RhieChow correction.
164  // The correction can be partial or total depending on the scale factor
165  divSigmaExp = fvc::div(sigma, "div(sigma)")
166  - scaleFactor*fvc::div(twoMuLambda_f*mesh().Sf() & gradDf)
167  - (1 - scaleFactor)*fvc::laplacian(twoMuLambda_f, D, laplacianName);
168  }
169  else
170  {
171  word laplacianName("laplacian(D" + D.name() + "," + D.name() + ")");
172 
173  // Divergence of the explicit component without RhieChow correction
174  divSigmaExp = fvc::div(sigma, "div(sigma)")
175  - fvc::laplacian(twoMuLambda_f, D, laplacianName);
176  }
177 
178  return tdivSigmaExp;
179 }
