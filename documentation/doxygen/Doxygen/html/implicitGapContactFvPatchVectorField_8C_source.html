43 Foam::implicitGapContactFvPatchVectorField::boundaryStiffness() const
44 {

53  scalarField volumes(patch.size());
54 
55  forAll(volumes, cellI)

66  }
67 
68  return (K)*faceAreas/volumes;

72 Foam::scalarField
73 Foam::implicitGapContactFvPatchVectorField::boundaryShearStiffness() const
74 {
75  const fvPatch& patch = this->patch();
76 
77  const fvPatchField<scalar>& mu =
78  patch.lookupPatchField<volScalarField, scalar>("mu");
79 
80  // Initialize face height with sqrt of area
81  scalarField faceHeights(Foam::sqrt(patch.magSf()));
82 
83  const pointField& localPoints = patch.patch().localPoints();
84  forAll(patch.patch().localFaces(), faceI)
85  {
86  if(mag(slip_[faceI]) > SMALL)
87  {
88  const labelList& pointLabelList = patch.patch().localFaces()[faceI];
89  scalar maxHeight = -GREAT;
90  scalar minHeight = GREAT;
91 
92  forAll(pointLabelList, pointI)
93  {
94  const label& pointIndex = pointLabelList[pointI];
95 
96  maxHeight = max
97  (
98  maxHeight,
99  localPoints[pointIndex] & (slip_[faceI]/mag(slip_[faceI]))
100  );
101 
102  minHeight = min
103  (
104  minHeight,
105  localPoints[pointIndex] & (slip_[faceI]/mag(slip_[faceI]))
106  );
107  }
108 
109  faceHeights[faceI] = maxHeight - minHeight;
110  }
111 
112  }
113 
114  return mu/faceHeights;
115 }

143 void Foam::implicitGapContactFvPatchVectorField::correctWeights()
144 {
145  // Correction factors for non matching surfaces (e.g. cylinders with a gap)
146  scalarField correctionFactor(this->size(), 1);
147 
148  if (regionCoupledPatch_.owner())
149  {
150  const AMIInterpolation& ami
151  = regionCoupledPatch_.regionCoupledPatch().AMI();
152 
153  correctionFactor = ami.srcWeightsSum();
154  }
155  else
156  {
157  const AMIInterpolation& ami
158  = regionCoupledPatch_.nbrPatch().regionCoupledPatch().AMI();
159 
160  correctionFactor = ami.tgtWeightsSum();
161  }
162 
163  // Reference to patch and neighbor patch
164  const fvPatch& patch = this->patch();
165  const fvPatch& nbrPatch = regionCoupledPatch_.neighbFvPatch();
166  const implicitGapContactFvPatchVectorField& nbr = this->neighbour();
167 
168 
169  // Patch quantities
170  // (P stands for owner cell/face, N stands for nbr)
171 
172  // Patch face mag
173  const scalarField magSfP(patch.magSf());
174  const scalarField magSfN(regionCoupledPatch_.regionCoupledPatch().interpolate(
175  nbrPatch.magSf()));
176 
177  // Owner, neighbour and common normal vectors
178  vectorField nP(patch.nf());
179  vectorField nN(-regionCoupledPatch_.regionCoupledPatch().interpolate(
180  nbrPatch.nf()));
181  vectorField n((nP + nN )/mag(nP + nN));
182 
183  // Face centers
184  vectorField cfP(patch.Cf());
185  vectorField cfN(regionCoupledPatch_.regionCoupledPatch().interpolate(
186  nbrPatch.Cf()));
187 
188  // Calculate initial gap to correct Cf position
189  vectorField initialGap = (cfN - cfP);
190 
191  // Cell centers
192  vectorField cpP(patch.boundaryMesh().mesh().C(), patch.faceCells());
193  vectorField cpN(regionCoupledPatch_.regionCoupledPatch().interpolate(
194  vectorField(patch.boundaryMesh().mesh().C(), nbrPatch.faceCells())));
195 
196  // Delta vectors
197  vectorField deltaP(cfP - cpP);
198  vectorField deltaN(cpN - cfN);
199 
200  // // Alternative deltaN
201  // vectorField deltaN(cpN - (cfN));
202  // NOTE: nbrDelta is then adjusted to correct for the presence of the gap
203  // vectorField deltaN(cpN - cfP);
204 
205  // forAll(*this, faceI)
206  // {
207  // if(nbrPatch.size() > 0)
208  // {
209  // if(mag(nN[faceI]) > SMALL)
210  // {
211 
212  // plane facePlane(cfN[faceI], n[faceI]);
213  // scalar intersect(facePlane.normalIntersect(
214  // cfP[faceI], n[faceI]));
215 
216  // vector projectedDist = intersect*n[faceI];
217 
218  // if(intersect >= 0)
219  // {
220  // deltaN[faceI] -= projectedDist;
221  // }
222  // }
223  // }
224  // }
225 
226  // 1/deltaCoeff scalars
227  scalarField deltaCoeffsP(1/max((nP & deltaP), SMALL));
228  scalarField deltaCoeffsN(1/max((nN & deltaN), SMALL));
229 
230 
231  // Patch fields
232 
233  // Calculate two*Mu+Lambda
234  const fvPatchField<scalar>& muP(
235  patch.lookupPatchField<volScalarField, scalar>("mu"));
236  const scalarField muN(regionCoupledPatch_.regionCoupledPatch().interpolate(
237  nbrPatch.lookupPatchField<volScalarField, scalar>("mu")));
238 
239  const fvPatchField<scalar>& lambdaP(
240  patch.lookupPatchField<volScalarField, scalar>("lambda"));
241  const scalarField lambdaN(regionCoupledPatch_.regionCoupledPatch().interpolate(
242  nbrPatch.lookupPatchField<volScalarField, scalar>("lambda")));
243 
244  const scalarField twoMuLambdaP = (2*muP + lambdaP);
245  const scalarField twoMuLambdaN = (2*muN + lambdaN);
246 
247  // Patch neighbour internal field and patch internal field
248  // NOTE: only taking pnf as a vectorField instead of patchNeighbourField
249  // works in multiprocessor
250  vectorField pnif = patchNeighbourField()();
251  vectorField pif = patchInternalField()();
252 
253  // Patch oldTime fields, and nbr patch current field
254  const vectorField pf0(patch.lookupPatchField<volVectorField, vector>("D_0"));
255  const vectorField pnf0(regionCoupledPatch_.regionCoupledPatch().interpolate(
256  nbrPatch.lookupPatchField<volVectorField, vector>("D_0")));
257  vectorField pnf(regionCoupledPatch_.regionCoupledPatch().interpolate(
258  this->neighbour()));
259 
260  // Explicit component of stress tensor
261  const tensorField sigmaExpP(
262  patch.lookupPatchField<volTensorField, tensor>("sigmaExp"));
263  const tensorField sigmaExpN(
264  regionCoupledPatch_.regionCoupledPatch().interpolate(
265  nbrPatch.lookupPatchField<volTensorField, tensor>("sigmaExp")));
266 
267  // Gradient patchField (necessary for non-orthogonal correction)
268  const tensorField gradP(
269  patch.lookupPatchField<volTensorField, tensor>("gradD"));
270  const tensorField gradN(
271  regionCoupledPatch_.regionCoupledPatch().interpolate(
272  nbrPatch.lookupPatchField<volTensorField, tensor>("gradD")));
273 
274  //- Normal components of the explicit stress tensor
275  vectorField Qp(nP & sigmaExpP);
276  vectorField Qn(nN & sigmaExpN);
277 
278 
279  // Non orthogonal correction
280 
281  // Non orthogonal vectors
282  vectorField kCorrP(pos(nP & deltaP)*(nP - (deltaP*deltaCoeffsP)));
283  vectorField kCorrN(pos(nN & deltaN)*(nN - (deltaN*deltaCoeffsN)));
284 
285  // Non orthogonal correction terms
286  vectorField nonOrthP = twoMuLambdaP * (kCorrP & gradP);
287  vectorField nonOrthN = twoMuLambdaN * (kCorrN & gradN);
288 
289  // Add non orthogonal correction to nbr explicit source term.
290  // No correction for owner to be consistent to how OpenFOAM handles
291  // boundary non-orthogonality
292  Qn += nonOrthN;
293 
294  // Increase slip (only when penetration is > 0)
295  // When gapWidth is zero total slip is equal to zero
296  vectorField slipPrev = slip_;
297  slip_ = slip0_ + neg(gapWidth_)*(
298  ((pnf - pnf0) - (*this - pf0)) & (I -nP*nP));
299  slip_ = neg(gapWidth_)* slip_;
300 
301  // Calculate "resistances" network
302 
303  // Interpolate back to this patch, so to avoid scalarFields of different lenght
304  const scalarField nbrBoundaryStiff(
305  regionCoupledPatch_.regionCoupledPatch().interpolate(nbr.boundaryStiffness()));
306  const scalarField nbrBoundaryShearStiff(
307  regionCoupledPatch_.regionCoupledPatch().interpolate(nbr.boundaryShearStiffness()));
308 
309  // Calculated new values for penalty factors
310  scalarField minDelta = min(1/deltaCoeffsP, 1/deltaCoeffsN);
311 
312  // Save previous penalty for augmented lag
313  scalarField penaltyFactPrev(penaltyFact_);
314 
315  penaltyFact_ = neg(gapWidth_)*penaltyScaleFact_*min(
316  boundaryStiffness(), nbrBoundaryStiff);
317  // scalarField newPenaltyFact = neg(gapWidth_)*pow(min(1/(avgDelta/10)*
318  // max(-gapWidth_, SMALL), 1.0)/0.03, 2.0)/4.0*0.03*penaltyScaleFact_*min(
319  // boundaryStiffness(), nbrBoundaryStiff);
320 
321  penaltyFactDashpot_ = neg(gapWidth_)*penaltyScaleFactDashpot_*min(
322  boundaryStiffness(), nbrBoundaryStiff);
323 
324  penaltyFact_t_ = neg(gapWidth_)*penaltyScaleFactFriction_*min(
325  boundaryShearStiffness(), nbrBoundaryShearStiff);
326  // scalarField newPenaltyFact_t = neg(gapWidth_)*pow(min(1/max(avgFaceHeighs/10, VSMALL)*
327  // mag(slip_), 1.0)/0.03, 2.0)/4.0*0.03*penaltyScaleFactFriction_*min(
328  // boundaryShearStiffness(), nbrBoundaryShearStiff);
329 
330  // Calculate spring (and dashpot) resistances
331  scalarField R_S = glued_ ? magSfP/GREAT : 1/max(penaltyFact_, SMALL);
332  scalarField R_S_t = glued_ ? magSfP/GREAT : 1/max(penaltyFact_t_, SMALL);
333  scalarField R_D = glued_ ? magSfP/SMALL : 1/max(penaltyFactDashpot_, SMALL);
334 
335  // Calculate normal "resistances"
336  const scalarField R_P = 1/twoMuLambdaP/deltaCoeffsP;
337  const scalarField R_N = 1/max(twoMuLambdaN, SMALL)/deltaCoeffsN;
338 
339  // Total local resistance (only due to dashpot and owner cell)
340  const scalarField RtotP = R_P + R_D;
341 
342  // Total coupled resistance (due to spring, nbr and owner cell)
343  const scalarField RtotC = R_P + R_N + R_S;
344 
345  // Local weights
346  wP_ = relax_*R_D/RtotP + (1 - relax_)*wP_;
347 
348  // Coupled weights
349  if (method_ == "augmentedLagrangian" and not(glued_))
350  {
351  forAll(penaltyFact_, faceI)
352  {
353  scalar oldValue = wC_[faceI];
354  scalar newValue = wC_[faceI];
355 
356  if(gapWidth_[faceI] > 0.0)
357  {
358  // This corresponds to open gap or gap re-opening
359  // Here we simply relax with a 0.1 factor.
360  // TODO: detaching relax factor to be chosen by user?
361  // or use the already existing relax factor?
362  newValue = 1;
363  wC_[faceI] = (0.9*oldValue + 0.1*newValue);
364  }
365  else if(-gapWidth_[faceI]/minDelta[faceI] > penetrationTolerance_)
366  {
367  // Here the multiplier is found as:
368  // (1 - wC)_old + (1 - wC)_new
369  // Extracting 1 from above and collecting the rest into wC_ to
370  // be used later in the rest of the algorithm we get:
371  // wC_new = wC_old + wC_new - 1
372  newValue = (1 - R_P[faceI]/RtotC[faceI]);
373  wC_[faceI] = max(min(oldValue + newValue - 1, 1.0), 0.0);
374  }
375 
376  // wC_[faceI] = max(min(0.5*(oldValue + mag(newValue)), 1.0), 0.0);
377  }
378  }
379  else
380  {
381  wC_ = relax_*(1 - R_P/RtotC) + (1 - relax_)*wC_;
382  }
383 
384  // Coupled weights for tangent component
385  wC_t_ = relax_*(1 - R_P/(R_P + R_N + R_S_t)) + (1 - relax_)*wC_t_;
386 
387 
388  // Initial gap correction
389  if(!glued_)
390  {
391  vectorField initialGapCorrection((initialGap)/R_N);
392  Qn += (initialGapCorrection & nP)*nP;
393  }
394 
395 
396  // Slip correction
397  if(!glued_)
398  {
399  vectorField slipCorrection((slip0_ + pf0 - pnf0)/R_N);
400  Qn += (slipCorrection & (I -nP*nP));
401  }
402 
403  // Set explicit components of gradient, value and laplacian vectors
404 
405  // Set gradient explicit components due to contact and pressure-dashpot
406  vectorField contactGradExp_n(deltaCoeffsP*(1 - wC_)*correctionFactor*(
407  (Qp*R_P + Qn*R_N + 0*pressure_*(R_P + R_N)*nP) & (nP*nP)));
408  vectorField contactGradExp_t(deltaCoeffsP*(1 - wC_t_)*correctionFactor*(
409  (Qp*R_P + Qn*R_N + 0*pressure_*(R_P + R_N)*nP) & (I - nP*nP)));
410  vectorField pressreDashpotGradExp(
411  (-pressure_*nP - Qp + *this/R_D)/twoMuLambdaP);
412 
413  // Full traction (trial/explicit for determining maximum tangent force)
414  vectorField traction_n( twoMuLambdaP*correctionFactor*(
415  deltaCoeffsP*(1 - wC_)*((pnif - pif) & (nP*nP))+ contactGradExp_n));
416  vectorField traction_t( twoMuLambdaP*correctionFactor*(
417  deltaCoeffsP*(1 - wC_t_)*((pnif - pif) & (I -nP*nP))+ contactGradExp_t));
418 
419  // Apply friction limiter for tangential traction.
420  scalarField magTractionT = min(
421  mag(traction_t), frictionCoeff_*mag(traction_n));
422 
423  // Calculate frictionToStickRatio
424  frictionToStickRatio_ = magTractionT/max(mag(traction_t), SMALL);
425 
426  // Adjust explicit gradient adding normal and tangent components
427  gradExp_ = wP_*(pressreDashpotGradExp + contactGradExp_n
428  + frictionToStickRatio_*contactGradExp_t);
429 
430  // Set explicit component of value
431  valueExp_ = gradExp_/deltaCoeffsP;
432 
433  // Set explicit component of laplacian that is added via manipulateMatrix
434  laplacianExp_ = gradExp_*twoMuLambdaP*magSfP;
435 }

862 (
863  const Pstream::commsTypes
864 )
865 {
866  if (!updated())
867  {
868  updateCoeffs();
869  }
870 
871  // Correction factors for non matching surfaces (e.g. cylinders with a gap)
872  scalarField correctionFactor(this->size(), 1);
873 
874  if (regionCoupledPatch_.owner())
875  {
876  const AMIInterpolation& ami
877  = regionCoupledPatch_.regionCoupledPatch().AMI();

881  else
882  {
883  const AMIInterpolation& ami
884  = regionCoupledPatch_.nbrPatch().regionCoupledPatch().AMI();
885 
886  correctionFactor = ami.tgtWeightsSum();

888 
889  // Reference to patch and neighbor patch
890  const fvPatch& patch = this->patch();
891 
892  // Normal vectors
893  vectorField n(patch.nf());

895  vectorField::operator=
896  (
897  wP_*patchInternalField()
898  + wP_*(1.0 - wC_)*correctionFactor*(
899  (this->patchNeighbourField() - this->patchInternalField()) & (n*n))
900  + wP_*(1.0 - wC_t_)*correctionFactor*frictionToStickRatio_*(
901  (this->patchNeighbourField() - this->patchInternalField()) & (I - n*n))
902  + valueExp_
903  );
