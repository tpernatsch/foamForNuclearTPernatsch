168 void Foam::FSHeatTransferCoefficientModels::multiRegimeBoiling::setDmdtW
169 (
170  const label& celli,
171  const scalar& qNBi,
172  const scalar& qFCi
173 ) const
174 {
175  //- No sub-cooled boiling possible if a model for the temperature of
176  // onest of nucleate boiling (ONB) is not provided
177  if (TONBPtr_.valid())
178  {
179  //- Fraction of the "real" (i.e. after superposition) pool boiling
180  // heat flux that results in net vapour generation
181  scalar f(1.0);
182  if (SCBFPtr_.valid())
183  {
184  f = SCBFPtr_->value(celli, qNBi);
185  }
186 
187  //- Heat flux that results in vapour generation
188  scalar qSCDmdti(f*(qNBi-qFCi));
189 
190  //- If fluid1 is liquid and 2 is vapour then L > 0 and
191  // this sub-cooled boiling term is also > 0. If fluid2
192  // is liquid and fluid1 is vapour then L < 0 and
193  // everything still works out in terms of the dmdtW
194  // sign, as I recall that it is positive for phase
195  // changes from fluid1 to fluid2 and negative
196  // vice-versa
197  (*dmdtWPtr_)[celli] =
198  pair_.structureRef().iAact()[celli]*qSCDmdti/
199  FFPairPtr_->L()[celli];
200  }

203 Foam::scalar Foam::FSHeatTransferCoefficientModels::multiRegimeBoiling::qNB
204 (
205  const label& celli,
206  const scalar& Twi1
207 ) const
208 {
209  const scalar& Tfi(Tf_[celli]);
210  const scalar& Tsati(FFPairPtr_->iT()[celli]);
211 
212  //- Cache and overwrite current wall temperature in celli if requested
213  const scalar& Twi(Tw_[celli]);
214  scalar Twi0(Twi);
215  bool cacheValues(Twi1 == Twi);
216  if (!cacheValues)
217  const_cast<scalar&>(Twi) = Twi1;
218 
219  //- Enhanced two-phase forced convection heat flux, pool boiling htc and
220  // its heat flux
221  scalar qFCi(htc2pFCi_*(Twi-Tfi));
222  scalar htcPBi(htcPBPtr_->value(celli));
223  scalar qPBi(htcPBi*(Twi-Tsati));
224 
225  //- Overall nucleate boiling htc and heat flux;
226  scalar qNBi(0.0);
227 
228  //- If computing the nucleate boiling heat flux via the TRACE approach
229  // (i.e. superposing the heat fluxes, not the htcs)
230  if (qMode_)
231  {
232  if (SPtr_.valid()) //- If a suppression factor model is the one that
233  // is used to turn-off pool boiling as the flow
234  // becomes increasingly annular
235  qNBi =
236  pow
237  (
238  pow(qFCi, exp_) + pow(SPtr_->value(celli)*qPBi, exp_),
239  oneByExp_
240  );
241  else if (TONBPtr_.valid()) //- If doing things the TRACE way, i.e.
242  // avoiding discontinuities at the onest
243  // of boiling via a qBI term
244  {
245  //- Calc pool boiling heat flux at the onset of boiling (ONB,
246  // needless to say, cannot do that if a TONB model has not been
247  // specified)
248  scalar Twi00(Twi);
249  const_cast<scalar&>(Twi) = TONBi_;
250  scalar hPBONBi(htcPBPtr_->value(celli));
251  const_cast<scalar&>(Twi) = Twi00;
252  scalar qBIi(hPBONBi*(TONBi_-Tsati));
253  qNBi = pow(pow(qFCi, exp_) + pow(qPBi-qBIi, exp_), oneByExp_);
254  }
255  else //- If previous methods are not applicable
256  qNBi = pow(pow(qFCi, exp_) + pow(qPBi, exp_), oneByExp_);
257  if (cacheValues)
258  {
259  scalar dT(Twi-Tfi);
260  dT = (dT >= 0.0) ? max(dT, 1e-3) : min(-dT, -1e-3);
261  htcNBi_ = qNBi/dT;
262  }
263  }
264  //- If computing the nucleate boiling heat flux via other approaches that
265  // superpose the htcs directly
266  else
267  {
268  scalar htcNBi(0.0);
269  if (SPtr_.valid())
270  htcNBi =
271  (
272  pow
273  (
274  pow(htc2pFCi_, exp_)+pow(SPtr_->value(celli)*htcPBi, exp_),
275  oneByExp_
276  )
277  );
278  else
279  htcNBi =
280  (
281  pow(pow(htc2pFCi_, exp_) + pow(htcPBi, exp_), oneByExp_)
282  );
283  qNBi = htcNBi_*(Twi-Tfi);
284  if (cacheValues)
285  htcNBi_ = htcNBi;
286  }
287 
288  //- Calculate subcooled boiling term
289  if (cacheValues) // and Tfi < Tsati)
290  setDmdtW(celli, qNBi, qFCi);
291  else //- Reset wall temperature in celli and return qNB
292  const_cast<scalar&>(Twi) = Twi0;
293 
294  return qNBi;
295 }

299 Foam::scalar
300 Foam::FSHeatTransferCoefficientModels::multiRegimeBoiling::value
301 (
302  const label& celli
303 ) const
304 {
305  if (FFPairPtr_ == nullptr)
306  {
307  HashTable<const FFPair*> FFPairs(pair_.mesh().lookupClass<FFPair>());
308  FFPairPtr_ = FFPairs[FFPairs.toc()[0]];
309  }
310  if (dmdtWPtr_ == nullptr)
311  {
312  //- Ptr to wall mass transfer term (i.e. due to subcooled boiling)
313  dmdtWPtr_ =
314  &
315  (
316  pair_.mesh().lookupObjectRef<volScalarField>
317  (
318  "dmdtW."+FFPairPtr_->name()
319  )
320  );
321  }
322 
323  //- Refs
324  const scalar& alphai(1.0-this->pair_.fluidRef().normalized()[celli]);
325  const scalar& Twi(Tw_[celli]);
326  const scalar& Tsati(FFPairPtr_->iT()[celli]);
327 
328  //- Two-phase forced-convection heat transfer coefficient
329  // and explicit heat flux, used later throughout the map
330  htc2pFCi_ = (htcFCPtr_->value(celli)*FPtr_->value(celli));
331 
332  /*-----------------------------------------------------------------------*/
333  /* HTC MAP FOLLOWS TRACE PHILOSOPHY, BUT SUPPORTS USER-SELECTABLE MODELS */
334  /*-----------------------------------------------------------------------*/
335  if (Twi < Tsati) //- Wall below saturation, either condensing or
336  // nothing special happens
337  {
338  if (htcCndPtr_.valid()) //- If a film condensation model has been
339  // specified
340  {
341  if (alphai <= 0.8) //- Nothing special
342  return htc2pFCi_;
343  else //- Film condensation
344  {
345  //- Film condensation heat transfer coefficient
346  scalar htcCndi(htcCndPtr_->value(celli));
347 
348  //- Linear interpolation if 0.8 < alphai < 0.9
349  if (alphai < 0.9)
350  {
351  scalar f(0.9-alphai/0.1);
352  return f*htc2pFCi_+(1.0-f)*htcCndi;
353  }
354  else //- Film condensation only
355  return htcCndi;
356  }
357  }
358  else
359  return htc2pFCi_;
360  }
361  else
362  {
363  //- If a TONB model has not been provided, no sub-cooled boiling
364  // and set it equal to Tsati
365  TONBi_ =
366  (
367  (TONBPtr_.valid()) ?
368  TONBPtr_->value(celli, htc2pFCi_) :
369  Tsati
370  );
371  if (Twi < TONBi_) //- Wall above saturation but below onset of
372  // nucleate boiling, nothing special
373  {
374  return htc2pFCi_;
375  }
376  else //- Wall above saturation and onset of nucleate boiling
377  {
378  scalar TCHFi(1e69); //- Needs dedicated model for its setting
379  if (Twi < TCHFi) //- Below CHF, i.e. either nucleate boiling
380  // or subcooled boiling
381  {
382  //- Calling qNBi also calculates the htcNBi_ so that is why
383  // it is called here
384  scalar qNBi(qNB(celli, Twi));
385  (void) qNBi; //- Suppress unused variable warining
386  return htcNBi_;
387  }
388  else //- Post-CHF heat transfer, currently missing
389  // implementation
390  {
391  return 1.0;
392  }
393  }
394  }
395 }
