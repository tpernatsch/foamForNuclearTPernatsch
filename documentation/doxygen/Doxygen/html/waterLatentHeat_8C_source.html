81 Foam::scalar Foam::latentHeatModels::water::value(const label& celli)
82 const
83 {
84  //- The critical T of water is 647.3 K, but the fit spew out 681.718 as
85  // a limit temperature for the fit. Understandable, since the data range
86  // used for the fit is 0.01-350 *C, and that above that the latent heat
87  // drops very, very rapidly to 0, which I do not want to deal with as I do
88  // not deal with supercritical transitions and shit like that. This is
89  // fine. Latent heat here is in SI units (J/kg)
90 
91  scalar T(min(max(iT_[celli], 273.16), 680));
92  return 1e6*(-2.44+0.82*log(681.718-T));
93 }
