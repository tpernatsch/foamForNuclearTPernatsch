109 void topCladRingPressureFvPatchVectorField::updateTraction()
110 {
111  // Set fuelTopPatchID_
112  if(fuelTopPatchID_ == -1)
113  {
114  const volVectorField& vfD =
115  this->db().lookupObject<volVectorField>(internalField().name());
116 
117  int nFuelTopPatch(0);
118 
119  forAll(vfD.boundaryField(), patchID)
120  {
121  if(isA<plenumSpringBase>(vfD.boundaryField()[patchID]))
122  {
123  fuelTopPatchID_ = patchID;
124  nFuelTopPatch += 1;
125  }
126  }
127 
128  if(nFuelTopPatch > 1)
129  {
130  FatalErrorInFunction()
131  << "More than one plenumSpring type patch was found." << nl
132  << "topCladRingPressure patchField requires one and "
133  << "only one plenum spring patch-type in the model."
134  << abort(FatalError);
135  }
136 
137  if(nFuelTopPatch == 0)
138  {
139  FatalErrorInFunction()
140  << "No plenumSpring type patch was found." << nl
141  << "topCladRingPressure patchField requires one and "
142  << "only one plenum spring patch-type in the model."
143  << abort(FatalError);
144  }
145  }
146 
147  // Pi
148  scalar pi = Foam::constant::mathematical::pi;
149 
150  const globalOptions& globalOpt
151  (db().lookupObject<globalOptions>("globalOptions"));
152 
153  scalar angularFraction = globalOpt.angularFraction();
154 
155  // Include the gap gas pressure (only when the cap is not explicitly modeled)
156  const gapGasModel& gapGas
157  = db().lookupObject<gapGasModel>("gapGas");
158 
159  // Reference to total displacement field
160  const volVectorField& D =
161  db().lookupObject<volVectorField>(internalField().name());
162 
163  // Request spring elongation and spring modulus from fuel top patch
164  const plenumSpringBase& fuelTopD(
165  refCast<const plenumSpringBase>(D.boundaryField()[fuelTopPatchID_]));
166 
167  scalar Dtot(fuelTopD.springElongation());
168  scalar springModulus(fuelTopD.springModulus());
169 
170  // Top clad ring area
171  scalar Atot = gSum(patch().magSf())/angularFraction;
172 
173  // Top cap area fluid side.
174  scalar topCapArea = pi*pow(topCapRadius_, 2.0);
175 
176  // Top cap area plenum side.
177  scalar topCapPlenumArea = pi*pow(topCapPlenumRadius_, 2.0);
178 
179  // Set pressures
180 
181  // Inner gap gas pressure (negative because it acts in the opposite direction
182  // on the cladding top ring)
183  scalarField gapGasPressure
184  (patch().size(), -gapGas.p()*topCapPlenumArea/Atot);
185 
186  // Spring pressure (negative because it acts in the opposite direction
187  // on the cladding top ring)
188  scalarField springPressure
189  (patch().size(), -springModulus*Dtot/Atot);
190 
191  // Fluid pressure
192  if(coolantPressureList_.valid())
193  {
194  coolantPressure_ =
195  coolantPressureList_->value
196  (
197  this->db().time().timeToUserTime
198  (
199  this->db().time().value()
200  )
201  );
202  }
203 
204  // NOTE: cannot multiply directly coolantPressure_ --> create new scalarField
205  scalarField coolantPressure(coolantPressure_*topCapArea/Atot);
206 
207  // Update total pressure
208  pressure_ = (gapGasPressure + springPressure + coolantPressure);
209 }
