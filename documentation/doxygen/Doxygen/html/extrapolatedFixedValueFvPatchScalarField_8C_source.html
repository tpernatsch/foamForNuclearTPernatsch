88 void extrapolatedFixedValueFvPatchScalarField::updateCoeffs()
89 {
90  if (this->updated())
91  {
92  return;
93  }
94 
95  // Name of the gradient field
96  const word gradFieldName("grad(" + internalField().name() + ")");
97 
98  // If available, use the gradient for non-orthogonal correction
99  if (db().foundObject<volVectorField>(gradFieldName))
100  {
101  // Lookup gradient of the field
102  const fvPatchField<vector>& gradField =
103  patch().lookupPatchField<volVectorField, vector>(gradFieldName);
104 
105  // Unit normals
106  const vectorField n(patch().nf());
107 
108  // Delta vectors
109  const vectorField delta(patch().delta());
110 
111  // Linearly extrapolate from the internal field and apply
112  // under-relaxation
113  fvPatchField<scalar>::operator==
114  (
115  (1.0 - relaxFac_)*(*this)
116  + relaxFac_
117  *(
118  patchInternalField() + (delta & gradField.patchInternalField())
119  )
120  );
121  }
122  else
123  {
124  // Zero gradient
125  fvPatchField<scalar>::operator==
126  (
127  (1.0 - relaxFac_)*(*this) + relaxFac_*patchInternalField()
128  );
129  }

131  fixedValueFvPatchScalarField::updateCoeffs();
132 }
