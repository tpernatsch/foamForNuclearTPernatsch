51 Foam::regionSolvers::regionSolvers(const Time& runTime)
52 :
53  meshHandler_(runTime)
54 {
55  List<Pair<word>> regionSolverNames;
56 
57  //Read mapping dict. Will be used later
58  IOdictionary multiRegionCouplingDict
59  (
60  IOobject
61  (
62  "multiRegionCouplingDict",
63  runTime.time().constant(),
64  runTime.db(),
65  IOobject::MUST_READ,
66  IOobject::NO_WRITE
67  )
68  );
69 
70  IOdictionary controlDict
71  (
72  IOobject
73  (
74  "controlDict",
75  runTime.system(),
76  runTime.db(),
77  IOobject::MUST_READ,
78  IOobject::NO_WRITE
79  )
80  );
81 
82 
83  if (controlDict.found("regionSolvers"))
84  {
85  const dictionary& regionSolversDict =
86  controlDict.subDict("regionSolvers");
87 
88  forAllConstIter(dictionary, regionSolversDict, iter)
89  {
90  const word regionName(iter().keyword());
91  const word solverName(iter().stream());
92 
93  regionSolverNames.append(Pair<word>(regionName, solverName));
94  }
95  }
96  else //no backward compatibility for now
97  {
98  FatalIOErrorInFunction(runTime.controlDict())
99  << "regionSolvers list missing from "
100  << runTime.controlDict().name()
101  << exit(FatalIOError);
102  }
103 
104 
105  solvers_.setSize(regionSolverNames.size());
106  prefixes_.setSize(regionSolverNames.size());
107 
108  string::size_type nRegionNameChars = 0;
109 
110  forAll(regionSolverNames, i)
111  {
112  const word& regionName = regionSolverNames[i].first();
113  const word& solverName = regionSolverNames[i].second();
114 
115  Info << "Creating solver for region "<< regionName<<nl<<endl;
116 
117  if (solverName != "multiPhysicsSolver")
118  {
119  solvers_.set(i, solver::New(solverName, meshHandler_.returnMesh(regionName)));
120  }
121  else
122  {
123  solvers_.set(i, solver::New(solverName, meshHandler_.returnMesh("dummy")));
124  Foam::solvers::multiPhysicsSolver* multiPhysicsSolverPtr = dynamic_cast<Foam::solvers::multiPhysicsSolver*>(&solvers_[i]);
125  multiPhysicsSolverPtr->getMapper(meshHandler_);
126  multiPhysicsSolverPtr->createSolvers(regionName);
127  multiPhysicsSolverPtr = nullptr;
128  }
129 
130  prefixes_[i] = regionName;
131  nRegionNameChars = max(nRegionNameChars, regionName.size());
132  }
133 
134 
135  nRegionNameChars++;
136 
137  prefix0_.append(nRegionNameChars, ' ');
138 
139  forAll(regionSolverNames, i)
140  {
141  prefixes_[i].append(nRegionNameChars - prefixes_[i].size(), ' ');
142  }
143 
144  // Create coupling fields
145 
146  meshHandler_.createCouplingFields(runTime);
147 }
